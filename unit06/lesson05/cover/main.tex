\begin{learningtargetbox}
\small
\begin{enumerate}[leftmargin=1.5em, itemsep=6pt]
  \item I can explain how the $y$-coordinate of a point on the unit circle produces the
        graph of $f(\theta)=\sin\theta$.
  \item I can explain how the $x$-coordinate of a point on the unit circle produces the
        graph of $f(\theta)=\cos\theta$.
  \item I can identify the domain, range, period, and key features (maximum, minimum, zeros)
        of $f(\theta)=\sin\theta$ and $f(\theta)=\cos\theta$ from their graphs.
\end{enumerate}
\end{learningtargetbox}
