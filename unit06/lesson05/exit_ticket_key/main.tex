\begin{enumerate}[leftmargin=1.5em, itemsep=14pt]

  \item State the \textbf{domain}, \textbf{range}, and \textbf{period} of
        $f(\theta) = \cos\theta$.

        \vspace{4pt}
        Domain: \ans{all real numbers ($-\infty, \infty$)} \quad Range: \ans{$[-1,\,1]$}

        \vspace{8pt}
        Period: \ans{$2\pi$}

  \item A student says, ``$\sin\theta$ and $\cos\theta$ have the same graph.''
        Correct the student's claim by giving \textbf{one specific value of $\theta$}
        that shows the two functions have different outputs at that angle.

        \vspace{6pt}
        At $\theta = $ \ans{$0$}, $\quad \sin\theta = $ \ans{$0$}
        $\quad$ but $\quad \cos\theta = $ \ans{$1$}.

        \vspace{8pt}
        Therefore the graphs are \ans{different} (the same / different).

        \vspace{6pt}
        \ansline{Any valid example earns full credit (e.g.\ $\theta=\pi/2$: $\sin=1$, $\cos=0$).}
        \ansline{The two graphs have the same shape but different starting values.}

\end{enumerate}
