\begin{enumerate}[leftmargin=1.5em, itemsep=14pt]

  \item The function $f(\theta) = \sin\theta$ completes one full cycle on $[0, 2\pi]$.

        \begin{enumerate}[leftmargin=1.5em, itemsep=8pt]
            \item How many complete cycles does $f(\theta) = \sin\theta$ complete on
                  $[0, 6\pi]$? \smash{\gradedblank[id=hw1a, type=short, points=1, answer={3}]{3cm}{13pt}}

            \item How many zeros does $f(\theta) = \sin\theta$ have on $[0, 4\pi]$?
                  (Count $\theta=0$ and $\theta=4\pi$.) \smash{\gradedblank[id=hw1b, type=short, points=1, answer={5}]{3cm}{13pt}} ($0, \pi, 2\pi, 3\pi, 4\pi$)

            \item On which interval(s) within $[0, 2\pi]$ is $\sin\theta > 0$?

                  \smash{\gradedblank[id=hw1c, type=short, check=text, points=1, answer={(0, \pi)}, display={$(0,\pi)$}]{8cm}{13pt}}
        \end{enumerate}

  \item \textbf{True or False?} The range of $f(\theta) = \cos\theta$ is $[0, 1]$.
        In the box, write TRUE or FALSE, then explain.

        \rubricbox[6pt]{\gradedblank[id=hw2, type=rubric, points=2,
  display={False. The range of $\cos\theta$ is $[-1,1]$: it takes negative values, e.g.\ $\cos\pi = -1$.}]{\linewidth}{31.5pt}}

  \item Without computing, explain why $\sin(\theta + 2\pi) = \sin\theta$ for any angle
        $\theta$. Your explanation should reference the unit circle.

        \rubricbox[4pt]{\gradedblank[id=hw3, type=rubric, points=2,
  display={Adding $2\pi$ is one full revolution, so the terminal ray returns to the same position and the unit-circle point (and its $y$-coordinate) is the same: $\sin(\theta+2\pi)=\sin\theta$.}]{\linewidth}{45.5pt}}

  \newpage
  \item \textbf{AP-Style Multiple Choice.} Which of the following intervals contains a
        zero of $f(\theta) = \cos\theta$?

        \vspace{4pt}
        \begin{enumerate}[label=(\Alph*), leftmargin=1.5em, itemsep=4pt]
            \item $\left(0,\;\dfrac{\pi}{4}\right)$
            \item $\left(\dfrac{\pi}{4},\;\dfrac{\pi}{2}\right)$
            \item $\left(\dfrac{\pi}{2},\;\pi\right)$ \textcolor{keyred}{\textbf{$\leftarrow$ correct}}
            \item $\left(\pi,\;\dfrac{3\pi}{2}\right)$ \hfill\textit{Letter:}~\gradedblank[id=hw4, type=choice, points=1, answer=C]{0.42in}{13pt}
        \end{enumerate}

        \begin{itemize}[leftmargin=1.5em, itemsep=3pt]
            \item $\cos\dfrac{\pi}{2} = 0$, which lies in interval (C) since
                  $\dfrac{\pi}{4} < \dfrac{\pi}{2} < \pi$. \quad
                  (The zero at $\dfrac{\pi}{2}$ is the left endpoint of (C) and strictly
                   inside the open interval.)
        \end{itemize}

  \item The maximum value of $f(\theta) = \sin\theta$ is \smash{\gradedblank[id=hw5a, type=short, points=1, answer={1}]{2cm}{13pt}} and it first occurs
        at $\theta = $ \smash{\gradedblank[id=hw5b, type=short, points=1, answer={\frac{\pi}{2}}, display={$\pi/2$}]{2cm}{13pt}}. The next time the function reaches that same
        maximum is at $\theta = $ \smash{\gradedblank[id=hw5c, type=short, points=1, answer={\frac{5\pi}{2}}, display={$5\pi/2$}]{2cm}{13pt}}.

  \item On the interval $[0, 2\pi]$, the function $f(\theta) = \cos\theta$ is
        \textbf{decreasing} on the interval \smash{\gradedblank[id=hw6a, type=short, check=text, points=1, answer={[0, \pi]}, display={$[0,\pi]$}]{5cm}{13pt}} and \textbf{increasing} on
        the interval \smash{\gradedblank[id=hw6b, type=short, check=text, points=1, answer={[\pi, 2\pi]}, display={$[\pi,2\pi]$}]{5cm}{13pt}}.

\end{enumerate}

\begin{extensionbox}
\textbf{Optional Extension:} On what interval(s) in $[0, 2\pi]$ is $f(\theta) = \cos\theta$
concave down (bending downward)? Relate your answer to the arc of the unit circle: where is
the unit-circle point moving in a way that causes the $x$-coordinate to decrease at an
increasing rate?

\vspace{6pt}
\ansline{$\cos\theta$ is concave down on $(0,\,\pi)$.}
\ansline{On this interval the unit-circle point moves from $(1,0)$ counterclockwise to $(-1,0)$.}
\ansline{The $x$-coordinate decreases, and the rate of decrease speeds up near $\theta=\pi/2$}
\ansline{(where the point moves most steeply leftward), so the curve bends downward.}
\end{extensionbox}
