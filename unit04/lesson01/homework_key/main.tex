\item The table below shows values of a sequence. Identify the sequence as arithmetic or
      geometric. Find $d$ or $r$, and write the explicit formula.

      \vspace{0.06in}
      \renewcommand{\arraystretch}{1.5}
      \begin{tabular}{|c|c|c|c|c|c|}
        \hline
        $n$    & 0  & 1  & 2   & 3   & 4   \\
        \hline
        $a(n)$ & 4  & 12 & 36  & 108 & 324 \\
        \hline
      \end{tabular}

      \vspace{0.06in}
      Type: \smash{\gradedblank[id=hw1a, type=short, check=text, points=1, answer={geometric}]{3cm}{13pt}} \quad $d$ or $r$: \smash{\gradedblank[id=hw1b, type=short, points=1, answer={r = 3}]{2.5cm}{13pt}}

      Explicit formula: $a_n = $ \smash{\gradedblank[id=hw1c, type=short, points=2, answer={4 \cdot 3^n}]{5cm}{13pt}}

\item Two terms of a sequence are given: $a_2 = 7$ and $a_6 = 19$.

      \begin{enumerate}[label=(\alph*), itemsep=8pt]
        \item Assuming the sequence is arithmetic, find $d$.

              $d = \dfrac{a_6 - a_2}{6 - 2} = $ \smash{\gradedblank[id=hw2a, type=short, points=1, answer={3}]{4cm}{13pt}} \quad \ans{$\dfrac{12}{4} = 3$}

        \item Write the explicit formula $a_n$.

              $a_n = $ \smash{\gradedblank[id=hw2b, type=short, points=2, answer={3n + 1}]{6cm}{13pt}} \quad \ans{$7 + 3(n-2) = 3n + 1$}

        \item Find $a_0$ and $a_{10}$.

              $a_0 = $ \smash{\gradedblank[id=hw2c1, type=short, points=1, answer={1}]{2.5cm}{13pt}} \quad $a_{10} = $ \smash{\gradedblank[id=hw2c2, type=short, points=1, answer={31}]{2.5cm}{13pt}}
      \end{enumerate}

\item A cell phone plan charges \$20 for month 0 (setup fee) and \$15 per month thereafter.

      \begin{enumerate}[label=(\alph*), itemsep=8pt]
        \item Write the explicit formula for the total cost $C_n$ after $n$ months.

              $C_n = $ \smash{\gradedblank[id=hw3a, type=short, points=1, answer={20 + 15n}]{5cm}{13pt}}

        \item Find the total cost after 6 months: $C_6 = $ \smash{\gradedblank[id=hw3b, type=short, points=1, answer={110}, display={\$110}]{3cm}{13pt}}

        \item In which month does the total cost first exceed \$200? \smash{\gradedblank[id=hw3c, type=short, points=2, answer={13}, display={Month 13}]{3cm}{13pt}}

      \begin{teachernote}
      $20 + 15n > 200 \Rightarrow 15n > 180 \Rightarrow n > 12$, so $n = 13$.
      $C_{12} = \$200$ exactly (not exceeding), $C_{13} = \$215$ (first exceeds).
      Accept month 12 if student interprets ``exceed'' as $\geq$.
      \end{teachernote}
      \end{enumerate}

\item For the arithmetic sequence $a_n = 3 + 5n$, find the smallest value of $n$ such that
      $a_n > 250$.

      \vspace{0.06in}
      Setup: \smash{\gradedblank[id=hw4a, type=short, check=text, points=1, answer={3 + 5n > 250}, display={$3 + 5n > 250$}]{6cm}{13pt}} \quad \ans{$\Rightarrow 5n > 247 \Rightarrow n > 49.4$}

      Smallest $n$: \smash{\gradedblank[id=hw4b, type=short, points=2, answer={50}]{2.5cm}{13pt}} \quad Term value: $a_n = $ \smash{\gradedblank[id=hw4c, type=short, points=1, answer={253}]{2.5cm}{13pt}}

\item \textbf{(AP-Style Multiple Choice)} A sequence is defined by $a_n = 4 + 3n$.
      Which of the following correctly describes this sequence?

      \medskip
      \begin{enumerate}[label=(\Alph*), itemsep=4pt]
        \item A geometric sequence with common ratio 3
        \item An arithmetic sequence with common difference 4
        \item \ans{An arithmetic sequence with common difference 3 and initial value 4}
        \item A geometric sequence with initial value 4 and common ratio $\frac{1}{3}$
      \end{enumerate}

      \vspace{0.04in}
      \leavevmode\rlap{\textit{Letter:}}\phantom{Answer:}\spacefactor2000{} \smash{\gradedblank[id=hw5a, type=choice, points=1, answer=C]{1.5cm}{13pt}} \quad Explain:

      \vspace{0.04in}
      \writegap\\
      \writebox{\gradedblank[id=hw5b, type=rubric, points=2,
  display={The formula $a_n = 4 + 3n$ matches $a_n = a_0 + dn$ with $a_0 = 4$ and $d = 3$: consecutive terms differ by the constant 3, so it is arithmetic, not geometric.}]{\linewidth}{26pt}}\\

\item \textbf{(Extension)} Consider the sequence $1,\ 1,\ 2,\ 3,\ 5,\ 8,\ 13,\ \ldots$

      \begin{enumerate}[label=(\alph*), itemsep=8pt]
        \item Is this sequence arithmetic? Explain.

              \writebox{\raisebox{-0.75pt}[0pt][0pt]{\gradedblank[id=hw6a, type=rubric, points=1,
  display={No. The differences $0, 1, 1, 2, 3, 5, \ldots$ are not constant.}]{\linewidth}{15.5pt}}}

        \item Is this sequence geometric? Explain.

              \writebox{\raisebox{-0.75pt}[0pt][0pt]{\gradedblank[id=hw6b, type=rubric, points=1,
  display={No. The ratios $1, 2, 1.5, 1.67, 1.6, \ldots$ are not constant.}]{\linewidth}{15.5pt}}}

        \item What rule does this sequence follow? (Hint: look at pairs of consecutive terms.)

              \writegap\\
              \writebox{\gradedblank[id=hw6c, type=rubric, points=2,
  display={Each term is the sum of the two preceding terms: $a_n = a_{n-1} + a_{n-2}$ (Fibonacci); neither arithmetic nor geometric.}]{\linewidth}{24pt}}\\
      \end{enumerate}

\end{enumerate}
