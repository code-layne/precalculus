\begin{enumerate}[itemsep=18pt]

\item Let $f(x) = 3x - 1$. Complete the table and state the rate of change.

      \vspace{0.08in}
      \renewcommand{\arraystretch}{1.6}
      \begin{tabular}{|c|c|c|c|c|}
        \hline
        $x$    & $0$ & $1$ & $2$ & $3$ \\
        \hline
        $f(x)$ & \ans{$-1$} & \ans{$2$} & \ans{$5$} & \ans{$8$} \\
        \hline
      \end{tabular}

      \vspace{0.06in}
      Rate of change of $f$: \ans{3}

\item State the domain of each function.

      \begin{enumerate}[label=(\alph*), itemsep=8pt]
        \item $f(x) = \sqrt{x}$ \quad Domain: \ans{$[0, \infty)$}
        \item $g(n) = 2 \cdot 3^n$, where $n$ is a whole number \quad Domain: \ans{$\{0, 1, 2, 3, \ldots\}$}
      \end{enumerate}

\item Let $f(n) = 2 \cdot 3^n$. Evaluate $f(0)$, $f(1)$, and $f(2)$.

      \vspace{0.06in}
      $f(0) = $ \ans{2} \quad $f(1) = $ \ans{6} \quad $f(2) = $ \ans{18}

\end{enumerate}
