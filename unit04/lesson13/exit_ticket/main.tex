\begin{notesbox}
\textbf{1.} A linear transformation $L$ maps $\hat{\imath}\mapsto\langle0,1\rangle$
and $\hat{\jmath}\mapsto\langle-1,0\rangle$.

\vspace{0.06in}
\textbf{(a)} Write the matrix $A$ associated with $L$.

\vspace{0.08in}
\hspace{1em}$A = \begin{bmatrix}\hspace{1.4cm}&\hspace{1.4cm}\\[10pt]\hspace{1.4cm}&\hspace{1.4cm}\end{bmatrix}$

\vspace{0.12in}
\textbf{(b)} Compute $L\bigl(\langle3,2\rangle\bigr)$. Show your work.

\writelines{3}

$L\bigl(\langle3,2\rangle\bigr) = \begin{bmatrix}\hspace{1.6cm}\\[8pt]\hspace{1.6cm}\end{bmatrix}$
\end{notesbox}
