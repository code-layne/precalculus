\begin{objectivebox}
By the end of this lesson, I will be able to:
\begin{itemize}[leftmargin=1.5em, itemsep=3pt]
  \item Find the transformation matrix from the images of $\hat{\imath}$ and $\hat{\jmath}$.
  \item Apply the rotation matrix $R(\theta)$.
  \item Compose two linear transformations by multiplying matrices in the correct order.
  \item Find the inverse of a linear transformation using $A^{-1}$.
\end{itemize}
\end{objectivebox}

\vspace{0.06in}

\begin{vocabbox}
\termblanklong{linear transformation}
\ansline{A function $L:\mathbb{R}^2\to\mathbb{R}^2$ satisfying $L(\vec u+\vec v)=L(\vec u)+L(\vec v)$ and $L(c\vec v)=cL(\vec v)$.}

\termblanklong{transformation matrix}
\ansline{The matrix $A$ with $L(\vec v)=A\vec v$; columns are $L(\hat{\imath})$ and $L(\hat{\jmath})$.}

\termblanklong{rotation matrix}
\ansline{$R(\theta)=\bigl[\begin{smallmatrix}\cos\theta&-\sin\theta\\\sin\theta&\cos\theta\end{smallmatrix}\bigr]$; rotates every vector counterclockwise by $\theta$.}

\termblanklong{dilation (area scale factor)}
\ansline{$|\det A|$; the factor by which $L$ multiplies the area of any region.}

\termblanklong{composition of transformations}
\ansline{Applying $L_1$ then $L_2$; the composition matrix equals $BA$.}

\termblanklong{inverse transformation}
\ansline{Undoes $L$; matrix is $A^{-1}$.}
\end{vocabbox}

\vspace{0.06in}

\begin{hookbox}
Rotating $30°$ then $60°$ is the same as rotating $90°$ directly, because the composition of
rotation matrices is the rotation matrix for the sum of the angles.
\end{hookbox}

\vspace{0.06in}

\begin{notesbox}{Section 1 --- Reading the Matrix from Unit-Vector Images --- Key}
\small
\textbf{Rule.}
\[
  A = \begin{bmatrix} \ans{$a_{11}$} & \ans{$a_{12}$} \\ \ans{$a_{21}$} & \ans{$a_{22}$} \end{bmatrix}
\]

\textbf{Key idea:} the columns of $A$ are the images of \ans{$\hat{\imath}$} and \ans{$\hat{\jmath}$}.

\vspace{0.08in}
\textbf{Example.} $L(\hat{\imath})=\langle3,1\rangle$, $L(\hat{\jmath})=\langle-1,2\rangle$.

\hspace{1em} $A = \begin{bmatrix} \ans{3} & \ans{-1} \\ \ans{1} & \ans{2} \end{bmatrix}$
\qquad $A\begin{bmatrix}2\\-1\end{bmatrix} = \begin{bmatrix}3(2)+(-1)(-1)\\1(2)+2(-1)\end{bmatrix} = $ \ans{$\begin{bmatrix}7\\0\end{bmatrix}$}
\end{notesbox}

\vspace{0.06in}

\begin{notesbox}{Section 2 --- The Rotation Matrix --- Key}
\small
\[
  R(\theta) = \begin{bmatrix} \ans{$\cos\theta$} & \ans{$-\sin\theta$} \\ \ans{$\sin\theta$} & \ans{$\cos\theta$} \end{bmatrix}
\]

\textbf{Why?} $R(\theta)$ maps $\hat{\imath} \to $ \ans{$\langle\cos\theta,\sin\theta\rangle$}
and $\hat{\jmath} \to $ \ans{$\langle-\sin\theta,\cos\theta\rangle$}.

\vspace{0.08in}
\textbf{Example.}
$R(90°) = \begin{bmatrix} \ans{0} & \ans{-1} \\ \ans{1} & \ans{0} \end{bmatrix}$
\qquad $R(90°)\begin{bmatrix}3\\1\end{bmatrix} = \begin{bmatrix}0(3)+(-1)(1)\\1(3)+0(1)\end{bmatrix} = $ \ans{$\begin{bmatrix}-1\\3\end{bmatrix}$}

\vspace{0.08in}
\textbf{Dilation.} $|\det A| = $ \ans{$|ad-bc|$} gives the area scale factor.

\hspace{1em} For $A=\bigl[\begin{smallmatrix}2&0\\0&3\end{smallmatrix}\bigr]$: $\det A = $ \ans{$6$},
so area scales by \ans{$6$}.
\end{notesbox}

\vspace{0.06in}

\begin{notesbox}{Section 3 --- Composing Transformations --- Key}
\small
\[
  (L_2\circ L_1)(\vec v) = B(A\vec v) = (BA)\vec v
  \qquad\Rightarrow\qquad \text{composition matrix} = \ans{$BA$}
\]

\textbf{Example.} Verify $R(60°)\cdot R(30°) = R(90°)$:

\[
  \begin{bmatrix}\tfrac{1}{2}&-\tfrac{\sqrt3}{2}\\\tfrac{\sqrt3}{2}&\tfrac{1}{2}\end{bmatrix}
  \begin{bmatrix}\tfrac{\sqrt3}{2}&-\tfrac{1}{2}\\\tfrac{1}{2}&\tfrac{\sqrt3}{2}\end{bmatrix}
  =
  \begin{bmatrix}\tfrac{\sqrt3}{4}-\tfrac{\sqrt3}{4}&-\tfrac{1}{4}-\tfrac{3}{4}\\
  \tfrac{3}{4}+\tfrac{1}{4}&-\tfrac{\sqrt3}{4}+\tfrac{\sqrt3}{4}\end{bmatrix}
  = \ans{$\begin{bmatrix}0&-1\\1&0\end{bmatrix} = R(90°)$. \checkmark}
\]
\end{notesbox}

\vspace{0.06in}

\begin{notesbox}{Section 4 --- Inverse Transformations --- Key}
\small
$L$ and $M$ are inverse transformations iff the composition matrix equals \ans{$I$}.
\[
  \text{If }L(\vec v)=A\vec v,\text{ then }L^{-1}(\vec v)=\ans{$A^{-1}\vec v$}.
\]

\textbf{Example.} $A=\bigl[\begin{smallmatrix}2&1\\1&1\end{smallmatrix}\bigr]$.

$\det A = $ \ans{$1$} \qquad
$A^{-1} = \dfrac{1}{1}\begin{bmatrix}1&-1\\-1&2\end{bmatrix} = $ \ans{$\begin{bmatrix}1&-1\\-1&2\end{bmatrix}$}

Verify: $\begin{bmatrix}2&1\\1&1\end{bmatrix}\begin{bmatrix}1&-1\\-1&2\end{bmatrix}
= \begin{bmatrix}1&0\\0&1\end{bmatrix}$ \ans{\checkmark}
\end{notesbox}

\vspace{0.06in}

\begin{practicebox}
\small
\textbf{Practice.} $L(\hat{\imath})=\langle0,-1\rangle$, $L(\hat{\jmath})=\langle1,0\rangle$.

(a) $A = $ \ans{$\begin{bmatrix}0&1\\-1&0\end{bmatrix}$} \qquad
(b) Rotation angle: \ans{$270°$ (or $-90°$) counterclockwise} \qquad

(c) $A^{-1} = $ \ans{$\begin{bmatrix}0&-1\\1&0\end{bmatrix}$} \qquad
(d) $L^{-1}$ rotates every vector \ans{$90°$ counterclockwise (or $270°$ clockwise).}
\end{practicebox}

\begin{teachernote}
In Section 3, emphasize the order: ``apply $A$ first'' means $A$ is on the \emph{right} in the
product $BA$. A common error is to write $AB$. In Section 4, note that $\det A = 1$ for all
rotation matrices --- rotations preserve area. This is a good conceptual anchor.
\end{teachernote}
