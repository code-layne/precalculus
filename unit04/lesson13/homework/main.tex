\textbf{1.} A transformation maps $\hat{\imath}\to\langle3,1\rangle$ and
$\hat{\jmath}\to\langle-2,4\rangle$.

\vspace{0.06in}
\textbf{(a)} Write the matrix $A$. \hspace{1em}
$A = \begin{bmatrix}\hspace{1.4cm}&\hspace{1.4cm}\\[10pt]\hspace{1.4cm}&\hspace{1.4cm}\end{bmatrix}$

\vspace{0.12in}
\textbf{(b)} Compute $L\bigl(\langle2,-1\rangle\bigr)$. Show your work.

\writelines{3}

$L\bigl(\langle2,-1\rangle\bigr) = \begin{bmatrix}\hspace{1.6cm}\\[8pt]\hspace{1.6cm}\end{bmatrix}$
\end{notesbox}

\begin{notesbox}
\textbf{4.} Let $L(\vec{v})=A\vec{v}$ with $A=\begin{bmatrix}3&1\\-1&2\end{bmatrix}$.

\vspace{0.06in}
\textbf{(a)} Compute $\det A$. \hspace{1em} $\det A = $ \blank{3cm}

\vspace{0.10in}
\textbf{(b)} Find $A^{-1}$ using the $2\times2$ inverse formula
$A^{-1}=\dfrac{1}{\det A}\begin{bmatrix}d&-b\\-c&a\end{bmatrix}$.

\writelines{3}

$A^{-1} = \begin{bmatrix}\hspace{1.4cm}&\hspace{1.4cm}\\[10pt]\hspace{1.4cm}&\hspace{1.4cm}\end{bmatrix}$

\vspace{0.10in}
\textbf{(c)} Apply $L^{-1}$ (i.e., multiply by $A^{-1}$) to $\langle7,1\rangle$.
Show your work.

\writelines{3}

$L^{-1}\bigl(\langle7,1\rangle\bigr) = \left\langle\hspace{1.2cm},\hspace{1.2cm}\right\rangle$
\end{notesbox}
