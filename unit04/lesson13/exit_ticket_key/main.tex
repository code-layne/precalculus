\begin{notesbox}
\textbf{1.} A linear transformation $L$ maps $\hat{\imath}\mapsto\langle0,1\rangle$
and $\hat{\jmath}\mapsto\langle-1,0\rangle$.

\vspace{0.06in}
\textbf{(a)} Write the matrix $A$ associated with $L$.

\vspace{0.06in}
\hspace{1em}$A = $ \ans{$\begin{bmatrix}0&-1\\1&0\end{bmatrix}$}

\vspace{0.12in}
\textbf{(b)} Compute $L\bigl(\langle3,2\rangle\bigr)$. Show your work.

\vspace{0.06in}
\ans{$\begin{bmatrix}0&-1\\1&0\end{bmatrix}\begin{bmatrix}3\\2\end{bmatrix}
= \begin{bmatrix}0(3)+(-1)(2)\\1(3)+0(2)\end{bmatrix}
= \begin{bmatrix}-2\\3\end{bmatrix}$}

\vspace{0.04in}
$L\bigl(\langle3,2\rangle\bigr) = $ \ans{$\begin{bmatrix}-2\\3\end{bmatrix}$}
\end{notesbox}

\vspace{0.10in}

\begin{notesbox}
\textbf{2.} The rotation matrix for angle $\theta$ is
$R_\theta = \begin{bmatrix}\cos\theta&-\sin\theta\\\sin\theta&\cos\theta\end{bmatrix}$.

\vspace{0.06in}
Write $R_{90°}$ using exact values and compute $\det(R_{90°})$.

\vspace{0.06in}
$R_{90°} = $ \ans{$\begin{bmatrix}0&-1\\1&0\end{bmatrix}$}
\qquad $\det(R_{90°}) = $ \ans{1}

\vspace{0.04in}
\ans{$\cos90°=0$, $\sin90°=1$, so $\det = (0)(0)-(-1)(1)=1$.}
\end{notesbox}

\vspace{0.10in}

\begin{notesbox}
\textbf{3.} Let $L_1$ have matrix $A=\begin{bmatrix}2&0\\0&2\end{bmatrix}$ and
$L_2$ have matrix $B=\begin{bmatrix}0&-1\\1&0\end{bmatrix}$.
Write the matrix for $L_2\circ L_1$ (apply $L_1$ first, then $L_2$).

\vspace{0.06in}
\ans{$BA = \begin{bmatrix}0&-1\\1&0\end{bmatrix}\begin{bmatrix}2&0\\0&2\end{bmatrix}
= \begin{bmatrix}0&-2\\2&0\end{bmatrix}$}

\vspace{0.04in}
Matrix for $L_2\circ L_1 = BA = $ \ans{$\begin{bmatrix}0&-2\\2&0\end{bmatrix}$}
\end{notesbox}
