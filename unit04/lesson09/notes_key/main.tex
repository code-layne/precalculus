\begin{objectivebox}
By the end of this lesson, I will be able to:
\begin{itemize}[leftmargin=1.5em, itemsep=3pt]
  \item \ans{Express a parametric function as a vector-valued position function and
        compute its magnitude to find distance from the origin.}
  \item \ans{Write and interpret a velocity vector; identify direction of motion from
        component signs; compute speed as the magnitude of the velocity vector.}
\end{itemize}
\end{objectivebox}

\vspace{0.08in}

\begin{vocabbox}
\termblanklong{vector-valued function}
\ansline{A function whose output is a vector; e.g.\ $\vec{p}(t)=\langle x(t),\,y(t)\rangle$.}

\termblanklong{position vector}
\ansline{$\vec{p}(t)$; its magnitude equals the particle's distance from the origin at time $t$.}

\termblanklong{velocity vector}
\ansline{$\vec{v}(t)=\langle x'(t),\,y'(t)\rangle$; components indicate direction of motion.}

\termblanklong{speed}
\ansline{Magnitude of the velocity vector, $\|\vec{v}(t)\|$.}
\end{vocabbox}

\vspace{0.08in}

\begin{hookbox}
A drone moves so that $x(t)=3t$ and $y(t)=t^{2}-4$ (meters, seconds).
How far from the origin is the drone at $t=2$?

\ans{At $t=2$: $x=6$, $y=0$. Distance $=\sqrt{36+0}=6$ m.
A single vector $\vec{p}(t)=\langle 3t,\,t^{2}-4\rangle$ encodes both coordinates.}
\end{hookbox}

\vspace{0.08in}

\begin{notesbox}{Section 1 --- Position as a Vector-Valued Function --- Key}
\small

\textbf{Key Idea.}\quad
\[
  \vec{p}(t) = \langle \ans{$x(t)$},\;\ans{$y(t)$}\rangle
             = x(t)\,\hat{\imath} + y(t)\,\hat{\jmath}.
\]

\[
  \|\vec{p}(t)\| = \sqrt{\ans{$[x(t)]$}^{2} + \ans{$[y(t)]$}^{2}}.
\]

\vspace{0.1in}
\textbf{Example.}\quad Drone: $x(t)=3t$, $y(t)=t^{2}-4$.

\begin{tabularx}{\linewidth}{@{}Xl@{}}
Vector-valued function: & $\vec{p}(t) = $ \ans{$\langle 3t,\;t^{2}-4\rangle$} \\[8pt]
$\vec{p}(2)$: & \ans{$\langle 6,\;0\rangle$} \\[8pt]
Distance from origin at $t=2$: & $\|\vec{p}(2)\| = \sqrt{36+0} = $ \ans{$6$ m}
\end{tabularx}
\end{notesbox}

\vspace{0.08in}

\begin{notesbox}{Section 2 --- Velocity Vector and Speed --- Key}
\small

\textbf{Key Idea.}\quad
\[
  \vec{v}(t) = \langle x'(t),\; y'(t)\rangle.
\]

\renewcommand{\arraystretch}{1.5}
\begin{tabular}{ll}
If $x'(t) > 0$: moving \ans{right} &
  \hspace{1cm} If $x'(t) < 0$: moving \ans{left} \\
If $y'(t) > 0$: moving \ans{up} &
  \hspace{1cm} If $y'(t) < 0$: moving \ans{down}
\end{tabular}

\vspace{0.1in}
\textbf{Speed} $= \|\vec{v}(t)\| = $ \ans{$\sqrt{[x'(t)]^2+[y'(t)]^2}$}

\vspace{0.1in}
\textbf{Example (continued).}\quad Drone: $x'(t)=3$, $y'(t)=2t$.

\begin{tabularx}{\linewidth}{@{}Xl@{}}
$\vec{v}(2)$: & \ans{$\langle 3,\;4\rangle$} \\[8pt]
Direction: & $x'(2)=3>0$ $\Rightarrow$ moving \ans{right};
  $y'(2)=4>0$ $\Rightarrow$ moving \ans{up} \\[8pt]
Speed at $t=2$: & $\sqrt{9+16} = $ \ans{$5$ m/s}
\end{tabularx}
\end{notesbox}

\vspace{0.08in}

\begin{notesbox}{Practice --- Key}
\small

$x(t)=t^{2}-4$, $y(t)=3t$.

\textbf{(a)} $\vec{p}(t) = $ \ans{$\langle t^{2}-4,\;3t\rangle$}

\vspace{0.08in}
\textbf{(b)} $\vec{p}(2) = $ \ans{$\langle 0,\;6\rangle$} \qquad
$\|\vec{p}(2)\| = \sqrt{0+36} = $ \ans{$6$}

\vspace{0.08in}
\textbf{(c)} Speed $= \sqrt{16+9} = $ \ans{$5$}; direction:
\ans{moving right ($x'=4>0$) and up ($y'=3>0$)}
\end{notesbox}
