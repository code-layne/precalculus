\begin{scenariobox}[Tracking a Particle]{sky}
\small
In each tier below you will work with vector-valued functions to describe the
position and motion of a particle in the plane.  Use vector notation
$\vec{p}(t)=\langle x(t),\,y(t)\rangle$ and the magnitude formula
$\|\vec{p}\|=\sqrt{[x(t)]^2+[y(t)]^2}$.
\end{scenariobox}

\vspace{0.1in}

%----------------------------------------------------------------------
% Tier R
%----------------------------------------------------------------------
\begin{tcolorbox}[colback=white, colframe=black!40,
    title=\textbf{Tier R --- Remediate},
    fonttitle=\bfseries, arc=2mm, left=3mm, right=3mm, top=2mm, bottom=2mm]
\small

A particle moves with position vector $\vec{p}(t)=\langle 3t,\,2t\rangle$.

\vspace{0.06in}
\textbf{(a)} Complete the table.

\renewcommand{\arraystretch}{1.6}
\vspace{0.04in}
\hspace{1em}
\begin{tabular}{|c|c|c|c|}
\hline
$t$ & $x(t)=3t$ & $y(t)=2t$ & $\vec{p}(t)=\langle x,y\rangle$ \\
\hline
$0$ & \blank{1.5cm} & \blank{1.5cm} & \blank{3cm} \\
\hline
$1$ & \blank{1.5cm} & \blank{1.5cm} & \blank{3cm} \\
\hline
$2$ & \blank{1.5cm} & \blank{1.5cm} & \blank{3cm} \\
\hline
\end{tabular}

\vspace{0.1in}
\textbf{(b)} Find $\|\vec{p}(2)\|$.

\vspace{0.06in}
\hspace{1em} $\|\vec{p}(2)\| = \sqrt{\blank{1.5cm}^{2}+\blank{1.5cm}^{2}}
= \sqrt{\blank{2cm}} = $ \blank{3cm}

\end{tcolorbox}

\vspace{0.1in}

%----------------------------------------------------------------------
% Tier A
%----------------------------------------------------------------------
\begin{tcolorbox}[colback=white, colframe=black!40,
    title=\textbf{Tier A --- Apply},
    fonttitle=\bfseries, arc=2mm, left=3mm, right=3mm, top=2mm, bottom=2mm]
\small

A particle moves so that $f(t)=(t^{2}-4,\;3t)$.

\vspace{0.06in}
\textbf{(a)} Express $f(t)$ as a vector-valued position function:
$\vec{p}(t) = \langle \blank{4.5cm}\rangle$

\vspace{0.1in}
\textbf{(b)} When is the particle on the $y$-axis?  Set $x(t)=0$ and solve for $t$.

\vspace{0.06in}
\hspace{1em} $t^{2}-4=0$ $\Rightarrow$ $t=$ \blank{2.5cm}

\hspace{1em} Corresponding position(s) on $y$-axis: \blank{4cm}

\vspace{0.1in}
\textbf{(c)} At $t=1$ the rates of change are $x'(1)=2$ and $y'(1)=3$.
Find the speed at $t=1$.

\vspace{0.06in}
\hspace{1em} $\vec{v}(1) = \langle \blank{3cm}\rangle$ \qquad
Speed $= \sqrt{\blank{2cm}} = $ \blank{3cm}

\end{tcolorbox}

\vspace{0.1in}

%----------------------------------------------------------------------
% Tier E
%----------------------------------------------------------------------
\begin{tcolorbox}[colback=white, colframe=black!40,
    title=\textbf{Tier E --- Extend},
    fonttitle=\bfseries, arc=2mm, left=3mm, right=3mm, top=2mm, bottom=2mm]
\small

A particle moves with position vector $\vec{p}(t)=\langle 2\cos t,\;2\sin t\rangle$.

\vspace{0.06in}
\textbf{(a)} Show algebraically that $\|\vec{p}(t)\|=2$ for all values of $t$.

\writelines{3}

\vspace{0.06in}
\textbf{(b)} The velocity vector is $\vec{v}(t)=\langle -2\sin t,\;2\cos t\rangle$.
Evaluate $\vec{v}(0)$.

\vspace{0.06in}
\hspace{1em} $\vec{v}(0) = \langle \blank{3cm}\rangle$

\vspace{0.06in}
\textbf{(c)} At $t=0$ the particle is at position \blank{2.5cm}.
Using $\vec{v}(0)$, describe the direction in which the particle is moving at that instant.

\writelines{2}

\end{tcolorbox}
