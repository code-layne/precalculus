\textbf{1.} Find the distance between the points $P_1 = (1, 3)$ and $P_2 = (5, 6)$.
Show your work using the distance formula.

\vspace{0.08in}
\hspace{1em} $d = \sqrt{(x_2 - x_1)^2 + (y_2 - y_1)^2} = $
\ans{$\sqrt{(5-1)^2+(6-3)^2} = \sqrt{16+9} = \sqrt{25}$}

\vspace{0.12in}
\hspace{1em} Distance: \ans{$5$}

\vspace{0.22in}

\textbf{2.} A point on the unit circle corresponds to angle $\theta = \dfrac{2\pi}{3}$.

\vspace{0.08in}
\hspace{1em} (a)\quad $\cos\!\left(\dfrac{2\pi}{3}\right) = $ \ans{$-\dfrac{1}{2}$}
\qquad
(b)\quad $\sin\!\left(\dfrac{2\pi}{3}\right) = $ \ans{$\dfrac{\sqrt{3}}{2}$}

\vspace{0.18in}
\hspace{1em} (c)\quad Write the coordinates of the point on the unit circle at
angle $\theta = \dfrac{2\pi}{3}$: \ans{$\left(-\dfrac{1}{2},\,\dfrac{\sqrt{3}}{2}\right)$}

\vspace{0.22in}

\textbf{3.} A drone travels with magnitude $r = 10$ at direction angle
$\theta = 60^\circ$ from the positive $x$-axis.
Find the horizontal and vertical components of the displacement.

\vspace{0.08in}
\hspace{1em} Horizontal component $= r\cos\theta = $
\ans{$10\cos 60^\circ = 10\cdot\dfrac{1}{2} = 5$}

\vspace{0.1in}
\hspace{1em} Vertical component $= r\sin\theta = $
\ans{$10\sin 60^\circ = 10\cdot\dfrac{\sqrt{3}}{2} = 5\sqrt{3}$}
