\begin{tcolorbox}[colback=white, colframe=black!40,
    title=\textbf{Tier R --- Remediate},
    fonttitle=\bfseries, arc=2mm, left=3mm, right=3mm, top=2mm, bottom=2mm]

\textbf{Directions:} For each equation below, identify the type of conic,
the center or vertex, and the key measurements. No equation-writing is required.

\vspace{6pt}
\textbf{Card A:}\quad $y - 4 = 3(x + 2)^2$

Type: \ans{parabola} \quad Vertex: \ans{$(-2,\,4)$} \quad $a =$ \ans{$3$} \quad Opens: \ans{up}

\vspace{10pt}
\textbf{Card B:}\quad $\dfrac{(x-3)^2}{25} + \dfrac{(y+1)^2}{9} = 1$

Type: \ans{ellipse} \quad Center: \ans{$(3,\,-1)$} \quad $a =$ \ans{$5$} \quad $b =$ \ans{$3$}

\vspace{10pt}
\textbf{Card C:}\quad $(x+4)^2 + (y-2)^2 = 36$

Type: \ans{circle} \quad Center: \ans{$(-4,\,2)$} \quad $r =$ \ans{$6$}

\vspace{10pt}
\textbf{Card D:}\quad $\dfrac{(x-1)^2}{9} - \dfrac{(y-2)^2}{4} = 1$

Type: \ans{hyperbola} \quad Center: \ans{$(1,\,2)$} \quad $a =$ \ans{$3$} \quad $b =$ \ans{$2$} \quad Opens: \ans{left and right}

\end{tcolorbox}

\begin{teachernote}
Card D: the positive $x$-term means the hyperbola opens left and right.
Students who say ``up and down'' are reading the wrong term as positive --- redirect.
\end{teachernote}

\begin{tcolorbox}[colback=white, colframe=black!40,
    title=\textbf{Tier A --- Approaching Proficiency},
    fonttitle=\bfseries, arc=2mm, left=3mm, right=3mm, top=2mm, bottom=2mm]

\textbf{Directions:} Write the standard form equation for each card below.

\vspace{6pt}
\textbf{Card A:} Opens right, vertex $(-1, 2)$, passes through $(3, 4)$.

Use $x - h = a(y - k)^2$: $\;x - (-1) = a(y - 2)^2$.
Substitute $(3, 4)$: $3 + 1 = a(4-2)^2 \Rightarrow 4 = 4a \Rightarrow a = 1$.

\ans{$x + 1 = (y - 2)^2$}

\vspace{8pt}
\textbf{Card B:} Ellipse, center $(0,0)$, $a = 4$ (horizontal), $b = 3$ (vertical).

\ans{$\dfrac{x^2}{16} + \dfrac{y^2}{9} = 1$}

\vspace{8pt}
\textbf{Card C:} Hyperbola, center $(1,0)$, $a = 3$, $b = 2$, opens left/right.

\ans{$\dfrac{(x-1)^2}{9} - \dfrac{y^2}{4} = 1$}

\vspace{8pt}
\textbf{Card D:} Circle, center $(-2,5)$, radius $3$.

\ans{$(x+2)^2 + (y-5)^2 = 9$}

\end{tcolorbox}

\begin{teachernote}
Card A: students may try $y - k = a(x-h)^2$ (the wrong orientation). Redirect:
``The problem says 'opens to the right,' so which variable is squared?''
Card C: center is $(1, 0)$, so $k = 0$ and the $y$-term has no shift.
\end{teachernote}

\begin{tcolorbox}[colback=white, colframe=black!40,
    title=\textbf{Tier E --- Extension},
    fonttitle=\bfseries, arc=2mm, left=3mm, right=3mm, top=2mm, bottom=2mm]

\textbf{Part 1.} $x$-intercepts and $y$-intercepts of $\dfrac{x^2}{16} + \dfrac{y^2}{9} = 1$.

$x$-intercepts (set $y = 0$): $\dfrac{x^2}{16} = 1 \Rightarrow x = \pm 4$.
\ans{$x$-intercepts: $(-4,\,0)$ and $(4,\,0)$}

$y$-intercepts (set $x = 0$): $\dfrac{y^2}{9} = 1 \Rightarrow y = \pm 3$.
\ans{$y$-intercepts: $(0,\,-3)$ and $(0,\,3)$}

\vspace{8pt}
\textbf{Part 2.} Asymptotes of $\dfrac{(x-1)^2}{9} - \dfrac{y^2}{4} = 1$.

Center $(1, 0)$, $a = 3$, $b = 2$.
\ans{$y = \dfrac{2}{3}(x-1)$ \quad and \quad $y = -\dfrac{2}{3}(x-1)$}

\vspace{8pt}
\textbf{Part 3.} Card E --- student-created. Accept any valid conic with matching equation.

\ans{Answers will vary. Check: the stated features match the equation, the correct standard form is used, and the equation is fully written.}

\end{tcolorbox}

\begin{teachernote}
Part 3 is a creativity and synthesis check. Require students to verify their equation
reproduces the features they stated (substitute the center/vertex into the equation to
confirm it works).
\end{teachernote}
