%----------------------------------------------------------------------
% Vocabulary
%----------------------------------------------------------------------
\begin{vocabbox}
\begin{description}[leftmargin=0pt, itemsep=6pt]
  \item[\termblanklong{conic section}]
    A curve formed by the intersection of a plane and a double cone:
    a \ans{parabola}, an \ans{ellipse}, or a \ans{hyperbola}.

  \item[\termblanklong{vertex (parabola)}]
    The point $(h, k)$ in the standard form $y - k = a(x-h)^2$ or
    $x - h = a(y-k)^2$. It is the \ans{turning} point of the parabola.

  \item[\termblanklong{center (ellipse / hyperbola)}]
    The point $(h, k)$ at the \ans{center} of an ellipse or hyperbola;
    read from $h$ and $k$ in the standard form.

  \item[\termblanklong{semi-major / semi-minor axis}]
    For an ellipse: the larger radius is $a$ (\ans{semi-major}) and the
    smaller is $b$ (\ans{semi-minor}).

  \item[\termblanklong{asymptote (hyperbola)}]
    Lines the hyperbola approaches but never reaches:
    $y - k = \pm\dfrac{\;\;}{\;\;}(x - h)$. \quad Fill in $b$ over $a$: \ans{$b/a$}
\end{description}
\end{vocabbox}

\begin{hookbox}
\textbf{Entry Question:} Planetary orbits, telescope mirrors, and navigation
systems all use conic sections.

\vspace{4pt}
\textbf{Q1.} Name one real-world application of a parabola and one of an ellipse.
\ansline{Parabola: satellite dish / telescope mirror / bridge arch. Ellipse: planetary orbit / whispering gallery.}

\textbf{Q2.} What do you predict all three conic shapes have in common
algebraically?
\ansline{All can be described by second-degree (quadratic) equations in $x$ and $y$; all arise from distance relationships.}
\end{hookbox}

%----------------------------------------------------------------------
% Section 1 — Parabolas
%----------------------------------------------------------------------
\begin{notesbox}{Section 1 --- Parabolas (LO 4.6.A, EK 4.6.A.1)}

\textbf{Standard forms:}

\vspace{4pt}
\renewcommand{\arraystretch}{1.8}
\begin{tabularx}{\linewidth}{@{} p{0.38\linewidth} p{0.30\linewidth} X @{}}
\textbf{Form} & \textbf{Axis} & \textbf{Direction} \\
\midrule
$y - k = a(x - h)^2$ & vertical & $a > 0$: opens \ans{up};
  $a < 0$: opens \ans{down} \\[4pt]
$x - h = a(y - k)^2$ & horizontal & $a > 0$: opens \ans{right};
  $a < 0$: opens \ans{left} \\
\end{tabularx}

\vspace{8pt}
\textbf{Worked Example:}\quad $y + 3 = 2(x - 1)^2$

Rewrite: $y - ($ \ans{$-3$} $) = 2(x - $ \ans{$1$} $)^2$

Vertex: \ans{$(1,\,-3)$} \quad $a =$ \ans{$2$}, so opens: \ans{up}

\vspace{10pt}
\textbf{Student Practice:}\quad $y - 5 = -(x + 2)^2$

Vertex: \ans{$(-2,\,5)$} \quad $a =$ \ans{$-1$}, so opens: \ans{down}

\end{notesbox}

\begin{teachernote}
For the practice problem, students must rewrite $(x+2)^2$ as $(x-(-2))^2$ to read
the vertex correctly. Cold-call: ``What is the value of $h$ here?''
\end{teachernote}

% Section 2 — Ellipses and Circles
%----------------------------------------------------------------------
\begin{notesbox}{Section 2 --- Ellipses and Circles (LO 4.6.A, EK 4.6.A.2)}

\textbf{Standard form of an ellipse:}
\[
  \frac{(x-h)^2}{a^2} + \frac{(y-k)^2}{b^2} = 1
\]

\textbf{Worked Example:}\quad
$\dfrac{(x+1)^2}{16} + \dfrac{(y-3)^2}{9} = 1$

\vspace{6pt}
Center: \ans{$(-1,\,3)$} \quad
$a^2 =$ \ans{$16$} $\Rightarrow a =$ \ans{$4$} (horizontal) \quad
$b^2 =$ \ans{$9$} $\Rightarrow b =$ \ans{$3$} (vertical)

\vspace{10pt}
\textbf{Circle (special case):}\quad When $a = b = r$, the equation becomes

\[
  (x - h)^2 + (y - k)^2 = r^2
\]

\textbf{Student Practice:}\quad
$\dfrac{(x-3)^2}{25} + \dfrac{(y+1)^2}{4} = 1$

Center: \ans{$(3,\,-1)$} \quad $a =$ \ans{$5$} \quad $b =$ \ans{$2$}

\end{notesbox}

\vspace{0.08in}

%----------------------------------------------------------------------
% Section 3 — Hyperbolas
%----------------------------------------------------------------------
\begin{notesbox}{Section 3 --- Hyperbolas (LO 4.6.A, EK 4.6.A.3)}

\textbf{Standard forms:}

\vspace{4pt}
\renewcommand{\arraystretch}{1.8}
\begin{tabularx}{\linewidth}{@{} p{0.44\linewidth} X @{}}
\textbf{Form} & \textbf{Opens} \\
\midrule
$\dfrac{(x-h)^2}{a^2} - \dfrac{(y-k)^2}{b^2} = 1$ &
  left and right ($x$-term positive) \\[4pt]
$\dfrac{(y-k)^2}{b^2} - \dfrac{(x-h)^2}{a^2} = 1$ &
  up and down ($y$-term positive) \\
\end{tabularx}

\vspace{8pt}
\textbf{Asymptotes:}\quad $y - k = \pm\dfrac{b}{a}(x - h)$

\vspace{10pt}
\textbf{Worked Example:}\quad
$\dfrac{(x-2)^2}{9} - \dfrac{(y+1)^2}{4} = 1$

Center: \ans{$(2,\,-1)$} \quad $a =$ \ans{$3$} \quad $b =$ \ans{$2$} \quad
Opens: \ans{left and right}

Asymptotes: $y + 1 = \pm$ \ans{$\dfrac{2}{3}$} $(x - 2)$

\vspace{10pt}
\textbf{Student Practice:}\quad
$\dfrac{(y+3)^2}{16} - \dfrac{(x-1)^2}{9} = 1$

Center: \ans{$(1,\,-3)$} \quad $a =$ \ans{$4$} \quad $b =$ \ans{$3$} \quad
Opens: \ans{up and down}

Asymptotes: $y + 3 = \pm$ \ans{$\dfrac{4}{3}$} $(x - 1)$

\end{notesbox}

\begin{teachernote}
For the practice hyperbola, $b^2 = 16$ gives $b = 4$ and $a^2 = 9$ gives $a = 3$.
The $y$-term is positive, so the hyperbola opens up and down. The asymptote slope
is $b/a = 4/3$ (not $a/b$). This is the most common error --- clarify by writing
the formula $y - k = \pm(b/a)(x-h)$ with the denominator labeled.
\end{teachernote}
