\begin{enumerate}[itemsep=18pt]

\item A substance decays at a rate of 8\% per year. Its initial amount is 300 grams.

      \begin{enumerate}[label=(\alph*), itemsep=6pt]
        \item Write a model $f(t) = ab^t$ for the amount remaining after $t$ years.

              \vspace{0.06in}
              $a = $ \smash{\gradedblank[id=hw1a1, type=short, points=1, answer={300}]{3cm}{13pt}} \quad $b = $ \smash{\gradedblank[id=hw1a2, type=short, points=1, answer={0.92}]{3cm}{13pt}} \quad $f(t) = $ \smash{\gradedblank[id=hw1a3, type=short, points=1, answer={300 \cdot (0.92)^t}]{5cm}{13pt}}

        \item Is this growth or decay? How do you know from the value of $b$?
              \rubricbox{\gradedblank[id=hw1b, type=rubric, points=2, display={Decay; $b = 0.92 < 1$.}]{\linewidth}{13pt}}
      \end{enumerate}

\item An exponential function passes through $(2,\,20)$ and $(5,\,540)$. Find $a$ and $b$.
      Show all algebraic steps.

      \vspace{0.08in}
      Equation 1: \ans{$20 = ab^2$} \quad Equation 2: \ans{$540 = ab^5$}

      \vspace{0.06in}
      Divide to eliminate $a$: \ans{$540/20 = b^{5-2}$, so $b^3 = 27$}

      \vspace{0.06in}
      $b = $ \smash{\gradedblank[id=hw2a, type=short, points=2, answer={3}]{3cm}{13pt}} \quad then $a = $ \smash{\gradedblank[id=hw2b, type=short, points=2, answer={\frac{20}{9}}, display={$20/9$}]{4cm}{13pt}} \quad \ans{$20/3^2 = 20/9 \approx 2.22$}

      \vspace{0.04in}
      Model: $f(x) = $ \smash{\gradedblank[id=hw2c, type=short, points=1, answer={\frac{20}{9} \cdot 3^x}, display={$(20/9) \cdot 3^x$}]{7cm}{13pt}}

\item A radioactive sample has a half-life of 10 years. Its initial mass is 80 grams.

      \begin{enumerate}[label=(\alph*), itemsep=8pt]
        \item Write the exponential model $M(t)$ for the mass remaining after $t$ years.

              \vspace{0.06in}
              $M(t) = $ \smash{\gradedblank[id=hw3a, type=short, points=1, answer={80 \cdot (0.5)^{t/10}}]{7cm}{13pt}}

        \item Predict the mass after 25 years. Show your work.

              \vspace{0.06in}
              $M(25) = $ \ans{$80 \cdot (0.5)^{2.5} = 80 \cdot 0.1768$} $\approx$ \smash{\gradedblank[id=hw3b, type=short, check=numeric, tolerance=0.05, points=2, answer={14.1}]{2cm}{13pt}} grams

        \item Is $t = -5$ a meaningful input in this context? Explain.

              \rubricbox[0.06in]{\gradedblank[id=hw3c, type=rubric, points=2,
  display={No: $t$ is the time elapsed since the start; a negative time has no meaning here.}]{\linewidth}{17pt}}
      \end{enumerate}

\item The function $f(d) = 5 \cdot 3^d$ models a quantity that grows daily, where $d$ is the
      number of days.

      \begin{enumerate}[label=(\alph*), itemsep=8pt]
        \item Rewrite $f$ as a function of weeks $w$ (where $d = 7w$).

              \vspace{0.06in}
              $g(w) = $ \smash{\gradedblank[id=hw4a, type=short, points=2, answer={5 \cdot 2187^w}]{8cm}{13pt}} \quad \ans{$5 \cdot (3^7)^w$}

        \item What does the base of $g(w)$ tell you about the weekly growth of the quantity?

              \rubricbox[0.04in]{\gradedblank[id=hw4b, type=rubric, points=1,
  display={The quantity multiplies by 2187 every week.}]{\linewidth}{15.5pt}}

        \item Rewrite $f$ as a function of 2-day periods $n$ (where $d = 2n$).

              \vspace{0.06in}
              $h(n) = $ \smash{\gradedblank[id=hw4c1, type=short, points=1, answer={5 \cdot 9^n}]{8cm}{13pt}} \quad Base: \smash{\gradedblank[id=hw4c2, type=short, points=1, answer={9}]{2cm}{13pt}}
      \end{enumerate}

\item \textbf{(AP-Style Multiple Choice)} Write the letter of the single best answer in its Letter box. A population is modeled by $f(t) = 500 e^{0.04t}$,
      where $t$ is in years. Which of the following best describes the approximate annual growth rate?

      \medskip
      \begin{enumerate}[label=(\Alph*), itemsep=4pt]
        \item 0.04\%
        \item 4\%
        \item 4.08\%
        \item 40\%
      \end{enumerate}

      \vspace{0.06in}
      \leavevmode\rlap{\textit{Letter:}}\phantom{Answer:}\spacefactor2000{} \gradedblank[id=hw5a, type=choice, points=1, answer=C]{1.5cm}{13pt} \quad Reasoning:
      \rubricbox[1pt]{\gradedblank[id=hw5b, type=rubric, points=2,
  display={$e^{0.04} \approx 1.0408$, a growth factor of 1.0408: about $4.08\%$ per year.}]{\linewidth}{13pt}}

\item \textbf{(Extension)} The table below shows temperature readings (in ${}^\circ$C) at time
      $t$ minutes in a cooling experiment.

      \begin{center}
      \renewcommand{\arraystretch}{1.5}
      \begin{tabular}{c|c}
      $t$ (min) & Temperature (${}^\circ$C) \\ \hline
      0  & 80 \\
      5  & 55 \\
      10 & 42.5 \\
      15 & 36.25 \\
      \end{tabular}
      \end{center}

      The temperature is approaching a room temperature of 25${}^\circ$C. Let $y = T - 25$
      (subtract room temperature).

      \begin{enumerate}[label=(\alph*), itemsep=6pt]
        \item Compute the adjusted values $y = T - 25$ and check whether they form an
              exponential pattern (compute successive ratios).

              \vspace{0.06in}
              \renewcommand{\arraystretch}{1.6}
              \begin{tabular}{c|c|c}
              $t$ & $y = T - 25$ & Ratio (successive) \\ \hline
              0  & \smash{\gradedblank[id=hw6a1, type=short, points=1, answer={55}]{2.5cm}{14pt}} & --- \\
              5  & \smash{\gradedblank[id=hw6a2, type=short, points=1, answer={30}]{2.5cm}{14pt}} & \smash{\gradedblank[id=hw6r2, type=short, check=numeric, tolerance=0.005, points=1, answer={0.545}, display={$30/55 \approx 0.545$}]{2.5cm}{14pt}} \\
              10 & \smash{\gradedblank[id=hw6a3, type=short, points=1, answer={17.5}]{2.5cm}{14pt}} & \smash{\gradedblank[id=hw6r3, type=short, check=numeric, tolerance=0.005, points=1, answer={0.583}, display={$17.5/30 \approx 0.583$}]{2.5cm}{14pt}} \\
              15 & \smash{\gradedblank[id=hw6a4, type=short, points=1, answer={11.25}]{2.5cm}{14pt}} & \smash{\gradedblank[id=hw6r4, type=short, check=numeric, tolerance=0.005, points=1, answer={0.643}, display={$11.25/17.5 \approx 0.643$}]{2.5cm}{14pt}} \\
              \end{tabular}

              \vspace{0.04in}
              \ans{The ratios are approximately equal ($\approx 0.5$ to $0.6$), suggesting an exponential decay pattern (slight variation may be due to rounding).}

        \item Write an exponential model for $y(t)$, then write the model for $T(t)$.

              \vspace{0.06in}
              $y(t) = $
              \rubricbox{\gradedblank[id=hw6b1, type=rubric, points=1, display={$55 \cdot (0.55)^{t/5}$}]{\linewidth}{13pt}}
              $T(t) = $
              \rubricbox{\gradedblank[id=hw6b2, type=rubric, points=1, display={$55 \cdot (0.55)^{t/5} + 25$}]{\linewidth}{13pt}}
      \end{enumerate}

\end{enumerate}
