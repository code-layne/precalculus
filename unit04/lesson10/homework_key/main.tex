% ---- Problem 1 ----
\item Consider the matrix $M = \begin{bmatrix}4 & -1 & 0 \\ 2 & 7 & 3\end{bmatrix}$.
\begin{enumerate}[label=(\alph*), itemsep=6pt]
  \item Dimensions of $M$: \ans{$2 \times 3$}
  \item $m_{12} = $ \ans{$-1$} \quad (row 1, column 2)
  \item $m_{23} = $ \ans{$3$} \quad (row 2, column 3)
\end{enumerate}

\vspace{0.2in}

% ---- Problem 2 ----
\item Compatibility:

\medskip
(a) $4\times 3$ and $3\times 2$: \ans{Defined; product is $4 \times 2$.}

\smallskip
(b) $2\times 5$ and $3\times 1$: \ans{Not defined; inner dimensions $5 \neq 3$.}

\smallskip
(c) $1\times 4$ and $4\times 3$: \ans{Defined; product is $1 \times 3$.}

\smallskip
(d) $3\times 3$ and $3\times 3$: \ans{Defined; product is $3 \times 3$.}

\vspace{0.2in}

% ---- Problem 5 ----
\item Nutrition context.
\begin{enumerate}[label=(\alph*), itemsep=6pt]
  \item $N$ is $2\times 3$ and $\mathbf{s}$ is $2\times 1$. \ans{Not defined; inner dimensions $3 \neq 2$.}\\[4pt]
        \textit{Instructor note: $N^{\top}\mathbf{s}$ (or $\mathbf{s}^{\top}N$) would be the meaningful product. Accept either interpretation as long as the student correctly checks compatibility.}
  \item \ans{$N^{\top}\mathbf{s}$: $N^{\top}$ is $3\times 2$ and $\mathbf{s}$ is $2\times 1$, defined.\\[3pt]
        $N^{\top}\mathbf{s} = \begin{bmatrix}0.25(200)+0.30(150)\\0.10(200)+0.05(150)\\0.60(200)+0.55(150)\end{bmatrix} = \begin{bmatrix}95\\27.5\\202.5\end{bmatrix}$\\[4pt]
        Protein: 95 g, Fat: 27.5 g, Carbohydrates: 202.5 g}
\end{enumerate}

\vspace{0.2in}

% ---- Problem 6 ----
\item \textbf{AP-Style MC:} $P$ is $3\times k$, $Q$ is $2\times 4$, and $PQ$ is defined.
For $PQ$ to be defined the columns of $P$ must equal the rows of $Q$, so $k = 2$.
\ans{(A) $k = 2$}

\end{enumerate}
