\begin{tcolorbox}[colback=white, colframe=black!40,
    title=\textbf{Tier R --- Remediate: Matrix-Vector Product},
    fonttitle=\bfseries, arc=2mm, left=3mm, right=3mm, top=2mm, bottom=2mm]

Let $A = \begin{bmatrix}1 & 2 \\ 3 & 4\end{bmatrix}$ and $\mathbf{v} = \begin{bmatrix}5 \\ 6\end{bmatrix}$.

\medskip
\textbf{Step 1.} Dimensions of $A$: \ans{$2 \times 2$}
Dimensions of $\mathbf{v}$: \ans{$2 \times 1$}
$A\mathbf{v}$ defined? \ans{Yes}

\medskip
\textbf{Step 2.}

Entry $(1,1)$: $1(5) + 2(6) = $ \ans{$17$}

\medskip
Entry $(2,1)$: $3(5) + 4(6) = $ \ans{$39$}

\medskip
$A\mathbf{v} = $ \ans{$\begin{bmatrix}17 \\ 39\end{bmatrix}$}

\end{tcolorbox}

\vspace{0.1in}

\begin{tcolorbox}[colback=white, colframe=black!40,
    title=\textbf{Tier A --- Apply: Does Order Matter?},
    fonttitle=\bfseries, arc=2mm, left=3mm, right=3mm, top=2mm, bottom=2mm]

Let $A = \begin{bmatrix}2 & 1 \\ 0 & 3\end{bmatrix}$ and $B = \begin{bmatrix}1 & 4 \\ 2 & 0\end{bmatrix}$.

\medskip
\textbf{(a)} $AB = \begin{bmatrix}2(1)+1(2) & 2(4)+1(0) \\ 0(1)+3(2) & 0(4)+3(0)\end{bmatrix}
= $ \ans{$\begin{bmatrix}4 & 8 \\ 6 & 0\end{bmatrix}$}

\medskip
\textbf{(b)} $BA = \begin{bmatrix}1(2)+4(0) & 1(1)+4(3) \\ 2(2)+0(0) & 2(1)+0(3)\end{bmatrix}
= $ \ans{$\begin{bmatrix}2 & 13 \\ 4 & 2\end{bmatrix}$}

\medskip
\textbf{(c)} $AB = BA$? \ans{No}

Matrix multiplication is \ans{not commutative} in general: $AB \neq BA$.

\end{tcolorbox}

\vspace{0.1in}

\begin{tcolorbox}[colback=white, colframe=black!40,
    title=\textbf{Tier E --- Extend: Revenue Matrix},
    fonttitle=\bfseries, arc=2mm, left=3mm, right=3mm, top=2mm, bottom=2mm]

$p = \begin{bmatrix}10 & 5 & 8\end{bmatrix}$, \quad
$Q = \begin{bmatrix}3 & 1 \\ 2 & 4 \\ 0 & 3\end{bmatrix}$

\medskip
\textbf{(a)} Dimensions of $p$: \ans{$1 \times 3$} \quad
Dimensions of $Q$: \ans{$3 \times 2$} \quad
$pQ$ defined? \ans{Yes} \quad
Dimensions of $pQ$: \ans{$1 \times 2$}

\medskip
\textbf{(b)}
$pQ = \begin{bmatrix}10(3)+5(2)+8(0) & 10(1)+5(4)+8(3)\end{bmatrix}
= $ \ans{$\begin{bmatrix}40 & 54\end{bmatrix}$}

\medskip
\textbf{(c)} \ans{The first entry, 40, is total revenue on Day 1.\\[3pt]
The second entry, 54, is total revenue on Day 2.}

\end{tcolorbox}
