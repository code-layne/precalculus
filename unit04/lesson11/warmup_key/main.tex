  \item Compute the matrix product.
        \[
          \begin{bmatrix}2 & 1 \\ 0 & 3\end{bmatrix}
          \begin{bmatrix}1 & 0 \\ 0 & 1\end{bmatrix}
          = \ans{$\begin{bmatrix}2 & 1 \\ 0 & 3\end{bmatrix}$}
        \]
        \vspace{0.1in}

  \item What does it mean for a real number to have a multiplicative inverse?
        Give an example.

        \ansline{A number $a$ has a multiplicative inverse $a^{-1}$ such that $a \cdot a^{-1} = 1$.}
        \ansline{Example: $6 \cdot \tfrac{1}{6} = 1$, so $\tfrac{1}{6}$ is the inverse of $6$.}

  \item Evaluate the expression $3(4) - 2(5)$.

        \vspace{0.05in}
        \ans{$3(4) - 2(5) = 12 - 10 = 2$}

\end{enumerate}
