\begin{learningtargetbox}
\small
\begin{enumerate}[leftmargin=1.5em, itemsep=5pt]
  \item I can compute the determinant of a $2\times2$ matrix using $\det(A) = ad - bc$.
  \item I can determine whether a $2\times2$ matrix has an inverse by checking whether
        $\det(A) \ne 0$.
  \item I can find the inverse of an invertible $2\times2$ matrix using the formula
        $A^{-1} = \dfrac{1}{\det(A)}\begin{bmatrix}d&-b\\-c&a\end{bmatrix}$.
\end{enumerate}
\end{learningtargetbox}
