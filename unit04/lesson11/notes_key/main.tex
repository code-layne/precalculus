\begin{objectivebox}
By the end of this lesson, I will be able to:
\begin{enumerate}[leftmargin=1.5em, itemsep=4pt]
    \item \ansline{Compute the determinant of a $2\times2$ matrix.}
    \item \ansline{Determine whether a $2\times2$ matrix has an inverse.}
    \item \ansline{Find the inverse of an invertible $2\times2$ matrix using the formula.}
\end{enumerate}
\end{objectivebox}

\vspace{0.08in}

\begin{vocabbox}
\termblanklong{identity matrix}
\ansline{A square matrix with 1s on the main diagonal and 0s elsewhere; $I_2 = \begin{bmatrix}1&0\\0&1\end{bmatrix}$.}
\termblanklong{inverse matrix}
\ansline{$A^{-1}$: the unique matrix such that $A \cdot A^{-1} = A^{-1} \cdot A = I$.}
\termblanklong{determinant}
\ansline{A scalar $\det(A) = ad - bc$ associated with a square matrix $A = \begin{bmatrix}a&b\\c&d\end{bmatrix}$.}
\termblanklong{invertible / singular}
\ansline{Invertible: $\det(A)\ne 0$ (inverse exists). Singular: $\det(A)=0$ (no inverse).}
\end{vocabbox}

\vspace{0.08in}

\begin{hookbox}
\textbf{Think:} What number times 6 gives 1? \ans{$\tfrac{1}{6}$}
That is the \textbf{multiplicative inverse} of 6.\\[6pt]
Can a \emph{matrix} have an inverse --- a matrix $M^{-1}$ such that
$M \cdot M^{-1} = I$?\\[6pt]
What condition do you think must hold? \ansline{The determinant must be nonzero.}
\end{hookbox}

\vspace{0.08in}

\begin{notesbox}{The Identity Matrix}
The $2\times2$ identity matrix is $I_2 = \ans{$\begin{bmatrix}1 & 0 \\ 0 & 1\end{bmatrix}$}$.\\[8pt]

For any $2\times2$ matrix $A$: \quad $A \cdot I = $ \ans{$A$} \quad and \quad $I \cdot A = $ \ans{$A$}.\\[8pt]

Verify: $\begin{bmatrix}3&1\\2&5\end{bmatrix}\begin{bmatrix}1&0\\0&1\end{bmatrix} = $
\ans{$\begin{bmatrix}3&1\\2&5\end{bmatrix}$}.
\end{notesbox}

\vspace{0.08in}

\begin{notesbox}{The Inverse of a Matrix}
The \textbf{inverse} of matrix $A$ is the matrix $A^{-1}$ such that:
\[
    A \cdot A^{-1} = \ans{$I$} \qquad \text{and} \qquad A^{-1} \cdot A = \ans{$I$}
\]

\textbf{Inverse formula for a $2\times2$ matrix:} If $A = \begin{bmatrix}a&b\\c&d\end{bmatrix}$, then
\[
    A^{-1} = \dfrac{1}{\ans{$\det(A)$}}\begin{bmatrix}\ans{$d$} & \ans{$-b$} \\[8pt]
    \ans{$-c$} & \ans{$a$}\end{bmatrix}
\]

\textbf{Example:} Find $A^{-1}$ for $A = \begin{bmatrix}3&2\\1&4\end{bmatrix}$.\\[6pt]
\ans{$\det(A) = 3(4)-2(1) = 10$. \quad
$A^{-1} = \dfrac{1}{10}\begin{bmatrix}4&-2\\-1&3\end{bmatrix}$.}
\end{notesbox}

\vspace{0.08in}

\begin{notesbox}{The Determinant}
For $A = \begin{bmatrix}a&b\\c&d\end{bmatrix}$: \quad
$\det(A) = $ \ans{$ad - bc$}\\[10pt]

$A$ is \textbf{invertible} if and only if $\det(A)$ \ans{$\ne 0$}\\[6pt]

\textbf{Example:} $\det\begin{bmatrix}3&2\\1&4\end{bmatrix} = $
\ans{$3(4)-2(1) = 12-2 = 10 \ne 0$, so $A$ is invertible.}
\end{notesbox}

\vspace{0.08in}

\begin{notesbox}{Geometric Interpretation}
If the columns of $A$ are two vectors in $\mathbb{R}^2$, then $|\det(A)|$ equals the
\ans{area} of the parallelogram they span.\\[6pt]

If $\det(A) = 0$, the two column vectors are \ans{parallel} (the parallelogram is degenerate).\\[8pt]

\textbf{Example:} $A = \begin{bmatrix}2&0\\1&3\end{bmatrix}$. \quad
Area $= |\det(A)| = $ \ans{$|2(3)-0(1)| = 6$}.
\end{notesbox}
