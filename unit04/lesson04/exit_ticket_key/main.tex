\begin{enumerate}[itemsep=20pt]

\item Rewrite $f(x) = 3^{x+4}$ in the form $a \cdot b^x$. Show your work.

      \vspace{0.06in}
      $3^{x+4} = 3^x \cdot 3^{{\color{keyred}\mathbf{4}}} = $ \ans{81} $\cdot\; 3^x$

      So $a = $ \ans{81} and $b = $ \ans{3}.

\item Rewrite $f(x) = 5^{2x}$ as a single base raised to the $x$ power. Show your work.

      \vspace{0.06in}
      $5^{2x} = (5^{{\color{keyred}\mathbf{2}}})^x = $ \ans{$25^x$}

\item Simplify each expression. Write your answer without negative or fractional exponents
      where possible.

      \vspace{0.06in}
      $2^{-3} = $ \ans{$\dfrac{1}{8}$} \qquad\qquad $8^{1/3} = $ \ans{2}

      \begin{teachernote}
        $2^{-3} = 1/2^3 = 1/8$. $8^{1/3} = \sqrt[3]{8} = 2$ since $2^3 = 8$.
        Students who write $8^{1/3} = 2.67$ are applying a calculator shortcut without
        understanding the cube-root connection --- re-teach EK 2.4.A.4.
      \end{teachernote}

\end{enumerate}
