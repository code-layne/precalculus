\begin{notesbox}{Parametrically Defined Circles and Lines Practice --- Key}

\textbf{1.}\quad Fountain: radius 6, center $(-1,\,3)$, counterclockwise from $(5,\,3)$.

\vspace{4pt}
(a)\quad \ans{$x(t) = -1 + 6\cos t$, \quad $y(t) = 3 + 6\sin t$, \quad $0 \le t \le 2\pi$}

(b)\quad At $t = \pi/2$: $x = -1 + 6(0) = $ \ans{$-1$}, $y = 3 + 6(1) = $ \ans{$9$}. Position: \ans{$(-1,\;9)$}.

At $t = \pi$: $x = -1 + 6(-1) = $ \ans{$-7$}, $y = 3 + 6(0) = $ \ans{$3$}. Position: \ans{$(-7,\;3)$}.

\vspace{0.08in}
\textbf{2.}\quad Circle: $x(t) = -2 + 5\cos t$, $y(t) = 4 + 5\sin t$.

(a)\quad \ans{Center $(-2,\,4)$, radius $5$.}

(b)\quad $x(t) = -2$: $-2 + 5\cos t = -2 \;\Rightarrow\; \cos t = 0 \;\Rightarrow\; t = \pi/2$ or $t = 3\pi/2$.

At $t=\pi/2$: $y = 4+5(1) = $ \ans{$9$}. Point: \ans{$(-2,\;9)$}.

At $t=3\pi/2$: $y = 4+5(-1) = $ \ans{$-1$}. Point: \ans{$(-2,\;-1)$}.

\vspace{0.08in}
\textbf{3.}\quad Clockwise, radius 7, center $(2,\,-3)$ (replace $\sin t$ with $-\sin t$):

\ans{$x(t) = 2 + 7\cos t$, \quad $y(t) = -3 - 7\sin t$, \quad $0 \le t \le 2\pi$}

At $t = \pi/2$: $x = 2 + 7(0) = $ \ans{$2$}, $y = -3 - 7(1) = $ \ans{$-10$}. Position: \ans{$(2,\;-10)$}.

\vspace{0.08in}
\textbf{4.}\quad Hiker: $A = (1,\,4)$ to $B = (9,\,0)$.

(a)\quad \ans{$x(t) = 1 + 8t$, \quad $y(t) = 4 - 4t$, \quad $0 \le t \le 1$}

(b)\quad At $t = 0.25$: $x = 1 + 2 = $ \ans{$3$}, $y = 4 - 1 = $ \ans{$3$}. Position: \ans{$(3,\;3)$}.

At $t = 0.75$: $x = 1 + 6 = $ \ans{$7$}, $y = 4 - 3 = $ \ans{$1$}. Position: \ans{$(7,\;1)$}.

(c)\quad $y(t) = 0$: $4 - 4t = 0 \;\Rightarrow\; t = $ \ans{$1$}. The path reaches $y=0$ at the endpoint $t=1$, position \ans{$(9,\;0)$}.

\end{notesbox}

\vspace{0.08in}

\begin{notesbox}{AP-Style Practice --- Key}

\textbf{5.}\quad $x(t) = 3 + 4\cos t = 3 \;\Rightarrow\; 4\cos t = 0 \;\Rightarrow\; \cos t = 0$,
so $t = \pi/2$ or $t = 3\pi/2$.

At $t = \pi/2$: $y = -2 + 4(1) = 2$.
At $t = 3\pi/2$: $y = -2 + 4(-1) = -6$.

Both values are attained, so the correct answer is \textbf{(C)}.

\textcolor{keyred}{\textbf{(C) $2$ and $-6$ $\leftarrow$ correct}}

\begin{itemize}[leftmargin=1.4em, itemsep=2pt]
  \item (A) $2$ --- only one of the two values; incomplete.
  \item (B) $-6$ --- only one of the two values; incomplete.
  \item (D) $-2$ --- that is the $y$-coordinate of the center, not a point where $x=3$.
\end{itemize}

\begin{teachernote}
  The AP item tests whether students recognize that $x(t) = 3$ is solved by
  $\cos t = 0$, which has \emph{two} solutions in $[0,2\pi)$. Students who
  report only one value likely stopped after finding $t = \pi/2$. Emphasize:
  always check for all solutions within the domain.
\end{teachernote}

\end{notesbox}

\begin{extensionbox}
\textbf{Extension (optional) --- Key:}

From $x(t) = h + r\cos t$: $\cos t = (x-h)/r$.

From $y(t) = k + r\sin t$: $\sin t = (y-k)/r$.

Use $\cos^2 t + \sin^2 t = 1$: $\left(\dfrac{x-h}{r}\right)^2 + \left(\dfrac{y-k}{r}\right)^2 = 1$,
so \ans{$(x-h)^2 + (y-k)^2 = r^2$.}
\end{extensionbox}
