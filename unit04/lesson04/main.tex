\begin{tabularx}{\linewidth}{@{} >{\leavevmode\bfseries\color{navy}}p{0.30\linewidth} X @{}}
Product property of exponents & $b^m \cdot b^n = b^{m+n}$; adds exponents when bases match. \\
Power property of exponents   & $(b^m)^n = b^{mn}$; multiplies exponents. \\
Negative exponent             & $b^{-n} = \dfrac{1}{b^n}$; flips to the reciprocal base. \\
Fractional exponent           & $b^{1/k} = \sqrt[k]{b}$; $b^{m/n} = (\sqrt[n]{b})^m$. \\
$k$th root                    & $\sqrt[k]{b} = b^{1/k}$; the value whose $k$th power equals $b$. \\
Equivalent form               & Two expressions that produce the same output for every input. \\
Vertical dilation             & Multiplying the output by a constant $a$ (stretches or compresses). \\
Horizontal translation        & Replacing $x$ with $x+k$ shifts the graph left ($k>0$) or right. \\
\end{tabularx}

\begin{skillbox}[Activate Prior Knowledge \& Spiral Review]{sky}
    Spiral review (warm-up) covers Lesson 2.3: identifying $a$ and $b$ in $f(x) = ab^x$,
    classifying growth vs.\ decay, evaluating an exponential function at a given input, and
    writing $f(x)$ from a table of values. Students work three short problems; teacher
    circulates to assess readiness for exponent manipulation.
\end{skillbox}

\begin{skillbox}[Hook]{sky}
Write two functions on the board:
\[
  f(x) = 2^{x+3} \qquad\qquad g(x) = 8 \cdot 2^x
\]
Ask: \textbf{``Are these the same function? Try $x = 0$, $x = 1$, and $x = -3$.''}
Give students 90 seconds. After discussion, reveal: \textbf{yes} --- because
$2^{x+3} = 2^x \cdot 2^3 = 8 \cdot 2^x$.
Frame: \textit{``Today we master the algebra that connects these forms --- and see how each
form tells a different story about the same graph.''}
\end{skillbox}

\begin{skillbox}[Lesson]{sky}
\begin{multicols}{2}
\textbf{Part 1 (15 min): Product Property.}

State: $b^m \cdot b^n = b^{m+n}$.
Graphical interpretation: a horizontal translation $f(x) = b^{x+k}$ equals a vertical
dilation $f(x) = b^k \cdot b^x = ab^x$ where $a = b^k$.

Model on the board:
\begin{itemize}[leftmargin=1.2em, itemsep=1pt, topsep=2pt]
  \item $5^{x+2} = 5^x \cdot 5^2 = 25 \cdot 5^x$
  \item $3^{x-1} = 3^x \cdot 3^{-1} = \tfrac{1}{3} \cdot 3^x$
  \item Reverse: $4 \cdot 2^x = 2^2 \cdot 2^x = 2^{x+2}$
\end{itemize}

Pose: ``Which form reveals the $y$-intercept? Which shows the horizontal shift?''

\columnbreak

\textbf{Part 2 (15 min): Power Property.}

State: $(b^m)^n = b^{mn}$.
Graphical interpretation: a horizontal dilation $f(x) = b^{cx}$ equals a base change
$f(x) = (b^c)^x$.

Model:
\begin{itemize}[leftmargin=1.2em, itemsep=1pt, topsep=2pt]
  \item $2^{3x} = (2^3)^x = 8^x$
  \item $9^{x/2} = (9^{1/2})^x = 3^x$
\end{itemize}

\textbf{Part 3 (10 min): Negative \& Fractional Exponents.}

$b^{-n} = \dfrac{1}{b^n}$ connects to decay:
$2^{-x} = \left(\dfrac{1}{2}\right)^x$.

$b^{1/k} = \sqrt[k]{b}$ and $b^{m/n} = (\sqrt[n]{b})^m$.
Examples: $8^{1/3} = 2$; $4^{3/2} = (\sqrt{4})^3 = 8$.
\end{multicols}
\end{skillbox}

\begin{skillbox}[Active Monitoring]{redbox}
\begin{itemize}[leftmargin=1.2em,itemsep=2pt,topsep=0pt]
  \item After Part 1: cold-call --- ``If $f(x) = 7^{x+2}$, what is $a$ in $ab^x$ form?''
        (Expect $a = 49$.) Watch for students who add the $k$ as a coefficient.
  \item During Part 2: listen for confusion between $b^{cx}$ and $(bc)^x$; the parentheses
        determine whether we change the base or multiply a coefficient.
  \item After Part 2: ``What single base replaces $4^{x/2}$?'' (Expect $2^x$.)
  \item Throughout: students collapsing $b^{x+k}$ to $(bk)^x$ need re-teaching of product
        vs.\ power property.
\end{itemize}
\end{skillbox}

\begin{skillbox}[Group Work \& Differentiation]{redbox}
\begin{itemize}[leftmargin=1.2em,itemsep=2pt,topsep=0pt]
  \item \textbf{Tier R (Remediate):} Identify the property needed, then rewrite each given
        expression one step at a time into $ab^x$ form: $2^{x+4}$, $3^{2x}$, $5^{-x}$, $4^{x/2}$.
  \item \textbf{Tier A (Apply):} Match three forms of the same function; rewrite $8^{x/3}$
        and $27 \cdot 3^{-x}$; show each step explicitly.
  \item \textbf{Tier E (Extend):} Rewrite $6^{2x+1}$ in three equivalent forms; explain which
        form is most useful in each of three mathematical contexts.
\end{itemize}
\end{skillbox}

\begin{skillbox}[Individual Work \& Assessment]{redbox}
Exit ticket (3 questions, 1 page): (1) rewrite $3^{x+4}$ in $a \cdot b^x$ form (product
property); (2) rewrite $5^{2x}$ as a single base to the $x$ power (power property);
(3) evaluate $2^{-3}$ and $8^{1/3}$ (negative and fractional exponents).

Sort tickets: students missing item 1 need product-property reinforcement (EK 2.4.A.1);
missing item 2 need power-property work (EK 2.4.A.2); missing item 3 need exponent-property
review (EK 2.4.A.3--A.4).
\end{skillbox}

\begin{skillbox}[Reinforcement \& Extension]{goldbox}
Homework (6 problems): product-property rewrites (horizontal translation $\to$ vertical
dilation); power-property rewrites (horizontal dilation $\to$ base change); negative and
fractional exponent simplification; show two functions equivalent; AP-style MC on equivalent
expressions for $16^{x/4}$; extension --- find $b$ such that $f(x) = b^{2x}$ equals
$g(x) = 9 \cdot 3^x$.

Preview of Lesson 2.5: logarithmic functions as inverses of exponentials --- connecting the
$ab^x$ form mastered today to logarithmic structure.
\end{skillbox}
