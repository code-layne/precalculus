\item \textbf{(Extension)}

      \begin{enumerate}[label=(\alph*), itemsep=8pt]
        \item $y_1 = ab^{x_1}$ \quad and \quad $y_2 = ab^{x_2}$.

        \item Dividing:
              \[
                \frac{y_2}{y_1} = \frac{ab^{x_2}}{ab^{x_1}} = b^{x_2 - x_1}.
              \]
              Raising both sides to the power $\dfrac{1}{x_2 - x_1}$:
              \[
                b = \left(\frac{y_2}{y_1}\right)^{1/(x_2 - x_1)}. \quad \checkmark
              \]
      \end{enumerate}
