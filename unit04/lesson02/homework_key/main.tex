\begin{enumerate}[itemsep=16pt]

\item

      \begin{enumerate}[label=(\alph*), itemsep=12pt]
        \item
        \renewcommand{\arraystretch}{1.2}
        \begin{tabular}{c|c}
          $x$ & $y$ \\ \hline
          0 & 80 \\ 1 & 40 \\ 2 & 20 \\ 3 & 10
        \end{tabular}

        \vspace{0.06in}
        Differences: \ans{$-40, -20, -10$ (not constant)} \quad Ratios: \ans{$0.5, 0.5, 0.5$}

        Type: \ans{Exponential} \quad Formula: $f(x) = $ \ans{$80 \cdot (0.5)^x$}

        \item
        \begin{tabular}{c|c}
          $x$ & $y$ \\ \hline
          1 & 5 \\ 3 & 11 \\ 5 & 17 \\ 7 & 23
        \end{tabular}

        \vspace{0.06in}
        Interval length = \ans{2} \quad Differences: \ans{$6, 6, 6$ (constant)}

        Type: \ans{Linear} \quad $m = 6/2 = 3$, Formula: $f(x) = $ \ans{$3x + 2$}
      \end{enumerate}

\item Points $(3,\ 4)$ and $(7,\ 20)$.

      Slope: $m = \dfrac{20 - 4}{7 - 3} = $ \ans{4}

      Function: $f(x) = $ \ans{$4 + 4(x - 3) = 4x - 8$}

\item Points $(1,\ 8)$ and $(4,\ 1000)$.

      $b = \left(\dfrac{1000}{8}\right)^{1/3} = \left(\ans{125}\right)^{1/3} = $ \ans{5}

      $a = y_1 / b^{x_1} = 8/5 = 1.6$

      Function: $f(x) = $ \ans{$1.6 \cdot 5^x$} \quad \textit{(or equivalently $8 \cdot 5^{x-1}$)}

\item

      \begin{enumerate}[label=(\alph*), itemsep=8pt]
        \item $A(t) = $ \ans{$500 + 50t$}

        \item $B(t) = $ \ans{$500 \cdot (1.05)^t$}

        \item $A(12) = \$$ \ans{1100} \quad $B(12) \approx \$$ \ans{897.93}

              Higher balance: \ans{Plan A (linear) after 12 months}

              \textit{(Note: exponential overtakes linear eventually, but not by month 12 at these parameters.)}
      \end{enumerate}

\item \textbf{(AP-Style MC)} Answer: \ans{B}

      Explanation: \ans{In table B, ratios of consecutive outputs are all 2 ($6/3=2$, $12/6=2$, $24/12=2$), so
      outputs change proportionally --- this is the defining property of an exponential function.
      Table A has constant differences (linear); Table C is $y=x^2$ (quadratic); Table D has
      $y=0$ at $x=0$, which is impossible for $f(x)=ab^x$ since $a = f(0) = 0$ would make
      $f$ identically zero.}

\item \textbf{(Extension)}

      \begin{enumerate}[label=(\alph*), itemsep=8pt]
        \item $y_1 = ab^{x_1}$ \quad and \quad $y_2 = ab^{x_2}$.

        \item Dividing:
              \[
                \frac{y_2}{y_1} = \frac{ab^{x_2}}{ab^{x_1}} = b^{x_2 - x_1}.
              \]
              Raising both sides to the power $\dfrac{1}{x_2 - x_1}$:
              \[
                b = \left(\frac{y_2}{y_1}\right)^{1/(x_2 - x_1)}. \quad \checkmark
              \]
      \end{enumerate}

\end{enumerate}
