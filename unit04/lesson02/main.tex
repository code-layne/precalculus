\begin{tcolorbox}[colback=sky, colframe=navy, boxrule=0.9pt, arc=2mm,

  \textbf{Primary Objective:} Students will identify key characteristics
  of parametric planar motion --- horizontal and vertical extrema from
  the maximum and minimum values of $x(t)$ and $y(t)$, and axis
  intercepts of the curve from zeros of $x(t)$ and $y(t)$ ---
  and apply these to describe a particle's position and range of motion.
  \textit{(Big Idea: Functions)}
\end{tcolorbox}

%----------------------------------------------------------------------
% Priority Ideas & Skills
%----------------------------------------------------------------------
\begin{skillbox}[Priority Ideas \& Skills]{goldbox}
\begin{tabularx}{\linewidth}{@{} p{0.46\linewidth} X @{}}
\textbf{AP Skills} & \textbf{Key Understandings (EKs)} \\
\midrule
\textbf{3.A} \textit{Describe characteristics} ---
  Identify horizontal and vertical extrema of a parametric curve from
  the maximum and minimum values of the component functions $x(t)$ and
  $y(t)$; describe what those extrema tell us about the range of motion
  in each direction.
&
\textbf{EK 4.2.A.1}: A parametric function $f(t) = (x(t),\, y(t))$
models particle motion in the plane. The graph indicates the position
of the particle at time $t$. \\[4pt]

\textbf{3.B} \textit{Apply numerical results} ---
  Use the max/min values of $x(t)$ and $y(t)$ (found analytically or
  from a table) to bound the particle's motion, and use zeros of $x(t)$
  and $y(t)$ to locate where the curve crosses the axes.
&
\textbf{EK 4.2.A.2}: The horizontal and vertical extrema of the
particle's motion are the maximum and minimum values of $x(t)$ and
$y(t)$, respectively. \\[4pt]

&
\textbf{EK 4.2.A.3}: Real zeros of $x(t)$ correspond to $y$-intercepts
of the curve; real zeros of $y(t)$ correspond to $x$-intercepts. \\
\end{tabularx}
\end{skillbox}

%----------------------------------------------------------------------
% Vocabulary

\begin{skillbox}[{Vocabulary, Concepts, \& Theorems}]{greenbox}
\begin{tabularx}{\linewidth}{@{} p{0.28\linewidth} X @{}}
\textbf{Term} & \textbf{Definition / Note} \\
\midrule
horizontal extremum &
  The maximum or minimum value of $x(t)$ over the domain; determines
  the rightmost and leftmost positions of the particle. \\[4pt]
vertical extremum &
  The maximum or minimum value of $y(t)$ over the domain; determines
  the highest and lowest positions of the particle. \\[4pt]
$y$-intercept (of parametric curve) &
  A point on the curve where $x(t) = 0$; found by solving $x(t) = 0$
  for $t$, then evaluating $y(t)$ at those values. \\[4pt]
$x$-intercept (of parametric curve) &
  A point on the curve where $y(t) = 0$; found by solving $y(t) = 0$
  for $t$, then evaluating $x(t)$ at those values. \\
\end{tabularx}
\end{skillbox}

\begin{skillbox}[Activate Prior Knowledge \& Spiral Review]{sky}
\begin{tabularx}{\linewidth}{@{}X@{}}
\textbf{Spiral topics activated during Warm-Up (first 8 minutes):}
\begin{itemize}[leftmargin=1.4em, itemsep=2pt]
  \item \textbf{Evaluating component functions}: Substitute $t$-values
    into $x(t)$ and $y(t)$ --- the core computation from Lesson 4.1,
    now used to build intuition about extrema from a table.
  \item \textbf{Maximum and minimum of a quadratic}: Use the vertex
    formula $t = -b/(2a)$ to locate the max/min of $at^2 + bt + c$;
    this is the primary analytical tool for finding extrema.
  \item \textbf{Solving quadratic equations}: Set $x(t) = 0$ or
    $y(t) = 0$ and solve by factoring or the quadratic formula to find
    axis intercepts of the parametric curve.
\end{itemize}
\end{tabularx}
\end{skillbox}

%----------------------------------------------------------------------
% Hook
%----------------------------------------------------------------------
\begin{skillbox}[Hook --- Entry Event]{sky}
\textbf{Scenario:} A ball is launched and its position (in feet) at
time $t$ seconds is modeled by:
\[
  x(t) = -16t^2 + 48t, \qquad y(t) = 12t - t^2.
\]

\textbf{Discussion prompts:}
\begin{enumerate}[leftmargin=1.4em, itemsep=3pt]
  \item \textit{``How high does the ball get? Which component function
    do you look at to answer that?''}
  \item \textit{``How far left or right does the ball travel? Which
    component tells you that?''}
  \item \textit{``When does the ball hit the ground (return to $y = 0$)?
    What equation do you solve?''}
\end{enumerate}

\textbf{Key tension to surface:} Each question maps to one of three
tasks --- finding max/min of $x(t)$, finding max/min of $y(t)$, or
solving $y(t) = 0$. These are exactly the skills of LO 4.2.A.

\textbf{Transition:} ``Today we develop a systematic way to find
horizontal extrema from $x(t)$, vertical extrema from $y(t)$, and axis
intercepts from the zeros of each component.''
\end{skillbox}

\vspace{0.08in}

%----------------------------------------------------------------------
% Lesson
%----------------------------------------------------------------------
\begin{skillbox}[Lesson --- Instruction Sequence]{sky}
\begin{multicols}{2}

\textbf{Step 1 --- Review \& Planar Motion (5 min)}

Briefly recall that $f(t) = (x(t),\, y(t))$ gives a particle's
position at time $t$. Now ask: \emph{how far} does it travel in each
direction? Write EK 4.2.A.1 on the board. Use the hook scenario to
frame the three questions the lesson answers.

\columnbreak

\textbf{Step 2 --- Horizontal Extrema from $x(t)$ (12 min)}

The leftmost and rightmost positions are the min and max of $x(t)$.
Model with the hook: $x(t) = -16t^2 + 48t$ is a downward parabola.
Vertex at $t = -48/(2 \cdot (-16)) = 1.5$ s;
$x(1.5) = 36$ ft (rightmost). EK 4.2.A.2. At $t = 0$, $x = 0$
(leftmost on $[0,3]$).

\end{multicols}

\begin{multicols}{2}

\textbf{Step 3 --- Vertical Extrema from $y(t)$ (10 min)}

Same technique for $y(t)$. Model: $y(t) = 12t - t^2$; vertex at
$t = 6$, $y(6) = 36$ ft (highest). Identify $y(0) = 0$ as the
starting height. Students practice on the second example in guided
notes. EK 4.2.A.2.

\columnbreak

\textbf{Step 4 --- Axis Intercepts from Zeros (15 min)}

EK 4.2.A.3: zeros of $x(t)$ give $y$-intercepts; zeros of $y(t)$
give $x$-intercepts. Model: Solve $y(t) = t(12-t) = 0 \Rightarrow
t = 0$ or $t = 12$; find coordinates of each point. Solve
$x(t) = -16t(t-3) = 0 \Rightarrow t = 0$ or $t = 3$; students find
coordinates. Period 1: Steps 1--2. Period 2: Steps 3--4 + activity.

\end{multicols}

\smallskip
\textbf{Pacing note:} Steps 1--2 in Period 1; Steps 3--4 + group
activity in Period 2.
\end{skillbox}

\vspace{0.08in}

%----------------------------------------------------------------------
% Active Monitoring
%----------------------------------------------------------------------
\begin{skillbox}[Active Monitoring]{redbox}
\begin{itemize}[leftmargin=1.4em, itemsep=3pt]
  \item \textbf{Mixing up components}: Students may find the max of
    $y(t)$ when asked for horizontal extrema. Prompt: \textit{``Horizontal
    means left-right. Which axis is that? Which component controls it?''}
  \item \textbf{Reporting $t$ instead of position}: Students stop at
    the $t$-value. Prompt: \textit{``$t = 2$ is a time, not a position.
    What are the coordinates at that moment?''}
  \item \textbf{Sign errors in vertex formula}: Require students to
    write $a$, $b$, $c$ explicitly before applying $t = -b/(2a)$.
  \item \textbf{Zeros vs.\ intercepts}: Students confuse zeros of $x(t)$
    with $x$-intercepts. Post the rule: \textit{zero of $x(t)$ →
    $y$-intercept; zero of $y(t)$ → $x$-intercept.}
\end{itemize}
\end{skillbox}

\vspace{0.08in}

%----------------------------------------------------------------------
% Group Work & Differentiation
%----------------------------------------------------------------------
\begin{skillbox}[Group Work \& Differentiation]{redbox}
Students work in groups of 3--4 on the tiered activity (remote-control
car scenario).

\begin{itemize}[leftmargin=1.4em, itemsep=4pt]
  \item \textbf{Tier R (Remediate)}: Given $f(t) = (2t,\; -t^2 + 4)$
    for $0 \le t \le 3$, evaluate both components at $t = 0, 1, 2, 3$
    and identify the leftmost/rightmost $x$-values and the
    highest/lowest $y$-values from the table. No algebra required.
  \item \textbf{Tier A (Accelerate)}: Given $f(t) = (t^2 - 4t,\;
    3\sin(\pi t/2))$ for $0 \le t \le 4$, find horizontal extrema of
    $x(t)$ analytically (vertex) and vertical extrema of $y(t)$ by
    reasoning about the sine function's range.
  \item \textbf{Tier E (Extend)}: Compare two parametrizations of the
    same circular path --- $f(t) = (\cos t, \sin t)$ and
    $g(t) = (\cos(2t), \sin(2t))$ --- and determine whether they
    produce the same extrema, the same intercepts, and the same range
    of motion. Explain what changes and what stays the same.
\end{itemize}
\end{skillbox}

\vspace{0.08in}

%----------------------------------------------------------------------
% Individual Work & Assessment
%----------------------------------------------------------------------
\begin{skillbox}[Individual Work \& Assessment]{redbox}
Exit ticket administered independently (final 5 minutes of Period 2).

\begin{enumerate}[leftmargin=1.4em, itemsep=4pt]
  \item Given $f(t) = (t^2 - 1,\; 2t - t^2)$ for $0 \le t \le 3$,
    complete a table at $t = 0, 1, 2, 3$.
  \item Find the horizontal extrema. Show work using the vertex formula
    for $x(t) = t^2 - 1$.
  \item For what values of $t$ does the particle cross the $y$-axis?
    State the coordinates.
\end{enumerate}

\textbf{Data use:} Students who cannot apply the vertex formula to
$x(t)$ or $y(t)$ need re-teaching on quadratic max/min. Students who
identify zeros of $y(t)$ as $y$-intercepts have the rule reversed ---
correct individually.
\end{skillbox}

\begin{skillbox}[Reinforcement \& Extension]{goldbox}

horizontal and vertical extrema from component functions, locating
axis intercepts of parametric curves, one contextual motion problem,
and one AP-style multiple-choice item.
