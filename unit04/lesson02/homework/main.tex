\begin{notesbox}{Contextual Problem}

\textbf{3.}\quad A ball is hit and moves with
$x(t) = 4t$ (meters east) and $y(t) = -5t^2 + 20t$ (meters high)
for $0 \le t \le 4$.

\vspace{4pt}
(a)\quad At what time does the ball reach its maximum height? What is
that height?
\writelines{3}

\vspace{0.04in}
(b)\quad The ball hits the ground when $y(t) = 0$. Find the time this
occurs (other than $t = 0$) and state the ball's horizontal position
at that moment.
\writelines{3}

\end{notesbox}

\begin{notesbox}{AP-Style Practice}

\textbf{4.}\quad A particle moves with
$h(t) = (t^2 - 2t - 3,\; t^2 - 4)$ for $-1 \le t \le 4$.
Which of the following correctly identifies the $y$-intercepts of
the parametric curve?

\vspace{6pt}
\begin{enumerate}[label=(\Alph*), leftmargin=2em, itemsep=4pt]
  \item $(-4,\; 0)$ and $(5,\; 0)$
  \item $(0,\; -3)$ and $(0,\; 5)$
  \item $(0,\; -3)$ only
  \item $(0,\; -4)$ and $(0,\; 12)$
\end{enumerate}

\end{notesbox}
