\begin{enumerate}[itemsep=20pt]

\item The table below shows inputs and outputs for a function.

      \medskip
      \renewcommand{\arraystretch}{1.3}
      \begin{tabular}{c|c}
        $x$ & $f(x)$ \\ \hline
        0 & 5 \\ 1 & 10 \\ 2 & 20 \\ 3 & 40
      \end{tabular}

      \medskip
      \ans{Exponential}

      \vspace{0.08in}
      Check: \ans{Ratios: $10/5=2,\ 20/10=2,\ 40/20=2$ --- constant ratio $=2$.}

      \vspace{0.06in}
      Write the formula: $f(x) = $ \ans{$5 \cdot 2^x$}

\item Write the exponential function through $(0,\ 3)$ and $(2,\ 75)$.

      \vspace{0.08in}
      Find the base $b$: $b = \left(\dfrac{75}{3}\right)^{1/2} = $ \ans{$\sqrt{25} = 5$}

      \vspace{0.08in}
      Write the function: $f(x) = $ \ans{$3 \cdot 5^x$}

\item \textbf{True or False:} A linear function and an exponential function can each be
      uniquely determined by knowing just two input-output pairs. \\[6pt]
      \ans{True}

      \vspace{0.06in}
      Explain: \ans{Two points determine a unique slope (for linear) or a unique base and
      initial value (for exponential), so both function types are fully specified by two
      input-output pairs (EK 2.2.B.3).}

\end{enumerate}
