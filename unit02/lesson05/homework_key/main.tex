\begin{headlinebox}{lilac}
\textbf{\color{plum}Part D --- Putting it together, and the headline}
\end{headlinebox}

\begin{enumerate}[leftmargin=1.9em, itemsep=9pt, start=8]

\item Jae says, ``$p$ is concave up on $(1, 5)$, so $p$ is increasing on $(1, 5)$.'' Use the
      graph of $p$ to show that Jae is wrong.
      \writespace{1.6cm}{On $(1, 3)$, $p$ is concave up but \emph{decreasing} --- it falls from 1 to $-2$, more and more slowly. It only increases on $(3, 5)$. Bend and direction are separate questions.}

\item In your own words: what is the difference between a function's \emph{maximum value} and
      \emph{where} the maximum occurs? Which axis do you read each one on --- and which axis
      gives every increasing or decreasing interval?
      \writespace{2.0cm}{The maximum value is the greatest output --- a height, read on the $y$-axis. Where it occurs is the input that produces it, read on the $x$-axis. Every increasing or decreasing interval is a stretch of $x$-values, so it is read on the $x$-axis too.}

\end{enumerate}
