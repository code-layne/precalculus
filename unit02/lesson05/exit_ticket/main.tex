\item Let $q(x) = 3(x-1)^2(x+4)^2$. Do not expand.

\begin{work}
  \text{degree of } q &= 2 + 2 \\
                      &= 4 \\
  \text{leading term} &= 3 \cdot x^{2} \cdot x^{2} \\
                      &= 3x^{4}
\end{work}

\begin{enumerate}[label=(\alph*), itemsep=6pt]
  \item Degree: \blank{1.6cm} \qquad Leading coefficient: \blank{1.6cm}
  \item $\lim_{x \to \infty} q(x) = $ \blank{2cm} \qquad
        $\lim_{x \to -\infty} q(x) = $ \blank{2cm}
\end{enumerate}

\item A polynomial $r$ satisfies $\lim_{x \to \infty} r(x) = \infty$ and
      $\lim_{x \to -\infty} r(x) = -\infty$.

\begin{enumerate}[label=(\alph*), itemsep=6pt]
  \item Is the degree of $r$ even or odd? \quad \blank{2.4cm}
  \item Is the leading coefficient positive or negative? \quad \blank{2.8cm}
\end{enumerate}

\end{enumerate}
