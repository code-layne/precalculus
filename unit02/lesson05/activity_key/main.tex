\boxguard
\begin{tcolorbox}[colback=white, colframe=black!40,
    title=\textbf{Tier A --- Approaching Proficiency},
    fonttitle=\bfseries, arc=2mm, left=3mm, right=3mm, top=2mm, bottom=2mm]

\begin{enumerate}[itemsep=8pt]
  \item Find the leading term yourself, then write both limit statements.
        Read the exponents, not the order the terms appear in.

        \begin{enumerate}[label=(\alph*), itemsep=6pt]
          \item $f(x) = 3x^4 - 9x^2 + 1$

                Leading term: \ans{$3x^4$} \quad
                $\lim_{x \to \infty} f(x) = $ \ans{$\infty$} \quad
                $\lim_{x \to -\infty} f(x) = $ \ans{$\infty$}

          \item $g(x) = 12 - 5x^3 + 2x^6 - x^7$

                Leading term: \ans{$-x^7$} \quad
                $\lim_{x \to \infty} g(x) = $ \ans{$-\infty$} \quad
                $\lim_{x \to -\infty} g(x) = $ \ans{$\infty$}
        \end{enumerate}

  \item Now in factored form. Add the exponents for the degree, multiply the
        leading pieces for the leading term --- do not expand.

        \begin{enumerate}[label=(\alph*), itemsep=6pt]
          \item $h(x) = (x-4)(x+1)(x-7)$

                \begin{work}
                  \text{degree of } h &= 1 + 1 + 1 \\
                                      &= 3 \\
                  \text{leading term} &= x \cdot x \cdot x \\
                                      &= x^{3}
                \end{work}

                $\lim_{x \to \infty} h(x) = $ \ans{$\infty$} \quad
                $\lim_{x \to -\infty} h(x) = $ \ans{$-\infty$}

          \item $k(x) = -2(x+3)^2(x^2+1)$

                \begin{work}
                  \text{degree of } k &= 2 + 2 \\
                                      &= 4 \\
                  \text{leading term} &= -2 \cdot x^{2} \cdot x^{2} \\
                                      &= -2x^{4}
                \end{work}

                $\lim_{x \to \infty} k(x) = $ \ans{$-\infty$} \quad
                $\lim_{x \to -\infty} k(x) = $ \ans{$-\infty$}
        \end{enumerate}

  \item Which of the four functions above are guaranteed to cross the
        $x$-axis, and what is it about them that guarantees it?

        \vspace{0.05in}
        \ansline{$g$ and $h$. Both have odd degree, so one end runs to $\infty$ and the other to}
        \ansline{$-\infty$; a continuous graph cannot get from one to the other without crossing.}
\end{enumerate}
\end{tcolorbox}

\boxguard
\begin{tcolorbox}[colback=white, colframe=black!40,
    title=\textbf{Tier E --- Extension},
    fonttitle=\bfseries, arc=2mm, left=3mm, right=3mm, top=2mm, bottom=2mm]

\begin{enumerate}[itemsep=8pt]
  \item Read Graphs P and Q at the top of the page. Neither equation is given
        --- the ends are enough.

        \begin{enumerate}[label=(\alph*), itemsep=4pt]
          \item Graph P: \; $\lim_{x \to \infty} y = $ \ans{$-\infty$},
                \; $\lim_{x \to -\infty} y = $ \ans{$-\infty$}, \;
                so the degree is \ans{even} and $a_n$ is \ans{negative}.
          \item Graph Q: \; $\lim_{x \to \infty} y = $ \ans{$-\infty$},
                \; $\lim_{x \to -\infty} y = $ \ans{$\infty$}, \;
                so the degree is \ans{odd} and $a_n$ is \ans{negative}.
        \end{enumerate}

  \item For $p(x) = x^4 - 500x^3$, find the exact input past which the leading
        term is larger in size than the rest.

        \begin{work}
          \frac{-500x^3}{x^4} &= \frac{-500}{x} \\
          \left|\frac{-500}{x}\right| &< 1 \\
          x &> 500
        \end{work}

        The leading term takes over for $x > $ \ans{$500$}. Fill in the
        ratio at three inputs:

        \smallskip
        \renewcommand{\arraystretch}{1.5}
        \begin{center}
        \begin{tabular}{|c|c|c|c|}
        \hline
        $x$ & $50$ & $500$ & $5{,}000$ \\
        \hline
        rest $\div$ leading & \ans{$-10$} & \ans{$-1$} & \ans{$-0.1$} \\
        \hline
        \end{tabular}
        \end{center}

  \item Write a polynomial in \textbf{factored form} that has even degree, a
        negative leading coefficient, and exactly three distinct real zeros.

        \vspace{0.05in}
        $p(x) = $ \ans{$-(x+2)(x-1)^2(x-4)$ --- degree 4, $a_n = -1$, zeros $-2$, $1$, $4$}

  \item Explain why $p(x) = x^5 - 900x^4 + 12$ and $q(x) = x^5 + 7x$ have
        exactly the same end behavior, even though their graphs look nothing
        alike on a standard window.

        \vspace{0.05in}
        \ansline{Both have leading term $x^5$. The $-900x^4$ term is huge on a small window, but}
        \ansline{its ratio to $x^5$ is $-900/x$, which shrinks to $0$ --- so past a few thousand it is}
        \ansline{negligible. Only the leading term survives at the ends, and theirs is identical.}
\end{enumerate}
\end{tcolorbox}
