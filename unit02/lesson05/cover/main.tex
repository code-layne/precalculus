\begin{enumerate}[leftmargin=1.5em, itemsep=5pt]
  \item I can describe the end behavior of a polynomial function using limit notation
        ($\lim_{x \to \infty}$ and $\lim_{x \to -\infty}$). \hfill\textit{(LO 1.6.A)}
  \item I can determine the end behavior of a polynomial from its degree and the sign of
        its leading coefficient, and explain why the leading term dominates for large $|x|$. \hfill\textit{(EK 1.6.A.3)}
\end{enumerate}
