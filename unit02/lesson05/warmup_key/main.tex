\begin{enumerate}[itemsep=18pt]

\item Rewrite $f(x) = 2^{x+5}$ in the form $a \cdot 2^x$, where $a$ is a constant.

      \vspace{0.08in}
      $f(x) = $ \ans{$32 \cdot 2^x$}

\item Rewrite $f(x) = 4^{3x}$ as a single base raised to the $x$ power.
      \textit{Hint: } $4^{3x} = (4^3)^x$.

      \vspace{0.08in}
      $f(x) = $ \ans{$64^x$}

\item Simplify $9^{3/2}$. Show your reasoning.

      \vspace{0.08in}
      $9^{3/2} = $ \ans{27}

\end{enumerate}

\begin{teachernote}
Item 1: $2^{x+5} = 2^x \cdot 2^5 = 32 \cdot 2^x$.

Item 2: $4^{3x} = (4^3)^x = 64^x$. Some students may write $(2^2)^{3x} = 2^{6x} = 64^x$ --- accept either.

Item 3: $9^{3/2} = (\sqrt{9})^3 = 3^3 = 27$, or $(9^3)^{1/2} = 729^{1/2} = 27$.
\end{teachernote}
