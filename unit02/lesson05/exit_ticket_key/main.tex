\begin{enumerate}[itemsep=14pt]

\item Let $p(x) = -2x^5 + 7x^3 - x + 3$.

\begin{enumerate}[label=(\alph*), itemsep=6pt]
  \item Leading term: \ans{$-2x^5$}
  \item $\lim_{x \to \infty} p(x) = $ \ans{$-\infty$} \qquad
        $\lim_{x \to -\infty} p(x) = $ \ans{$\infty$}
\end{enumerate}

\item Let $q(x) = 3(x-1)^2(x+4)^2$. Do not expand.

\begin{work}
  \text{degree of } q &= 2 + 2 \\
                      &= 4 \\
  \text{leading term} &= 3 \cdot x^{2} \cdot x^{2} \\
                      &= 3x^{4}
\end{work}

\begin{enumerate}[label=(\alph*), itemsep=6pt]
  \item Degree: \ans{$4$} \qquad Leading coefficient: \ans{$3$}
  \item $\lim_{x \to \infty} q(x) = $ \ans{$\infty$} \qquad
        $\lim_{x \to -\infty} q(x) = $ \ans{$\infty$}
\end{enumerate}

\item A polynomial $r$ satisfies $\lim_{x \to \infty} r(x) = \infty$ and
      $\lim_{x \to -\infty} r(x) = -\infty$.

\begin{enumerate}[label=(\alph*), itemsep=6pt]
  \item Is the degree of $r$ even or odd? \quad \ans{Odd}
  \item Is the leading coefficient positive or negative? \quad \ans{Positive}
\end{enumerate}

\end{enumerate}
