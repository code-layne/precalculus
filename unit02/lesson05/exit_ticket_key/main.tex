\begin{enumerate}[itemsep=16pt]

\item A population grows at 6\% per year and starts at 5{,}000.

      Write the model $P(t)$ in $ab^t$ form: \quad $P(t) = $ \ans{$5000 \cdot (1.06)^t$}

      \vspace{0.06in}
      Find $P(10)$. Round to the nearest whole number.

      \vspace{0.06in}
      $P(10) = $ \ans{$5000 \cdot (1.06)^{10} \approx 5000 \cdot 1.7908 \approx 8{,}954$}

\item An exponential model passes through $(1,\,6)$ and $(3,\,54)$. Find $a$ and $b$.

      \vspace{0.06in}
      Show your work: \ans{$6 = ab^1$ and $54 = ab^3$. Divide: $54/6 = b^2$, so $b^2 = 9$, $b = 3$.\\[3pt]
      Then $a = 6/3 = 2$.}

      \vspace{0.06in}

      $a = $ \ans{2} \quad $b = $ \ans{3}

\item For the model in \#1, what does the base $b = 1.06$ tell you about the population?
      Answer in a complete sentence.

      \vspace{0.06in}
      \ans{The base $b = 1.06$ means the population grows by a factor of 1.06 each year, which represents a 6\% increase per year.}

\end{enumerate}

\begin{teachernote}
Item 1: $a = 5000$, $b = 1.06$. $P(10) = 5000(1.06)^{10} \approx 8954$.

Item 2: From dividing the two equations, $b^2 = 9$ so $b = 3$ (reject negative base). Then $a = 6/b = 2$. Model: $f(x) = 2 \cdot 3^x$.

Item 3: Accept any complete sentence that connects $b = 1.06$ to a 6\% annual growth rate.
\end{teachernote}
