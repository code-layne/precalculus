\begin{enumerate}[itemsep=16pt]

\item Let $p(x) = -2x^5 + 7x^3 - x + 3$.

\begin{enumerate}[label=(\alph*), itemsep=8pt]
  \item What is the leading term of $p(x)$? \ans{$-2x^5$}

  \item Write the limit notation for the end behavior as $x \to \infty$:

  \vspace{0.1in}
  $\displaystyle\lim_{x \to \infty} p(x) = $ \ans{$-\infty$}

  \item Write the limit notation for the end behavior as $x \to -\infty$:

  \vspace{0.1in}
  $\displaystyle\lim_{x \to -\infty} p(x) = $ \ans{$\infty$}
\end{enumerate}

\item A polynomial function $q$ has degree 4 and a positive leading coefficient.
Complete the sentence using $\infty$ or $-\infty$:

\vspace{0.15in}
As $x \to \infty$, \; $q(x) \to$ \ans{$\infty$} \quad and \quad as $x \to -\infty$, \; $q(x) \to$ \ans{$\infty$}

\end{enumerate}
