      \end{tikzpicture}
      \end{center}
      {\footnotesize $E(d)$ = elevation in feet, $d$ miles from the trailhead.}

    \column{0.46\textwidth}
      \begin{block}{Direction}
        \textbf{Increasing:} as $x$ gets larger, the outputs get larger --- the graph
        \textbf{rises}.\\[3pt]
        \textbf{Decreasing:} the graph \textbf{falls}. \textbf{Constant:} it is flat.
      \end{block}

      \medskip
      \textbf{An interval names $x$-values.}
      \begin{itemize}\setlength{\itemsep}{4pt}
        \item Increasing on $(0, 2)$ and $(6, 8)$
        \item Decreasing on $(2, 6)$
        \item Miles, not feet --- never $(300, 600)$
      \end{itemize}
  \end{columns}
\end{frame}

      \end{tikzpicture}
      \end{center}

    \column{0.46\textwidth}
      \begin{block}{Relative and absolute}
        \textbf{Relative max / min:} a peak / a valley, where the graph turns.\\[3pt]
        \textbf{Absolute max / min:} the greatest / least output anywhere, endpoints included.
      \end{block}

      \medskip
      \textbf{The value is $y$. It occurs at $x$.}
      \begin{itemize}\setlength{\itemsep}{4pt}
        \item Relative max: \textbf{600 ft}, at $d = 2$
        \item Absolute max: \textbf{700 ft}, at $d = 8$ --- the end beats the peak
      \end{itemize}
  \end{columns}
\end{frame}

    \column{0.44\textwidth}
      \begin{block}{Cup and cap}
        \textbf{Concave up:} bends like a cup, $\cup$.\\[2pt]
        \textbf{Concave down:} bends like a cap, $\cap$.\\[2pt]
        \textbf{Point of inflection:} where the bend switches.
      \end{block}

      \medskip
      On the trail: concave down on $(0, 4)$, concave up on $(4, 8)$; inflection point
      $(4, 400)$ --- where the trail drops most steeply.
  \end{columns}

    \column{0.50\textwidth}

        \item Sam: ``$f$ is decreasing on $(-2, 4)$.'' \quad Which axis did Sam read?

% DEBRIEF
\begin{frame}[t, plain]
  \plumheader{Debrief --- The Headline}
  \sectionlabel[goldacc]{Share out 15--18, one answer each $\cdot$ then say it back}

  \begin{columns}[T]
    \column{0.48\textwidth}
      \begin{block}{Which axis?}
        Intervals and locations are \textbf{$x$-values}.\\[3pt]
        A maximum or minimum \textbf{value} is a \textbf{$y$-value}.
      \end{block}
    \column{0.48\textwidth}
      \begin{block}{Which question?}
        Direction: does it \textbf{go} up or down?\\[3pt]
        Concavity: does it \textbf{bend} like a cup or a cap?
      \end{block}
  \end{columns}

  \bigskip
  \textbf{Next time (2.6):} we stop asking \emph{which way} and start measuring \emph{how
  fast} --- the average rate of change, and concavity re-read as a rate that changes.
\end{frame}
