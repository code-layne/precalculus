\begin{enumerate}[itemsep=22pt]

\item Let $f(x) = 2^x$. What whole-number value of $x$ makes $f(x) = 32$?

      Set up and solve: $2^x = 32$, so $x = $ \ans{5}

\item Let $f(x) = 10^x$. What value of $x$ makes $f(x) = 1{,}000$?

      Set up and solve: $10^x = 1{,}000$, so $x = $ \ans{3}

\item Complete the sentence:

      \medskip
      ``A logarithm asks: what \ans{exponent} do I raise the \ans{base} to,
      in order to get \ans{the result}?''

\end{enumerate}
