\begin{enumerate}[itemsep=10pt]

\item The graph of a polynomial $f$ on the restricted domain $[0, 5]$ is
      shown.

\begin{center}
\begin{tikzpicture}[x=0.75cm, y=0.5cm]
  \draw[linegray, very thin, step=1] (0,0) grid (5,5);
  \draw[->, thick] (0,0) -- (5.6,0) node[below right] {\small $x$};
  \draw[->, thick] (0,0) -- (0,5.6) node[above] {\small $f(x)$};
  \foreach \x in {1,2,3,4,5} \node[below] at (\x,0) {\small \x};
  \foreach \y in {1,2,3,4,5} \node[left] at (0,\y) {\small \y};
  \draw[very thick, plum] plot [smooth] coordinates
    {(0,2) (0.5,3.3) (1,4) (1.5,3.3) (2,2.2) (2.5,1.3) (3,1)
     (3.5,1.5) (4,2.6) (4.5,3.8) (5,5)};
  \fill[goldacc] (1,4) circle (2.7pt);
  \fill[goldacc] (3,1) circle (2.7pt);
\end{tikzpicture}
\end{center}

\begin{enumerate}[label=(\alph*), itemsep=4pt]
  \item $f$ is increasing on \ans{$(0,1)$} and on \ans{$(3,5)$} , and
        decreasing on \ans{$(1,3)$} .
  \item $f$ has a local maximum at $x = $ \ans{1} and a local
        minimum at $x = $ \ans{3} .
\end{enumerate}

\item Using the same graph:

\begin{enumerate}[label=(\alph*), itemsep=4pt]
  \item The global maximum value is \ans{5} , at $x = $
        \ans{5} .
  \item The global minimum value is \ans{1} , at $x = $
        \ans{3} .
\end{enumerate}

\item A different polynomial $g$ has these average rates of change over the
      consecutive intervals $[0,1], [1,2], [2,3], [3,4], [4,5]$:

      \begin{center}
      $3,\quad 7,\quad 12,\quad 9,\quad 5$
      \end{center}

\begin{enumerate}[label=(\alph*), itemsep=4pt]
  \item Does $g$ have a local maximum or minimum on $[0, 5]$? \quad
        \textbf{Yes} / \ans{No} \quad Why? \ans{every AROC is positive}
  \item A point of inflection occurs near $x = $ \ans{3} , where the
        graph changes from concave \ans{up} to concave \ans{down} .
\end{enumerate}

\end{enumerate}
