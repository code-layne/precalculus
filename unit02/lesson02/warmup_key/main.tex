\item Evaluate each parent function from the library at the given input.

\vspace{4pt}
\begin{tabular}{@{}l@{\quad}l@{\quad}l@{\quad}l@{}}
(a) $\sqrt{x}$ at $x = 9$: \ans{$3$} &
(b) $|x|$ at $x = -4$: \ans{$4$} &
(c) $x^3$ at $x = -2$: \ans{$-8$} &
(d) $\tfrac{1}{x}$ at $x = \tfrac{1}{2}$: \ans{$2$}
\end{tabular}

\item Let $f(x) = x^2$ and $g(x) = x + 3$. Find $f(g(-3))$ and $f(g(-5))$.

\begin{work}
  f(g(-3)) &= f(0) \\
           &= 0 \\
  f(g(-5)) &= f(-2) \\
           &= (-2)^2 = 4
\end{work}

$f(g(-3)) = $ \ans{$0$} \qquad $f(g(-5)) = $ \ans{$4$}

\item Solve each equation for $x$.

\vspace{4pt}
\begin{tabular}{@{}l@{\qquad\qquad}l@{\qquad\qquad}l@{}}
(a) $x + 3 = 0$ \qquad $x = $ \ans{$-3$} &
(b) $x - 5 = 2$ \qquad $x = $ \ans{$7$} &
(c) $2x = 6$ \qquad $x = $ \ans{$3$}
\end{tabular}

\item Which of these points lie on the graph of $y = \sqrt{x - 1}$? Circle every one that does.

\vspace{6pt}
\qquad \ans{$(1, 0)$} \qquad\qquad $(4, 2)$ \qquad\qquad \ans{$(5, 2)$} \qquad\qquad \ans{$(10, 3)$} \qquad\qquad $(0, -1)$

\begin{work}
  \sqrt{1 - 1}  &= 0 \\
  \sqrt{4 - 1}  &= \sqrt{3} \ne 2 \\
  \sqrt{5 - 1}  &= 2 \\
  \sqrt{10 - 1} &= 3
\end{work}

\end{enumerate}
