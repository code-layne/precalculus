\item From Lesson 2.0: let $p(x) = -2x^4 + 5x^3 - x + 8$.

\begin{enumerate}[label=(\alph*), itemsep=4pt]
  \item Degree: \ans{4}
  \item Leading term: \ans{$-2x^4$}
  \item Leading coefficient: \ans{$-2$}
\end{enumerate}

\item From Lesson 2.1: the table gives values of a function $g$. Find the
      average rate of change of $g$ over $[1, 3]$.

\smallskip
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{|c|c|c|c|}
\hline
$x$ & 1 & 3 & 5 \\
\hline
$g(x)$ & 4 & 16 & 40 \\
\hline
\end{tabular}
\smallskip

\begin{work}
  \text{AROC} &= \frac{g(3) - g(1)}{3 - 1} \\
              &= \frac{16 - 4}{2} \\
              &= 6
\end{work}

AROC $= $ \ans{6}

\item From Lesson 2.1: a function's average rates of change over consecutive
      equal-length intervals are $14,\ 10,\ 6,\ 2$.

\begin{enumerate}[label=(\alph*), itemsep=4pt]
  \item The AROCs are \textbf{increasing} / \ans{decreasing} (circle one).
  \item So the graph of the function is concave \ans{down}.
\end{enumerate}

\end{enumerate}
