\begin{enumerate}[leftmargin=1.9em, itemsep=12pt]

\item Which function's graph is the graph of $y = \sqrt{x}$ translated 2 units to the
      \textbf{left} and 5 units \textbf{down}?

      \begin{enumerate}[leftmargin=2.6em, itemsep=2pt]
        \item[(A)] $y = \sqrt{x - 2} - 5$
        \item[(B)] $y = \sqrt{x - 5} + 2$
        \item[(C)] $y = \sqrt{x + 2} - 5$
        \item[(D)] $y = \sqrt{x + 5} - 2$ \hfill\textit{Letter:}~\gradedblank[id=ap1, type=choice, points=1, answer=C]{0.42in}{13pt}
      \end{enumerate}

\item The point $(12, -6)$ lies on the graph of the function $f$. Let $g(x) = f(3x)$. Which
      point must lie on the graph of $g$?

      \smallskip
      \textbf{(A)} $(36, -6)$ \hspace{2.2em} \textbf{(B)} $(12, -18)$ \hspace{2.2em} \textbf{(C)} $(4, -2)$ \hspace{2.2em} \textbf{(D)} $(4, -6)$ \hfill\textit{Letter:}~\gradedblank[id=ap2, type=choice, points=1, answer=D]{0.42in}{13pt}

\item Let $g(x) = \dfrac{1}{x - 2} + 3$, a transformation of $y = \dfrac{1}{x}$. Which gives the
      vertical and horizontal asymptotes of the graph of $g$?
      \begin{enumerate}[leftmargin=2.6em, itemsep=2pt]
        \item[(A)] $x = -2$ and $y = 3$
        \item[(B)] $x = 2$ and $y = 3$
        \item[(C)] $x = 3$ and $y = 2$
        \item[(D)] $x = 2$ and $y = -3$ \hfill\textit{Letter:}~\gradedblank[id=ap3, type=choice, points=1, answer=B]{0.42in}{13pt}
      \end{enumerate}

\item \begin{minipage}[t]{0.56\linewidth}
      The graph of a transformation of $y = x^2$ is shown. Its vertex is $(1, -2)$, and it
      passes through $(0, -4)$ and $(2, -4)$. Which is an equation for the graph?
      \begin{enumerate}[leftmargin=2.6em, itemsep=3pt]
        \item[(A)] $y = -2(x - 1)^2 - 2$
        \item[(B)] $y = -2(x + 1)^2 - 2$
        \item[(C)] $y = 2(x - 1)^2 - 2$
        \item[(D)] $y = -\tfrac{1}{2}(x - 1)^2 - 2$ \hfill\textit{Letter:}~\gradedblank[id=ap4, type=choice, points=1, answer=A]{0.42in}{13pt}
      \end{enumerate}
      \end{minipage}\hfill
      \begin{minipage}[t]{0.38\linewidth}
      \vspace{-4pt}
      \centering
      \begin{tikzpicture}[x=0.5cm, y=0.3cm, baseline=(current bounding box.north)]
        \draw[linegray, very thin, step=1] (-2,-9) grid (4,2);
        \draw[->, thick] (-2,0) -- (4.5,0) node[right] {\scriptsize $x$};
        \draw[->, thick] (0,-9) -- (0,2.5) node[above] {\scriptsize $y$};
        \foreach \x in {-1,1,2,3} \node[below, font=\tiny] at (\x,0) {\x};
        \foreach \y in {-8,-6,-4,-2} \node[left, font=\tiny] at (0,\y) {\y};
        \draw[very thick, plum, domain=-1:3, samples=50] plot (\x, {-2*(\x-1)*(\x-1) - 2});
        \fill[plum] (1,-2) circle (2.2pt);
        \fill[plum] (0,-4) circle (2.0pt);
        \fill[plum] (2,-4) circle (2.0pt);
      \end{tikzpicture}
      \end{minipage}

\item The range of $f(x) = \sqrt{x}$ is $y \ge 0$. What is the range of
      $g(x) = -3\sqrt{x} + 4$?

      \smallskip
      \textbf{(A)} $y \ge 4$ \hspace{2.2em} \textbf{(B)} $y \ge 0$ \hspace{2.2em} \textbf{(C)} $y \le 4$ \hspace{2.2em} \textbf{(D)} $y \le -3$ \hfill\textit{Letter:}~\gradedblank[id=ap5, type=choice, points=1, answer=C]{0.42in}{13pt}

\end{enumerate}

\begin{enumerate}[leftmargin=1.9em, itemsep=10pt, start=6]

\item The temperature in Greenhouse A, in degrees Celsius, is $f(t)$, where $t$ is hours after
      noon (a negative $t$ is before noon). A few values are given. Greenhouse B runs on a
      different timer and heater, and its temperature is $g(t) = 2f(t - 3) + 1$.

      \begin{center}
      {\renewcommand{\arraystretch}{1.3}%
      \begin{tabular}{|c|c|c|c|c|}
      \hline
      $t$ (hours) & $-2$ & $0$ & $2$ & $4$ \\ \hline
      $f(t)$ ($^\circ$C) & $3$ & $-1$ & $1$ & $5$ \\ \hline
      \end{tabular}}
      \end{center}

      \textbf{(a)} Each point $(t, f(t))$ in the table becomes a point of $g$. Complete the
      table.

      \begin{center}
      {\renewcommand{\arraystretch}{1.35}%
      \begin{tabular}{|c|c|c|c|c|}
      \hline
      Point of $f$ & $(-2, 3)$ & $(0, -1)$ & $(2, 1)$ & $(4, 5)$ \\ \hline
      Point of $g$ & \smash{\gradedblank[id=ap6a1, type=short, check=text, points=0.5, answer={(1, 7)}, display={$(1, 7)$}]{1.4cm}{13pt}} & \smash{\gradedblank[id=ap6a2, type=short, check=text, points=0.5, answer={(3, -1)}, display={$(3, -1)$}]{1.4cm}{13pt}} & \smash{\gradedblank[id=ap6a3, type=short, check=text, points=0.5, answer={(5, 3)}, display={$(5, 3)$}]{1.4cm}{13pt}} & \smash{\gradedblank[id=ap6a4, type=short, check=text, points=0.5, answer={(7, 11)}, display={$(7, 11)$}]{1.4cm}{13pt}} \\ \hline
      \end{tabular}}
      \end{center}

      \textbf{(b)} Find $g(5)$, and name the value of $f$ it used. Interpret $g(5)$ in the
      context of the problem, with units.
      \rubricbox{\gradedblank[id=ap6b, type=rubric, points=2,
        display={$g(5) = 2f(2) + 1 = 2(1) + 1 = 3$; it used $f(2)$. At 5 p.m., the temperature in Greenhouse B is 3 degrees Celsius.}]{\linewidth}{\dimexpr 2.0cm-4pt\relax}}

      \textbf{(c)} Describe, in order, the transformations that turn the graph of $f$ into the
      graph of $g$.
      \rubricbox{\gradedblank[id=ap6c, type=rubric, points=2,
        display={Right 3 (the inside is $t - 3$); then a vertical stretch by a factor of 2; then up 1.}]{\linewidth}{\dimexpr 2.0cm-4pt\relax}}

      \textbf{(d)} A third greenhouse runs on a faster cycle, $h(t) = f(2t)$. Kai claims the
      graph of $h$ passes through $(8, 5)$, because $f(4) = 5$. Is Kai right? Find the correct
      point and explain.
      \rubricbox{\gradedblank[id=ap6d, type=rubric, points=2,
        display={No. $h(t)$ uses $f(4)$ when $2t = 4$, so at $t = 2$: the point is $(2, 5)$. The inside runs backwards --- $f(2t)$ compresses the graph by a factor of $\tfrac{1}{2}$, so the same temperatures come twice as fast.}]{\linewidth}{\dimexpr 2.4cm-4pt\relax}}

      \textbf{(e)} Write a function $j$, in terms of $f$, whose graph is the graph of $f$ reflected
      over the $y$-axis and then moved up 4. Which point of $j$ comes from $(4, 5)$?
      \rubricbox{\gradedblank[id=ap6e, type=rubric, points=2,
        display={$j(t) = f(-t) + 4$. The point $(4, 5)$ lands at $(-4, 9)$: $-t = 4$ at $t = -4$, and $5 + 4 = 9$.}]{\linewidth}{\dimexpr 2.2cm-4pt\relax}}

\end{enumerate}
