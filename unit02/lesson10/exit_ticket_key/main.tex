\begin{enumerate}[leftmargin=1.8em, itemsep=14pt]

\item The point $(2,\, 9)$ is on the graph of $g(x) = b^x$. What point must be on the
      graph of the inverse function $f(x) = \log_b x$?

\vspace{4pt}
Point on $f$: \ans{$(9,\; 2)$}

\begin{teachernote}
Students who write $(b, 2)$ or $(2, b)$ are confusing the base with the input/output.
Reinforce: the inverse swaps the ordered pair; $(s,t)$ on $g$ gives $(t,s)$ on $f$.
\end{teachernote}

\vspace{6pt}

\item Verify that $f(x) = \log_3 x$ and $g(x) = 3^x$ are inverse functions by computing
      $g(f(27))$. Show each step.

\vspace{4pt}
\begin{enumerate}[leftmargin=1.5em, itemsep=6pt, label=\alph*.]
  \item $f(27) = \log_3 27 = $ \ans{3} \quad (since $3^3 = 27$)
  \item $g(f(27)) = 3^{3} = $ \ans{27}
  \item Does $g(f(27)) = 27$? \ans{Yes}
\end{enumerate}

\vspace{6pt}

\item Complete the table for $f(x) = \log_2 x$:

\vspace{6pt}
\begin{center}
\renewcommand{\arraystretch}{1.9}
\begin{tabular}{|c|c|}
\hline
\rowcolor{sky}
$x$ & $f(x) = \log_2 x$ \\
\hline
$1$   & \ans{0} \\
\hline
$2$   & \ans{1} \\
\hline
$8$   & \ans{3} \\
\hline
$\tfrac{1}{2}$ & \ans{$-1$} \\
\hline
\end{tabular}
\end{center}

\begin{teachernote}
The row $x = \tfrac{1}{2}$ tests negative logarithm outputs. Students who leave it blank
likely have not yet internalized that $2^{-1} = \tfrac{1}{2}$, so $\log_2(\tfrac{1}{2}) = -1$.
Address in the next warm-up.
\end{teachernote}

\end{enumerate}
