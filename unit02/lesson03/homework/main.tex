\begin{enumerate}[leftmargin=1.9em, itemsep=10pt, start=3]

\item Let $r(x) = x^4 - 6x^2 + 1$. Find $r(-x)$, simplify, and decide: even, odd, or neither.

\begin{work}
  r(-x) &= (-x)^4 - 6(-x)^2 + 1 \\
        &= x^4 - 6x^2 + 1 \\
        &= r(x)
\end{work}

      $r$ is \blank{2.2cm}

\item Let $w(x) = x^3 - x + 2$. A classmate says $w$ must be odd, because $x^3$ and $x$ are
      odd powers. Test it.

\begin{work}
  w(-x) &= (-x)^3 - (-x) + 2 \\
        &= -x^3 + x + 2 \\
  -w(x) &= -x^3 + x - 2
\end{work}

      Is the classmate right? Explain using your work.
      \writespace{1.3cm}{}

\end{enumerate}

\item \begin{minipage}[t]{0.47\linewidth}
      Let $f(x) = 4x + 8$. Find $f^{-1}(x)$.

\begin{work}
  x &= 4y + 8 \\
  x - 8 &= 4y \\
  f^{-1}(x) &= \dfrac{x - 8}{4}
\end{work}
      \end{minipage}\hfill
      \begin{minipage}[t]{0.49\linewidth}
