\begin{skillbox}[Priority Ideas \& Skills]{goldbox}
\begin{tabularx}{\linewidth}{@{}X X@{}}
\textbf{Priority Skills} &
\textbf{Key Understandings} \\
\midrule
\textbf{AP Skill 2.B} --- Construct equivalent graphical, numerical, and analytical
representations: decide even, odd, or neither from a graph and from a formula. &
\begin{itemize}[leftmargin=1.2em,itemsep=2pt,topsep=0pt]
  \item Even: symmetric over the $y$-axis, and $f(-x) = f(x)$. Odd: symmetric about the
        origin, and $f(-x) = -f(x)$. Most functions are neither. (EK 1.5.B.1--2)
  \item $f(-x)$ is the $y$-axis reflection from 2.2: an even function is one that reflection
        leaves unchanged.
  \item \textbf{Second trap:} ``it has an odd power, so it is odd.'' $x^3 + 1$ is neither ---
        the constant never flips sign.
\end{itemize} \\
\textbf{AP Skill 2.B} --- Read the input-output pairs of an inverse from a table or a graph. &
\begin{itemize}[leftmargin=1.2em,itemsep=2pt,topsep=0pt]
  \item A function is invertible on a domain where each output comes from exactly one input ---
        the horizontal line test. (EK 2.8.A.1)
  \item If $f(a) = b$, then $f^{-1}(b) = a$: the pair $(a, b)$ becomes $(b, a)$, a reflection
        over $y = x$. Evaluating $f^{-1}$ at $b$ is solving $f(x) = b$. (EK 2.8.A.2, 2.8.B.3)
  \item The domain of $f$ is the range of $f^{-1}$, and the range of $f$ is the domain of
        $f^{-1}$. (EK 2.8.B.2)
\end{itemize} \\
\textbf{AP Skill 1.A} --- Solve equations: find $f^{-1}$ by swapping $x$ and $y$ and solving. &
\begin{itemize}[leftmargin=1.2em,itemsep=2pt,topsep=0pt]
  \item Swap and solve; then verify by composition: $f(f^{-1}(x)) = f^{-1}(f(x)) = x$, the
        1.3 idea of one machine undoing another. (EK 2.8.B.1, 2.8.B.4)
  \item Restricting the domain makes a function invertible: $x^2$ on $x \ge 0$ has inverse
        $\sqrt{x}$; a context can restrict an inverse further. (EK 2.8.A.1, 2.8.B.5)
  \item \textbf{Target misconception:} $f^{-1}(x)$ means $\dfrac{1}{f(x)}$. The $-1$ names
        the inverse --- it is not an exponent. $g^{-1}(10) = 3$ for $g(x) = 2x + 4$, while
        $1/g(10) = 1/24$.
\end{itemize} \\
\end{tabularx}
\end{skillbox}

\begin{skillbox}[{Vocabulary, Concepts, \& Theorems}]{greenbox}
\renewcommand{\arraystretch}{1.4}
\begin{tabularx}{\linewidth}{@{} >{\leavevmode\bfseries\color{plum}}p{0.24\linewidth} X @{}}
Even function        & $f(-x) = f(x)$ for every $x$ in the domain; the graph is symmetric over the $y$-axis. \\
Odd function         & $f(-x) = -f(x)$ for every $x$ in the domain; the graph is symmetric about the origin. \\
One-to-one function  & A function in which each output comes from exactly one input. \\
Horizontal line test & A graph is one-to-one exactly when no horizontal line crosses it more than once. \\
Inverse function     & $f^{-1}$, the function that runs $f$ backward: $f(a) = b$ means $f^{-1}(b) = a$. Its graph is the reflection of $f$'s over $y = x$. \\
Restricted domain    & A smaller domain chosen so a function becomes one-to-one, such as $x^2$ on $x \ge 0$. \\
Identity check       & \TallMath{f\big(f^{-1}(x)\big) = f^{-1}\big(f(x)\big) = x} \\
\end{tabularx}
\end{skillbox}

\begin{skillbox}[Activate Prior Knowledge \& Spiral Review (5 min)]{lilac}
    Please see attached warm-up (spiral review).
    Skills reviewed: simplifying $(-x)^2$, $(-x)^3$ and the negative of an expression --- the
    sign work every even/odd test runs on; solving $y = 2x + 5$ for $x$ --- the move that finds
    an inverse; evaluating a composition $f(g(5))$ from 1.3 --- which, for this pair, comes back
    to 5 and previews the inverse check; and moving a point under the 2.2 reflections
    $y = f(-x)$ and $y = -f(x)$.
\end{skillbox}

\begin{skillbox}[Hook (2 min)]{lilac}
    \textbf{Three letters.} On the board, large: \textsf{\textbf{A \quad S \quad H}}.
    \begin{enumerate}[itemsep=2pt]
        \item Ask which letters look the same in a mirror held along a vertical line through
              the middle. A and H. Then which look the same after a half turn --- upside down.
              S and H.
        \item Name the two tests out loud: the \textit{fold} and the \textit{half turn}. Graphs
              get the same two tests today, and they have names --- \textbf{even} and
              \textbf{odd}.
    \end{enumerate}
    \textit{Discussion prompt:} then add a third mirror, tilted: the diagonal line $y = x$.
    A function and its inverse are mirror images across it. That is the second half of the
    period.
\end{skillbox}

\begin{skillbox}[Lesson --- I Do / We Do / You Do]{lilac}
\begin{multicols}{2}

\textbf{I DO --- Part 1: The Fold and the Half Turn} (10 min)

\textit{Notes rows SYMMETRY and EVEN OR ODD TEST: read each definition, mark up the display,
work problems 1 and 3. Teacher works; students watch and annotate. Nothing is cold-called in
this phase.}
\begin{itemize}
  \item Read the SYMMETRY sentences. On Graph A, fold along the $y$-axis with your hand and
        point to the gold pair $(2, -2)$ and $(-2, -2)$; on Graph B, spin the page a half
        turn. Fill both label boxes yourself: $(-x, y)$ and $(-x, -y)$.
  \item Work \textbf{problem 1} aloud: A is even, and say \textit{why} --- the fold matches.
  \item Read the EVEN OR ODD TEST sentences and the three steps. Tie $f(-x)$ to 2.2 in one
        sentence: it is the $y$-axis reflection, and an even function is one it leaves alone.
  \item Work \textbf{problem 3}, $f(x) = x^4 - 5x^2$: write $(-x)$ in parentheses
        \textit{first}, narrate ``an even power erases the sign,'' then compare to $f(x)$.
\end{itemize}

\textbf{WE DO --- Part 2: Even, Odd, or Neither} (6 min)

\textit{Problems 2 and 4--6. Class fills the blanks together; cold-call every one.}
\begin{itemize}
  \item \textbf{Problem 2}: B is odd, C is neither. Push on C: it \textit{looks} like B moved
        up, and that is exactly why it fails.
  \item \textbf{Problems 4 and 5}: one odd, one even. On 4, make the class factor out the
        $-1$ to \textit{see} $-g(x)$.
  \item \textbf{Problem 6 is the second trap}: ``$x^3 + 1$ has an odd power, so it is odd.''
        Write $h(-x) = -x^3 + 1$ and $-h(x) = -x^3 - 1$ side by side and let the class find
        the sign that disagrees. Graph C was this function's cousin.
\end{itemize}

\columnbreak

\textbf{WE DO --- Part 3: Inverses from Tables and Graphs} (8 min)

\textit{Notes row INVERSES. Class fills the blanks together; cold-call every one. Teacher
works problem 7; the class works 8--10.}
\begin{itemize}
  \item Read the definitions; on the left graph, draw nothing --- just point at the dashed
        line $y = 4$ crossing twice. That is ``not one-to-one.''
  \item On the right graph, fill the two label boxes: the line $y = x$, and $(1, 3) \to
        (3, 1)$. Then fill the $f^{-1}$ table as a class, one cold-call per cell.
  \item Work \textbf{problem 7}: $f(3) = 7$ but $f^{-1}(3) = 1$. Say it as a sentence: ``3
        came \textit{out} of $f$ when 1 went in.'' The two readings disagree on purpose.
  \item \textbf{Problem 8} is the evaluate-versus-solve distinction: solving $f(x) = 9$
        \textit{is} evaluating $f^{-1}(9)$. \textbf{Problem 10} swaps domain and range.
\end{itemize}

\textbf{WE DO --- Part 4: Finding the Inverse} (8 min)

\textit{Notes row FINDING THE INVERSE --- the crux. Class fills the blanks together; cold-call
every one. Teacher works problem 11; the class works 12--14.}
\begin{itemize}
  \item Read the four steps, then the display: the INVERSE box against the RECIPROCAL box.
        Read the plum sentence aloud --- the $-1$ is a name, not an exponent.
  \item Work \textbf{problem 11}, $f(x) = 3x - 6$, swap then solve. \textbf{Problem 12}
        checks it by composition --- the 1.3 move, and the ``in your own words'' line.
  \item \textbf{Problem 13 is the crux}: $g^{-1}(10) = 3$ against $1/g(10) = 1/24$. Make the
        class put \textit{both} into $g$; only one comes back to 10.
  \item \textbf{Problem 14} restricts $x^2$ to $x \ge 0$: without it, $f^{-1}(9)$ would need
        two answers --- the left graph again.
\end{itemize}

\textbf{YOU DO --- Part 5: Mirror and Undo} (16 min)

\textit{Notes row MIRROR AND UNDO, problems 15--18 --- the whole independent phase. Students
work alone; circulate and say nothing a neighbor could say instead.}
\begin{itemize}
  \item \textbf{Problem 15} tests $r(x) = 2x^3 - x$ (odd) and $q(x) = x^3 - 2$ (neither): the
        second trap transferred. \textbf{16} is the crux transferred to a table.
  \item \textbf{Problem 17} finds and checks $h^{-1}$ for $h(x) = 4x - 3$.
  \item \textbf{Problem 18} is the finisher's question: restrict $(x - 2)^2$ to $x \ge 2$, find
        $k^{-1}(x) = \sqrt{x} + 2$, and swap domain and range.
\end{itemize}
\end{multicols}
\end{skillbox}

\begin{skillbox}[Active Monitoring]{redbox}
Circulate during the \textbf{You Do} phase --- the sixteen minutes where students are working
problems 15--18 without you --- and check. The crux is \textbf{problem 13} ($g^{-1}(10) = 3$,
not $1/24$); its You Do transfer is \textbf{problem 16} ($s^{-1}(4) = 3$, not $1/2$).
\begin{itemize}
  \item \textbf{Common error (the target misconception):} reading $f^{-1}$ as $1/f$. On
        problem 16 a student writes $\tfrac{1}{2}$. Push: ``Put your answer into $s$. Do you
        get 4 back?'' Point at the INVERSE and RECIPROCAL boxes from Part 4.
  \item \textbf{Common error (the second trap):} calling $q(x) = x^3 - 2$ odd because of the
        cube. Ask: ``What is $-q(x)$? Write it next to $q(-x)$. Do they match?''
  \item \textbf{Common error:} dropping the parentheses, so $(-x)^3$ becomes $x^3$ or
        $-x^2$ becomes $x^2$. Cold-call: ``Is the power even or odd? So does the sign stay?''
  \item \textbf{Common error:} solving for $x$ without swapping, then naming the result
        $h^{-1}(y)$ --- or swapping and never solving. Ask: ``Which letter is the input of
        $h^{-1}$? Is $y$ alone yet?''
  \item \textbf{Evaluate versus solve:} on a table, reading $s(4)$ when asked for $s^{-1}(4)$.
        Ask: ``Is 4 going in, or did 4 come out?''
\end{itemize}
\end{skillbox}

\begin{skillbox}[Differentiation --- During You Do (16 min)]{redbox}
There is no separate activity sheet: students work the notes' You Do row (problems 15--18)
alone while you circulate. Differentiate by \textit{where you stand}, not by handing out
different paper.

\begin{itemize}
  \item \textbf{Support.} Sit with a struggling student on \textbf{problem 15} and have them
        write $r(-x) = 2(-x)^3 - (-x)$ with every parenthesis before simplifying. Then make
        them write $-r(x)$ on the line below and compare term by term. The comparison, not
        the simplifying, is where they are lost.
  \item \textbf{On level.} \textbf{Problems 16 and 17} are the inverse in two representations.
        On 16 do not answer; ask ``which row is 4 in?'' and walk away. On 17, make them check
        with composition rather than asking you.
  \item \textbf{Extend.} \textbf{Problem 18} is the finisher's question --- the swap gives
        $x = (y - 2)^2$, and choosing $+\sqrt{x}$ rather than $\pm\sqrt{x}$ is exactly what the
        restriction $x \ge 2$ buys. A finisher who says \textit{why} the domain of $k^{-1}$ is
        $x \ge 0$ (it is the range of $k$) presents in the Debrief.
\end{itemize}
\end{skillbox}

\boxguard[20]
\begin{skillbox}[Debrief (5 min)]{redbox}
Whole class, teacher-led, packets still open. Three moves, in order:
\begin{enumerate}[itemsep=3pt]
  \item \textbf{Share out the You Do} (3 min) --- one answer per problem, not a full review.
        Problem 15: $r$ odd, $q$ neither --- name the $-2$ that refuses to flip. Problem 16:
        $s^{-1}(4) = 3$, with the check $s(3) = 4$. Problem 17: $h^{-1}(x) = \frac{x + 3}{4}$.
        Problem 18: $k^{-1}(x) = \sqrt{x} + 2$, domain $x \ge 0$ --- a finisher presents why.
  \item \textbf{Name the headline} (1 min) --- say it, then make the class say it back:
        \textit{an inverse swaps every input with its output, so $f^{-1}$ runs $f$ backward ---
        it undoes $f$; it is not $1/f$.} Point at the dashed line $y = x$ while you say it,
        then at the three symmetries on the slide: the fold, the half turn, the diagonal
        mirror.
  \item \textbf{Point forward} (1 min) --- the DeltaMath assignment, \textit{Lesson 2.3}, and
        its due date: write it on the board and have students copy it into the cover's
        Homework row (row 4). Then: next class is a quiz on 2.1--2.3; after it, 2.4 reads
        domain, range, intercepts, and end behavior from any graph --- and problem 10's
        swap of domain and range is the first of those.
\end{enumerate}
\textbf{Do not} re-teach the algebraic test here or take new questions. There is no exit
ticket: the You Do was the period's formative read. If it showed the crux
did not land, that is the quiz-review warm-up's job, not this minute's.
\end{skillbox}

\begin{teachernote}[Warm-Up]
Five minutes, silent start, no calculators. Item 1 is the only rehearsal of the sign work in
$(-x)^n$ --- a student who writes $(-x)^2 = -x^2$ will call every even function odd within the
hour, so note names before the notes start. Item 2 is the swap-and-solve move without the swap;
item 3 is a composition that comes back to 5 on purpose --- if anyone notices, tell them to
hold the thought until Part 4. Item 4 sets up the even/odd point pairs: $(-2, -4)$ is the
$y$-axis partner an even function would need.
\end{teachernote}

\begin{teachernote}[Guided Notes]
The notes are one Main Ideas / Notes table, four instruction rows and a You Do row; in each
instruction row you read the printed definition, mark up the display, and work the first
problem of the grid (1, 3, 7, 11), and the class works the rest with the pen in their hand,
every problem cold-called. Keep the theme audible: \textit{three symmetries} --- the fold
(even), the half turn (odd), the diagonal mirror (inverse). The SYMMETRY row is read from
pictures only; resist doing algebra there. In EVEN OR ODD TEST, the parentheses around $-x$
are the whole technique, and problem 6 ($x^3 + 1$) is a planned trap --- let a student say
``odd'' before anyone tests it.

INVERSES is where the evaluate-versus-solve distinction now lives: $f(3) = 7$ but
$f^{-1}(3) = 1$ (problem 7), and solving $f(x) = 9$ is evaluating $f^{-1}(9)$ (problem 8).
FINDING THE INVERSE carries the crux. Problem 13 puts $g^{-1}(10)$ and $1/g(10)$ side by side
so the two readings \emph{disagree} --- 3 against $\frac{1}{24}$ --- and the check (put each into
$g$) settles it without your authority. Do not accept ``the inverse is the flip''; make the
class say ``runs $g$ backward.''

The You Do row (15--18) is the entire independent phase and carries sixteen minutes. Problem
16 is the crux with a table instead of a formula; problem 18 is the finisher's question. If the
clock slips, cut 18 and let the Debrief carry its answer; restriction returns in the last
DeltaMath topic and AP Practice 6(e).
\end{teachernote}

\begin{teachernote}[Debrief]
Five minutes, and they are at risk: the guided phase runs 22 minutes across two topics and will
try to borrow them. Borrow from problem 18 instead --- restriction reappears in the last
DeltaMath topic. The share-out is one answer per You Do problem so that a student who stalled
on 15 still hears $s^{-1}(4) = 3$ with its check --- the exact shape of the diagnostic
DeltaMath topic, which they will otherwise meet alone at home.
\end{teachernote}

\begin{teachernote}[AP Practice]
Extra credit, optional, and not a substitute for the DeltaMath assignment --- say so when you
hand out the packet; the cover gives it a $+$ cell outside the total. Multiple-choice answers are B, C, A, D,
B. Item 2 is the most useful one to glance at: (B) $-\frac{1}{4}$ is $1/f(6)$ --- the target
misconception --- and (A) $-4$ is $f(6)$, evaluating instead of solving. Item 1's (A),
$x^3 + 1$, is the second trap. In the free response, credit (d) for any answer saying that
$1/C(50)$ is a reciprocal of a cost while $C^{-1}$ turns a cost back into a number of
cupcakes; (e) is the deepest part --- credit any answer saying half a cupcake cannot be
ordered, so only certain costs are valid inputs of $C^{-1}$.
\end{teachernote}

\begin{teachernote}[Homework]
DeltaMath, graded, assigned as \textit{Lesson 2.3}. The diagnostic is \textbf{Evaluating
Inverse Functions (Tables and Graphs)}: a miss there means the student is evaluating $f$ where
the item asks for $f^{-1}$ --- reading the wrong row of the table --- or taking a reciprocal,
and either one will cost them on the quiz. Check that topic's scores before the quiz-review
warm-up. DeltaMath cannot set $f^{-1}(b)$ beside $1/f(b)$ and ask which one undoes $f$, nor
ask \textit{why} $x^3 + 1$ is not odd; You Do problems 16 and 15 carry those, so read them
during the You Do rather than waiting on scores. The second topic (algebraic even/odd test)
also catches the odd-power trap when the function has a constant term. The restricted-domain
topic predicts 2.4: a student who states the domain of $k^{-1}$ as the range
of $k$ is ready to read domain and range from any graph. If that topic or the diagnostic slips
for more than a few students, open the quiz-review warm-up with one table item ($f^{-1}(b)$
from a table) and one swap of domain and range.
\end{teachernote}
