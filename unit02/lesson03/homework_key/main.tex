\begin{enumerate}[itemsep=16pt]

\item For each exponential function, identify $a$, $b$, classify as growth or decay, and state
      the range.

      \begin{enumerate}[label=(\alph*), itemsep=10pt]
        \item $f(x) = 6 \cdot 2^x$

              $a = $ \ans{6} \quad $b = $ \ans{2} \quad
              Growth or decay: \ans{Growth} \quad Range: \ans{$(0,\infty)$}

        \item $g(x) = 12 \cdot (0.25)^x$

              $a = $ \ans{12} \quad $b = $ \ans{0.25} \quad
              Growth or decay: \ans{Decay} \quad Range: \ans{$(0,\infty)$}

        \item $h(x) = 0.5 \cdot 4^x$

              $a = $ \ans{0.5} \quad $b = $ \ans{4} \quad
              Growth or decay: \ans{Growth} \quad Range: \ans{$(0,\infty)$}
      \end{enumerate}

\item Determine whether each table represents exponential, linear, or neither.

      \begin{enumerate}[label=(\alph*), itemsep=12pt]
        \item Ratios: $6/2=3$, $18/6=3$, $54/18=3$ --- constant.

              Work: \ans{Constant ratio of 3} \quad Type: \ans{Exponential}

        \item Differences: $9-5=4$, $13-9=4$, $17-13=4$ --- constant.

              Work: \ans{Constant difference of 4} \quad Type: \ans{Linear}

        \item Ratios: $3/1=3$, $8/3\approx2.67$, $16/8=2$ --- not constant.
              Differences: $2$, $5$, $8$ --- not constant.

              Work: \ans{Neither constant ratios nor constant differences} \quad Type: \ans{Neither}
      \end{enumerate}

\item $q$ outputs 4, 20, 100, 500 for inputs 0, 1, 2, 3. Ratios: $20/4=5$, $100/20=5$, $500/100=5$.

      $a = $ \ans{4} \quad $b = $ \ans{5} \quad $q(x) = $ \ans{$4 \cdot 5^x$}

\item End behavior:

      \begin{enumerate}[label=(\alph*), itemsep=8pt]
        \item $f(x) = 7 \cdot 3^x$ (growth, $b=3>1$):

              $\displaystyle\lim_{x\to\infty} f(x) = $ \ans{$\infty$} \qquad
              $\displaystyle\lim_{x\to-\infty} f(x) = $ \ans{$0$}

        \item $g(x) = 8 \cdot (0.6)^x$ (decay, $0<b<1$):

              $\displaystyle\lim_{x\to\infty} g(x) = $ \ans{$0$} \qquad
              $\displaystyle\lim_{x\to-\infty} g(x) = $ \ans{$\infty$}
      \end{enumerate}

\item \textbf{(AP-Style Multiple Choice)} Answer: \ans{C}

      Justification: \ans{$h(x) = 5\cdot(0.8)^x$ has $a=5>0$ and $0 < b=0.8 < 1$, so it is always
      decreasing and has horizontal asymptote $y=0$; option (B) has $a<0$ so the asymptote is still
      $y=0$ but the range is $(-\infty,0)$ and it is always increasing; option (D) is a constant
      function, not exponential.}

\item \textbf{(Extension)} \ans{If $b=1$, then $f(x) = a\cdot 1^x = a$ for all $x$, which is a
      constant function. It fails the defining property of exponential functions in two ways:
      (1) its outputs are not proportional over equal-length input intervals (they are all equal,
      so every ratio is 1 trivially, but the function neither increases nor decreases), and
      (2) it has no end behavior tending to $\infty$ or $0$; the limit in both directions is
      simply $a$. Excluding $b=1$ ensures the function is strictly monotone.}

\end{enumerate}
