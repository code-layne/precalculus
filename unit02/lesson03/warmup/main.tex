\item Factor completely: $x^2 - 7x + 12$.

\begin{work}
  x^2 - 7x + 12 &= (x - 3)(x - 4) \\
  \text{check: } (-3)(-4) &= 12 \text{ and } -3 + (-4) = -7
\end{work}

$x^2 - 7x + 12 = $ \blank{4cm}

\item Solve $(x + 5)(2x - 3) = 0$. (Zero-product property: if a product is
      zero, at least one factor is zero.)

\begin{work}
  x + 5 &= 0 \quad \text{or} \quad 2x - 3 = 0 \\
      x &= -5 \quad \text{or} \quad 2x = 3 \\
      x &= -5 \quad \text{or} \quad x = \tfrac{3}{2}
\end{work}

$x = $ \blank{2cm} or $x = $ \blank{2cm}

\item Let $p(x) = x^3 - 2x^2 - 5x + 6$. Find $p(1)$.

\begin{work}
  p(1) &= (1)^3 - 2(1)^2 - 5(1) + 6 \\
       &= 1 - 2 - 5 + 6 \\
       &= 0
\end{work}

\begin{enumerate}[label=(\alph*), itemsep=4pt]
  \item $p(1) = $ \blank{2cm}
  \item Is $x = 1$ a solution of the equation $p(x) = 0$? \quad
        \textbf{Yes} / \textbf{No}
\end{enumerate}

\end{enumerate}
