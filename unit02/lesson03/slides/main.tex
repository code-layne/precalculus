% ROW 2 — THE ALGEBRAIC TEST
\begin{frame}[t, plain]
  \plumheader{Even or Odd --- The Algebraic Test}
  \sectionlabel[goldacc]{Notes row EVEN OR ODD TEST $\cdot$ I Do: problem 3 $\cdot$ We Do: 4--6 (6 is the trap)}

  \begin{block}{Replace \(x\) with \((-x)\), simplify, compare}
    \(f(-x) = f(x)\) \(\Rightarrow\) \textbf{even} \qquad
    \(f(-x) = -f(x)\) \(\Rightarrow\) \textbf{odd} \qquad
    neither \(\Rightarrow\) \textbf{neither}
  \end{block}

  \medskip
  \begin{columns}[T]
    \column{0.48\textwidth}
      \textbf{Work it: \(f(x) = x^4 - 5x^2\)}
      \begin{align*}
        f(-x) &= (-x)^4 - 5(-x)^2 \\
              &= x^4 - 5x^2 = f(x)
      \end{align*}
      Even. An even power erases the sign.

    \column{0.48\textwidth}
      \textbf{Your turn}
      \begin{itemize}\setlength{\itemsep}{5pt}
        \item \(g(x) = x^3 - 4x\)
        \item \(k(x) = 6 - x^2\)
        \item \(h(x) = x^3 + 1\) --- it has an odd power. \textbf{Is it odd?}
      \end{itemize}
  \end{columns}

  \bigskip
  \textbf{The signs:} \((-x)^2 = x^2\), but \((-x)^3 = -x^3\).
\end{frame}

% ROW 3 — INVERSES FROM TABLES AND GRAPHS
\begin{frame}[t, plain]
  \plumheader{One-to-One, and the Inverse}
  \sectionlabel[goldacc]{Notes row INVERSES $\cdot$ I Do: problem 7 $\cdot$ We Do: problems 8--10}

  \begin{columns}[T]
    \column{0.52\textwidth}
      \begin{block}{Runs \(f\) backward}
        If \(f(a) = b\), then \(f^{-1}(b) = a\).\\[3pt]
        Each point \((a, b)\) becomes \((b, a)\): a reflection over \(y = x\).\\[3pt]
        Domain of \(f\) = range of \(f^{-1}\), and the reverse.
      \end{block}

      \medskip
      \textbf{Only a one-to-one function has one.} Horizontal line test: no horizontal
      line crosses the graph twice. \(y = x^2\) fails --- \(-2\) and \(2\) both give 4.

      \medskip
      \begin{tabular}{|c|c|c|c|c|c|}
      \hline \(x\) & 0 & 1 & 2 & 3 & 4 \\ \hline \(f(x)\) & 1 & 3 & 5 & 7 & 9 \\ \hline
      \end{tabular}

      \smallskip
      \(f(3) = 7\), \quad but \(f^{-1}(3) = 1\).

    \column{0.44\textwidth}
      \centering
      \begin{tikzpicture}[x=0.75cm, y=0.75cm]
        \draw[linegray, very thin, step=1] (0,0) grid (5,5);

% ROW 4 — FINDING THE INVERSE, AND THE CRUX
\begin{frame}[t, plain]
  \plumheader{Finding \(f^{-1}\) --- and What It Is Not}
  \sectionlabel[redacc]{Notes row FINDING THE INVERSE $\cdot$ I Do: 11 $\cdot$ We Do: 12--14 (13 is the crux)}

  \begin{columns}[T]
    \column{0.48\textwidth}
      \begin{block}{Swap and solve}
        \textbf{1.} Write \(y = f(x)\).\quad \textbf{2.} Swap \(x\) and \(y\).\\[2pt]
        \textbf{3.} Solve for \(y\).\quad \textbf{4.} Rename it \(f^{-1}(x)\).
      \end{block}
      \textbf{\(f(x) = 3x - 6\)}
      \begin{align*}
        x &= 3y - 6 \\
        f^{-1}(x) &= \frac{x + 6}{3}
      \end{align*}
      \textbf{Check:} \(f(f^{-1}(x)) = x\) and \(f^{-1}(f(x)) = x\).

    \column{0.48\textwidth}
      \begin{block}{\(f^{-1}(x)\) is NOT \(\dfrac{1}{f(x)}\)}
        \(g(x) = 2x + 4\). Find \(g^{-1}(10)\) and \(\dfrac{1}{g(10)}\).\\[3pt]
        Which one undoes \(g\)? Put each into \(g\) and see.
      \end{block}
      The \(-1\) names the inverse --- it is \textbf{not} an exponent.

      \medskip
      \textbf{Not one-to-one?} Restrict the domain: \(x^2\) on \(x \ge 0\) has
      inverse \(\sqrt{x}\).
  \end{columns}
\end{frame}

  \sectionlabel[goldacc]{Share out 15--18, one answer each $\cdot$ then say it back}

  \begin{columns}[T]
    \column{0.31\textwidth}
      \begin{block}{Over the \(y\)-axis}
        \textbf{Even:} \(f(-x) = f(x)\)
      \end{block}
    \column{0.31\textwidth}
      \begin{block}{About the origin}
        \textbf{Odd:} \(f(-x) = -f(x)\)
      \end{block}
    \column{0.31\textwidth}
      \begin{block}{Over \(y = x\)}
        \textbf{Inverse:} \((a, b) \to (b, a)\)
      \end{block}
  \end{columns}

  \bigskip
  \begin{center}
    \large\textit{An inverse swaps every input with its output, so \(f^{-1}\) runs \(f\)
    backward. It undoes \(f\) --- it is not \(1/f\).}
  \end{center}
