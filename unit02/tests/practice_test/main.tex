\parthead{Part D --- Average Rate of Change and Secant Lines (6 pts)}

Let $h(x) = \sqrt[3]{x + 1} - 2$.

\begin{work}
                     h(-9) &= \sqrt[3]{-8} - 2 \\
                           &= -4 \\
                      h(7) &= \sqrt[3]{8} - 2 \\
                           &= 0 \\
  \frac{h(7) - h(-9)}{7 - (-9)} &= \frac{0 - (-4)}{16} \\
                           &= \frac{1}{4} \\
                       y - 0 &= \tfrac{1}{4}(x - 7) \\
                           y &= \tfrac{1}{4}x - \tfrac{7}{4}
\end{work}

\qn{19} Find the average rate of change of $h$ from $x = -9$ to $x = 7$. Write your answer
as a simplified fraction. \hfill
\gradedblank[id=p19, type=short, points=3, answer=1/4]{2cm}{1.1cm}
