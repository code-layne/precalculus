\cshead{Transformations}
\[ g(x) = a\,f\big(b(x - h)\big) + k \]
\begin{itemize}
  \item $h$: shift \textbf{right} $h$ (inside: opposite sign)
  \item $k$: shift \textbf{up} $k$
  \item $|a|$: vertical stretch ($>1$) or shrink ($<1$);
        $a<0$: reflect over the $x$-axis
  \item $b$: horizontal shrink by $\tfrac{1}{|b|}$;
        $b<0$: reflect over the $y$-axis
  \item Factor $b$ out first: $f(2x+6) = f\big(2(x+3)\big)$
\end{itemize}
\textbf{Moving a point:} $(x,\,y) \to \left(\tfrac{x}{b} + h,\ a y + k\right)$

\cshead{Symmetry}
\begin{itemize}
  \item \textbf{Even:} $f(-x) = f(x)$. Mirror over the $y$-axis.
  \item \textbf{Odd:} $f(-x) = -f(x)$. Symmetric about the origin.
  \item Check a table: $x$ and $-x$ give the \emph{same} $y$ (even) or
        \emph{opposite} $y$ (odd). Otherwise: neither.
\end{itemize}

\cshead{Functions \& Inverses}
\begin{itemize}
  \item \textbf{Vertical line test} passes $\Rightarrow$ a function.
  \item \textbf{Horizontal line test} passes $\Rightarrow$ one-to-one
        $\Rightarrow$ invertible.
  \item \textbf{Find $f^{-1}$:} write $y = f(x)$, swap $x$ and $y$, solve for $y$.
        Undo a cube root by cubing.
  \item Domain of $f^{-1}$ = range of $f$;\\ range of $f^{-1}$ = domain of $f$.
\end{itemize}

\cshead{Domain \& Range}
\begin{itemize}
  \item $\sqrt{\ \text{inside}\ }$: set inside $\ge 0$, solve.
        Dividing by a negative \textbf{flips} the inequality.
  \item $\sqrt[3]{\ }$, $|\ |$, lines: domain $(-\infty, \infty)$.
  \item Range: the lowest to highest $y$ on the graph.
  \item $[\ ]$ includes the endpoint; $(\ )$ does not; always $(\ )$ at $\pm\infty$.
\end{itemize}
\end{minipage}%

\cshead{Reading a Graph}
\begin{itemize}
  \item \textbf{Increasing / decreasing:} read left to right; give \emph{$x$-intervals}
        in $(\ )$. Join pieces with $\cup$.
  \item \textbf{Concave up} $\cup$ (holds water); \textbf{concave down} $\cap$.
  \item \textbf{Inflection point:} where concavity changes.
  \item \textbf{Max / min:} $y = a|x - h| + k$ has vertex $(h, k)$: minimum $k$ if
        $a>0$, maximum $k$ if $a<0$. The value is $y$; it occurs at $x$.
\end{itemize}

\cshead{Intercepts}
\begin{itemize}
  \item $x$-int: set $f(x) = 0$. If $|\text{stuff}| = c$ with $c>0$, then
        stuff $= c$ \textbf{or} stuff $= -c$.
  \item $y$-int: find $f(0)$; write $(0,\,f(0))$.
\end{itemize}

\cshead{End Behavior}
Left end: $\lim_{x\to-\infty} f(x)$\par Right end: $\lim_{x\to\infty} f(x)$\par
Each is where $y$ heads: $\infty$ or $-\infty$.

\cshead{Rate of Change \& Secant Line}
\[ \text{ARC} = \frac{f(b) - f(a)}{b - a} \]
Secant line through $(a, f(a))$:\par\centerline{$y - f(a) = m(x - a)$}then solve for $y$.
