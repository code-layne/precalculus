% HOOK
\begin{frame}[t, plain]
  \plumheader{Hook: What Does ``50 mph'' Mean?}
  \sectionlabel[goldacc]{Think About It}

  A car passes \textbf{mile marker 30 at 1:00\,pm} and \textbf{mile marker 180 at 4:00\,pm}.

  \bigskip
  \begin{center}
    {\Large\bfseries\color{plum} \(\dfrac{180 - 30}{4 - 1} = \dfrac{150 \text{ miles}}{3 \text{ hours}} = 50 \text{ mph}\)}
  \end{center}

  \bigskip
  \begin{columns}[T]
    \column{0.48\textwidth}
      \begin{block}{Ask}
        Was the speedometer reading 50 the \textbf{whole} time?
      \end{block}
    \column{0.48\textwidth}
      \begin{block}{Then ask}
        If not, what does the number 50 actually tell you about the trip?
      \end{block}
  \end{columns}

  \bigskip
  \textit{50 is the one steady speed that covers the same distance in the same time.}
\end{frame}

% ROW 1 — AVERAGE RATE OF CHANGE
\begin{frame}[t, plain]
  \plumheader{Average Rate of Change}
  \sectionlabel[goldacc]{Notes row AVERAGE RATE $\cdot$ I Do: problem 1 $\cdot$ We Do: problems 2--4}

  \begin{block}{Change in output over change in input}
    \[ \text{average rate of change on } [a, b] = \frac{f(b) - f(a)}{b - a} \]
    Units: \textbf{output units per input unit}.
  \end{block}

  \begin{columns}[T]
    \column{0.52\textwidth}
      \begin{tabular}{|l|c|c|c|c|c|}
      \hline
      \(t\) (months) & 0 & 2 & 4 & 6 & 8 \\ \hline
      \(B(t)\) (\$)  & 100 & 160 & 260 & 400 & 580 \\ \hline
      \end{tabular}

      \medskip
      On \([0, 4]\): \(\dfrac{260 - 100}{4 - 0} = 40\) \textbf{dollars per month}.

    \column{0.44\textwidth}
      \begin{block}{The trap}
        \(\dfrac{4 - 0}{260 - 100}\) divides by the change in \textbf{output}. Its units
        would be months per dollar --- upside down.
      \end{block}
  \end{columns}
\end{frame}

\begin{frame}[t, plain]
  \plumheader{The Secant Line}
  \sectionlabel[goldacc]{Notes row SECANT LINE $\cdot$ I Do: problem 5 $\cdot$ We Do: problems 6--8}

  \begin{columns}[T]
    \column{0.44\textwidth}
      \begin{center}
      \begin{tikzpicture}[x=0.85cm, y=0.27cm]
        \draw[linegray, very thin, step=1] (-0.5,0) grid (5,11);
        \draw[->, thick] (-0.5,0) -- (5.4,0) node[below right] {\small \(x\)};
        \draw[->, thick] (0,0) -- (0,11.6) node[above] {\small \(f(x)\)};
        \foreach \x in {1,2,3,4} \node[below] at (\x,0) {\scriptsize \x};
        \foreach \y in {2,4,6,8,10} \node[left] at (0,\y) {\scriptsize \y};
        \draw[very thick, plum, domain=-0.45:4.1, samples=60] plot (\x, {\x*\x - 2*\x + 2});
        \foreach \p in {(0,2),(1,1),(2,2),(3,5),(4,10)} \fill[plum] \p circle (2.2pt);
        \draw[very thick, goldacc, dashed] (0.6,-0.2) -- (4.4,11.2);
        \draw[thick, redacc] (1,1) -- (4,1) -- (4,10);
      \end{tikzpicture}
      \end{center}

    \column{0.52\textwidth}
      \begin{block}{Same number, two names}
        The secant line through \((a, f(a))\) and \((b, f(b))\) has slope equal to the
        average rate of change on \([a, b]\).
      \end{block}

      \medskip
      On \([1, 4]\): \quad \(\dfrac{10 - 1}{4 - 1} = \dfrac{9}{3} = 3\)

      \medskip
      Point-slope form: \quad \(y - 1 = 3(x - 1)\)
  \end{columns}
\end{frame}

% ROW 4 — DIFFERENCE QUOTIENT
\begin{frame}[t, plain]
  \plumheader{The Difference Quotient}
  \sectionlabel[redacc]{Notes row DIFFERENCE QUOTIENT $\cdot$ I Do: problem 11 $\cdot$ We Do: 12--14}

  \begin{block}{The average rate on \([x, x + h]\)}
    \[ \frac{f(x + h) - f(x)}{h}, \qquad h \ne 0 \]
    Build \(f(x + h)\) as in 1.3: replace \textbf{every} \(x\) with \((x + h)\).
  \end{block}

  \begin{columns}[T]
    \column{0.48\textwidth}
      \textbf{\(f(x) = 3x - 2\)}
      \begin{align*}
        f(x + h) &= 3x + 3h - 2 \\
        \text{top} &= 3h \\
        \text{quotient} &= 3
      \end{align*}
    \column{0.48\textwidth}
      \textbf{\(f(x) = x^2\)}
      \begin{align*}
        f(x + h) &= x^2 + 2xh + h^2 \\
        \text{top} &= 2xh + h^2 \\
        \text{quotient} &= 2x + h
      \end{align*}
  \end{columns}
\end{frame}

% THE CRUX
\begin{frame}[t, plain]
  \plumheader{The Crux: \(f(x + h)\) Is Not \(f(x) + h\)}
  \sectionlabel[redacc]{Problem 13 $\cdot$ \(f(x) = x^2\), \(x = 3\), \(h = 2\)}

  \begin{columns}[T]
    \column{0.31\textwidth}
      \begin{block}{\(f(x + h)\)}
        \(f(5) = 25\)
      \end{block}
    \column{0.31\textwidth}
      \begin{block}{\(f(x) + h\)}
        \(9 + 2 = 11\)
      \end{block}
    \column{0.31\textwidth}
      \begin{block}{\(x^2 + h^2\)}
        \(9 + 4 = 13\)
      \end{block}
  \end{columns}

  \bigskip
  Only one of these is \(f(5)\). The input is \(x + h\), so the \textbf{whole} \((x + h)\) is
  squared: \((x + h)^2 = x^2 + 2xh + h^2\).

  \bigskip
  \textbf{Check (problem 14):} \(2x + h = 2(3) + 2 = 8\), and
  \(\dfrac{25 - 9}{5 - 3} = 8\). The quotient \emph{is} the average rate.
\end{frame}
