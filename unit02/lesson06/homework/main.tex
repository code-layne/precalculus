\begin{headlinebox}{lilac}
\textbf{\color{plum}Part C --- The difference quotient}
\end{headlinebox}

\begin{enumerate}[leftmargin=1.9em, itemsep=9pt, start=6]

\item Let $f(x) = x^2 + 5$.

      \begin{minipage}[t]{0.48\linewidth}
      \textbf{(a)} Write $f(x + h)$, expanded.

\begin{work}
  f(x + h) &= (x + h)^2 + 5 \\
           &= x^2 + 2xh + h^2 + 5
\end{work}

      $f(x + h) = $ \blank{3.6cm}
      \end{minipage}\hfill
      \begin{minipage}[t]{0.48\linewidth}
      \textbf{(b)} Find and simplify the difference quotient.

\begin{work}
  \frac{f(x + h) - f(x)}{h} &= \frac{2xh + h^2}{h} \\
                            &= 2x + h
\end{work}

      Quotient: \blank{2.4cm}
      \end{minipage}

\item \begin{minipage}[t]{0.48\linewidth}
      Let $g(x) = 4x - 7$. Find and simplify the difference quotient of $g$.

\begin{work}
  g(x + h) &= 4x + 4h - 7 \\
  \frac{g(x + h) - g(x)}{h} &= \frac{4h}{h} = 4
\end{work}

      Quotient: \blank{2.0cm}
      \end{minipage}\hfill
      \begin{minipage}[t]{0.48\linewidth}
      Why does it make sense that no $h$ is left in problem 7's answer?
      \writespace{1.7cm}{}
      \end{minipage}

\item Let $f(x) = x^2 + 3x$.

      \begin{minipage}[t]{0.52\linewidth}
      \textbf{(a)} Find and simplify the difference quotient.

\begin{work}
  f(x + h) &= x^2 + 2xh + h^2 + 3x + 3h \\
  \frac{f(x + h) - f(x)}{h} &= 2x + h + 3
\end{work}
      \end{minipage}\hfill
      \begin{minipage}[t]{0.44\linewidth}
      \textbf{(b)} Evaluate (a) at $x = 1$, $h = 2$; check against the average rate of change on $[1, 3]$.

\begin{work}
  2(1) + 2 + 3 &= 7 \\
  \frac{f(3) - f(1)}{3 - 1} &= \frac{18 - 4}{2} = 7
\end{work}
      \end{minipage}

      Quotient: \blank{2.6cm} \qquad Value at $x = 1$, $h = 2$: \blank{1.4cm}

\end{enumerate}
