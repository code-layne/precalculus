\begin{enumerate}[itemsep=10pt]

\item Find the slope of the line through $(2, 5)$ and $(6, 17)$.

\begin{work}
  m &= \frac{17 - 5}{6 - 2} \\
    &= \frac{12}{4} = 3
\end{work}

Slope: \ans{$3$}

\item Write the equation of the line with slope $-2$ through the point $(3, 4)$ in
point-slope form, $y - y_1 = m(x - x_1)$.

\vspace{0.1in}
Equation: \ans{$y - 4 = -2(x - 3)$}

\item Expand and simplify $(x + 4)^2$. \textit{Hint: write it as $(x + 4)(x + 4)$ first.}

\begin{work}
  (x + 4)^2 &= (x + 4)(x + 4) \\
            &= x^2 + 4x + 4x + 16 \\
            &= x^2 + 8x + 16
\end{work}

$(x + 4)^2 = $ \ans{$x^2 + 8x + 16$} \ \ans{(not $x^2 + 16$)}

\item Let $f(x) = 2x + 5$ and $g(x) = x - 1$. Build $f(g(x))$ by replacing \emph{every} $x$ in
$f$ with $g(x)$, then simplify.

\begin{work}
  f(g(x)) &= 2(x - 1) + 5 \\
          &= 2x - 2 + 5 \\
          &= 2x + 3
\end{work}

$f(g(x)) = $ \ans{$2x + 3$}

\item Simplify, assuming $h \ne 0$: \quad \TallMath{\frac{6h + h^2}{h}}

\begin{work}
  \frac{6h + h^2}{h} &= \frac{h(6 + h)}{h} \\
                     &= 6 + h
\end{work}

Simplified: \ans{$6 + h$}

\end{enumerate}
