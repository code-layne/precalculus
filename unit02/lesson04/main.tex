\begin{skillbox}[Priority Ideas \& Skills]{goldbox}
\begin{tabularx}{\linewidth}{@{}X X@{}}
\textbf{Priority Skills} &
\textbf{Key Understandings} \\
\midrule
\textbf{AP Skill 2.A} --- Identify information from graphical, numerical, analytical, and
verbal representations: read domain, range, intercepts, and end behavior off a pre-drawn graph. &
\begin{itemize}[leftmargin=1.2em,itemsep=2pt,topsep=0pt]
  \item Domain is read left to right (inputs); range is read bottom to top (outputs). A filled
        dot is included, an open circle is not, an arrow keeps going (1.1.A.1).
  \item Interval notation: $[\ ]$ includes, $(\ )$ excludes, $\infty$ always gets $(\ )$;
        $\cup$ joins two pieces.
\end{itemize} \\
\textbf{AP Skill 1.A} --- Solve equations and inequalities: radicand $\ge 0$, denominator
$\ne 0$, and $f(x) = 0$. &
\begin{itemize}[leftmargin=1.2em,itemsep=2pt,topsep=0pt]
  \item Every real number is in the domain unless a denominator would be 0 or an even root
        would take a negative. Both tests can apply to one formula.
  \item The $y$-intercept is $(0, f(0))$ --- input 0. A zero $a$ solves $f(a) = 0$ and gives
        the $x$-intercept $(a, 0)$ --- output 0 (1.5.A.3).
\end{itemize} \\
\textbf{AP Skill 3.A} --- Describe the characteristics of a function: end behavior in limit
notation. &
\begin{itemize}[leftmargin=1.2em,itemsep=2pt,topsep=0pt]
  \item As the inputs increase or decrease without bound, a nonconstant polynomial's outputs
        increase or decrease without bound; the leading term's degree and sign decide which
        (1.6.A.1--1.6.A.3).
  \item $\lim_{x \to \infty} r(x) = b$ means the outputs get and stay close to $b$: a
        horizontal asymptote $y = b$ (1.7.A.6).
  \item \textbf{Target misconception:} mixing up input and output --- writing a range with
        $x$-values (the vertical asymptote's $x = 1$ instead of the horizontal asymptote's
        $y = 2$), or reading $\lim_{x \to \infty} f(x) = 2$ as ``$x$ goes to 2.'' The $x \to$
        under ``lim'' is the input's trip; the value after the equals sign is the output's
        destination.
\end{itemize} \\
\end{tabularx}
\end{skillbox}

\begin{skillbox}[{Vocabulary, Concepts, \& Theorems}]{greenbox}
\renewcommand{\arraystretch}{1.4}
\begin{tabularx}{\linewidth}{@{} >{\leavevmode\bfseries\color{plum}}p{0.24\linewidth} X @{}}
Interval notation & A set of numbers written by its endpoints: $[\ ]$ includes an endpoint, $(\ )$ leaves it out, $\infty$ always gets $(\ )$; $\cup$ joins two intervals. \\
Radicand          & The expression under a root sign; under an even root it must be $\ge 0$. \\
$y$-intercept     & The point where the graph crosses the $y$-axis: $(0, f(0))$. A function has at most one. \\
Zero of a function & An input $a$ with $f(a) = 0$; the graph has an $x$-intercept at $(a, 0)$. \\
End behavior      & What the outputs do as the inputs increase without bound ($x \to \infty$) or decrease without bound ($x \to -\infty$). \\
Limit notation    & \TallMath{\lim_{x \to \infty} f(x) = L}: as $x$ increases without bound, $f(x)$ approaches $L$ --- where $L$ is $\infty$, $-\infty$, or a number. \\
Horizontal asymptote & A line $y = b$ the graph levels off toward: $\lim_{x \to \infty} f(x) = b$ or $\lim_{x \to -\infty} f(x) = b$. \\
Leading term      & The term of a polynomial with the highest power; it decides the end behavior. \\
\end{tabularx}
\end{skillbox}

\begin{skillbox}[Hook (2 min)]{lilac}
    \textbf{The coffee on the desk.} A cup of coffee at $180^\circ$F is set down in a
    $70^\circ$F room; $T(t)$ is its temperature $t$ minutes later.
    \begin{enumerate}[itemsep=2pt]
        \item Ask: if you leave it on the desk all day, about what temperature will it read?
              Take two or three guesses; land on ``about 70, room temperature.''
        \item Follow up: which quantity keeps growing without stopping, and which one settles
              down? Make a student say both halves --- \textit{time} keeps growing; the
              \textit{temperature} settles near 70.
    \end{enumerate}
    \textit{Discussion prompt:} the input runs off forever while the output settles. By the end
    of the period that sentence is written $\lim_{t \to \infty} T(t) = 70$ --- and the 70 is a
    temperature, not a time. Leave the symbol unwritten for now; it arrives in Part 4.
\end{skillbox}

\begin{skillbox}[Lesson --- I Do / We Do / You Do]{lilac}
\begin{multicols}{2}

\textbf{I DO --- Part 1: Reading the Edges of a Graph} (10 min)

\textit{Notes rows DOMAIN \& RANGE and DOMAIN (from a formula): read each definition, mark up
the display, work problems 1 and 5. Teacher works; students watch and annotate. Nothing is
cold-called in this phase.}
\begin{itemize}
  \item Read the DOMAIN \& RANGE sentences aloud. Trace the graph of $f$ with a finger left to
        right for the domain, then bottom to top for the range --- two different directions
        for two different questions.
  \item Fill the three label boxes yourself: filled dot \textit{included}, open circle
        \textit{left out}, arrow \textit{keeps going}. Work \textbf{problem 1}, $[-4, 4)$,
        saying which bracket each end gets and why.
  \item Read the DOMAIN row and its three steps. Mark up the number line for
        $k(x) = \sqrt{x - 3}/(x - 5)$: the filled dot at 3 is the radicand's test, the open
        circle at 5 is the denominator's. Write $[\,3, 5) \cup (5, \infty)$ in the label box
        and name $\cup$ out loud.
  \item Work \textbf{problem 5}, $\sqrt{x - 2}$: set the radicand $\ge 0$ and solve.
\end{itemize}

\textbf{WE DO --- Part 2: Graph and Formula Practice} (8 min)

\textit{Problems 2--4 and 6--8. Class fills the blanks together; cold-call every one.}
\begin{itemize}
  \item \textbf{Problem 2}, the range of $f$: the lowest point $(-1, -2)$ is on the graph; 3 is
        reached only at the open circle, so $[-2, 3)$.
  \item \textbf{Problem 3} reads the arrow on $g$; \textbf{problem 4} drills the notation ---
        all real numbers is $(-\infty, \infty)$.
  \item \textbf{Problem 7}, $\sqrt{6 - 2x}$, flips the inequality on division by $-2$; make the
        error visible once, on purpose. \textbf{Problem 8} runs both tests at once.
\end{itemize}

\columnbreak

\textbf{WE DO --- Part 3: Intercepts} (5 min)

\textit{Notes row INTERCEPTS. Class fills the blanks together; cold-call every one. Teacher
works problem 9; the class works 10.}
\begin{itemize}
  \item Read the sentences, then fill the two label boxes on the parabola: at an $x$-intercept
        the \textit{output} is 0; at the $y$-intercept the \textit{input} is 0. This is the
        lesson's distinction in its simplest form --- name it.
  \item Work \textbf{problem 9} aloud and check each point against the picture.
        \textbf{Problem 10}, $x^2 - 16$, uses square roots and must produce two points.
\end{itemize}

\textbf{WE DO --- Part 4: End Behavior} (9 min)

\textit{Notes row END BEHAVIOR --- the spine of the lesson. Class fills the blanks together;
cold-call every one. Teacher works problem 11; the class works 12--16.}
\begin{itemize}
  \item Read the definition and write the hook's sentence on the board in symbols:
        $\lim_{t \to \infty} T(t) = 70$. Underline the $t \to \infty$ in one colour
        (\textit{input}) and the 70 in another (\textit{output}).
  \item Graph C: fill both label boxes with 2 and say the sentence both ways.
        Work \textbf{problem 11} on graph A. \textbf{Problem 12} is graph B.
  \item Read the leading-term sentence and table. \textbf{Problem 13} is direct;
        \textbf{problem 14}, $4x^2 - 2x^3$, is the order trap --- the leading term is not the
        one written first.
  \item Read the boxed rule aloud. \textbf{Problems 15 and 16 are the crux}: $x$ is going to
        $\infty$, the outputs to 2, and $r(2) = 3$; the domain is missing the input 1 and the
        range the output 2. Ask for the \textit{reason}, not the number.
\end{itemize}

\textbf{YOU DO --- Part 5: All Four} (16 min)

\textit{Notes row ALL FOUR, problems 17--19 --- the whole independent phase. Students work
alone; circulate and say nothing a neighbor could say instead.}
\begin{itemize}
  \item \textbf{Problem 17} reads graph D --- arrows on both ends, so the domain is
        $(-\infty, \infty)$ and the range stops at the peak, $(-\infty, 3\,]$.
  \item \textbf{Problem 18} is the algebra: a denominator for the domain, a numerator for the
        zeros, $g(0)$ for the $y$-intercept.
  \item \textbf{Problem 19} is the finisher's question and the crux transferred: graph E with
        no formula, and a classmate's range that is really a domain fact.
\end{itemize}
\end{multicols}
\end{skillbox}

\begin{skillbox}[Active Monitoring]{redbox}
Circulate during the \textbf{You Do} phase --- the sixteen minutes where students are working
problems 17--19 without you --- and check. The crux is \textbf{problems 15 and 16} (the limit 2
is an output; $x = 1$ is the input the domain skips); its You Do transfer is \textbf{problem
19} (graph E, asymptotes $x = -2$ and $y = -1$, no formula).
\begin{itemize}
  \item \textbf{Watch for (the target misconception):} input and output mixed up --- a range
        written with the vertical asymptote's $x$-value, or a limit read as ``$x$ goes to
        $-1$.'' Cold-call: ``Point to the $x$ in that statement. What is it doing? Now point to
        the output.''
  \item \textbf{Common error:} the range of $d$ in problem 17 written as $(-\infty, \infty)$
        because both arrows point away. Ask: ``Bottom to top --- does the graph ever get above
        3?''
  \item \textbf{Common error:} brackets on infinity, $[-2, \infty]$, or a round parenthesis on a
        filled dot. Ask: ``Does the graph include that endpoint?''
  \item \textbf{Common error:} in problem 18, setting the denominator equal to 0 and calling
        $x = 1$ a zero. Ask: ``A zero makes which part equal 0 --- the input or the output?''
  \item \textbf{Common error:} the $y$-intercept written as a single number or as $(4, 0)$.
        Ask: ``At the $y$-intercept, which coordinate is 0?''
\end{itemize}
\end{skillbox}

\begin{skillbox}[Differentiation --- During You Do (16 min)]{redbox}
There is no separate activity sheet: students work the notes' You Do row (problems 17--19)
alone while you circulate. Differentiate by \textit{where you stand}, not by handing out
different paper.

\begin{itemize}
  \item \textbf{Support.} Sit with a struggling student on \textbf{problem 17} and have them
        trace graph D with a finger: left to right for the domain, bottom to top for the range,
        saying ``input'' or ``output'' each time. Then point at each axis crossing and ask which
        coordinate is 0 there.
  \item \textbf{On level.} \textbf{Problem 18} is the algebra the class should clear unaided.
        If a student stalls, ask only which piece of the formula each question uses --- the
        denominator, the numerator, or $g(0)$ --- and walk away.
  \item \textbf{Extend.} \textbf{Problem 19} is the finisher's question. Push a student who
        finishes it: ``Write the coffee's end behavior in symbols, and say which number in your
        statement is a temperature.'' A student who can explain part (c) presents it in the
        Debrief.
\end{itemize}
\end{skillbox}

\boxguard[20]
\begin{skillbox}[Debrief (5 min)]{redbox}
Whole class, teacher-led, packets still open. Three moves, in order:
\begin{enumerate}[itemsep=3pt]
  \item \textbf{Share out the You Do} (2 min) --- one answer per problem, not a full review.
        Problem 17: the range $(-\infty, 3\,]$ and the three intercepts. Problem 18: $x \ne 1$,
        zeros $\pm 2$, $(0, 4)$. Problem 19: both limits $-1$, and part (c) --- $-2$ is the input
        the graph skips. If a student wrote 19(c) cleanly, they present it; it is the only
        time the rest of the class hears the distinction put into words.
  \item \textbf{Name the headline} (1 min) --- say it, then make the class say it back:
        \textit{every characteristic is a question about inputs or about outputs. Under
        ``lim,'' $x \to$ is the input's trip; the number after the equals sign is where the
        output goes.} Point at graph C's two dashed lines while you say it.
  \item \textbf{Point forward} (2 min) --- the DeltaMath assignment \textit{Lesson 2.4} and its
        due date: write the date on the board and have students copy it into the cover's
        Homework row (row 4). Then: today was the edges of a graph; 2.5 reads its middle ---
        where it rises and falls, its highest and lowest points, and which way it bends.
\end{enumerate}
\textbf{Do not} re-teach interval notation here or take new questions. There is no exit
ticket: the You Do was the period's formative read. If circulating showed the
brackets did not land, that is the 2.5 warm-up's job, not this minute's.
\end{skillbox}

\textbf{AP Practice --- extra credit (2 pages).} Optional, last in the packet.
Section I is five AP-style multiple-choice items whose distractors are the lesson's errors: a
domain that drops the denominator, the radicand, or the radicand's bracket (1); the vertical

first taken as the leading term (3); the $y$-intercept or swapped coordinates given as zeros
(4); domain and range swapped, or an open circle ignored (5). Section II is one free-response
question on an online game's player count: the $y$-intercept in context (a), domain and range
(b), the limit interpreted (c), a reporter's input/output error (d), and whether the asymptote's
value is ever reached (e) --- the finisher's question.

\begin{teachernote}[Warm-Up]
Five minutes, silent start, no calculators. Each item is the algebra one notes row runs on:
items 1--2 are the zeros of row 3, item 3 is the radicand test of row 2, item 4 is the
interval notation every row uses, and item 5 is the first look at end behavior. Item 4 is the
one to read over shoulders: 1.1 met interval notation only lightly, and a student who writes
$[4, \infty]$ will carry the bracket on infinity into every answer today --- flag them before
the notes start. Item 5 matters more than it looks: a student who writes $f(-10) = 100$ has
read $-x^2$ as $(-x)^2$, and that same student will read the end
behavior of $-x^2$ wrongly at the left end.
\end{teachernote}

\begin{teachernote}[Guided Notes]
The notes are one Main Ideas / Notes table, four instruction rows and a You Do row; in each
instruction row you read the printed definition, mark up the display, and work the first
problem of the grid (1, 5, 9, 11), and the class works the rest with the pen in their hand,
every problem cold-called. The two domain rows are fast on purpose --- 1.1 introduced domain and
range from a graph, and the warm-up rehearsed the inequality --- so spend the saved minutes in
END BEHAVIOR. Insist on interval notation from the first problem; inequalities are no longer the
answer form.

INTERCEPTS is the lesson's distinction in miniature: the two label boxes (\textit{output} is 0
at an $x$-intercept, \textit{input} is 0 at the $y$-intercept) set up exactly the language the
crux needs. END BEHAVIOR is the spine. Write the hook's coffee statement in symbols before
problem 11 so the first limit students see has a temperature in it, then colour-code the input
and the output every time. Problems 15 and 16 are built so the two readings disagree: a
student who reads ``$x$ goes to 2'' finds $r(2) = 3$ on the same graph, and the domain and range
are missing different numbers. Do not accept ``because the line is there''; make them say
\textit{input} or \textit{output}.

The You Do row (17--19) is the entire independent phase and carries sixteen minutes. Problem
19 is the crux with the formula taken away. If the clock slips, cut problem 4 and let the
DeltaMath topic \textit{Interval Notation} carry the notation drill.
\end{teachernote}

\begin{teachernote}[Debrief]
Five minutes, and on this lesson they are at risk --- four characteristics in one guided phase
will try to borrow them. Borrow from problem 4 or problem 10 instead; both have a twin among
the DeltaMath topics. Never borrow from the You Do: with no exit ticket, circulating it is the
last look you get before the work leaves the room. The share-out is one answer per You Do
problem so that a student who stalled on 17 still hears 19(c) --- the input/output sentence
that no DeltaMath topic asks for, and that they will otherwise never hear said aloud.
\end{teachernote}

\begin{teachernote}[AP Practice]
Extra credit, optional, and not a substitute for the DeltaMath assignment --- say so when you
hand out the packet. Score it however extra credit works in your gradebook; the cover gives it a $+$ cell
outside the total. Multiple-choice answers are C, A, B, A, A. Item 2 is the most useful one to
glance at: a student who circles (B) reported the vertical asymptote's $x = 3$ as the limit,
and a student who circles (C) read the $-2$ as an input --- those are the two halves of the
lesson's misconception, and either one means problems 15--16 of the notes did not land. Item 3's
(A) is the order trap from problem 14. In the free response, accept $[0, \infty)$ with or without
a stated upper limit on the game's life in (b); part (e) is the deepest question on the page ---
credit any answer that reaches $8/(t + 1) = 0$ and says it has no solution, so 9 is not in the
range.
\end{teachernote}

\begin{teachernote}[Homework]
DeltaMath, graded, assigned as \textit{Lesson 2.4}, due on the date you set. The diagnostic is
\textbf{End Behavior from a Graph (Limit Notation)}: a student who misses it with the vertical
asymptote's $x$-value in the answer box, or with $\pm\infty$ where the graph levels off, is
reading the input where the output belongs --- the lesson's misconception, not a careless
slip. Check it student by student before 2.5. DeltaMath cannot test whether a student can
\textit{say} which part of a limit statement is the input and which the output; You Do problem
19(c) carries that, so read 19(c) as you circulate and note who could not answer it. On
\textbf{Domain of a Function Algebraically}, a wrong answer with the right endpoints is the
bracket error, not the domain error --- see \textbf{Interval Notation} for the same student. Two
topics predict 2.5, where every answer is again an interval of inputs: \textbf{Domain and Range
from a Graph} and the diagnostic. If either is weak across the class, open the 2.5 warm-up with a
graph-C item asking for the range and $\lim_{x \to \infty} r(x)$ side by side; if
\textbf{End Behavior of Polynomials} shows the order trap ($4x^2 - 2x^3$), reuse notes problem 14
there too.
\end{teachernote}
