\begin{notesbox}{LO 2.4.A --- Product Property (EK 2.4.A.1)}
\textbf{Product Property:} \quad $b^m \cdot b^n = $ \blank{3cm}

\medskip
\textbf{Graphical connection:} A horizontal translation $f(x) = b^{x+k}$ is equivalent to a
vertical dilation $f(x) = \blank{1.5cm} \cdot b^x$, where $a = $ \blank{2cm}.

\medskip
\textbf{Examples --- rewrite in the form $ab^x$:}

\renewcommand{\arraystretch}{2.2}
\begin{tabular}{@{} l @{\quad$=$\quad} l @{\quad$=$\quad} l @{}}
$5^{x+2}$ & $5^x \cdot 5^{\blank{0.6cm}}$ & \blank{2cm} $\cdot\, 5^x$ \\
$3^{x-1}$ & $3^x \cdot 3^{\blank{0.6cm}}$ & \blank{2cm} $\cdot\, 3^x$ \\
$4 \cdot 2^x$ & $2^{\blank{0.6cm}} \cdot 2^x$ & $2^{x + \blank{0.6cm}}$ \\
\end{tabular}
\end{notesbox}

\vspace{0.06in}

\begin{notesbox}{LO 2.4.A --- Power Property (EK 2.4.A.2)}
\textbf{Power Property:} \quad $(b^m)^n = $ \blank{3cm}

\medskip
\textbf{Graphical connection:} A horizontal dilation $f(x) = b^{cx}$ is equivalent to a base
change $f(x) = (\blank{2cm})^x$, where the new base is $b^c$.

\medskip
\textbf{Examples --- rewrite as a single base to the $x$ power:}

\renewcommand{\arraystretch}{2.2}
\begin{tabular}{@{} l @{\quad$=$\quad} l @{\quad$=$\quad} l @{}}
$2^{3x}$   & $(2^{\blank{0.6cm}})^x$ & $\blank{2cm}^x$ \\
$9^{x/2}$  & $(9^{\blank{0.6cm}})^x$ & $\blank{2cm}^x$ \\
$4^{x/2}$  & $(4^{\blank{0.6cm}})^x$ & $\blank{2cm}^x$ \\
\end{tabular}
\end{notesbox}

\vspace{0.06in}

\begin{notesbox}{LO 2.4.A --- Negative \& Fractional Exponents (EK 2.4.A.3, A.4)}
\renewcommand{\arraystretch}{1.5}
\begin{tabular}{@{} >{\bfseries}p{0.38\linewidth} l @{}}
Negative exponent rule:     & $b^{-n} = \dfrac{1}{\blank{2cm}}$ \\
Fractional exponent rule:   & $b^{1/k} = \blank{2.5cm}$ \\
General fractional exponent: & $b^{m/n} = (\sqrt[n]{b})^{\blank{0.8cm}}$ \\
\end{tabular}

\medskip
\textbf{Decay connection:} $f(x) = 2^{-x} = \left(\dfrac{\blank{1cm}}{\blank{1cm}}\right)^x$

\medskip
\textbf{Evaluate:}
\quad $2^{-3} = $ \blank{2cm}
\quad $8^{1/3} = $ \blank{2cm}
\quad $4^{3/2} = $ \blank{2cm}
\end{notesbox}
