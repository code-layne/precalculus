% SECTION 1 — COMPLEX NUMBER REFRESHER
\begin{frame}[t, plain]
  \plumheader{1. A Complex-Number Refresher}

  \begin{block}{From Algebra 2}
    \(i^2 = -1\). A \textbf{complex number} is \(a + bi\) with \(a\) and \(b\)
    real --- real part \(a\), imaginary part \(b\).
    Every real number is complex (\(b = 0\)); \textbf{non-real} means
    \(b \ne 0\).
  \end{block}

  \medskip
  \begin{columns}[T]
    \column{0.48\textwidth}
      \begin{block}{Is \(i\) a zero of \(p(x) = x^2 + 1\)?}
        \[ p(i) = i^2 + 1 = -1 + 1 = 0 \]
        Yes --- same definition of ``zero'' as always.
      \end{block}

    \column{0.48\textwidth}
      \begin{block}{Conjugate pairs}
        \[ (3+2i)(3-2i) = 9 + 4 = 13 \]
        The \(i\) terms cancel. The product of a conjugate pair is
        \textbf{real}.
      \end{block}
  \end{columns}

  \bigskip
  The conjugate of \(a + bi\) is \(a - bi\): only the sign in front of the
  \(i\) changes.
\end{frame}

% SECTION 2 — FUNDAMENTAL THEOREM OF ALGEBRA
\begin{frame}[t, plain]
  \plumheader{2. The Fundamental Theorem of Algebra}

  \begin{block}{The theorem}
    A polynomial function of degree \(n\) has \textbf{exactly \(n\)} complex
    zeros, counting multiplicities.
  \end{block}

  \bigskip
  \begin{itemize}\setlength{\itemsep}{8pt}
    \item ``Exactly,'' not ``at most'' --- once non-real inputs are allowed,
          the count is guaranteed.
    \item \textbf{Real} zeros are the \(x\)-intercepts you can see.
    \item \textbf{Non-real} zeros are not on the graph at all. No graph can
          ever show them.
  \end{itemize}

  \bigskip
  \begin{block}{The hook, resolved}
    \(p(x) = (x-2)(x+2)(x^2+1)\), and \(x^2 + 1 = 0\) gives \(x = \pm i\).
    All four zeros: \(2,\ -2,\ i,\ -i\). The graph could only show two.
  \end{block}
\end{frame}

% SECTION 2, CONT. — ZERO CENSUS
\begin{frame}[t, plain]
  \plumheader{The Zero Census}

  \begin{block}{The move to memorize}
    degree \(-\) (real zeros, counted with multiplicity) \(=\) number of
    non-real zeros
  \end{block}

  \bigskip
  \begin{center}
  \begin{tabular}{|l|c|c|c|c|}
  \hline
  \textbf{Factored form} & \textbf{Degree} & \textbf{\# real} & \textbf{\# non-real} & \textbf{Total} \\
  \hline
  \((x-2)(x+2)(x^2+1)\) & 4 & 2 & 2 & 4 \\
  \hline
  \((x+1)^3(x^2+4)\) & 5 & 3 & 2 & 5 \\
  \hline
  \(x(x^2+9)\) & 3 & 1 & 2 & 3 \\
  \hline
  \((x-1)^2(x^2+1)^2\) & 6 & 2 & 4 & 6 \\
  \hline
  \end{tabular}
  \end{center}

  \bigskip
  \textit{Multiplicity counts. \((x+1)^3\) contributes \textbf{three} real
  zeros, not one.}
\end{frame}

% SECTION 3 — CONJUGATE PAIRS
\begin{frame}[t, plain]
  \plumheader{3. Non-Real Zeros Come in Pairs}

  \begin{block}{The conjugate pair rule}
    If \(a + bi\) is a zero of a polynomial with \textbf{real coefficients},
    then \(a - bi\) is a zero too.
  \end{block}

  \bigskip
  \begin{itemize}\setlength{\itemsep}{10pt}
    \item The number of non-real zeros is always \textbf{even} --- they cannot
          arrive one at a time.
    \item An \textbf{odd-degree} polynomial has an odd number of zeros, so
          they cannot all be non-real: it must have at least one
          \textbf{real} zero.
    \item Consequence: the graph of an odd-degree polynomial \textit{has} to
          cross the \(x\)-axis.
  \end{itemize}

  \bigskip
  \textit{Why? Because only the conjugate partner cancels the \(i\) terms and
  leaves real coefficients behind.}
\end{frame}

% SECTION 3, CONT. — BUILDING FROM A ZERO LIST
\begin{frame}[t, plain]
  \plumheader{Building a Polynomial from Its Zeros}

  \textbf{Given:} degree 3, real coefficients, leading coefficient 1, zeros
  \(x = 2\) and \(x = 1 + 3i\).

  \bigskip
  \begin{block}{Step 1 --- the pairing rule hands you the third zero}
    \(1 + 3i\) is a zero, so \(1 - 3i\) is a zero. That is all three.
  \end{block}

  \medskip
  \begin{block}{Step 2 --- multiply the conjugate factors first}
    \[
      \big((x-1) - 3i\big)\big((x-1) + 3i\big)
      = (x-1)^2 - 9i^2
      = x^2 - 2x + 10
    \]
  \end{block}

  \medskip
  \begin{block}{Step 3 --- finish}
    \[ p(x) = (x-2)(x^2 - 2x + 10) = x^3 - 4x^2 + 14x - 20 \]
    Every coefficient came out \textbf{real} --- exactly as promised.
  \end{block}
\end{frame}

% SECTION 4 — EVEN AND ODD
\begin{frame}[t, plain]
  \plumheader{4. Even and Odd Polynomial Functions}

  \begin{columns}[T]
    \column{0.48\textwidth}
      \begin{block}{Even}
        \(f(-x) = f(x)\) for every \(x\).\\[4pt]
        Symmetric over the line \(x = 0\) --- a mirror across the \(y\)-axis.
      \end{block}

    \column{0.48\textwidth}
      \begin{block}{Odd}
        \(f(-x) = -f(x)\) for every \(x\).\\[4pt]
        Symmetric about the point \((0,0)\) --- a half-turn about the origin.
      \end{block}
  \end{columns}

  \bigskip
  \begin{block}{Verify by substitution: \(g(x) = x^3 - 4x\)}
    \[ g(-x) = (-x)^3 - 4(-x) = -x^3 + 4x = -(x^3 - 4x) = -g(x) \]
    So \(g\) is \textbf{odd}.
  \end{block}

  \bigskip
  \textit{``Odd'' is not the same as ``not even.'' Most functions are
  \textbf{neither}.}
\end{frame}

% SECTION 4, CONT. — THE EXPONENT SHORTCUT AND THE GRAPHS
\begin{frame}[t, plain]
  \plumheader{The Exponent Shortcut}

  \begin{columns}[T]
    \column{0.46\textwidth}
      \begin{itemize}\setlength{\itemsep}{8pt}
        \item All exponents \textbf{even} \(\Rightarrow\) even function.
        \item All exponents \textbf{odd} \(\Rightarrow\) odd function.
        \item A \textbf{mix} \(\Rightarrow\) neither.
      \end{itemize}

      \medskip
      \textbf{The trap:} a constant term is \(x^0\), an \textbf{even} power.
      So \(x^3 + 5\) is \textbf{neither}.

      \medskip
      \(x^4 - 3x^2 - 4\) from the hook? All even --- \textbf{even}.

  \sectionlabel[redacc]{Try It}

  \begin{enumerate}\setlength{\itemsep}{10pt}
    \item Let \(p(x) = (x-6)(x^2 + 2x + 5)\). Give the degree and list all

    \item A polynomial with real coefficients has degree 6 and exactly two

    \item Classify each as even, odd, or neither:
          \(x^4 - 7x^2 + 1\), \quad \(x^5 - x\), \quad \(x^4 + x\).
    \item Explain in one sentence why a degree-7 polynomial must cross the
          \(x\)-axis.
  \end{enumerate}
