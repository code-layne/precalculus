% ROW 2 — DOMAIN FROM A FORMULA
\begin{frame}[t, plain]
  \plumheader{Domain --- From a Formula}
  \sectionlabel[goldacc]{Notes row DOMAIN $\cdot$ I Do: problem 5 $\cdot$ We Do: problems 6--8}

  \begin{columns}[T]
    \column{0.48\textwidth}
      \begin{block}{Two forbidden moves}
        \textbf{1.} Denominator \(= 0\)? Cross those inputs out.\\[3pt]
        \textbf{2.} Even root of a negative? Set the radicand \(\ge 0\).\\[3pt]
        \textbf{3.} Keep the inputs that pass \textit{both}.
      \end{block}

    \column{0.48\textwidth}
      \[ k(x) = \frac{\sqrt{x - 3}}{x - 5} \]
      \begin{itemize}\setlength{\itemsep}{4pt}
        \item radicand: \(x - 3 \ge 0\), so \(x \ge 3\)
        \item denominator: \(x - 5 \ne 0\), so \(x \ne 5\)
      \end{itemize}
  \end{columns}

  \medskip
  \begin{center}
  \begin{tikzpicture}[x=0.9cm]
    \draw[<->, thick] (-0.3,0) -- (9.3,0);
    \foreach \x in {0,...,9} {\draw (\x,0.1) -- (\x,-0.1); \node[below, font=\scriptsize] at (\x,-0.1) {\x};}
    \draw[line width=3pt, plum] (3,0) -- (4.88,0);
    \draw[line width=3pt, plum, ->] (5.12,0) -- (9.25,0);
    \fill[plum] (3,0) circle (3.5pt);
    \draw[very thick, plum, fill=white] (5,0) circle (3.5pt);
  \end{tikzpicture}

  \smallskip
  Domain: \([\,3, 5) \cup (5, \infty)\) \quad --- \(\cup\) joins two intervals
  \end{center}
\end{frame}

% ROW 3 — INTERCEPTS
\begin{frame}[t, plain]
  \plumheader{Intercepts --- Input 0 or Output 0}
  \sectionlabel[goldacc]{Notes row INTERCEPTS $\cdot$ I Do: problem 9 $\cdot$ We Do: problem 10}

  \begin{columns}[T]
    \column{0.42\textwidth}
      \begin{center}
      \begin{tikzpicture}[x=0.55cm, y=0.3cm]
        \draw[linegray, very thin, step=1] (-3,-5) grid (5,6);
        \draw[->, thick] (-3,0) -- (5.5,0) node[right] {\small \(x\)};
        \draw[->, thick] (0,-5) -- (0,6.5);
        \draw[very thick, plum, <->, samples=60, domain=-1.85:3.85] plot (\x, {\x*\x-2*\x-3});
        \fill[goldacc] (-1,0) circle (3pt);
        \fill[goldacc] (3,0) circle (3pt);
        \fill[redacc] (0,-3) circle (3pt);
      \end{tikzpicture}

      \(f(x) = x^2 - 2x - 3\)
      \end{center}

    \column{0.54\textwidth}
      \begin{block}{\(y\)-intercept: the INPUT is 0}
        Compute \(f(0) = -3\). \quad Point \((0, -3)\).
      \end{block}
      \smallskip
      \begin{block}{\(x\)-intercepts (zeros): the OUTPUT is 0}
        Solve \(x^2 - 2x - 3 = 0\):\\
        \((x - 3)(x + 1) = 0\), so \(x = 3\) or \(x = -1\).\\
        Points \((3, 0)\) and \((-1, 0)\).
      \end{block}
  \end{columns}
\end{frame}

% ROW 4 — END BEHAVIOR
\begin{frame}[t, plain]
  \plumheader{End Behavior --- Off the Edges}
  \sectionlabel[redacc]{Notes row END BEHAVIOR $\cdot$ I Do: problem 11 $\cdot$ We Do: 12--14}

  \begin{block}{Limit notation}
    \(\displaystyle\lim_{x \to \infty} f(x) = \infty\) \quad reads \quad ``as \(x\) increases
    without bound, \(f(x)\) increases without bound.''
  \end{block}

  \medskip
  \begin{columns}[T]
    \column{0.48\textwidth}
      The answer after the equals sign is one of three things:
      \begin{itemize}\setlength{\itemsep}{4pt}
        \item \(\infty\) --- the outputs climb forever
        \item \(-\infty\) --- the outputs fall forever
        \item a number \(b\) --- the outputs level off at the \textbf{horizontal asymptote}
              \(y = b\)
      \end{itemize}

    \column{0.48\textwidth}
      \textbf{Polynomials: the leading term decides.}

      \smallskip
      \renewcommand{\arraystretch}{1.3}
      \begin{tabular}{|l|c|c|}
        \hline
        & \(n\) even & \(n\) odd \\ \hline
        \(a > 0\) & \(\infty\), \(\infty\) & \(-\infty\), \(\infty\) \\ \hline
        \(a < 0\) & \(-\infty\), \(-\infty\) & \(\infty\), \(-\infty\) \\ \hline
      \end{tabular}

      \smallskip
      {\small (left end, right end) for \(ax^n\)}
  \end{columns}
\end{frame}

\begin{frame}[t, plain]
  \plumheader{Input or Output? --- The Crux}
  \sectionlabel[redacc]{Notes row END BEHAVIOR $\cdot$ We Do: problems 15--16, the crux}

  \begin{columns}[T]
    \column{0.44\textwidth}
      \begin{center}
      \begin{tikzpicture}[x=0.38cm, y=0.32cm]
        \draw[linegray, very thin, step=1] (-6,-5) grid (7,6);
        \draw[->, thick] (-6,0) -- (7.4,0) node[right] {\scriptsize \(x\)};
        \draw[->, thick] (0,-5) -- (0,6.4);
        \foreach \x in {-4,1,4} \node[below, font=\tiny] at (\x,0) {\x};
        \node[left, font=\tiny] at (0,2) {2};
        \draw[redacc, very thick, dashed] (1,-5) -- (1,6);
        \draw[redacc, very thick, dashed] (-6,2) -- (7,2);
        \draw[very thick, plum, <->, samples=50, domain=-5.9:0.86] plot (\x, {1/(\x-1)+2});
        \draw[very thick, plum, <->, samples=50, domain=1.25:6.9] plot (\x, {1/(\x-1)+2});
      \end{tikzpicture}

      \(r(x) = \dfrac{1}{x - 1} + 2\)
      \end{center}

    \column{0.52\textwidth}
      \begin{itemize}\setlength{\itemsep}{6pt}
        \item Sam reads \(\lim_{x \to \infty} r(x) = 2\) as ``\(x\) goes to 2.''
              What is \(x\) really doing? What gets close to 2?
        \item Find \(r(2)\). Is it 2?
        \item Domain and range of \(r\): which number is missing from each?
      \end{itemize}

      \medskip
      \begin{block}{Read them separately}
        Under ``lim,'' \(x \to\) is the \textbf{input}. After the \(=\) is the
        \textbf{output}.
      \end{block}
  \end{columns}
\end{frame}
