\item From Lesson 2.3: $p(x) = (x+3)(x-1)^2(x-4)$.

\begin{enumerate}[label=(\alph*), itemsep=6pt]
  \item List the real zeros of $p$ with the multiplicity of each.

        \vspace{0.05in}
        Zeros: \ans{$x = -3$ (mult.\ 1), $x = 1$ (mult.\ 2), $x = 4$ (mult.\ 1)}
  \item What is the degree of $p$? \quad \ans{4}
\end{enumerate}

\item State the degree of each polynomial from its factored form.

\begin{enumerate}[label=(\alph*), itemsep=6pt]
  \item $q(x) = 5(x-2)^3(x+7)^2$ \quad degree $= $ \ans{5}
  \item $r(x) = (x^2+1)(x-6)^2$ \quad degree $= $ \ans{4}
\end{enumerate}

\item From Algebra 2: simplify each expression.

\begin{enumerate}[label=(\alph*), itemsep=6pt]
  \item $i^2 = $ \ans{$-1$}
  \item $(2+3i)(2-3i)$

        \begin{work}
          (2+3i)(2-3i) &= 4 - 6i + 6i - 9i^2 \\
                       &= 4 - 9(-1) \\
                       &= 13
        \end{work}

        $(2+3i)(2-3i) = $ \ans{13}
\end{enumerate}

\end{enumerate}
