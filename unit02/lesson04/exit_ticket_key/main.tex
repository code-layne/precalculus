\begin{enumerate}[itemsep=12pt]

\item Let $p(x) = (x - 4)(x^2 + 25)$.

\begin{work}
  x^2 + 25 &= 0 \\
       x^2 &= -25 \\
         x &= \pm 5i
\end{work}

\begin{enumerate}[label=(\alph*), itemsep=4pt]
  \item Degree of $p$: \ans{3}
  \item All complex zeros of $p$: \ans{$x = 4$, $x = 5i$, $x = -5i$}
  \item How many $x$-intercepts does the graph of $p$ have? \quad
        \ans{1 (only $x = 4$)}
\end{enumerate}

\item A polynomial $q$ has real coefficients, degree 4, and $2 - i$ is one of
      its zeros.

\begin{enumerate}[label=(\alph*), itemsep=4pt]
  \item Name another zero of $q$. \quad \ans{$2 + i$}
  \item Counting multiplicity, how many \textbf{real} zeros does $q$ have?
        \quad \ans{2}
\end{enumerate}

\item Classify each polynomial function as \textbf{even}, \textbf{odd}, or
      \textbf{neither}.

\begin{enumerate}[label=(\alph*), itemsep=4pt]
  \item $f(x) = x^4 - 6x^2$ \quad \ans{even}
  \item $g(x) = x^5 + 2x^3 - x$ \quad \ans{odd}
  \item $h(x) = x^3 + x^2$ \quad \ans{neither}
\end{enumerate}

\end{enumerate}
