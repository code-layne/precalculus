\begin{enumerate}[leftmargin=1.5em, itemsep=12pt]

\item A polynomial $p$ has degree 4, real coefficients, and the following known zeros:
      $x = -1$ (multiplicity 2) and $x = 2 + i$.

      \begin{enumerate}[leftmargin=1.5em, itemsep=6pt]
        \item List \emph{all four} zeros of $p$.
              \ansline{$x=-1$ (mult.\ 2), $x=2+i$, and $x=2-i$ (conjugate pair).}

        \item At which real zero is the graph tangent to the $x$-axis?
              \ansline{At $x=-1$, because its multiplicity is 2 (even), so the graph touches but does not cross.}
      \end{enumerate}

\item Determine whether $f(x) = x^5 - 3x^3 + x$ is even, odd, or neither.

      \vspace{0.06in}
      $f(-x) = $ \ans{$(-x)^5 - 3(-x)^3 + (-x)$}

      \vspace{0.08in}
      Simplified: \ans{$-x^5 + 3x^3 - x = -(x^5 - 3x^3 + x) = -f(x)$}

      \vspace{0.08in}
      Therefore $f$ is (circle one): \quad \textbf{Even} \quad {\color{keyred}\textbf{Odd}} \quad \textbf{Neither}

      \vspace{0.06in}
      Because: \ansline{$f(-x) = -f(x)$ for all $x$, which is the definition of an odd function.}

\end{enumerate}
