% I DO — THE IMAGINARY UNIT
\begin{frame}[t, plain]
  \plumheader{The Imaginary Unit \(i\)}
  \sectionlabel[goldacc]{Notes row IMAGINARY UNIT $\cdot$ I Do: problem 1 $\cdot$ We Do: problems 2--4}

  \begin{columns}[T]
    \column{0.52\textwidth}
      \begin{block}{The definition}
        \(i = \sqrt{-1}\), so \(i^2 = -1\).\\[3pt]
        For \(n > 0\): \(\sqrt{-n} = i\sqrt{n}\). Pull the \(i\) out \textbf{first}.
      \end{block}
      \medskip
      \textbf{Examples}
      \begin{itemize}\setlength{\itemsep}{4pt}
        \item \(\sqrt{-25} = 5i\) \qquad \(\sqrt{-12} = i\sqrt{12} = 2i\sqrt{3}\)
        \item \(x^2 = -9\): \quad \(x = \pm 3i\)
      \end{itemize}
      \medskip
      \textbf{The trap:} \(\sqrt{-4}\cdot\sqrt{-9} \ne \sqrt{36}\).\\
      Rewrite first: \((2i)(3i) = 6i^2 = -6\).

    \column{0.44\textwidth}
      \begin{center}
      \begin{tikzpicture}[>=stealth]
        \node (a) at (0,1.3) {\(i^1 = i\)};
        \node (b) at (2.1,0) {\(i^2 = -1\)};
        \node (c) at (0,-1.3) {\(i^3 = -i\)};
        \node (d) at (-2.1,0) {\(i^4 = 1\)};
        \draw[->, very thick, plum] (a) to[bend left=25] (b);
        \draw[->, very thick, plum] (b) to[bend left=25] (c);
        \draw[->, very thick, plum] (c) to[bend left=25] (d);
        \draw[->, very thick, plum] (d) to[bend left=25] (a);
        \node[font=\scriptsize\itshape] at (0,0) {\(\times\, i\) each step};
      \end{tikzpicture}
      \end{center}
      \smallskip
      {\small The powers cycle every 4. Divide the exponent by 4; the remainder says where
      you land.}
  \end{columns}
\end{frame}

% I DO — COMPLEX ARITHMETIC
\begin{frame}[t, plain]
  \plumheader{Complex Numbers \(a + bi\)}
  \sectionlabel[goldacc]{Notes row COMPLEX NUMBERS $\cdot$ I Do: problems 5 and 7 $\cdot$ We Do: 6 and 8}

  \begin{block}{The form}
    \(a + bi\): \(a\) is the \textbf{real part}, \(b\) is the \textbf{imaginary part}. In
    \(7 - 2i\), the real part is 7 and the imaginary part is \(-2\).
  \end{block}

  \medskip
  \begin{columns}[T]
    \column{0.27\textwidth}
      \textbf{Add: combine like parts}
      \[ (3 + 2i) + (5 - 4i) = 8 - 2i \]
    \column{0.40\textwidth}
      \textbf{Multiply: FOIL, then \(i^2 = -1\)}
      \small
      \begin{align*}
        (2 + 3i)(4 - i) &= 8 - 2i + 12i - 3i^2 \\
                        &= 8 + 10i + 3 \\
                        &= 11 + 10i
      \end{align*}
    \column{0.28\textwidth}
      \textbf{Conjugates}\\[3pt]
      \(a + bi\) and \(a - bi\):
      \[ (a + bi)(a - bi) = a^2 + b^2 \]
      always a \textbf{real} number.
  \end{columns}
\end{frame}

% WE DO — READING DOMAIN AND RANGE
\begin{frame}[t, plain]
  \plumheader{Reading the Library: Domain, Range, Degree}
  \sectionlabel[goldacc]{Domain left to right $\cdot$ range bottom to top}

  \begin{columns}[T]
    \column{0.48\textwidth}
      \begin{block}{Three to model}
        Constant \(y = 2\): domain all reals; range only \(y = 2\).\\[3pt]
        Linear \(y = x\): domain and range all reals.\\[3pt]
        Quadratic \(y = x^2\): domain all reals; range \(y \ge 0\).
      \end{block}
      \medskip
      \textbf{Never reaches the axis:} \(\frac{1}{x}\) and \(\frac{1}{x^2}\) get closer and
      closer to an axis but never touch it --- so \(x \ne 0\).
    \column{0.48\textwidth}
      \begin{block}{Even vs.\ odd degree}
        \textbf{Even} (\(x^2, x^4, x^6, \dots\)): both arms point \textbf{up}.\\[3pt]
        \textbf{Odd} (\(x^3, x^5, \dots\)): left arm \textbf{down}, right arm \textbf{up}.
      \end{block}
      \medskip
      \textbf{Problem 9:} through \((-8,-2), (-1,-1), (0,0), (1,1), (8,2)\) --- which card?
  \end{columns}
\end{frame}

% WE DO — ABSOLUTE VALUE AS PIECEWISE
\begin{frame}[t, plain]
  \plumheader{Absolute Value Is a Piecewise Function}
  \sectionlabel[redacc]{Notes row ABSOLUTE VALUE --- the hinge of this lesson $\cdot$ I Do: problem 11}

  \begin{columns}[T]
    \column{0.50\textwidth}
      \[ |x| = \begin{cases} -x & x < 0 \\[2pt] x & x \ge 0 \end{cases} \]
      \begin{block}{Read \(-x\) as ``the opposite of \(x\)''}
        The rule \(-x\) only ever gets \textbf{negative} inputs --- and the opposite of a
        negative is \textbf{positive}: \(-(-3) = 3\).
      \end{block}
    \column{0.46\textwidth}
      \begin{center}
      \begin{tikzpicture}[x=0.32cm, y=0.32cm]
        \draw[linegray, very thin, step=1] (-5,0) grid (5,5);
        \draw[->, thick] (-5.4,0) -- (5.6,0) node[below right] {\small \(x\)};
        \draw[->, thick] (0,-0.4) -- (0,5.6) node[above] {\small \(|x|\)};
        \draw[very thick, redacc] (-5,5) -- (0,0);
        \draw[very thick, plum] (0,0) -- (5,5);
        \fill[redacc] (-3,3) circle (2.2pt);
        \node[font=\scriptsize\bfseries\color{redacc}, align=center, left] at (-5.2,2.5) {\(-x\) owns\\ \(x < 0\)};
        \node[font=\scriptsize\bfseries\color{plum}, align=center, right] at (5.4,2.5) {\(x\) owns\\ \(x \ge 0\)};
      \end{tikzpicture}
      \end{center}
  \end{columns}

  \medskip
  \textbf{Move the break:} \(x - 2\) is negative when \(x < 2\), so \quad
  \( |x - 2| = \begin{cases} 2 - x & x < 2 \\ x - 2 & x \ge 2 \end{cases} \)
\end{frame}

% WE DO — THE CRUX
\begin{frame}[t, plain]
  \plumheader{``But \(-x\) Is Negative'' --- Test It}
  \sectionlabel[redacc]{We Do: problems 12--14 $\cdot$ 13 is the crux}

  \begin{columns}[T]
    \column{0.48\textwidth}
      \begin{itemize}\setlength{\itemsep}{7pt}
        \item Find \(-x\) for \(x = -5, -1, 0, 2\). When is \(-x\) positive?
        \item A classmate: ``\(|x| = -x\) for \(x < 0\) can't be right, because \(-x\) is
              negative.'' Test it at \(x = -6\).
        \item \(|x - 2|\) at \(x = -1\) and \(x = 5\). What would \(x - 2\) alone give at
              \(x = -1\)?
      \end{itemize}
    \column{0.48\textwidth}
      \begin{block}{Two readings, one input}
        At \(x = -1\):\\[3pt]
        the rule that owns it: \(2 - (-1) = 3\)\\[3pt]
        the wrong rule: \(-1 - 2 = -3\)\\[3pt]
        An absolute value is a distance --- it is \textbf{never} negative.
      \end{block}
  \end{columns}
\end{frame}

% DEBRIEF
\begin{frame}[t, plain]
  \plumheader{Debrief --- The Headline}
  \sectionlabel[goldacc]{Share out 15--18, one answer each $\cdot$ then say it back}

  \begin{columns}[T]
    \column{0.48\textwidth}
      \begin{block}{Rewrite with \(i\) first}
        \(\sqrt{-n} = i\sqrt{n}\) --- \textbf{before} you multiply, add, or simplify.
        Then every \(i^2\) becomes \(-1\).
      \end{block}
    \column{0.48\textwidth}
      \begin{block}{\(-x\) means ``the opposite of \(x\)''}
        On the inputs it owns, \(x < 0\), the opposite is \textbf{positive}. That is why
        \(|x|\) is never negative.
      \end{block}
  \end{columns}
