\begin{enumerate}[itemsep=20pt]

\item The sequence $7,\ 11,\ 15,\ 19,\ \ldots$ starts at $n = 0$.

      Identify the type: \ans{arithmetic} \quad Common difference: $d = $ \ans{4}

      \vspace{0.06in}
      Explicit formula: $a_n = $ \ans{$7 + 4n$}

\item The sequence $3,\ 6,\ 12,\ 24,\ \ldots$ starts at $n = 0$.

      Identify the type: \ans{geometric} \quad Common ratio: $r = $ \ans{2}

      \vspace{0.06in}
      Explicit formula: $g_n = $ \ans{$3 \cdot 2^n$}

\item \textbf{True or False:} Both sequences in \#1 and \#2 are functions from the whole
      numbers to the real numbers. Circle your answer and explain in one sentence.

      \medskip
      \ans{\textbf{True}} \qquad\quad \textbf{False}

      \vspace{0.06in}
      \ans{Each sequence assigns exactly one real-number output to every whole-number input $n$, which is the definition of a function with domain the whole numbers.}

\end{enumerate}
