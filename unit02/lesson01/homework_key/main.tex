\begin{enumerate}[itemsep=10pt]

\item Read each feature straight off the form given.

\begin{enumerate}[label=(\alph*), itemsep=6pt]
  \item $g(x) = (x+1)^2 - 9$.

        Vertex: ( \ans{$-1$} , \ans{$-9$} ) \quad Axis of symmetry:
        $x = $ \ans{$-1$} \quad Concave \ans{up} .

  \item $k(x) = -2(x-4)(x+2)$. \quad Zeros: \ans{4} and \ans{$-2$}

        Since $a = -2$, the parabola opens \ans{downward} .
\end{enumerate}

\item Let $f(x) = x^2 + 2x$. Find the average rate of change over each
      interval.

\begin{enumerate}[label=(\alph*), itemsep=6pt]
  \item Over $[0, 3]$:

        \begin{work}
          f(0) &= 0 \\
          f(3) &= (3)^2 + 2(3) \\
               &= 15 \\
          \text{AROC} &= \frac{15 - 0}{3 - 0} \\
                      &= 5
        \end{work}

        AROC $= $ \ans{5}

  \item Over $[3, 5]$:

        \begin{work}
          f(5) &= (5)^2 + 2(5) \\
               &= 35 \\
          \text{AROC} &= \frac{35 - 15}{5 - 3} \\
                      &= 10
        \end{work}

        AROC $= $ \ans{10}

  \item The AROC grew from (a) to (b), so the graph of $f$ is concave
        \ans{up} .
\end{enumerate}

\boxguard[18]
\item Let $q(x) = x^2 + x$. Complete the table and both difference rows.

\smallskip
\renewcommand{\arraystretch}{1.5}
\begin{tabular}{|c|c|c|c|c|c|}
\hline
$x$ & 0 & 1 & 2 & 3 & 4 \\
\hline
$q(x)$ & \ans{0} & \ans{2} & \ans{6} & \ans{12} & \ans{20} \\
\hline
\end{tabular}
\smallskip

First differences: \ans{2} , \ans{4} , \ans{6} , \ans{8}

Second differences: \ans{2} , \ans{2} , \ans{2}

The inputs step by $1$, so the second difference equals $2a$. That gives
$a = $ \ans{1} , which matches the formula.

\boxguard[18]
\item Two functions are recorded at inputs that step by $2$.

\smallskip
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{|c|c|c|c|c|c|}
\hline
$x$ & 0 & 2 & 4 & 6 & 8 \\
\hline
$f(x)$ & 3 & 11 & 19 & 27 & 35 \\
\hline
$g(x)$ & 1 & 5 & 17 & 37 & 65 \\
\hline
\end{tabular}
\smallskip

\begin{enumerate}[label=(\alph*), itemsep=4pt]
  \item First differences of $f$: \ans{8} , \ans{8} , \ans{8} ,
        \ans{8}
  \item First differences of $g$: \ans{4} , \ans{12} , \ans{20} ,
        \ans{28}

        Second differences of $g$: \ans{8} , \ans{8} , \ans{8}
  \item One of these is linear and the other is quadratic. Say which is
        which, and justify each choice with the numbers you just computed.

        \vspace{0.05in}
        \ansline{$f$ is linear: its first differences are a constant 8, so its rate of change never}
        \ansline{changes. $g$ is quadratic: its first differences grow, but its second differences}
        \ansline{are a constant 8, so its rates of change change at a constant rate.}
\end{enumerate}

\item A quadratic function is recorded at inputs that step by $1$, and its
      second differences are constantly $6$. What is its leading
      coefficient $a$?

\begin{enumerate}[label=\Alph*., itemsep=3pt]
  \item $3$ \quad \textcolor{keyred}{\textbf{$\leftarrow$ correct: $2a = 6$}}
  \item $6$
  \item $1.5$
  \item $12$
\end{enumerate}

\end{enumerate}

\boxguard
\begin{extensionbox}
Let $c(x) = x^3$, with outputs $0,\ 1,\ 8,\ 27,\ 64$ at $x = 0,1,2,3,4$. Its
first differences are $1, 7, 19, 37$ and its second differences are
$6, 12, 18$. Are the second differences constant --- and is $c$ quadratic?
Now take one more level of differences. What do you find?

\vspace{0.05in}
\ansline{The second differences are not constant, so $c$ is not quadratic. The third}
\ansline{differences are $6, 6$ --- constant. Degree three needs three levels of differences.}
\end{extensionbox}
