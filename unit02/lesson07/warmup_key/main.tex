\begin{enumerate}[itemsep=14pt]

\item Given $f(x) = 3x - 1$, evaluate the following.

      \vspace{0.06in}
      $f(4) = $ \ans{11} \qquad $f(-2) = $ \ans{$-7$}

\item Given $f(x) = x^2$ and $g(x) = x + 3$, find $f(g(1))$ step by step.

      \vspace{0.06in}
      \textbf{Step 1:} Find $g(1)$. \quad $g(1) = $ \ans{4}

      \vspace{0.06in}
      \textbf{Step 2:} Use your answer from Step 1 as the input for $f$.

      \quad $f\bigl(g(1)\bigr) = f\!\left(\ans{4}\right) = $ \ans{16}

\item Is $f(g(1)) = g(f(1))$? Compute $g(f(1))$ below, then compare.

      \vspace{0.06in}
      $f(1) = $ \ans{1} \quad so \quad $g(f(1)) = g\!\left(\ans{1}\right) = $ \ans{4}

      \vspace{0.06in}
      Are they equal? \ans{No. $f(g(1)) = 16$ but $g(f(1)) = 4$. Order matters.}

\end{enumerate}

\begin{teachernote}
Item 1: $f(4) = 3(4)-1 = 11$; $f(-2) = 3(-2)-1 = -7$.

Item 2: $g(1) = 1+3 = 4$; $f(4) = 4^2 = 16$. Guide students through the two-step process.

Item 3: $f(1) = 1^2 = 1$; $g(1) = 1+3 = 4$. The two results are different, motivating the
non-commutativity discussion in today's lesson.
\end{teachernote}
