\item From Lesson 2.6: factor each polynomial \textbf{completely}. Pull out the
greatest common factor first.

\begin{enumerate}[label=(\alph*), itemsep=6pt]
  \item $x^3 - 9x = $ \ans{$x(x-3)(x+3)$}
  \item $x^3 + 2x^2 - 8x = $ \ans{$x(x+4)(x-2)$}
\end{enumerate}

\item From Lesson 2.3: $p(x) = (x+1)^2(x-4)$.

\begin{enumerate}[label=(\alph*), itemsep=6pt]
  \item List the real zeros of $p$ with the multiplicity of each.

        \vspace{0.05in}
        Zeros: \ans{$x = -1$ (mult.\ 2), $x = 4$ (mult.\ 1)}
  \item At which zero is the graph \textbf{tangent} to the $x$-axis?
        \quad \ans{$x = -1$}
\end{enumerate}

\item Evaluate $V(x) = x(20-2x)(30-2x)$ at $x = 2$.

\begin{work}
  V(2) &= 2\big(20 - 2(2)\big)\big(30 - 2(2)\big) \\
       &= 2(16)(26) \\
       &= 832
\end{work}

$V(2) = $ \ans{832}

\end{enumerate}
