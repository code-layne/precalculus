\begin{enumerate}[itemsep=14pt]

\item \textbf{Numerical Composition.} Use the table to evaluate each composition.

      \vspace{0.06in}
      \renewcommand{\arraystretch}{1.6}
      \begin{tabular}{c|c|c}
      $x$ & $f(x)$ & $g(x)$ \\ \hline
      $-1$ & 3 & 5 \\
      0    & 1 & $-1$ \\
      1    & 4 & 0 \\
      5    & 2 & 1 \\
      \end{tabular}

      \vspace{0.08in}
      (a) $f(g(1)) = $ \ans{$f(0) = 1$} \qquad (b) $g(f(1)) = $ \ans{$g(4)$ --- undefined from table}

      \vspace{0.06in}
      (c) $f(g(0)) = $ \ans{$f(-1) = 3$} \qquad (d) $g(f(-1)) = $ \ans{$g(3)$ --- undefined from table}

      \begin{teachernote}
      (a) $g(1)=0$, $f(0)=1$. (b) $f(1)=4$; 4 is not an $x$-input in the table for $g$ --- undefined.
      (c) $g(0)=-1$, $f(-1)=3$. (d) $f(-1)=3$; 3 is not an $x$-input for $g$ --- undefined.
      Emphasize this illustrates the domain restriction of compositions.
      \end{teachernote}

\item \textbf{Analytic Composition.} Let $f(x) = 5x - 2$ and $g(x) = x^2$.

      \vspace{0.06in}
      (a) Find a formula for $f(g(x))$. Simplify.

      \vspace{0.06in}
      $f(g(x)) = $ \ans{$5x^2 - 2$}

      \vspace{0.06in}
      (b) Find a formula for $g(f(x))$. Simplify.

      \vspace{0.06in}
      $g(f(x)) = $ \ans{$(5x-2)^2 = 25x^2 - 20x + 4$}

\item Using your formulas from \#2, evaluate $(f \circ g)(-1)$.

      \vspace{0.06in}
      $(f \circ g)(-1) = $ \ans{$5(-1)^2 - 2 = 5 - 2 = 3$}

\item \textbf{Decomposition.} Write $h(x) = (x-5)^3$ as $f(g(x))$. Identify $f$ and $g$.

      \vspace{0.06in}
      $g(x) = $ \ans{$x - 5$} \qquad $f(u) = $ \ans{$u^3$}

      \vspace{0.04in}
      Verify: $f(g(x)) = $ \ans{$(x-5)^3$}

\item \textbf{Real-World Application.}

      (a) Write a formula for $P(u(p))$. Simplify.

      \vspace{0.06in}
      $P(u(p)) = $ \ans{$4(500-10p) - 200 = 2000 - 40p - 200 = 1800 - 40p$}

      \vspace{0.06in}
      (b) Evaluate $P(u(30))$ and interpret your answer in context.

      \vspace{0.06in}
      $P(u(30)) = $ \ans{$1800 - 40(30) = 1800 - 1200 = 600$} \quad Meaning: \ansline{When the price is \$30, the company earns a profit of \$600.}

\item \textbf{AP-Style Multiple Choice.}

      \vspace{0.06in}
      \begin{tabular}{@{}ll@{\qquad}ll@{}}
      \textcolor{keyred}{\textbf{(A) $2(x+1)^2$}} & (B) $2x^2 + 1$ & (C) $2x^2 + 2$ & (D) $2x + 2$
      \end{tabular}

      \vspace{0.06in}
      Answer: \ans{A} \quad Show work: \ans{$f(g(x)) = f(x+1) = 2(x+1)^2$}

\end{enumerate}
