\item Solve $x^3 = 4x$. (Careful --- do not divide by $x$.)

\begin{work}
  x^3 - 4x &= 0 \\
  x\big(x^2 - 4\big) &= 0 \\
  x(x-2)(x+2) &= 0
\end{work}
