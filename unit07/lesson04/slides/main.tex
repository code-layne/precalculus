%----------------------------------------------------------------------
% SLIDE 1: Reciprocal Definitions
%----------------------------------------------------------------------
\begin{frame}[t, plain]
  \navyheader{Reciprocal Trigonometric Functions --- Definitions}

  \begin{columns}[T]
    \begin{column}{0.33\textwidth}
      \begin{block}{Secant}
        $$\sec\theta = \frac{1}{\cos\theta}$$
        Defined when $\cos\theta \neq 0$
      \end{block}
    \end{column}
    \begin{column}{0.33\textwidth}
      \begin{block}{Cosecant}
        $$\csc\theta = \frac{1}{\sin\theta}$$
        Defined when $\sin\theta \neq 0$
      \end{block}
    \end{column}
    \begin{column}{0.33\textwidth}
      \begin{block}{Cotangent}
        $$\cot\theta = \frac{\cos\theta}{\sin\theta}$$
        Defined when $\sin\theta \neq 0$
      \end{block}
    \end{column}
  \end{columns}

  \vspace{0.5cm}
  \begin{block}{Key Idea}
    Each reciprocal function is formed by ``flipping'' a familiar trig function.
    Zeros of the parent become \textbf{vertical asymptotes} of the reciprocal.
    The points where the parent equals $\pm 1$ are \textbf{invariant}: the reciprocal
    also equals $\pm 1$ there.
  \end{block}
\end{frame}

%----------------------------------------------------------------------
% SLIDE 2: Asymptote Locations
%----------------------------------------------------------------------
\begin{frame}[t, plain]
  \navyheader{Where Are the Asymptotes?}

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{$\sec\theta$ and $\csc\theta$}
        \begin{itemize}
          \item $\sec\theta$ is undefined when $\cos\theta = 0$
          \item[$\Rightarrow$] Asymptotes: $\theta = \dfrac{\pi}{2} + n\pi$
          \item[] (i.e., $\theta = \ldots, -\tfrac{\pi}{2}, \tfrac{\pi}{2}, \tfrac{3\pi}{2}, \ldots$)
          \item[\vspace{6pt}]{}
          \item $\csc\theta$ is undefined when $\sin\theta = 0$
          \item[$\Rightarrow$] Asymptotes: $\theta = n\pi$
          \item[] (i.e., $\theta = \ldots, -\pi, 0, \pi, 2\pi, \ldots$)
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{$\cot\theta$}
        \begin{itemize}
          \item $\cot\theta$ is undefined when $\sin\theta = 0$
          \item[$\Rightarrow$] Asymptotes: $\theta = n\pi$
          \item[] (same as $\csc\theta$)
          \item[\vspace{6pt}]{}
          \item Between consecutive asymptotes:\\
            $\cot\theta$ is \textbf{decreasing}\\
            (contrast: $\tan\theta$ is \textit{increasing})
          \item Period of $\cot\theta = \pi$
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}
