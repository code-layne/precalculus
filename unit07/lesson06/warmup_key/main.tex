\begin{notesbox}{Spiral Review --- Key}

\textbf{1.}\quad Find the distance from the origin to the point $(3, 4)$.

\vspace{4pt}
Distance $= $ \ans{$\sqrt{3^2+4^2} = \sqrt{9+16} = \sqrt{25} = 5$}

\vspace{0.14in}
\textbf{2.}\quad Evaluate each expression. Give exact values.

\vspace{4pt}
(a)\quad $\cos\!\left(\dfrac{\pi}{4}\right) = $ \ans{$\dfrac{\sqrt{2}}{2}$}
\hspace{1.5cm}
(b)\quad $\sin\!\left(\dfrac{\pi}{4}\right) = $ \ans{$\dfrac{\sqrt{2}}{2}$}

\vspace{0.14in}
\textbf{3.}\quad A right triangle has legs of length $5$ and $12$.

\vspace{4pt}
(a)\quad Hypotenuse $= $ \ans{$\sqrt{5^2+12^2} = \sqrt{169} = 13$}

\vspace{4pt}
(b)\quad Acute angle $\theta$ opposite the leg of length $5$:\quad
$\theta = \arctan\!\left(\dfrac{5}{12}\right) \approx$ \ans{$0.395$ rad}

\vspace{0.14in}
\textbf{4.}\quad $\arctan(1) = $ \ans{$\dfrac{\pi}{4}$}
