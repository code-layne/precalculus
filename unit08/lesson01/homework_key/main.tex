\begin{notesbox}{Parametric Functions Practice --- Key}

\textbf{1.}\quad $g(t) = (t^2 - 2t,\; t - 1)$, $-1 \le t \le 3$.

\vspace{6pt}
(a)\quad Complete the table of values.

\vspace{4pt}
\begin{center}
\renewcommand{\arraystretch}{1.6}
\begin{tabular}{|c|c|c|c|}
\hline
$t$ & $x(t) = t^2 - 2t$ & $y(t) = t - 1$ & $(x,\; y)$ \\
\hline
$-1$ & \ans{$3$}  & \ans{$-2$} & \ans{$(3,\;-2)$}  \\
\hline
$0$  & \ans{$0$}  & \ans{$-1$} & \ans{$(0,\;-1)$}  \\
\hline
$1$  & \ans{$-1$} & \ans{$0$}  & \ans{$(-1,\;0)$}  \\
\hline
$2$  & \ans{$0$}  & \ans{$1$}  & \ans{$(0,\;1)$}   \\
\hline
$3$  & \ans{$3$}  & \ans{$2$}  & \ans{$(3,\;2)$}   \\
\hline
\end{tabular}
\end{center}

\vspace{8pt}
(b)\quad Start and end points:

\ans{Start: $(3,\,-2)$ at $t = -1$. End: $(3,\,2)$ at $t = 3$.}

\vspace{0.10in}
\textbf{2.}

(a)\quad Solve $y(t) = 2$:

\ans{$t - 1 = 2 \;\Rightarrow\; t = 3$.
At $t = 3$: $x(3) = 9 - 6 = 3$.
Position: $(3,\,2)$ (the end point).}

(b)\quad Solve $y(t) = 0$:

\ans{$t - 1 = 0 \;\Rightarrow\; t = 1$.
At $t = 1$: $x(1) = 1 - 2 = -1$.
Position: $(-1,\,0)$.}

\vspace{0.10in}
\textbf{3.}

(a)\quad Solve $x(t) = 0$:

\ans{$t^2 - 2t = 0 \;\Rightarrow\; t(t-2) = 0 \;\Rightarrow\; t = 0$
or $t = 2$. Both are in $[-1,\,3]$.}

(b)\quad Coordinates of both crossing points:

\ans{At $t = 0$: $(0,\,-1)$. At $t = 2$: $(0,\,1)$.}

\vspace{0.10in}
\textbf{4.}\quad Why does the curve cross the $y$-axis twice?

\ans{$x(t) = t^2 - 2t = t(t-2)$ equals zero at two distinct values of
$t$ (namely $t = 0$ and $t = 2$). Because the two $t$-values give
different $y$-values ($-1$ and $1$), the curve passes through the
$y$-axis at two different heights. This means the curve is not a
function of $x$ on this domain.}

\end{notesbox}

\begin{notesbox}{AP-Style Practice --- Key}

\textbf{5.}\quad $h(t) = (2t - 1,\; t^2 - 4)$, $-1 \le t \le 2$.
Find the $y$-coordinate where $x(t) = 3$.

\ans{Set $x(t) = 2t - 1 = 3 \;\Rightarrow\; 2t = 4 \;\Rightarrow\; t = 2$.
Then $y(2) = 2^2 - 4 = 0$.}

Answer: \ans{(A) $0$}

\end{notesbox}
