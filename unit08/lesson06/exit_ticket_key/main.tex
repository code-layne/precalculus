\textbf{1.} The vector $\vec{v}$ has tail $P_1 = (2, 5)$ and head $P_2 = (6, 2)$.

\vspace{0.08in}
\hspace{1em} (a)\quad Write $\vec{v}$ in component form: \quad
$\vec{v} = \langle$ \ans{$4$}$,$ \ans{$-3$}$\rangle$

\vspace{0.12in}
\hspace{1em} (b)\quad Find the magnitude: \quad $\|\vec{v}\| = $ \ans{$\sqrt{4^2+(-3)^2}=\sqrt{16+9}=5$}

\vspace{0.22in}

\textbf{2.} Let $\vec{u} = \langle 4, 3\rangle$ and $\vec{w} = \langle 1, -2\rangle$.

\vspace{0.08in}
\hspace{1em} (a)\quad Compute the dot product: \quad
$\vec{u} \cdot \vec{w} = (4)(1) + (3)(-2) = $ \ans{$4 - 6 = -2$}

\vspace{0.12in}
\hspace{1em} (b)\quad Are $\vec{u}$ and $\vec{w}$ perpendicular?
Circle one: \quad \textbf{Yes} \quad \textbf{No}

\vspace{0.08in}
\hspace{1em} Explain: \ansline{No. The dot product is $-2 \ne 0$, so the vectors are not perpendicular.}

\vspace{0.22in}

\textbf{3.} Find the unit vector in the direction of $\vec{v} = \langle 5, 12\rangle$.

\vspace{0.08in}
\hspace{1em} $\|\vec{v}\| = $ \ans{$\sqrt{25+144}=\sqrt{169}=13$}

\vspace{0.12in}
\hspace{1em} $\hat{u} = \left\langle \dfrac{5}{\ans{13}},\; \dfrac{12}{\ans{13}} \right\rangle
= \langle$ \ans{$5/13$}$,$ \ans{$12/13$}$\rangle$
