\small
Let $\vec{u} = \langle 3, 4\rangle$ and $\vec{w} = \langle -3, 4\rangle$.

\textbf{(a)} Find the magnitude of each vector.

\vspace{0.06in}
\hspace{1em} $\|\vec{u}\| = \sqrt{3^2 + 4^2} = $ \blank{4cm}

\vspace{0.08in}
\hspace{1em} $\|\vec{w}\| = \sqrt{(-3)^2 + 4^2} = $ \blank{4cm}

\vspace{0.14in}
\textbf{(b)} Compute the vector sum $\vec{u} + \vec{w}$.

\vspace{0.06in}
\hspace{1em} $\vec{u} + \vec{w} = \langle 3+(-3),\; 4+4 \rangle = \langle$ \blank{1.8cm}$,$ \blank{1.8cm}$\rangle$

\vspace{0.14in}
\textbf{(c)} Compute the dot product $\vec{u} \cdot \vec{w}$.

\vspace{0.06in}
\hspace{1em} $\vec{u} \cdot \vec{w} = (3)(-3) + (4)(4) = $ \blank{4cm}

\vspace{0.14in}
\textbf{(d)} Are $\vec{u}$ and $\vec{w}$ perpendicular? Explain using the dot product.

\vspace{0.06in}
\writelines{2}
\end{tcolorbox}

A ship travels displacement $\vec{u} = \langle 6, 8\rangle$ km, then
$\vec{w} = \langle -2, 4\rangle$ km.

\textbf{(a)} Find the total displacement vector $\vec{r} = \vec{u} + \vec{w}$.

\vspace{0.06in}
\hspace{1em} $\vec{r} = \langle$ \blank{2cm}$,$ \blank{2cm}$\rangle$

\vspace{0.14in}
\textbf{(b)} Find the magnitude of the total displacement $\|\vec{r}\|$.

\vspace{0.06in}
\hspace{1em} $\|\vec{r}\| = $ \blank{6cm}

\vspace{0.14in}
\textbf{(c)} Find the unit vector $\hat{r}$ in the direction of $\vec{r}$.

\vspace{0.06in}
\hspace{1em} $\hat{r} = \dfrac{\vec{r}}{\|\vec{r}\|} = \left\langle \dfrac{\blank{1cm}}{\blank{1cm}},\; \dfrac{\blank{1cm}}{\blank{1cm}} \right\rangle$

\vspace{0.14in}
\textbf{(d)} Verify that $\|\hat{r}\| = 1$.

\vspace{0.06in}
\hspace{1em} $\|\hat{r}\| = $ \blank{6cm}
\end{tcolorbox}
