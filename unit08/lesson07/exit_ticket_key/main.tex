\begin{enumerate}[leftmargin=1.6em, itemsep=20pt]

\item Can a $3 \times 2$ matrix be multiplied by a $3 \times 1$ matrix? Explain your reasoning.

\medskip
\ans{No. The first matrix has 2 columns, but the second matrix has 3 rows.\\[3pt]
Since $2 \neq 3$, the inner dimensions do not match and the product is undefined.}

\vspace{0.7in}

\item Compute the matrix product:
\[
  \begin{bmatrix}2 & 1 \\ 0 & 3\end{bmatrix}
  \begin{bmatrix}4 \\ 2\end{bmatrix}
\]

\medskip
Entry $(1,1)$: $2(4)+1(2)=10$ \quad Entry $(2,1)$: $0(4)+3(2)=6$

\medskip
Product $=$ \ans{$\begin{bmatrix}10 \\ 6\end{bmatrix}$}

\end{enumerate}
