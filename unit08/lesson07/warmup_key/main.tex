\begin{enumerate}[leftmargin=1.6em, itemsep=18pt]

\item Compute the dot product $\langle 3,\, 2,\, 1\rangle \cdot \langle 4,\, 5,\, 6\rangle$.

\medskip
\ans{$3(4) + 2(5) + 1(6) = 12 + 10 + 6 = 28$}

\vspace{0.8in}

\item Matrix $A$ has 2 rows and 3 columns. Matrix $B$ has 3 rows and 1 column.
      What are the dimensions of the product $AB$?

\medskip
\ans{$2 \times 1$} \quad (the inner dimensions, both 3, match; product is $2 \times 1$)

\vspace{0.6in}

\item Simplify: $3\langle 2,\,-1\rangle + \langle 0,\,4\rangle$.

\medskip
\ans{$\langle 6, -3\rangle + \langle 0, 4\rangle = \langle 6,\, 1\rangle$}

\vspace{0.4in}

\end{enumerate}
