\begin{enumerate}[itemsep=12pt]

\item A function is given as a set of ordered pairs:
      $f = \{(1, 4),\ (2, 9),\ (3, 16)\}$.

\begin{enumerate}[label=(\alph*), itemsep=4pt]
  \item Write $f^{-1}$ as a set of ordered pairs: \ans{$\{(4,1),\ (9,2),\ (16,3)\}$}
  \item Domain of $f^{-1}$: \ans{$\{4, 9, 16\}$}
\end{enumerate}

\item Find $f^{-1}(x)$ for $f(x) = 2x - 5$.

\begin{work}
          y &= 2x - 5 \\
          x &= 2y - 5 \\
      x + 5 &= 2y \\
  \frac{x+5}{2} &= y
\end{work}

$f^{-1}(x) = $ \ans{$(x+5)/2$}

\item Verify your answer to problem 2 by computing $f\big(f^{-1}(x)\big)$.

\begin{work}
  f\big(f^{-1}(x)\big) &= 2\left(\frac{x+5}{2}\right) - 5 \\
                       &= (x + 5) - 5 \\
                       &= x
\end{work}

$f\big(f^{-1}(x)\big) = $ \ans{$x$}

\smallskip
Circle the statement that is \textbf{true}:

\begin{enumerate}[label=\Alph*., itemsep=3pt]
  \item $f^{-1}(x) = \dfrac{1}{f(x)}$
  \item $f^{-1}$ reverses $f$: if $f(a) = b$, then $f^{-1}(b) = a$
        \quad \textcolor{keyred}{\textbf{$\leftarrow$ correct}}
\end{enumerate}

\end{enumerate}
