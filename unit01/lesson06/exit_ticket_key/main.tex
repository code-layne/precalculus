\begin{enumerate}[itemsep=18pt]

\item Given $f = \{(1,4),\ (2,9),\ (3,16)\}$, write $f^{-1}$ as a set of ordered pairs.

      \vspace{0.06in}
      $f^{-1} = \big\{$\ans{$(4,1),\ (9,2),\ (16,3)$}$\big\}$

\item Find $f^{-1}(x)$ algebraically for $f(x) = 2x - 5$. Show all steps.

      \vspace{0.06in}
      Step 1 (write $y$): \ans{$y = 2x - 5$}

      \vspace{0.04in}
      Step 2 (swap $x$ and $y$): \ans{$x = 2y - 5$}

      \vspace{0.04in}
      Step 3 (solve for $y$): \ans{$x + 5 = 2y$, so $y = \dfrac{x+5}{2}$}

      \vspace{0.04in}
      $f^{-1}(x) = $ \ans{$\dfrac{x+5}{2}$}

\item Verify your answer to \#2: compute $f(f^{-1}(x))$ and confirm it equals $x$.

      \vspace{0.06in}
      $f(f^{-1}(x)) = 2\left({\color{keyred}\mathbf{\dfrac{x+5}{2}}}\right) - 5 = (x+5) - 5 = $ \ans{$x$} \quad \checkmark

\end{enumerate}
