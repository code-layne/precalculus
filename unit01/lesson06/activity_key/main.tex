\begin{scenariobox}[Reading the Thermometer Backwards]{plumlight}
\small
A lab thermometer reports in Celsius, but the safety manual is written in
Fahrenheit. The conversion rule is
\begin{center}
$F(x) = 1.8x + 32$,
\end{center}
where $x$ is a temperature in degrees Celsius and $F(x)$ is the same
temperature in degrees Fahrenheit. All day the lab needs it the \textit{other}
way: a Fahrenheit reading comes in and somebody has to say what it is in
Celsius. Every tier below works from this same rule and this same table.

\smallskip
\begin{center}
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{|c|c|c|c|c|c|}
\hline
$x$ (degrees C) & $-10$ & 0 & 10 & 20 & 100 \\
\hline
$F(x)$ (degrees F) & 14 & 32 & 50 & 68 & 212 \\
\hline
\end{tabular}
\end{center}
\end{scenariobox}

\vspace{0.1in}

\boxguard[30]
\begin{tcolorbox}[colback=white, colframe=black!40,
    title=\textbf{Tier R --- Remediate},
    fonttitle=\bfseries, arc=2mm, left=3mm, right=3mm, top=2mm, bottom=2mm]

\begin{enumerate}[itemsep=8pt]
  \item \textbf{Check two of the entries.} Use $F(x) = 1.8x + 32$ to confirm
        two values in the table above.

        \begin{work}
          F(10) &= 1.8(10) + 32 \\
                &= 50 \\
          F(20) &= 1.8(20) + 32 \\
                &= 68
        \end{work}

        $F(10) = $ \ans{50} \qquad $F(20) = $ \ans{68}

  \item \textbf{Turn the table around.} Build $F^{-1}$ by trading the two rows
        --- do not compute anything.

        \smallskip
        \renewcommand{\arraystretch}{1.5}
        \begin{center}
        \begin{tabular}{|c|c|c|c|c|c|}
        \hline
        $x$ (degrees F) & 14 & 32 & 50 & 68 & 212 \\
        \hline
        $F^{-1}(x)$ (degrees C) & \ans{$-10$} & \ans{0} & \ans{10}
          & \ans{20} & \ans{100} \\
        \hline
        \end{tabular}
        \end{center}

  \item \textbf{Read it off.} A reading of $68^\circ$F comes in.

        $F^{-1}(68) = $ \ans{20} degrees Celsius.

  \item \textbf{The swap.} Using only the two tables above:

        \begin{enumerate}[label=(\alph*), itemsep=4pt]
          \item Domain of $F$ (the top row of the first table):
                \ans{$\{-10, 0, 10, 20, 100\}$}
          \item Range of $F$: \ans{$\{14, 32, 50, 68, 212\}$}
          \item So the domain of $F^{-1}$ is the \ans{range} of $F$, and the
                range of $F^{-1}$ is the \ans{domain} of $F$.
        \end{enumerate}
\end{enumerate}
\end{tcolorbox}

\vspace{0.1in}

\boxguard[30]
\begin{tcolorbox}[colback=white, colframe=black!40,
    title=\textbf{Tier A --- Approaching Proficiency},
    fonttitle=\bfseries, arc=2mm, left=3mm, right=3mm, top=2mm, bottom=2mm]

The lab wants a formula, not a table --- one that turns \textit{any}
Fahrenheit reading into Celsius.

\begin{enumerate}[itemsep=8pt]
  \item \textbf{Swap and solve.} Find $F^{-1}(x)$ for $F(x) = 1.8x + 32$.

        \begin{work}
                            y &= 1.8x + 32 \\
                            x &= 1.8y + 32 \\
                       x - 32 &= 1.8y \\
            \frac{x - 32}{1.8} &= y
        \end{work}

        $F^{-1}(x) = $ \ans{$(x-32)/1.8$}

  \item \textbf{Verify it.} Show that running $F^{-1}$ and then $F$ returns the
        original input.

        \begin{work}
          F\big(F^{-1}(x)\big) &= 1.8\left(\frac{x - 32}{1.8}\right) + 32 \\
                               &= (x - 32) + 32 \\
                               &= x
        \end{work}

        $F\big(F^{-1}(x)\big) = $ \ans{$x$}, the \ans{identity} function ---
        so $F$ and $F^{-1}$ are inverses.

  \item \textbf{Use it.} Water boils at $212^\circ$F.

        \begin{work}
          F^{-1}(212) &= \frac{212 - 32}{1.8} \\
                      &= \frac{180}{1.8} \\
                      &= 100
        \end{work}

        $F^{-1}(212) = $ \ans{100} degrees Celsius.

  \item \textbf{Say what it does.} In one sentence, describe what the function
        $F^{-1}$ takes in and what it gives back.

        \ansline{It takes a Fahrenheit reading in and gives the Celsius temperature back.}
\end{enumerate}
\end{tcolorbox}

\vspace{0.1in}

\boxguard[30]
\begin{tcolorbox}[colback=white, colframe=black!40,
    title=\textbf{Tier E --- Extension},
    fonttitle=\bfseries, arc=2mm, left=3mm, right=3mm, top=2mm, bottom=2mm]

A greenhouse in the same lab warms and then cools again. Its temperature
$t$ hours after 6:00 a.m.\ is modeled by $T(t) = (t-6)^2 + 50$ for
$0 \le t \le 12$.

\begin{enumerate}[itemsep=8pt]
  \item \textbf{Why $T$ has no inverse on $[0, 12]$.} Evaluate $T$ at two
        different times.

        \begin{work}
          T(4) &= (4-6)^2 + 50 \\
               &= 54 \\
          T(8) &= (8-6)^2 + 50 \\
               &= 54
        \end{work}

        $T(4) = $ \ans{54} and $T(8) = $ \ans{54}. Two different
        inputs, one output --- so $T$ is not \ans{one-to-one} on $[0,12]$ and
        cannot be reversed there.

  \item \textbf{Restrict, then invert.} Keep only $6 \le t \le 12$, where the
        greenhouse is warming. Writing the input as $x$, find $T^{-1}(x)$ on
        that piece.

        \begin{work}
                        y &= (x-6)^2 + 50 \\
                        x &= (y-6)^2 + 50 \\
                   x - 50 &= (y-6)^2 \\
            \sqrt{x - 50} &= y - 6 \\
          6 + \sqrt{x-50} &= y
        \end{work}

        $T^{-1}(x) = $ \ans{$6 + \sqrt{x-50}$}

        Why did we keep only the \textbf{positive} square root?

        \ansline{On the restricted domain $t \ge 6$, so $y - 6$ is never negative.}

  \item \textbf{The notation trap, in numbers.} Fill in the table for
        $F(x) = 1.8x + 32$ and its inverse.

        \smallskip
        \renewcommand{\arraystretch}{1.6}
        \begin{center}
        \begin{tabular}{|c|c|c|}
        \hline
        & value at $x = 50$ & what it means \\
        \hline
        $F^{-1}(50)$ & \ans{10} & \ans{$50^\circ$F is $10^\circ$C} \\
        \hline
        $\dfrac{1}{F(50)}$ & \ans{$1/122$} & \ans{a reciprocal, not a temperature} \\
        \hline
        \end{tabular}
        \end{center}

        These two expressions are \ans{completely different}.
\end{enumerate}
\end{tcolorbox}
