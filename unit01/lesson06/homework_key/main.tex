\textbf{Polynomial} & \textbf{Degree} & \textbf{Lead Coeff.} & $\lim_{x \to \infty}$ & $\lim_{x \to -\infty}$ \\

$p(x) = 4x^3 - 9x + 1$         & \ans{3} & \ans{4}  & \ans{$\infty$}  & \ans{$-\infty$} \\

\hline

\item A polynomial has leading term $-3x^6$. Describe its end behavior in words (not limit notation).

\vspace{0.1in}
\ansline{As $x$ increases without bound (to the right) and as $x$ decreases without bound
(to the left), the output values decrease without bound (fall toward $-\infty$). Both ends of
the graph fall downward.}

\item Can a polynomial of \textit{even} degree have $\lim_{x \to \infty} p(x) = \infty$ and
$\lim_{x \to -\infty} p(x) = -\infty$? Explain your reasoning.

\vspace{0.1in}
\ansline{No. For an even-degree polynomial, $x^n$ and $(-x)^n$ have the same value (since the
exponent is even), so the leading term produces the same magnitude and sign on both sides. Both
limits must be either $\infty$ (positive leading coefficient) or $-\infty$ (negative leading coefficient).
Opposite end behaviors require an odd degree.}

\item The table below shows output values for a polynomial $f$.

\smallskip
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{|c|c|c|c|c|c|}
\hline
$x$    & $-100$ & $-10$ & $0$ & $10$ & $100$ \\
\hline
$f(x)$ & $-50{,}000{,}000$ & $-500$ & $3$ & $-490$ & $-49{,}999{,}997$ \\
\hline
\end{tabular}

\begin{enumerate}[label=(\alph*), itemsep=5pt]
  \item Based on the table, is the degree of $f$ most likely even or odd? \ans{even}
        Explain: \ans{Both ends ($x = \pm 100$) give large negative outputs, same direction.}

  \item Is the leading coefficient most likely positive or negative? \ans{negative}
        Explain: \ans{Both ends go to $-\infty$, which requires a negative leading coefficient with even degree.}

  \item Write the likely end behavior using limit notation.

        $\lim_{x \to \infty} f(x) = $ \ans{$-\infty$} \qquad $\lim_{x \to -\infty} f(x) = $ \ans{$-\infty$}
\end{enumerate}

\item \textbf{AP-Style Multiple Choice.} Which of the following polynomial functions has
$\lim_{x \to -\infty} f(x) = \infty$ and $\lim_{x \to \infty} f(x) = -\infty$?

\begin{enumerate}[label=(\Alph*), itemsep=3pt]
  \item $f(x) = 3x^4 - x^2 + 2$
  \item $f(x) = -2x^4 + 5x^3 - 1$
  \item $f(x) = x^5 + 3x^2 - 7$
  \item \ans{$f(x) = -x^5 + 4x^3 - x + 6$}
\end{enumerate}
\textit{(D: odd degree, negative leading coefficient $\Rightarrow$ left end $\to \infty$, right end $\to -\infty$.)}

\end{enumerate}

\begin{extensionbox}
\textbf{Extension (optional).} Complete the table for $p(x) = x^n$ at $x = 100$ and $x = -100$
for $n = 1, 2, 3, 4$.

\smallskip
\renewcommand{\arraystretch}{1.6}
\begin{tabular}{|c|c|c|}
\hline
$n$ & $p(100) = 100^n$ & $p(-100) = (-100)^n$ \\
\hline
1 & \ans{$100$}         & \ans{$-100$} \\
2 & \ans{$10{,}000$}    & \ans{$10{,}000$} \\
3 & \ans{$10^6$}        & \ans{$-10^6$} \\
4 & \ans{$10^8$}        & \ans{$10^8$} \\
\hline
\end{tabular}

\medskip
Pattern: when $n$ is even, $p(100)$ and $p(-100)$ have the \ans{same} sign.
When $n$ is odd, they have \ans{opposite} signs.
\end{extensionbox}

\textbf{Lesson Preview.} In Lesson 1.7, we study \textit{rational functions} --- ratios of polynomials.
End behavior takes a new form: instead of going to $\pm\infty$, rational functions can approach
finite horizontal asymptotes. Think about what happens when you divide a polynomial by another
polynomial of the same or higher degree.
\end{reflectionbox}
