\item Each table gives every value of its function.

\smallskip
\renewcommand{\arraystretch}{1.4}
\begin{center}
\begin{tabular}{|c|c|c|c|c|c|}
\hline
$x$ & 1 & 2 & 3 & 4 & 5 \\
\hline
$f(x)$ & 9 & 4 & 7 & 2 & 0 \\
\hline
\end{tabular}
\qquad
\begin{tabular}{|c|c|c|c|c|c|}
\hline
$x$ & 1 & 2 & 3 & 4 & 5 \\
\hline
$g(x)$ & 6 & 1 & 6 & 8 & 3 \\
\hline
\end{tabular}
\end{center}

\begin{enumerate}[label=(\alph*), itemsep=6pt]
  \item $f^{-1}(7) = $ \ans{3} \qquad $f^{-1}(0) = $ \ans{5}
        \qquad $f^{-1}(9) = $ \ans{1}
  \item Complete the $f^{-1}$ table by trading rows.

        \smallskip
        \renewcommand{\arraystretch}{1.4}
        \begin{center}
        \begin{tabular}{|c|c|c|c|c|c|}
        \hline
        $x$ & 0 & 2 & 4 & 7 & 9 \\
        \hline
        $f^{-1}(x)$ & \ans{5} & \ans{4} & \ans{2}
          & \ans{3} & \ans{1} \\
        \hline
        \end{tabular}
        \end{center}

  \item $g$ has no inverse. In one sentence, name the answer that causes the
        trouble and say what goes wrong.

        \vspace{0.05in}
        \ansline{The answer 6 came from both $x = 1$ and $x = 3$, so $g^{-1}(6)$ would have to be two numbers at once.}
\end{enumerate}

\item Find each inverse by swapping and solving.

\begin{work}
              y &= 6x + 5 \\
              x &= 6y + 5 \\
          x - 5 &= 6y \\
  \frac{x-5}{6} &= y \\
              y &= 3 - 2x \\
              x &= 3 - 2y \\
             2y &= 3 - x \\
  \frac{3-x}{2} &= y
\end{work}

\begin{enumerate}[label=(\alph*), itemsep=5pt]
  \item $f(x) = 6x + 5$: \quad $f^{-1}(x) = $ \ans{$\dfrac{x-5}{6}$}
  \item $g(x) = 3 - 2x$: \quad $g^{-1}(x) = $ \ans{$\dfrac{3-x}{2}$}
\end{enumerate}

\item Show that $f(x) = \dfrac{x}{2} + 4$ and $g(x) = 2x - 8$ are inverses by
      checking \textbf{both} directions.

\begin{work}
  f(g(x)) &= \frac{2x - 8}{2} + 4 \\
          &= (x - 4) + 4 \\
          &= x \\
  g(f(x)) &= 2\left(\frac{x}{2} + 4\right) - 8 \\
          &= (x + 8) - 8 \\
          &= x
\end{work}

$f(g(x)) = $ \ans{$x$} \qquad $g(f(x)) = $ \ans{$x$} \qquad
so $f$ and $g$ are \ans{inverses}.

\item A delivery app charges $D(x) = 1.5x + 4$ dollars for a $x$-mile
      delivery.

\begin{work}
                  y &= 1.5x + 4 \\
                  x &= 1.5y + 4 \\
              x - 4 &= 1.5y \\
  \frac{x-4}{1.5}   &= y
\end{work}

\begin{enumerate}[label=(\alph*), itemsep=5pt]
  \item A 6-mile delivery costs \$\ans{13}.
  \item $D^{-1}(x) = $ \ans{$\dfrac{x-4}{1.5}$}
  \item Your receipt says \$19 and does not say the distance. The delivery was
        \ans{10} miles.
  \item Compare the two: $D^{-1}(13) = $ \ans{6}, but
        $D(13) = $ \ans{23.5}, so $\dfrac{1}{D(13)} = $ \ans{$1/23.5$}.
        Which one is a number of miles? \ans{the 6}
\end{enumerate}

\item Let $h(x) = (x - 2)^2$.

\begin{enumerate}[label=(\alph*), itemsep=5pt]
  \item $h(0) = $ \ans{4} and $h(4) = $ \ans{4}, so $h$ is not
        \ans{one-to-one} on all real numbers.
  \item Restrict the domain to $x \ge 2$. Then
        $h^{-1}(x) = $ \ans{$2 + \sqrt{x}$}, with domain \ans{$x \ge 0$}.
  \item The graph of a one-to-one function $F$ passes through $(-3, 8)$.
        Name a point that must be on the graph of $F^{-1}$:
        $($ \ans{8} , \ans{$-3$} $)$.
  \item The graphs of $F$ and $F^{-1}$ are reflections of each other across the
        line \ans{$y = x$}.
\end{enumerate}

\item A function $f$ satisfies $f(3) = -2$. Which statement must be true?

\begin{enumerate}[label=\Alph*., itemsep=3pt]
  \item $f^{-1}(3) = -2$
  \item \ans{$f^{-1}(-2) = 3$}
  \item $f^{-1}(3) = -\dfrac{1}{2}$
  \item $f^{-1}(-2) = -\dfrac{1}{3}$
\end{enumerate}

\end{enumerate}

\boxguard
\begin{extensionbox}
\textbf{Shoes and socks.} Let $f(x) = 2x$ and $g(x) = x + 5$.

\begin{enumerate}[label=(\alph*), itemsep=4pt]
  \item Build $(f \circ g)(x)$, then find its inverse by swapping and solving.
  \item Now compute $g^{-1}\big(f^{-1}(x)\big)$ and compare. Then compute
        $f^{-1}\big(g^{-1}(x)\big)$ and compare again.
  \item One of the two orders undoes the chain. Say which, and why undoing a
        chain has to start at the \textit{last} step.
\end{enumerate}

\vspace{0.05in}
\ansline{(a) $(f \circ g)(x) = 2x + 10$, so its inverse is $(x-10)/2 = x/2 - 5$.}
\ansline{(b--c) $g^{-1}(f^{-1}(x)) = x/2 - 5$ matches; $f^{-1}(g^{-1}(x)) = (x-5)/2$ does not. Undo the last step first.}
\end{extensionbox}
