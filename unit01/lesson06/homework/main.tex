\item Find each inverse by swapping and solving.

\begin{work}
              y &= 6x + 5 \\
              x &= 6y + 5 \\
          x - 5 &= 6y \\
  \frac{x-5}{6} &= y \\
              y &= 3 - 2x \\
              x &= 3 - 2y \\
             2y &= 3 - x \\
  \frac{3-x}{2} &= y
\end{work}

\begin{enumerate}[label=(\alph*), itemsep=5pt]
  \item $f(x) = 6x + 5$: \quad $f^{-1}(x) = $ \blank{3cm}
  \item $g(x) = 3 - 2x$: \quad $g^{-1}(x) = $ \blank{3cm}
\end{enumerate}

\item Show that $f(x) = \dfrac{x}{2} + 4$ and $g(x) = 2x - 8$ are inverses by
      checking \textbf{both} directions.

\begin{work}
  f(g(x)) &= \frac{2x - 8}{2} + 4 \\
          &= (x - 4) + 4 \\
          &= x \\
  g(f(x)) &= 2\left(\frac{x}{2} + 4\right) - 8 \\
          &= (x + 8) - 8 \\
          &= x
\end{work}

$f(g(x)) = $ \blank{1.4cm} \qquad $g(f(x)) = $ \blank{1.4cm} \qquad
so $f$ and $g$ are \blank{2.4cm}.

\item A delivery app charges $D(x) = 1.5x + 4$ dollars for a $x$-mile
      delivery.

\begin{work}
                  y &= 1.5x + 4 \\
                  x &= 1.5y + 4 \\
              x - 4 &= 1.5y \\
  \frac{x-4}{1.5}   &= y
\end{work}

\begin{enumerate}[label=(\alph*), itemsep=5pt]
  \item A 6-mile delivery costs \$\blank{1.6cm}.
  \item $D^{-1}(x) = $ \blank{3cm}
  \item Your receipt says \$19 and does not say the distance. The delivery was
        \blank{1.6cm} miles.
  \item Compare the two: $D^{-1}(13) = $ \blank{1.4cm}, but
        $D(13) = $ \blank{1.6cm}, so $\dfrac{1}{D(13)} = $ \blank{1.8cm}.
        Which one is a number of miles? \blank{2.4cm}
\end{enumerate}

\item Let $h(x) = (x - 2)^2$.

\begin{enumerate}[label=(\alph*), itemsep=5pt]
  \item $h(0) = $ \blank{1.4cm} and $h(4) = $ \blank{1.4cm}, so $h$ is not
        \blank{2.6cm} on all real numbers.
  \item Restrict the domain to $x \ge 2$. Then
        $h^{-1}(x) = $ \blank{3cm}, with domain \blank{2.4cm}.
  \item The graph of a one-to-one function $F$ passes through $(-3, 8)$.
        Name a point that must be on the graph of $F^{-1}$:
        $($ \blank{1.2cm} , \blank{1.2cm} $)$.
  \item The graphs of $F$ and $F^{-1}$ are reflections of each other across the
        line \blank{2cm}.
\end{enumerate}

\item A function $f$ satisfies $f(3) = -2$. Which statement must be true?

\begin{enumerate}[label=\Alph*., itemsep=3pt]
  \item $f^{-1}(3) = -2$
  \item $f^{-1}(-2) = 3$
  \item $f^{-1}(3) = -\dfrac{1}{2}$
  \item $f^{-1}(-2) = -\dfrac{1}{3}$
\end{enumerate}

\end{enumerate}
