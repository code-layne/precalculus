\begin{frame}[t, plain]
  \navyheader{Hook: The Rubber Band Ball}
  \sectionlabel[goldacc]{Think About It}

  A student keeps wrapping rubber bands around a ball. The radius (in mm) is modeled by:
  \[
    r(t) = 2t^3 - 50t^2 + 100t
  \]

  \bigskip
  \begin{columns}[T]
    \column{0.52\textwidth}
      \begin{block}{At $t = 10$ minutes}
        \begin{itemize}\setlength{\itemsep}{5pt}
          \item $2t^3 = 2{,}000$
          \item $-50t^2 = -5{,}000$
          \item $100t = 1{,}000$
          \item $r(10) = -2{,}000$
        \end{itemize}
      \end{block}

    \column{0.42\textwidth}
      \begin{block}{At $t = 1{,}000$ minutes}
        \begin{itemize}\setlength{\itemsep}{5pt}
          \item $2t^3 = 2 \times 10^9$
          \item $-50t^2 = -5 \times 10^7$
          \item $100t = 10^5$
        \end{itemize}
      \end{block}
  \end{columns}

  \bigskip
  \textit{Discussion:} Which term dominates as $t \to \infty$? Does $-50t^2$ matter when $t$ is huge?
\end{frame}

% LEADING TERM DOMINANCE
\begin{frame}[t, plain]
  \navyheader{Leading Term Dominance}

  \begin{block}{Key Idea}
    For large $|x|$, the \textbf{leading term} dominates all lower-degree terms.\\
    To find end behavior: look \textbf{only at the leading term}.
  \end{block}

  \bigskip
  \textbf{Example:} $p(x) = x^3 - 100x^2$

  \smallskip
  \begin{tabular}{|c|r|r|r|}
  \hline
  $x$ & $p(x)$ & $x^3$ & $-100/x$ \\
  \hline
  $100$   & $0$           & $10^6$ & $-1$ \\
  $1{,}000$ & $9\times10^8$ & $10^9$ & $-0.1$ \\
  $10{,}000$ & $9.9\times10^{11}$ & $10^{12}$ & $-0.01$ \\
  \hline
  \end{tabular}

  \bigskip
  As $x \to \infty$: the ratio $-100x^2 / x^3 = -100/x \to 0$.

  \medskip
  $p(x)$ behaves like $x^3$. \quad So $\lim_{x \to \infty} p(x) = \infty$.
\end{frame}

% THE FOUR CASES
\begin{frame}[t, plain]
  \navyheader{The Four End-Behavior Cases}

  End behavior depends on: \textbf{degree (even/odd)} and \textbf{sign of leading coefficient}.

  \bigskip
  \renewcommand{\arraystretch}{1.6}
  \begin{tabular}{|c|c|c|c|}
  \hline
  \textbf{Degree} & \textbf{Lead Coeff.} & $\lim_{x \to \infty} p(x)$ & $\lim_{x \to -\infty} p(x)$ \\
  \hline
  Even & Positive & $\infty$  & $\infty$ \\
  Even & Negative & $-\infty$ & $-\infty$ \\
  Odd  & Positive & $\infty$  & $-\infty$ \\
  Odd  & Negative & $-\infty$ & $\infty$ \\
  \hline
  \end{tabular}

  \bigskip
  \begin{columns}[T]
    \column{0.48\textwidth}
      \begin{block}{Even Degree}
        Both ends go the \textbf{same direction}.\\
        Think: $y = x^2$ (both ends up).
      \end{block}
    \column{0.48\textwidth}
      \begin{block}{Odd Degree}
        Ends go \textbf{opposite directions}.\\
        Think: $y = x^3$ (left down, right up).
      \end{block}
  \end{columns}
\end{frame}
