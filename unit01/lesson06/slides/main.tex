% HOOK SLIDE
\begin{frame}[t, plain]
  \plumheader{Hook: Give Me Back the Celsius}
  \sectionlabel[goldacc]{Think About It}

  The weather app says \(68^\circ\)F. Your friend in Paris wants that in
  Celsius. The only rule on the board is

  \begin{center}
    {\Large \(F(x) = 1.8x + 32\)}
  \end{center}

  \dots\ and it converts Celsius \textbf{into} Fahrenheit --- the wrong
  direction.

  \bigskip
  \begin{columns}[T]
    \column{0.46\textwidth}
      \begin{block}{Question 1}
        What do you do to \(68\) to get back to Celsius --- and in what
        \textbf{order}?
      \end{block}
    \column{0.46\textwidth}
      \begin{block}{Question 2}
        You just followed a rule. Write it down. Is that new rule a function
        too?
      \end{block}
  \end{columns}

  \bigskip
  \textit{Every function whose outputs never repeat has a partner that runs it
  backwards. Today we build the partner --- and check it.}
\end{frame}

% SECTION 1 — UNDOING AS A REVERSED MAPPING
\begin{frame}[t, plain]
  \plumheader{1. ``Undoing'' Is a Reversed Mapping}

  \begin{columns}[T]
    \column{0.48\textwidth}
      \begin{block}{The definition, in one line}
        If \(f(a) = b\), then \(f^{-1}(b) = a\).
      \end{block}

      \medskip
      \begin{tabular}{|l|l|}
      \hline
      \(F\) sends & \(F^{-1}\) sends back \\
      \hline
      \(F(20) = 68\) & \(F^{-1}(68) = 20\) \\
      \hline
      \(F(0) = 32\)  & \(F^{-1}(32) = 0\) \\
      \hline
      \(F(a) = b\)   & \(F^{-1}(b) = a\) \\
      \hline
      \end{tabular}

    \column{0.48\textwidth}
      \begin{block}{Warning --- today's signature error}
        \[ f^{-1}(x) \ne \frac{1}{f(x)} \]
        The \(-1\) is a \textbf{name}, not an exponent.
      \end{block}

      \medskip
      \begin{itemize}\setlength{\itemsep}{5pt}
        \item \(F(10) = 50\), so \(F^{-1}(50) = 10\).
        \item But \(F(50) = 122\), so \(\frac{1}{F(50)} = \frac{1}{122}\).
        \item \(10\) and \(\tfrac{1}{122}\) are not close to the same number.
      \end{itemize}
  \end{columns}
\end{frame}

% SECTION 2 — TABLES AND THE DOMAIN/RANGE SWAP
\begin{frame}[t, plain]
  \plumheader{2. Inverses from Tables --- the Domain/Range Swap}

  Build \(F^{-1}\) by \textbf{trading the two rows}. Nothing is computed.

  \bigskip
  \begin{columns}[T]
    \column{0.48\textwidth}
      \begin{center}
      \begin{tabular}{|c|c|c|c|c|c|}
      \hline
      \(x\) (C) & \(-10\) & 0 & 10 & 20 & 100 \\
      \hline
      \(F(x)\) (F) & 14 & 32 & 50 & 68 & 212 \\
      \hline
      \end{tabular}

      \medskip
      \begin{tabular}{|c|c|c|c|c|c|}
      \hline
      \(x\) (F) & 14 & 32 & 50 & 68 & 212 \\
      \hline
      \(F^{-1}(x)\) (C) & \(-10\) & 0 & 10 & 20 & 100 \\
      \hline
      \end{tabular}
      \end{center}

    \column{0.48\textwidth}
      \begin{block}{The swap}
        Domain of \(f^{-1}\) \(=\) range of \(f\). \\[3pt]
        Range of \(f^{-1}\) \(=\) domain of \(f\).
      \end{block}

      \medskip
      \begin{block}{The catch}
        \(g = \{(1,5), (2,5), (3,8)\}\) reversed sends 5 to \textbf{two}
        outputs --- not a function.\\[3pt]
        Only a \textbf{one-to-one} function is invertible.
      \end{block}
  \end{columns}
\end{frame}

% SECTION 3 — SWAP AND SOLVE
\begin{frame}[t, plain]
  \plumheader{3. The Analytic Method: Swap and Solve}

  \begin{columns}[T]
    \column{0.34\textwidth}
      \begin{block}{Four steps}
        \begin{enumerate}\setlength{\itemsep}{4pt}
          \item Write \(y = f(x)\).
          \item \textbf{Swap} \(x\) and \(y\).
          \item Solve for \(y\).
          \item Rename \(f^{-1}(x)\).
        \end{enumerate}
        \medskip
        Step 2 is the inversion. The rest is algebra.
      \end{block}

    \column{0.30\textwidth}
      \begin{block}{\(f(x) = 4x - 1\)}
        \(y = 4x - 1\) \\[3pt]
        \(x = 4y - 1\) \\[3pt]
        \(x + 1 = 4y\) \\[3pt]
        \(f^{-1}(x) = \frac{x+1}{4}\)
      \end{block}

    \column{0.30\textwidth}
      \begin{block}{\(F(x) = 1.8x + 32\)}
        \(y = 1.8x + 32\) \\[3pt]
        \(x = 1.8y + 32\) \\[3pt]
        \(x - 32 = 1.8y\) \\[3pt]
        \(F^{-1}(x) = \frac{x-32}{1.8}\)
      \end{block}
  \end{columns}

  \bigskip
  \textit{Check: \(F^{-1}(68) = \frac{68-32}{1.8} = 20\) --- the same answer the
  table gave.}
\end{frame}

% SECTION 4 — VERIFICATION BY COMPOSITION
\begin{frame}[t, plain]
  \plumheader{4. Verification by Composition}

  \sectionlabel[goldacc]{This Is What Lesson 1.5 Was For}

  \begin{center}
    {\Large \(f\big(f^{-1}(x)\big) = f^{-1}\big(f(x)\big) = x\)}
  \end{center}

  \bigskip
  \begin{columns}[T]
    \column{0.48\textwidth}
      \begin{block}{Verify \(f(x) = 4x - 1\)}
        \(f\big(f^{-1}(x)\big) = 4\left(\frac{x+1}{4}\right) - 1 = x\)
        \\[5pt]
        \(f^{-1}\big(f(x)\big) = \frac{(4x-1)+1}{4} = x\)
      \end{block}

    \column{0.48\textwidth}
      \begin{block}{The identity function}
        \(I(x) = x\) returns its own input. \\[3pt]
        A function composed with its inverse --- in \textbf{either} order ---
        gives it.
      \end{block}
  \end{columns}

  \bigskip
  \textit{You already did this: in the warm-up, \(f(g(7)) = 7\) and
  \(g(f(7)) = 7\) with these exact functions. Algebra checks every input at
  once.}
\end{frame}

% SECTION 5 — NO INVERSE, AND THE MIRROR LINE
\begin{frame}[t, plain]
  \plumheader{5. No Inverse --- and the Mirror Line \(y = x\)}

  \begin{columns}[T]
    \column{0.40\textwidth}
      \begin{center}
      \begin{tikzpicture}[x=0.40cm, y=0.40cm]
        \draw[linegray, very thin, step=1] (-5,-5) grid (5,5);
        \draw[->, thick] (-5.4,0) -- (5.7,0) node[below right] {\small \(x\)};
        \draw[->, thick] (0,-5.4) -- (0,5.7) node[above] {\small \(y\)};
        \foreach \x in {-4,-2,2,4} \node[below, font=\tiny] at (\x,0) {\x};
        \foreach \y in {-4,-2,2,4} \node[left, font=\tiny] at (0,\y) {\y};
        \draw[dashed, thick, slate] (-5,-5) -- (5,5);
        \node[font=\tiny, slate, above left] at (4.6,4.6) {\(y=x\)};
        \draw[very thick, plum] (-3,-5) -- (2,5);
        \draw[very thick, goldacc] (-5,-3) -- (5,2);
        \fill[plum] (0,1) circle (2.2pt);
        \fill[goldacc] (1,0) circle (2.2pt);
        \fill[plum] (2,5) circle (2.2pt);
        \fill[goldacc] (5,2) circle (2.2pt);
      \end{tikzpicture}
      \end{center}
      \centering{\scriptsize plum \(= f(x) = 2x+1\) \quad
      gold \(= f^{-1}(x) = \frac{x-1}{2}\)}

    \column{0.54\textwidth}
      \begin{block}{When there is no inverse}
        \(f(x) = x^2\): \(f(-2) = f(2) = 4\). \\[3pt]
        Reversing would send 4 to two places, so \(f\) is not one-to-one on
        all reals.\\[3pt]
        \textbf{Restrict} to \(x \ge 0\): then \(f^{-1}(x) = \sqrt{x}\).
      \end{block}

      \medskip
      \begin{block}{Reading the figure}
        Every point \((a,b)\) on \(f\) becomes \((b,a)\) on \(f^{-1}\):
        \((0,1) \to (1,0)\), \ \((2,5) \to (5,2)\).\\[3pt]
        The graphs are reflections over \(y = x\).
      \end{block}
  \end{columns}
\end{frame}

% CLASS PRACTICE
\begin{frame}[t, plain]
  \plumheader{Class Practice}
  \sectionlabel[redacc]{Try It}

  \begin{enumerate}\setlength{\itemsep}{10pt}
    \item Let \(f(x) = 3x + 6\). Find \(f^{-1}(x)\), then verify by computing
          \(f^{-1}\big(f(x)\big)\).
    \item A function \(f\) satisfies \(f(-4) = 7\).
          \begin{enumerate}[(a)]\setlength{\itemsep}{4pt}
            \item What is \(f^{-1}(7)\)?
            \item Why is \(f^{-1}(7)\) \textbf{not} equal to \(\frac{1}{7}\)?
          \end{enumerate}
    \item A table gives \(h(1) = 5\), \(h(2) = 8\), \(h(3) = 8\).
          Is \(h\) invertible? If not, say exactly what goes wrong when you
          try to reverse the pairs.
  \end{enumerate}
\end{frame}
