\begin{enumerate}[itemsep=12pt]

\item The graph of $g(x) = f(x + 3) - 2$ is a transformation of the graph of
      $f$. Fill in both moves.

\begin{enumerate}[label=(\alph*), itemsep=4pt]
  \item Horizontal: \ans{3} units to the \ans{left}
  \item Vertical: \ans{2} units \ans{down}
\end{enumerate}

\item The table gives values of $f$. Complete the row for $g(x) = -f(x)$.

\smallskip
\renewcommand{\arraystretch}{1.5}
\begin{center}
\begin{tabular}{|c|c|c|c|c|}
\hline
$x$ & $-1$ & 0 & 1 & 2 \\
\hline
$f(x)$ & 2 & 5 & $-3$ & 4 \\
\hline
$g(x) = -f(x)$ & \ans{$-2$} & \ans{$-5$} & \ans{3}
  & \ans{$-4$} \\
\hline
\end{tabular}
\end{center}

\item The domain of $f$ is $[-2, 6]$. Find the domain of $g(x) = f(x - 4)$.

\begin{work}
  -2 &\le x - 4 \le 6 \\
   2 &\le x \le 10
\end{work}

Domain of $g$: \ans{$[2, 10]$}

\end{enumerate}
