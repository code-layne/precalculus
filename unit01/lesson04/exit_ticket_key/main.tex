\item Let $g(x) = f(x + 5) - 2$.

\begin{enumerate}[label=(\alph*), itemsep=5pt]
  \item The change to the \textbf{question} moves the graph \ans{5}
        units to the \ans{left}.
  \item The change to the \textbf{answer} moves the graph \ans{2}
        units \ans{down}.
  \item The lowest point on the graph of $f$ is $(0, 0)$. The lowest point on
        the graph of $g$ is ( \ans{$-5$} , \ans{$-2$} ).
\end{enumerate}

\item The table gives values of $f$. Complete the row for $g(x) = -f(x) + 1$.

\item The domain of $f$ is $[-2, 6]$ and the range of $f$ is $[0, 4]$. Let
      $g(x) = f(x - 4)$.

\begin{work}
  -2 &\le x - 4 \le 6 \\
   2 &\le x \le 10
\end{work}

\begin{enumerate}[label=(\alph*), itemsep=5pt]
  \item Domain of $g$: \ans{$[2,\,10]$}
  \item Range of $g$: \ans{$[0,\,4]$} \quad --- why? \ans{the answers never changed}
\end{enumerate}

\end{enumerate}
