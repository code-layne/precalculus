\begin{objectivebox}
By the end of this lesson, I will be able to\ldots
\begin{enumerate}[itemsep=2pt]
  \item Describe how $g(x) = af(x+h)+k$ transforms the graph of $f$ using
        \ans{translations} and \ans{dilations (and reflections)}. \hfill\textit{(LO 1.12.A)}
  \item Determine the \ans{domain} and \ans{range} of a transformed function
        from the parent's domain and range. \hfill\textit{(LO 1.12.A)}
\end{enumerate}
\end{objectivebox}

\vspace{0.06in}

\begin{vocabbox}
\termblanklong{parent function}
\ansline{The original, un-transformed function from which transformed versions are built.}
\termblanklong{transformation}
\ansline{An operation that shifts, scales, or reflects the graph of a function.}
\termblanklong{vertical translation}
\ansline{$g(x)=f(x)+k$: shifts the graph up ($k>0$) or down ($k<0$) by $|k|$ units.}
\termblanklong{horizontal translation}
\ansline{$g(x)=f(x+h)$: shifts the graph left ($h>0$) or right ($h<0$) by $|h|$ units.}
\termblanklong{vertical dilation}
\ansline{$g(x)=af(x)$: stretches ($|a|>1$) or compresses ($0<|a|<1$) outputs; $a<0$ also reflects over $x$-axis.}
\termblanklong{horizontal dilation}
\ansline{$g(x)=f(bx)$: compresses ($|b|>1$) or stretches ($0<|b|<1$) inputs; $b<0$ also reflects over $y$-axis.}
\termblanklong{reflection}
\ansline{A flip of the graph over an axis; $a<0$ flips over $x$-axis; $b<0$ flips over $y$-axis.}
\end{vocabbox}

\vspace{0.06in}

\begin{hookbox}
One function --- four disguises. Can you un-transform it back? \\[4pt]
Each transformation below produces the \textit{same} parent shape in a different location
or size. Your job today is to decode the rule so you can construct \textit{any} disguise
or reverse-engineer it from a table.
\end{hookbox}

\vspace{0.06in}

\begin{notesbox}{Type 1 — Vertical Translation: $g(x) = f(x) + k$}
\textbf{Rule (EK 1.12.A.1):} Adding $k$ to the output \ans{shifts (translates)} the graph by $k$ units.
\begin{itemize}[itemsep=2pt]
  \item $k > 0$: shift \ans{up} \hspace{1cm} $k < 0$: shift \ans{down}
\end{itemize}

\smallskip
\textbf{Example:} Let $f(x) = x^2$. Let $g(x) = f(x) + 3 = x^2 + 3$.

\smallskip
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{|c|c|c|}
\hline
$x$ & $f(x) = x^2$ & $g(x) = f(x)+3$ \\
\hline
$-2$ & $4$ & \ans{$7$} \\
$-1$ & $1$ & \ans{$4$} \\
$0$  & $0$ & \ans{$3$} \\
$1$  & $1$ & \ans{$4$} \\
$2$  & $4$ & \ans{$7$} \\
\hline
\end{tabular}

\medskip
\textbf{Effect on domain/range:} Domain \ans{unchanged}. Range shifts \ans{up by 3}.
\end{notesbox}

\vspace{0.06in}

\begin{notesbox}{Type 2 — Horizontal Translation: $g(x) = f(x + h)$}
\textbf{Rule (EK 1.12.A.2):} Replacing $x$ with $x+h$ shifts the graph
\ans{horizontally} by $|h|$ units.
\begin{itemize}[itemsep=2pt]
  \item $h > 0$: shift \ans{left} \hspace{1cm} $h < 0$: shift \ans{right}
\end{itemize}

\smallskip
\textbf{Caution:} $g(x) = f(x+3)$ shifts the graph \ans{left 3} (not right), because we
need $x + 3 = 0$, so $x = $ \ans{$-3$}.

\smallskip
\textbf{Example:} $f(x) = x^2$, $g(x) = f(x-2) = (x-2)^2$.

\smallskip
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{|c|c|c|}
\hline
$x$ & $f(x) = x^2$ & $g(x) = f(x-2)$ \\
\hline
$0$ & $0$ & \ans{$4$} \\
$1$ & $1$ & \ans{$1$} \\
$2$ & $4$ & \ans{$0$} \\
$3$ & $9$ & \ans{$1$} \\
$4$ & $16$ & \ans{$4$} \\
\hline
\end{tabular}

\medskip
\textbf{Effect on domain/range:} Domain shifts \ans{right by 2}. Range \ans{unchanged}.
\end{notesbox}

\vspace{0.06in}

\begin{notesbox}{Type 3 — Vertical Dilation: $g(x) = af(x)$, $a \ne 0$}
\textbf{Rule (EK 1.12.A.3):} Multiplying the \textit{output} by $a$ scales the graph
\ans{vertically} by $|a|$.
\begin{itemize}[itemsep=2pt]
  \item $|a| > 1$: \ans{vertical stretch} (stretches) \hspace{0.5cm}
        $0 < |a| < 1$: \ans{vertical compression} (compresses)
  \item $a < 0$: also reflects over the \ans{$x$-axis}
\end{itemize}

\smallskip
\textbf{Example:} $f(x) = x^2$; $g(x) = 2f(x) = 2x^2$; $h(x) = -f(x) = -x^2$.

\smallskip
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{|c|c|c|c|}
\hline
$x$ & $f(x) = x^2$ & $g(x) = 2f(x)$ & $h(x) = -f(x)$ \\
\hline
$-2$ & $4$ & \ans{$8$}  & \ans{$-4$} \\
$-1$ & $1$ & \ans{$2$}  & \ans{$-1$} \\
$0$  & $0$ & \ans{$0$}  & \ans{$0$}  \\
$1$  & $1$ & \ans{$2$}  & \ans{$-1$} \\
$2$  & $4$ & \ans{$8$}  & \ans{$-4$} \\
\hline
\end{tabular}

\medskip
\textbf{Effect on domain/range:} Domain \ans{unchanged}. Range is multiplied by \ans{$a$}.
\end{notesbox}

\vspace{0.06in}

\begin{notesbox}{Type 4 — Horizontal Dilation: $g(x) = f(bx)$, $b \ne 0$}
\textbf{Rule (EK 1.12.A.4):} Multiplying the \textit{input} by $b$ scales the graph
\ans{horizontally} by $\dfrac{1}{|b|}$.
\begin{itemize}[itemsep=2pt]
  \item $|b| > 1$: \ans{horizontal compression} (compresses horizontally) \hspace{0.3cm}
        $0 < |b| < 1$: \ans{horizontal stretch} (stretches horizontally)
  \item $b < 0$: also reflects over the \ans{$y$-axis}
\end{itemize}

\smallskip
\textbf{Example:} $f(x) = x^2$; $g(x) = f(2x) = (2x)^2 = 4x^2$.

\smallskip
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{|c|c|c|}
\hline
$x$ & $f(x) = x^2$ & $g(x) = f(2x)$ \\
\hline
$-1$ & $1$ & \ans{$4$} \\
$-\frac{1}{2}$ & $\frac{1}{4}$ & \ans{$1$} \\
$0$  & $0$ & \ans{$0$} \\
$\frac{1}{2}$ & $\frac{1}{4}$ & \ans{$1$} \\
$1$  & $1$ & \ans{$4$} \\
\hline
\end{tabular}
\end{notesbox}

\vspace{0.06in}

\begin{notesbox}{Combined Transformations: $g(x) = af(x+h)+k$}
\textbf{Rule (EK 1.12.A.5):} Apply transformations in this order: \\[4pt]
\textbf{Step 1:} Horizontal dilation/reflection ($b$) \quad
\textbf{Step 2:} Horizontal shift ($h$) \\
\textbf{Step 3:} Vertical dilation/reflection ($a$) \quad
\textbf{Step 4:} Vertical shift ($k$)

\medskip
\textbf{Example:} Describe all transformations applied to $f$ to produce
$g(x) = -2f(x+1) - 3$.

\medskip
\renewcommand{\arraystretch}{1.5}
\begin{tabular}{|l|l|}
\hline
\textbf{Parameter} & \textbf{Transformation} \\
\hline
$a = -2$  & \ans{Vertical stretch by factor 2; reflect over $x$-axis} \\
\hline
$h = 1$   & \ans{Horizontal shift left 1 unit} \\
\hline
$k = -3$  & \ans{Vertical shift down 3 units} \\
\hline
\end{tabular}

\medskip
\textbf{Effect on domain/range (EK 1.12.A.6):}
If the domain of $f$ is $[-2,\,4]$ and the range of $f$ is $[0,\,9]$, then:

\medskip
Domain of $g$: \ans{$[-3,\,3]$} \quad Range of $g$: \ans{$[-21,\,-3]$}

\begin{teachernote}
Domain shifts left 1: $[-2-1,\,4-1]=[-3,\,3]$. \\
Range: multiply by $-2$ gives $[-18,\,0]$, then shift down 3 gives $[-21,\,-3]$.
Note the interval flips because multiplying by $-2<0$ reverses the inequality.
\end{teachernote}
\end{notesbox}
