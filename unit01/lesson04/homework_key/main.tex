\begin{enumerate}[itemsep=18pt]

\item For each transformation rule below, identify the type of transformation (vertical
      translation, horizontal translation, vertical dilation, or horizontal dilation/reflection)
      and describe its effect on the graph of $f$.

      \begin{enumerate}[label=(\alph*), itemsep=10pt]
        \item $g(x) = f(x) - 5$

              Type: \ans{vertical translation} \quad Effect: \ans{shift down 5 units}

        \item $g(x) = f(x - 3)$

              Type: \ans{horizontal translation} \quad Effect: \ans{shift right 3 units}

        \item $g(x) = \dfrac{1}{3}f(x)$

              Type: \ans{vertical dilation} \quad Effect: \ans{vertical compression by factor $\frac{1}{3}$}

        \item $g(x) = f(-x)$

              Type: \ans{horizontal reflection} \quad Effect: \ans{reflect over the $y$-axis}
      \end{enumerate}

\item The table below gives values of $f$.

      \medskip
      \renewcommand{\arraystretch}{1.3}
      \begin{tabular}{|c|c|c|c|c|c|}
      \hline
      $x$    & $-2$ & $-1$ & $0$ & $1$ & $2$ \\
      \hline
      $f(x)$ & $8$  & $2$  & $0$ & $2$ & $8$ \\
      \hline
      \end{tabular}

      \medskip
      Complete a table of values for each transformed function.

      \medskip
      \textbf{(a)} $g(x) = f(x) + 4$

      \smallskip
      \renewcommand{\arraystretch}{1.3}
      \begin{tabular}{|c|c|c|c|c|c|}
      \hline
      $x$    & $-2$ & $-1$ & $0$ & $1$ & $2$ \\
      \hline
      $g(x)$ & \ans{$12$} & \ans{$6$} & \ans{$4$} & \ans{$6$} & \ans{$12$} \\
      \hline
      \end{tabular}

      \medskip
      \textbf{(b)} $h(x) = -f(x)$

      \smallskip
      \renewcommand{\arraystretch}{1.3}
      \begin{tabular}{|c|c|c|c|c|c|}
      \hline
      $x$    & $-2$ & $-1$ & $0$ & $1$ & $2$ \\
      \hline
      $h(x)$ & \ans{$-8$} & \ans{$-2$} & \ans{$0$} & \ans{$-2$} & \ans{$-8$} \\
      \hline
      \end{tabular}

\item Let $f$ have domain $[1,\,5]$ and range $[-3,\,6]$. For each transformed function,
      state the domain and range.

      \begin{enumerate}[label=(\alph*), itemsep=10pt]
        \item $g(x) = f(x) + 2$ \quad Domain: \ans{$[1,5]$} Range: \ans{$[-1,8]$}
        \item $g(x) = f(x + 4)$ \quad Domain: \ans{$[-3,1]$} Range: \ans{$[-3,6]$}
        \item $g(x) = 3f(x)$    \quad Domain: \ans{$[1,5]$} Range: \ans{$[-9,18]$}
        \item $g(x) = f(x-1)-2$ \quad Domain: \ans{$[2,6]$} Range: \ans{$[-5,4]$}
      \end{enumerate}

\item Describe all transformations applied to $f$ to obtain $g(x) = -3f(x+2) + 1$.
      List them in order of application.

      \vspace{0.08in}
      \ansline{Step 1: Horizontal shift left 2 units ($h=2>0$). Step 2: Vertical stretch by factor 3 and reflect over the $x$-axis ($a=-3$). Step 3: Vertical shift up 1 unit ($k=1$).}

\item \textbf{(AP-Style Multiple Choice)} Let $f$ be a function with domain $[-4,\,4]$ and
      range $[0,\,8]$. Which of the following gives the range of $g(x) = -2f(x) + 1$?

      \medskip
      \begin{enumerate}[label=(\Alph*), itemsep=4pt]
        \item \ans{$[-15,\,1]$} \textcolor{keyred}{\textbf{$\leftarrow$ correct}}
        \item $[-16,\,0]$
        \item $[1,\,17]$
        \item $[-15,\,-1]$
      \end{enumerate}

      \vspace{0.06in}
      Answer: \ans{A} \quad Show your reasoning:

      \vspace{0.08in}
      \ansline{Range of $f$ is $[0,8]$. Multiply by $-2$: $[0,8]\to[-16,0]$ (reversed). Add 1: $[-16+1,0+1]=[-15,1]$.}

\end{enumerate}

\begin{reflectionbox}
What is the trickiest part of horizontal transformations? Why does $f(x+h)$ shift
\textit{opposite} to the sign of $h$?

\vspace{0.06in}
\ansline{Because $f(x+h)$ evaluates $f$ at $x+h$, so the output at input $x$ equals what $f$ outputs at $x+h$.
To get $f(0)$, we need $x+h=0$, so $x=-h$. The graph shifts to where the input gives zero of the inner expression.}
\end{reflectionbox}
