\begin{enumerate}[itemsep=10pt]

\item From Lesson 1.1: let $f(x) = x^2 - 4$. Evaluate each value.

\begin{work}
  f(3) &= (3)^2 - 4 \\
       &= 9 - 4 \\
       &= 5 \\
  f(-1) &= (-1)^2 - 4 \\
        &= 1 - 4 \\
        &= -3
\end{work}

$f(3) = $ \ans{5} \qquad $f(-1) = $ \ans{$-3$}

\item From Lesson 1.1: the table gives every value of a function $g$.

\smallskip
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{|c|c|c|c|c|c|}
\hline
$x$ & $-2$ & $-1$ & 0 & 1 & 2 \\
\hline
$g(x)$ & 5 & 2 & 1 & 2 & 5 \\
\hline
\end{tabular}
\smallskip

\begin{enumerate}[label=(\alph*), itemsep=4pt]
  \item Domain of $g$: \ans{$\{-2, -1, 0, 1, 2\}$}
  \item Range of $g$: \ans{$\{1, 2, 5\}$}
\end{enumerate}
