\item Let $g(x) = x^2 - 4x$. Find the average rate of change over each
      interval.

\begin{enumerate}[label=(\alph*), itemsep=6pt]
  \item Over $[1, 4]$:

        \begin{work}
          g(1) &= (1)^2 - 4(1) \\
               &= -3 \\
          g(4) &= (4)^2 - 4(4) \\
               &= 0 \\
          \text{AROC} &= \frac{0 - (-3)}{4 - 1} \\
                      &= 1
        \end{work}

        AROC $= $ \blank{1.8cm}

  \item Over $[0, 2]$:

        \begin{work}
          g(0) &= 0 \\
          g(2) &= (2)^2 - 4(2) \\
               &= -4 \\
          \text{AROC} &= \frac{-4 - 0}{2 - 0} \\
                      &= -2
        \end{work}

        AROC $= $ \blank{1.8cm}
\end{enumerate}

AROCs over $[0,1], [1,2], [2,3], [3,4]$: \quad
\blank{1cm} , \blank{1cm} , \blank{1cm} , \blank{1cm}

\begin{enumerate}[label=(\alph*), itemsep=4pt]
  \item Each AROC changes by \blank{1.2cm} --- so $q$ is
