\item Let $f(x) = x^2 - 4x$. Find the average rate of change over each
      interval. Watch the signs --- they do not come out the same.

\begin{enumerate}[label=(\alph*), itemsep=6pt]
  \item Over $[0, 2]$:

        \begin{work}
          \text{AROC} &= \frac{f(2) - f(0)}{2 - 0} \\
                      &= \frac{-4 - 0}{2} \\
                      &= -2
        \end{work}

        Answer: \blank{2.4cm}

  \item Over $[2, 5]$:

        \begin{work}
          \text{AROC} &= \frac{f(5) - f(2)}{5 - 2} \\
                      &= \frac{5 - (-4)}{3} \\
                      &= 3
        \end{work}

        Answer: \blank{2.4cm}
\end{enumerate}
