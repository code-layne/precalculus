\item Find the slope of the line through $(2, 5)$ and $(6, 17)$.

\begin{work}
  m &= \frac{17 - 5}{6 - 2} \\
    &= \frac{12}{4} \\
    &= 3
\end{work}

$m = $ \ans{3}

\item Let $f(x) = x^2 + 3$. Find $f(1)$ and $f(4)$.

\begin{work}
  f(1) &= (1)^2 + 3 \\
       &= 4 \\
  f(4) &= (4)^2 + 3 \\
       &= 19
\end{work}

$f(1) = $ \ans{4} \qquad $f(4) = $ \ans{19}
