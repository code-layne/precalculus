\begin{enumerate}[leftmargin=1.5em, itemsep=10pt]

\item Given $f(x) = 3x - 2$, compute the average rate of change on each interval below.
What do you notice?

\smallskip
\begin{tabular}{|l|c|}
\hline
\textbf{Interval} & \textbf{AROC} \\
\hline
$[0, 2]$ & \ans{3} \\[4pt]
$[2, 4]$ & \ans{3} \\[4pt]
$[4, 6]$ & \ans{3} \\
\hline
\end{tabular}

\vspace{0.06in}
What do you notice? \ansline{The AROC is always 3 --- it is constant, equal to the slope of the linear function.}

\item The table below shows values for $g(x) = x^2$. Compute the AROC for each unit interval.

\smallskip
\begin{tabular}{|c|c|l|}
\hline
\textbf{Interval} & \textbf{Values used} & \textbf{AROC} \\
\hline
$[0, 1]$ & $g(0)=0,\; g(1)=1$    & \ans{1} \\[3pt]
$[1, 2]$ & $g(1)=1,\; g(2)=4$    & \ans{3} \\[3pt]
$[2, 3]$ & $g(2)=4,\; g(3)=9$    & \ans{5} \\[3pt]
$[3, 4]$ & $g(3)=9,\; g(4)=16$   & \ans{7} \\
\hline
\end{tabular}

\item The slope of the secant line from $(1,\, f(1))$ to $(4,\, f(4))$ for some function $f$ is 5.
What is $f(4) - f(1)$? Show your reasoning.

\vspace{0.1in}
The AROC formula is $\dfrac{f(4)-f(1)}{4-1} = 5$, so $f(4)-f(1) = 5 \times 3 = $
\ans{15}

\end{enumerate}
