\begin{enumerate}[itemsep=10pt]

\item Find the slope of the line through the points $(3, 5)$ and $(7, 17)$.

\begin{work}
  m &= \frac{17 - 5}{7 - 3} \\
    &= \frac{12}{4} \\
    &= 3
\end{work}

$m = $ \ans{3}

\item From Lesson 1.1: let $f(x) = x^2 - 2x$. Find $f(4)$.

\begin{work}
  f(4) &= (4)^2 - 2(4) \\
       &= 16 - 8 \\
       &= 8
\end{work}

$f(4) = $ \ans{8}
