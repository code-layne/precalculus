\begin{enumerate}[leftmargin=1.5em, itemsep=14pt]

\item For $f(x) = -2x + 5$, compute the AROC on each interval below.
Are the AROCs constant? What is the slope of $f$?

\smallskip
\begin{tabular}{|l|c|}
\hline
\textbf{Interval} & \textbf{AROC} \\
\hline
$[0, 1]$ & \ans{$-2$} \\
$[1, 2]$ & \ans{$-2$} \\
$[2, 3]$ & \ans{$-2$} \\
\hline
\end{tabular}

\vspace{0.05in}
Are the AROCs constant? \ans{Yes} \quad Slope of $f$: \ans{$-2$}

\vspace{0.1in}

\item For $g(x) = x^2 - 3x$, compute the AROC on $[0,1]$, $[1,2]$, and $[2,3]$. Then compute the second differences.

\smallskip
Values: $g(0)=0$, $g(1)=-2$, $g(2)=-2$, $g(3)=0$.

\begin{tabular}{|l|c|c|}
\hline
\textbf{Interval} & \textbf{AROC} & \textbf{Second Difference} \\
\hline
$[0, 1]$ & \ans{$-2$} & \\
$[1, 2]$ & \ans{$0$}  & \ans{$2$} \\
$[2, 3]$ & \ans{$2$}  & \ans{$2$} \\
\hline
\end{tabular}

\vspace{0.1in}

\item The table of values is shown below.

\smallskip
\begin{tabular}{|c|c|c|c|c|c|}
\hline
$x$ & 0 & 1 & 2 & 3 & 4 \\
\hline
$y$ & 3 & 7 & 13 & 21 & 31 \\
\hline
\end{tabular}

\medskip
First differences: $4, 6, 8, 10$. Second differences: $2, 2, 2$.

\ansline{The second differences are constant and nonzero, so the data represent a quadratic function.}

\item The AROC of a function is increasing as $x$ increases. Is the graph concave up or concave down?

\ansline{Concave up. When successive AROCs (first differences) increase, the graph bends upward like a bowl.}

\item \textbf{AP-style:}

\smallskip
\begin{tabular}{|c|c|c|c|c|}
\hline
$x$    & 1 & 2 & 3 & 4 \\
\hline
$f(x)$ & 5 & 8 & 11 & 14 \\
\hline
\end{tabular}

\medskip
First differences: $3, 3, 3$ --- constant. Second differences: $0, 0$.

\begin{enumerate}[label=(\Alph*), itemsep=3pt]
  \item $f$ is quadratic because the first differences are constant.
  \item $f$ is linear because the first differences are constant. \textcolor{keyred}{\textbf{$\leftarrow$ correct}}
  \item $f$ is quadratic because the second differences are constant.
  \item $f$ is neither linear nor quadratic because the second differences are not zero.
\end{enumerate}

\begin{teachernote}
(A) is a common distractor --- constant first differences mean \textit{linear}, not quadratic.
(C) is also a distractor: second differences of zero are consistent with linear, not quadratic.
\end{teachernote}

\end{enumerate}

\vspace{0.15in}

\begin{extensionbox}
For $f(x) = x^3$:

\smallskip
\begin{tabular}{|l|c|c|c|c|}
\hline
$x$          & 0 & 1 & 2 & 3 \\
\hline
$f(x)$       & 0 & 1 & 8 & 27 \\
\hline
1st diff.\ (AROC) & & \ans{1} & \ans{7} & \ans{19} \\
\hline
2nd diff.    &   & & \ans{6} & \ans{12} \\
\hline
3rd diff.    &   & & & \ans{6} \\
\hline
\end{tabular}

\medskip
\ansline{The third differences are constant (equal to 6). The second differences form a linear pattern. This confirms that it takes 3 rounds of differencing to reach a constant for a cubic function --- consistent with its degree.}
\end{extensionbox}
