\begin{notesbox}{Practice --- Key}

\textbf{1.} Table of $f$:
\quad $f(0)=6,\ f(2)=14,\ f(5)=26,\ f(8)=38,\ f(12)=50$.

\begin{enumerate}[label=(\alph*), itemsep=4pt]
    \item AROC over $[0,5]$: $\dfrac{26-6}{5-0} = \dfrac{20}{5} =$ \ans{4}
    \item AROC over $[5,12]$: $\dfrac{50-26}{12-5} = \dfrac{24}{7} \approx$ \ans{3.43}
    \item \ans{$[0,5]$ has the greater AROC (4 vs.\ $\approx$3.43).
          The function grew faster on the first interval.}
\end{enumerate}

\vspace{6pt}

\textbf{2.} Water tank draining from 500 gal to 320 gal in 2 hours.

\begin{enumerate}[label=(\alph*), itemsep=4pt]
    \item AROC $= \dfrac{320-500}{2} = \dfrac{-180}{2} =$ \ans{$-90$ gallons per hour}
    \item \ans{The negative sign means the water volume is \emph{decreasing} — as time
          passes, the tank loses water at an average rate of 90 gallons per hour.}
\end{enumerate}

\vspace{6pt}

\textbf{3.} $p(3)=41$, $p(3.1)=42.05$, $p(3.01)=41.1005$.

\begin{enumerate}[label=(\alph*), itemsep=4pt]
    \item AROC over $[3,3.1]$: $\dfrac{42.05-41}{0.1} = \dfrac{1.05}{0.1} =$ \ans{10.5}
    \item AROC over $[3,3.01]$: $\dfrac{41.1005-41}{0.01} = \dfrac{0.1005}{0.01} =$ \ans{10.05}
    \item \ans{Approximately 10} (the AROCs are converging toward 10 as the interval shrinks).
\end{enumerate}

\vspace{6pt}

\textbf{4.} \ans{\textbf{False.}} A negative AROC means the net change in output is
negative over the interval — but the function could increase on part of the interval
and decrease on another part, as long as the overall change is negative. For example,
$f$ could rise then fall sharply, giving a negative net change but not being everywhere
decreasing.

\vspace{6pt}

\textbf{5.} AROC of $q$ over $[1,9]$: $\dfrac{q(9)-q(1)}{9-1} = \dfrac{10-8}{8} = \dfrac{2}{8} = 0.25$.

Correct answer: \ans{(A) --- the AROC is $0.25$ and is positive.}

\begin{teachernote}
(B) has the subtraction reversed. (C) confuses AROC with a simple difference. (D) is
false — AROC is fully determined by the endpoint values.
\end{teachernote}

\end{notesbox}

\vspace{0.1in}

\begin{extensionbox}
\textbf{Extension (optional) --- Key:}

\[
    \text{AROC} = \frac{(2+h)^2 - 4}{h}
    = \frac{4h + h^2}{h}
    = \ans{4 + h}
\]

As $h \to 0$, the AROC approaches \ans{4}. This is the instantaneous rate of change (derivative)
of $f(x)=x^2$ at $x=2$.
\end{extensionbox}
