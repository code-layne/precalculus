\begin{notesbox}{AROC of Linear Functions}
Use $f(x) = 2x + 1$. Complete the table.

\smallskip
\renewcommand{\arraystretch}{1.5}
\begin{tabular}{|c|c|c|c|}
\hline
\textbf{Interval} & $f(\text{left})$ & $f(\text{right})$ & \textbf{AROC} \\
\hline
$[0, 1]$ & $f(0) = 1$ & $f(1) = 3$ & \blank{2cm} \\
$[1, 2]$ & $f(1) = 3$ & $f(2) = 5$ & \blank{2cm} \\
$[2, 3]$ & $f(2) = 5$ & $f(3) = 7$ & \blank{2cm} \\
$[3, 4]$ & $f(3) = 7$ & $f(4) = 9$ & \blank{2cm} \\
\hline
\end{tabular}
\renewcommand{\arraystretch}{1}

\smallskip
\textbf{Conclusion:} For a linear function, the AROC is \blank{4cm}.

This is because the slope of any secant line on a linear graph equals \blank{4cm}.
\end{notesbox}

\begin{notesbox}{Secant Lines \& Concavity}
\textbf{Secant line:} The line connecting two points $(a, f(a))$ and $(b, f(b))$ on a graph.

Its slope $= \dfrac{f(b)-f(a)}{b-a} =$ \blank{3cm}

\medskip
\textbf{Concavity from AROC pattern:}
\begin{itemize}[leftmargin=1.5em, itemsep=4pt]
  \item If the AROCs are \textit{increasing} as $x$ increases, the graph is \blank{3cm}.
  \item If the AROCs are \textit{decreasing} as $x$ increases, the graph is \blank{3cm}.
\end{itemize}

\medskip
\textit{Example:} For $f(x) = x^2$, the AROCs over unit intervals are $1, 3, 5, 7, \ldots$

These are \blank{2cm} (increasing / decreasing), so the graph is \blank{3cm}.
\end{notesbox}
