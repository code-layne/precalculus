\begin{minipage}[t]{0.48\linewidth}\vspace{0pt}
    \textbf{\color{navy}Key Understandings}
    \begin{itemize}
        \item AROC $= \dfrac{\Delta y}{\Delta x} = \dfrac{f(b)-f(a)}{b-a}$ is the constant
              rate that produces the same net change over $[a,b]$.
        \item The rate of change \emph{at a point} can be approximated by computing AROC
              over successively smaller intervals containing that point.
        \item A \emph{positive} AROC means the output increases as the input increases;
              a \emph{negative} AROC means the output decreases as the input increases.
    \end{itemize}
\end{minipage}

\begin{skillbox}[{Vocabulary, Concepts, \& Theorems}]{greenbox}
\begin{multicols}{2}
\begin{itemize}
    \item \textbf{average rate of change (AROC)} — the ratio $\dfrac{f(b)-f(a)}{b-a}$
          representing the constant rate that yields the same change in output over
          the interval $[a,b]$.
    \item \textbf{secant line} — the line through $(a,f(a))$ and $(b,f(b))$; its slope
          equals the AROC over $[a,b]$.
    \item \textbf{rate of change at a point} — the instantaneous rate of change, approximated
          by AROC over small intervals containing the point.
    \item \textbf{positive / negative rate of change} — a positive ROC means the two
          quantities change in the \emph{same} direction; a negative ROC means they change
          in \emph{opposite} directions.
\end{itemize}
\end{multicols}
\end{skillbox}

\begin{skillbox}[Hook]{sky}
    \textbf{Scenario (2 min):} ``You're driving on the highway. Your GPS says your average
    speed over the last hour was 62 mph. Were you always going exactly 62 mph? What does
    `average' mean here?''

    \medskip
    Facilitate a brief discussion: students will likely note that speed varied — sometimes
    faster, sometimes slower. The average speed is the \emph{one constant speed} that would
    cover the same distance in the same time. This is exactly what AROC captures.
    Transition: ``Today we'll make this idea precise and apply it to any function.''
\end{skillbox}

\begin{skillbox}[Lesson]{sky}
    \textbf{Part 1 — Derive and define AROC (10 min)}

    Write the slope formula on the board; ask: ``What if the two points lie on a curve, not
    a line?'' Build AROC $= \dfrac{f(b)-f(a)}{b-a} = \dfrac{\Delta f}{\Delta x}$.
    Label: numerator $=$ change in output, denominator $=$ change in input. Emphasize that
    AROC equals the \emph{slope of the secant line} connecting the two points.

    \medskip\textbf{Part 2 — Compute AROC from tables and graphs (12 min)}

    \begin{multicols}{2}
    Display a table of population data (as in the Group Activity). Model: compute AROC
    over 1990--2000, then 2000--2010. Ask: ``Which decade saw faster growth? How do you
    know?'' Have students compute AROC over 2010--2020 individually. Cold-call to share.

    \columnbreak
    Then show a graph of a function. Ask students to estimate AROC between two labeled
    points by reading coordinates. Discuss how a steeper secant line corresponds to a larger
    (in absolute value) AROC.
    \end{multicols}

    \textbf{Part 3 — Approximate ROC at a point (10 min)}

    Present the question: ``What is the rate of change of the population \emph{at} 2010,
    not just over an interval?'' Build a table of AROC over shrinking intervals:
    $[2005,2010]$, $[2008,2010]$, $[2009,2010]$, $[2009.5,2010]$. Students fill in the
    table from the Guided Notes. Observe: AROC approaches a limiting value. That limiting
    value approximates the ROC at a point. Connect to vocabulary: rate of change at a
    point, secant lines approaching a tangent.

    \medskip\textbf{Part 4 — Sign of rate of change (8 min)}

    Discuss: if AROC is positive, what does that tell you about the two quantities? Negative?
    Use real-world examples: temperature rising (positive), elevation decreasing as you walk
    downhill (negative). Students fill in the sign-of-ROC section of the notes.
\end{skillbox}

\begin{skillbox}[Active Monitoring]{redbox}
    \textbf{During Part 1:} Are students correctly identifying $a$, $b$, $f(a)$, $f(b)$
    from a table? Watch for subtraction-order errors ($f(b)-f(a)$ in numerator vs.\
    $b-a$ in denominator).

    \medskip\textbf{During Part 2:} Do students recognize that ``greater AROC'' means
    steeper secant line? Ask: ``What would a zero AROC tell you about the function?''

    \medskip\textbf{During Part 3:} Are students seeing the pattern of the AROC values
    converging? Ask: ``Why do smaller intervals give a better approximation?''

    \medskip\textbf{Cold-call prompts:}
    \begin{itemize}
        \item ``Give me the units of this AROC. What do those units mean in context?''
        \item ``If AROC is negative, what does that say about the shape of the function
              on that interval?''
    \end{itemize}
\end{skillbox}

\begin{skillbox}[Group Work \& Differentiation]{redbox}
    Students work in groups on the Group Activity (city population context).

    \textbf{Tier R (Remediate):} Compute AROC over two given intervals; determine which
    interval has greater growth rate. Scaffolded formula with blanks.

    \textbf{Tier A (Apply):} Compute AROC over multiple decades, compare growth rates,
    interpret negative AROC; identify the decade of fastest growth.

    \textbf{Tier E (Extend):} Approximate instantaneous ROC at a given year using
    successively smaller intervals; reflect on the limitations of the approximation.

    \medskip\textbf{Support:} Remind Tier R groups that AROC $= \dfrac{f(b)-f(a)}{b-a}$
    is just slope with function notation. Let them label $a, b, f(a), f(b)$ before computing.

    \textbf{Challenge:} Ask Tier E groups: ``What happens if you try to use an interval
    of length 0? What mathematical operation does that resemble?'' (Preview of limits.)
\end{skillbox}

\begin{skillbox}[Individual Work \& Assessment]{redbox}
    \textbf{Exit Ticket (5 min):} Two questions.
    \begin{enumerate}
        \item Given $f(2)=10$, $f(5)=25$, $f(5.1)=25.3$: compute AROC from $x=2$ to
              $x=5$; then approximate the ROC at $x=5$ using the interval $[5,5.1]$.
        \item AROC of $g$ from $x=0$ to $x=3$ is $-4$. Describe what this tells you
              about how $g$'s output changes.
    \end{enumerate}
    \textbf{Look for:} correct formula application, correct arithmetic, and whether
    students can articulate the meaning of a negative AROC in words.
\end{skillbox}

\begin{skillbox}[Reinforcement \& Extension]{goldbox}
    \textbf{Homework:} 5 problems covering AROC from tables and context, approximating
    ROC at a point, and a True/False interpretation question. An AP-style MC problem
    comparing rates of change from a table is included.

    \textbf{Extension:} For $f(x)=x^2$, compute AROC over $[2,2+h]$ and let $h\to 0$.
    What value does the AROC approach? (Connects to the derivative at $x=2$.)

    \textbf{Preview:} Next lesson (1.3) examines how AROC behaves differently for linear
    vs.\ quadratic functions — specifically, what ``constant rate of change'' and
    ``changing rate of change'' mean geometrically and numerically.
\end{skillbox}
