\begin{frame}[t, plain]
  \plumheader{Hook: The Road-Trip Average}
  \sectionlabel[goldacc]{Think About It}

  A family drives 120 miles in 3 hours. The GPS reports:

  \bigskip
  \begin{center}
    {\LARGE \textbf{average speed 40 mph}}
  \end{center}

  \bigskip
  \begin{itemize}\setlength{\itemsep}{8pt}
    \item Were they going exactly 40 the whole time?
    \item So what does the 40 \textbf{promise}?
  \end{itemize}

  \bigskip
  \textit{One steady 40 mph would cover the same 120 miles in the same
  3 hours --- that promise is today's definition.}
\end{frame}

% SECTION 1 — AROC DEFINITION
\begin{frame}[t, plain]
  \plumheader{1. Average Rate of Change --- Slope, Upgraded}

  \begin{block}{The definition}
    \[
      \text{AROC over } [a, b]
      \;=\; \frac{\text{change in output}}{\text{change in input}}
      \;=\; \frac{f(b) - f(a)}{b - a}
    \]
    The slope formula, wearing function notation --- and it works on
    \textbf{any} function, not just lines.
  \end{block}

  \medskip
  \begin{block}{Example: \(f(x) = x^2\) over \([1, 4]\)}
    \(f(1) = 1\), \quad \(f(4) = 16\)
    \[
      \text{AROC} = \frac{16 - 1}{4 - 1} = \frac{15}{3} = 5
    \]
  \end{block}

  \medskip
  The line through \((1,1)\) and \((4,16)\) is the \textbf{secant line};
  the AROC is its \textbf{slope}.
\end{frame}

% SECTION 2 — FROM A TABLE
\begin{frame}[t, plain]
  \plumheader{2. AROC from a Table}

  \(T(t)\) is the temperature in \(^\circ\)F \(t\) hours after sunrise.

  \bigskip
  \begin{tabular}{|c|c|c|c|c|}
  \hline
  \(t\) (hours) & 0 & 2 & 4 & 6 \\
  \hline
  \(T(t)\) (\(^\circ\)F) & 40 & 52 & 58 & 54 \\
  \hline
  \end{tabular}

  \bigskip
  \begin{itemize}\setlength{\itemsep}{8pt}
    \item AROC over \([0,2]\): \((52-40)/2 = 6\) \(^\circ\)F per hour
    \item AROC over \([2,4]\): \((58-52)/2 = 3\) \(^\circ\)F per hour
    \item AROC over \([4,6]\): \((54-58)/2 = -2\) \(^\circ\)F per hour
  \end{itemize}

  \bigskip
  \begin{block}{Watch the order}
    Output difference upstairs, input difference downstairs ---
    \textbf{subtracted in the same order on both floors}.
  \end{block}
\end{frame}

% SECTION 3 — SIGN AND UNITS
\begin{frame}[t, plain]
  \plumheader{3. What the Sign Says}

  \begin{block}{Sign = direction}
    \textbf{Positive} AROC: the quantities move in the \textbf{same}
    direction. \quad
    \textbf{Negative} AROC: as one increases, the other \textbf{decreases}.
  \end{block}

  \medskip
  \begin{itemize}\setlength{\itemsep}{8pt}
    \item The \(-2\) from the table: temperature \textbf{falling} 2 degrees
          per hour.
    \item Zero AROC: no \textbf{net} change --- careful, it does not mean
          nothing happened.
    \item Every rate carries \textbf{units}: output units per input unit
          (\(^\circ\)F per hour, feet per second, people per year).
  \end{itemize}

  \bigskip
  \textit{From now on, an answer without units is half an answer.}
\end{frame}

% SECTION 4 — LINEAR VS QUADRATIC
\begin{frame}[t, plain]
  \plumheader{4. The Pattern of the AROCs}

  Compute the AROC over each unit interval --- then watch the
  \textbf{pattern}.

  \medskip
  \begin{columns}[T]
    \column{0.48\textwidth}
      \begin{block}{Linear: \(f(x) = 2x + 1\)}
        Outputs: 1, 3, 5, 7, 9\\[4pt]
        AROCs: \(2,\ 2,\ 2,\ 2\)\\[4pt]
        The AROC is \textbf{constant} --- that is what one fixed
        ``slope'' means.
      \end{block}

    \column{0.48\textwidth}
      \begin{block}{Quadratic: \(g(x) = x^2\)}
        Outputs: 0, 1, 4, 9, 16\\[4pt]
        AROCs: \(1,\ 3,\ 5,\ 7\)\\[4pt]
        Not constant --- but each step climbs by exactly \textbf{2}:
        the AROCs change at a \textbf{constant rate}.
      \end{block}
  \end{columns}

  \bigskip
  \begin{block}{The concavity link (from 1.2)}
    AROCs \textbf{increasing} \(\Rightarrow\) concave \textbf{up}. \quad
    AROCs \textbf{decreasing} \(\Rightarrow\) concave \textbf{down}.
  \end{block}
\end{frame}

% CLASS PRACTICE
\begin{frame}[t, plain]
  \plumheader{Class Practice}
  \sectionlabel[redacc]{Try It}

  \begin{enumerate}\setlength{\itemsep}{10pt}
    \item Let \(h(x) = x^2 - 3x\). Find the AROC over \([2, 5]\).
    \item A function's AROCs over consecutive equal-length intervals are
          \(10,\ 7,\ 4,\ 1\).
          \begin{enumerate}\setlength{\itemsep}{5pt}
            \item By how much does each AROC change? Is the function linear
                  or quadratic?
            \item Is its graph concave up or concave down?
          \end{enumerate}
    \item A pool drains so that the AROC of the water volume is \(-50\)
          gallons per minute over \([0, 10]\). Say what that means in one
          sentence --- with units.
  \end{enumerate}
\end{frame}
