\begin{frame}[t, plain]
  \plumheader{Hook: Sixty Steps, Thirty Seconds}
  \sectionlabel[goldacc]{Vote Before You Compute}

  Ravi and Mira start at the bottom of the same 60-step stairway at the same
  moment. \textbf{Both} reach the top 30 seconds later.

  \smallskip
  \begin{columns}[T]
    \column{0.54\textwidth}
      \begin{center}
      \textbf{Ravi}

      \smallskip
      \begin{tabular}{|c|c|c|c|c|}
      \hline
      time (s) & 0 & 10 & 20 & 30 \\
      \hline
      steps & 0 & 40 & 55 & 60 \\
      \hline
      \end{tabular}

      \medskip
      \textbf{Mira}

      \smallskip
      \begin{tabular}{|c|c|c|c|c|}
      \hline
      time (s) & 0 & 10 & 20 & 30 \\
      \hline
      steps & 0 & 20 & 40 & 60 \\
      \hline
      \end{tabular}
      \end{center}

    \column{0.42\textwidth}
      \begin{block}{The question}
        Who climbed \textbf{faster}? Commit to an answer.
      \end{block}

      \medskip
      \begin{block}{Then compute the average}
        60 steps in 30 seconds --- for \textbf{both}. Two steps per second,
        each. The average calls them identical.
      \end{block}
  \end{columns}
\end{frame}

% SECTION 1 — THE STAND-IN
\begin{frame}[t, plain]
  \plumheader{1. The Stand-In}
  \sectionlabel[redacc]{The Whole Lesson in One Idea}

  \begin{block}{Average rate of change}
    The one \textbf{steady} rate that would produce the same change in output
    over the same change in input.
    \[
      \frac{f(b) - f(a)}{b - a}
      \;=\; \frac{\text{change in output}}{\text{change in input}}
    \]
  \end{block}

  \medskip
  \begin{columns}[T]
    \column{0.48\textwidth}
      \textbf{What it promises.} A steady 2-steps-per-second climber leaves at
      the same time, from the same step, and arrives at the same step 30
      seconds later.

    \column{0.48\textwidth}
      \textbf{What it costs.} The stand-in is pinned only at the
      \textbf{endpoints}. Everything that happened between them is thrown
      away --- on purpose.
  \end{columns}
\end{frame}

% SECTION 2A — FORMULA AND TABLE
\begin{frame}[t, plain]
  \plumheader{2. Computing It --- From a Formula, From a Table}

  \begin{columns}[T]
    \column{0.44\textwidth}
      \textbf{From a formula.} \(f(x) = x^2 + 3\) on \([1,4]\):
      \[
        \frac{f(4) - f(1)}{4 - 1} = \frac{19 - 4}{3} = 5
      \]

      \medskip
      \begin{block}{Order matters}
        Whichever input you subtract first \textit{downstairs}, its output
        must be subtracted first \textit{upstairs}.
      \end{block}

    \column{0.52\textwidth}
      \textbf{From a table.} \(T(t)\) in \(^\circ\)F, \(t\) hours after 6 a.m.

      \smallskip
      \begin{tabular}{|c|c|c|c|c|c|}
      \hline
      \(t\) & 0 & 2 & 4 & 6 & 8 \\
      \hline
      \(T(t)\) & 52 & 58 & 68 & 74 & 72 \\
      \hline
      AROC & --- & 3 & 5 & 3 & \(-1\) \\
      \hline
      \end{tabular}

      \medskip
      Fastest warming: between \(t = 2\) and \(t = 4\), at \textbf{5 degrees
      per hour}. And every rate carries \textbf{units}.
  \end{columns}
\end{frame}

% SECTION 2B — GRAPH
\begin{frame}[t, plain]
  \plumheader{2. Computing It --- From a Graph}
  \sectionlabel[goldacc]{The Secant Line}

  \begin{columns}[T, onlytextwidth]
    \column{0.50\textwidth}
      \begin{center}
      \begin{tikzpicture}[x=0.60cm, y=0.40cm]
        \draw[linegray, very thin, step=1] (0,0) grid (8,6);
        \draw[->, thick] (0,0) -- (8.6,0);
        \draw[->, thick] (0,0) -- (0,6.6) node[above] {\small \(g(t)\) (m)};
        \node[font=\small] at (4.3,-1.6) {\(t\) (min)};
        \foreach \x in {2,4,6,8} \node[below, font=\tiny] at (\x,0) {\x};
        \foreach \y in {1,3,5} \node[left, font=\tiny] at (0,\y) {\y};
        \draw[dashed, thick, goldacc] (0,1) -- (4,5);
        \draw[dashed, thick, goldacc] (4,5) -- (8,1);
        \draw[very thick, plum] plot[smooth] coordinates
          {(0,1) (1,2.6) (2,3.9) (3,4.7) (4,5) (5,4.7) (6,3.9) (7,2.6) (8,1)};
      \end{tikzpicture}
      \end{center}

    \column{0.42\textwidth}
      A kite's height \(g(t)\), in meters.

      \medskip
      \begin{itemize}\setlength{\itemsep}{6pt}
        \item Over \([0,4]\): from 1 m to 5 m, so \(+1\) \textbf{meter per
              minute}.
        \item Over \([4,8]\): \(-1\) \textbf{meter per minute}.
      \end{itemize}

      \medskip
      \begin{block}{The punch line}
        Each dashed line is a \textbf{secant line}. Its \textbf{slope}
        \textit{is} the average rate of change over that interval.
      \end{block}
  \end{columns}
\end{frame}

% SECTION 3 — SIGN AND ZERO
\begin{frame}[t, plain]
  \plumheader{3. What the Sign Says --- and What Zero Does Not Say}

  \begin{columns}[T]
    \column{0.46\textwidth}
      \begin{itemize}\setlength{\itemsep}{7pt}
        \item \textbf{Positive:} as the input increases, so does the output.
              The quantities move the \textbf{same} way.
        \item \textbf{Negative:} as the input increases, the output falls.
              They move \textbf{opposite} ways.
        \item \textbf{Units, every time:} output units \textit{per} input
              unit. ``5'' is not an answer.
      \end{itemize}

    \column{0.50\textwidth}
      \begin{block}{Zero}
        A hiker leaves the trailhead at 900 ft, climbs all afternoon, and is
        back at 900 ft four hours later.
        \[
          \frac{900 - 900}{4 - 0} = 0 \text{ ft per hour}
        \]
      \end{block}

      \medskip
      \textbf{Did nothing happen?} No. The average compares only the start and
      the end --- and they are the same. This is the stand-in's cost, made
      visible.
  \end{columns}
\end{frame}

% SECTION 4 — CHOP IT FINER
\begin{frame}[t, plain]
  \plumheader{4. Chop It Finer}
  \sectionlabel[redacc]{Settling the Vote}

  Same tool, smaller intervals --- and the story the whole-climb average
  erased comes back.

  \medskip
  \begin{center}
  \renewcommand{\arraystretch}{1.4}
  \begin{tabular}{|l|c|c|c|}
  \hline
  \textbf{split (seconds)} & \textbf{0 to 10} & \textbf{10 to 20} & \textbf{20 to 30} \\
  \hline
  Ravi's rate (steps/s) & 4 & 1.5 & 0.5 \\
  \hline
  Mira's rate (steps/s) & 2 & 2 & 2 \\
  \hline
  \end{tabular}
  \end{center}

  \medskip
  \begin{columns}[T]
    \column{0.48\textwidth}
      Ravi's rates are \textbf{shrinking} --- in 1.2's language, his climb is
      \textbf{concave down}.

    \column{0.48\textwidth}
      Mira's rates \textbf{never change}. Her climb is a straight line, and
      she \textit{is} her own stand-in.
  \end{columns}
\end{frame}

% SECTION 5 — THE PATTERN
\begin{frame}[t, plain]
  \plumheader{5. The Pattern Names the Function}
  \sectionlabel[redacc]{Topic 1.3}

  Average rate of change over each unit interval from \(x = 0\) to \(x = 4\):

  \medskip
  \begin{center}
  \renewcommand{\arraystretch}{1.4}
  \begin{tabular}{|l|c|c|c|c|c|}
  \hline
  \(x\) & 0 & 1 & 2 & 3 & 4 \\
  \hline
  \(f(x) = 2x + 1\) & 1 & 3 & 5 & 7 & 9 \\
  \hline
  AROC of \(f\) & --- & 2 & 2 & 2 & 2 \\
  \hline
  \(g(x) = x^2\) & 0 & 1 & 4 & 9 & 16 \\
  \hline
  AROC of \(g\) & --- & 1 & 3 & 5 & 7 \\
  \hline
  change in the AROC of \(g\) & --- & --- & 2 & 2 & 2 \\
  \hline
  \end{tabular}
  \end{center}

  \medskip
  \begin{columns}[T]
    \column{0.32\textwidth}
      \textbf{Linear.} Same rate over every interval --- that number is the
      slope.

    \column{0.32\textwidth}
      \textbf{Quadratic.} The rates march by a constant step; the rates are
      themselves linear.

    \column{0.32\textwidth}
      \textbf{Concavity.} Rates increasing \(\Rightarrow\) concave up;
      decreasing \(\Rightarrow\) concave down.
  \end{columns}
\end{frame}

% CLASS PRACTICE
\begin{frame}[t, plain]
  \plumheader{Class Practice}
  \sectionlabel[redacc]{Try It}

  \begin{enumerate}\setlength{\itemsep}{10pt}
    \item A delivery van's distance \(D\), in miles, after \(t\) hours:

          \smallskip
          \begin{tabular}{|c|c|c|c|c|c|}
          \hline
          \(t\) (hours) & 0 & 1 & 2 & 3 & 4 \\
          \hline
          \(D(t)\) (miles) & 0 & 60 & 100 & 160 & 180 \\
          \hline
          \end{tabular}

          \smallskip
          Find the average rate of change over \([0,2]\) and over \([2,4]\),
          with units. Which stretch was faster, and by how much?

    \item One function's average rates of change are \(7, 7, 7, 7\).
          Another's are \(2, 6, 10, 14\). Which one is linear, and what is its
          slope? What is the concavity of the other --- and how do you know
          without a graph?
  \end{enumerate}
\end{frame}
