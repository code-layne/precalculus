  \hline
  \end{tabular}

  \medskip
  \textbf{Compute:} AROC on $[0,1]$, $[1,2]$, $[2,3]$, $[3,4]$

  \medskip
  \begin{itemize}
    \item What pattern do you see in the AROCs?
    \item What do you predict for the AROC on $[4,5]$?
    \item What are the \textit{differences} of those AROCs?
  \end{itemize}

% LINEAR AROC SLIDE
\begin{frame}[t, plain]
  \navyheader{AROC of Linear Functions}
  \vspace{0.3cm}

  \sectionlabel[goldacc]{Key Idea: Linear $\Rightarrow$ Constant AROC}

  \medskip
  For $f(x) = 2x + 1$:

  \medskip
  \begin{tabular}{|c|c|c|}
  \hline
  \textbf{Interval} & \textbf{AROC} & \textbf{Second Difference} \\
  \hline
  $[0,1]$ & 2 & \\
  $[1,2]$ & 2 & 0 \\
  $[2,3]$ & 2 & 0 \\
  $[3,4]$ & 2 & 0 \\
  \hline
  \end{tabular}

  \medskip
  \begin{itemize}
    \item The AROC equals the slope of the secant line \textit{and} the slope of the function.
    \item Second differences are \textbf{zero} for linear functions.
  \end{itemize}
\end{frame}

% QUADRATIC AROC SLIDE
\begin{frame}[t, plain]
  \navyheader{AROC of Quadratic Functions: First \& Second Differences}
  \vspace{0.3cm}

  \sectionlabel[goldacc]{Key Idea: Quadratic $\Rightarrow$ Constant Second Differences}

  \medskip
  For $f(x) = x^2$:

  \medskip
  \begin{tabular}{|c|c|c|}
  \hline
  \textbf{Interval} & \textbf{AROC (1st diff.)} & \textbf{Change in AROC (2nd diff.)} \\
  \hline
  $[0,1]$ & 1 & \\
  $[1,2]$ & 3 & $+2$ \\
  $[2,3]$ & 5 & $+2$ \\
  $[3,4]$ & 7 & $+2$ \\
  \hline
  \end{tabular}

  \medskip
  \begin{itemize}
    \item First differences form a \textbf{linear} (arithmetic) sequence.
    \item Second differences are \textbf{constant and nonzero} for quadratic functions.
  \end{itemize}
\end{frame}

% SECANT LINES SLIDE
\begin{frame}[t, plain]
  \navyheader{Secant Lines}
  \vspace{0.3cm}

  \sectionlabel[sky]{Definition}

  The \textbf{secant line} through $(a, f(a))$ and $(b, f(b))$ has slope:
  \[
    m_{\text{sec}} = \frac{f(b) - f(a)}{b - a}
  \]
  This slope is exactly the \textbf{AROC} of $f$ on $[a,b]$.

  \medskip
  \textbf{Example:} For $f(x) = x^2$, the secant from $(1,1)$ to $(3,9)$:
  \[
    m = \frac{9-1}{3-1} = \frac{8}{2} = 4
  \]

  \medskip
  \begin{itemize}
    \item This is the AROC on $[1,3]$.
    \item As the interval shrinks, the secant approaches the tangent line.
  \end{itemize}
\end{frame}

% CONCAVITY SLIDE
\begin{frame}[t, plain]
  \navyheader{AROCs and Concavity}
  \vspace{0.3cm}

  \sectionlabel[greenacc]{Connecting the Pattern to the Graph}

  \medskip
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \textbf{Concave Up}\\[4pt]
      AROCs are \textit{increasing} as $x$ increases.\\[4pt]
      Second differences $> 0$.\\[4pt]
      Example: $f(x) = x^2$\\
      AROCs: $1, 3, 5, 7, \ldots$
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Concave Down}\\[4pt]
      AROCs are \textit{decreasing} as $x$ increases.\\[4pt]
      Second differences $< 0$.\\[4pt]
      Example: $f(x) = -x^2$\\
      AROCs: $-1, -3, -5, -7, \ldots$
    \end{column}
  \end{columns}

  \medskip
  \begin{block}{Summary}
    Linear: 1st differences constant, 2nd differences $= 0$.\\
    Quadratic: 2nd differences constant $\neq 0$.
  \end{block}
\end{frame}
