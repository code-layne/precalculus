\begin{enumerate}[itemsep=12pt]

\item Let $f(x) = x^2 + 1$. Find the average rate of change over $[1, 3]$.

\begin{work}
  f(1) &= 2 \\
  f(3) &= 10 \\
  \text{AROC} &= \frac{10 - 2}{3 - 1} \\
              &= 4
\end{work}

AROC $= $ \ans{4}

\item $T(t)$ gives the temperature in $^\circ$F $t$ hours after midnight.

\smallskip
\begin{tabular}{|c|c|c|c|}
\hline
$t$ (hours) & 0 & 2 & 4 \\
\hline
$T(t)$ ($^\circ$F) & 60 & 52 & 48 \\
\hline
\end{tabular}
\smallskip

\begin{enumerate}[label=(\alph*), itemsep=4pt]
  \item AROC over $[0, 4]$: \ans{$-3$ $^\circ$F per hour}
  \item Interpret your answer in a sentence --- include the sign and the
        units.

        \vspace{0.05in}
        \ansline{On average the temperature fell 3 degrees each hour between midnight and}
        \ansline{4 a.m. --- negative because temperature drops as time increases.}
\end{enumerate}
