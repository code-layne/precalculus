\boxguard
\begin{tcolorbox}[colback=white, colframe=black!40,
    title=\textbf{Tier R --- Remediate},
    fonttitle=\bfseries, arc=2mm, left=3mm, right=3mm, top=2mm, bottom=2mm]

\begin{enumerate}[itemsep=8pt]
  \item Fill in the blanks to compute the AROC over $[0, 1]$:

        \vspace{0.05in}
        AROC $= \dfrac{h(1) - h(0)}{1 - 0} = \dfrac{\text{\blank{1.2cm}} - \text{\blank{1.2cm}}}{1 - 0}
        = $ \blank{1.4cm} feet per second

  \item Now the AROC over $[1, 2]$:

        \vspace{0.05in}
        AROC $= \dfrac{h(2) - h(1)}{2 - 1} = $ \blank{1.4cm} feet per second

  \item During which second did the balloon climb \textbf{faster} --- the
        first ($0$ to $1$) or the second ($1$ to $2$)? \quad \blank{2.4cm}

  \item Between $t = 2$ and $t = 3$, is the balloon rising or falling? \quad
        \textbf{Rising} / \textbf{Falling}
\end{enumerate}
\end{tcolorbox}

\begin{enumerate}[itemsep=8pt]
  \item Complete the AROC table.

        \smallskip
        \renewcommand{\arraystretch}{1.5}
        \begin{tabular}{|c|c|c|c|c|}
        \hline
        \textbf{Interval} & $[0,1]$ & $[1,2]$ & $[2,3]$ & $[3,4]$ \\
        \hline
        \textbf{AROC (ft/s)} & \blank{1.2cm} & \blank{1.2cm} & \blank{1.2cm} & \blank{1.2cm} \\
        \hline
        \end{tabular}

  \item Write one sentence interpreting the AROC over $[2, 3]$ --- include
        its sign and its units.

        \vspace{0.05in}
        \writelines{2}

  \item Look at your table: each AROC changes by the same amount from one
        interval to the next. That step is \blank{1.4cm} ft/s per second ---
        so $h$ is a \blank{2.4cm} function, and its graph is concave
        \blank{1.4cm}.

  \item Compute the AROC over the whole flight, $[0, 4]$. \quad
        \blank{1.4cm} ft/s

        The balloon clearly moved --- why is this answer technically correct
        but misleading?

        \vspace{0.05in}
        \writelines{2}
\end{enumerate}
\end{tcolorbox}
