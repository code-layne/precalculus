\vspace{0.12in}

\begin{tcolorbox}[colback=white, colframe=black!40,
    title=\textbf{Tier R --- Remediate},
    fonttitle=\bfseries, arc=2mm, left=3mm, right=3mm, top=2mm, bottom=2mm]

\textbf{R1.} Compute the AROC of the population over the interval $[1970,\,1990]$.
Use the formula $\text{AROC} = \dfrac{P(b)-P(a)}{b-a}$.

\[
    \text{AROC} = \frac{P(\hphantom{0000}) - P(\hphantom{0000})}{\hphantom{0000} - \hphantom{0000}}
    = \frac{\hphantom{000} - \hphantom{000}}{\hphantom{0000}}
    = \hphantom{000} \text{ thousands per year}
\]

\vspace{6pt}
\textbf{R2.} Compute the AROC over $[1990,\,2010]$.

\[
    \text{AROC} = \frac{P(\hphantom{0000}) - P(\hphantom{0000})}{\hphantom{0000} - \hphantom{0000}}
    = \hphantom{0000} \text{ thousands per year}
\]

\vspace{6pt}
\textbf{R3.} During which time period did the population grow \emph{faster}?
How do you know?
