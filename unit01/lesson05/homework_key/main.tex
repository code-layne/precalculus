\begin{enumerate}[itemsep=10pt]

\item The table gives every value of two functions $f$ and $g$. Evaluate each
      composite by working from the inside out.

\smallskip
\renewcommand{\arraystretch}{1.5}
\begin{center}
\begin{tabular}{|c|c|c|c|c|c|}
\hline
$x$ & 1 & 2 & 3 & 4 & 5 \\
\hline
$f(x)$ & 5 & 1 & 4 & 2 & 3 \\
\hline
$g(x)$ & 3 & 5 & 1 & 5 & 2 \\
\hline
\end{tabular}
\end{center}

\begin{enumerate}[label=(\alph*), itemsep=6pt]
  \item $f(g(1)) = $ \ans{4} \qquad (inner value: \ans{3})
  \item $g(f(1)) = $ \ans{2} \qquad (inner value: \ans{5})
  \item $f(g(3)) = $ \ans{5} \qquad (inner value: \ans{1})
  \item $g(g(4)) = $ \ans{2} \qquad (inner value: \ans{5})
\end{enumerate}

\item Let $f(x) = x^2 + 1$ and $g(x) = x - 4$. Build both composites and
      simplify completely.

\begin{work}
  f(g(x)) &= f(x - 4) \\
          &= (x - 4)^2 + 1 \\
          &= x^2 - 8x + 16 + 1 \\
          &= x^2 - 8x + 17 \\
  g(f(x)) &= g(x^2 + 1) \\
          &= (x^2 + 1) - 4 \\
          &= x^2 - 3
\end{work}

$(f \circ g)(x) = $ \ans{$x^2 - 8x + 17$} \qquad $(g \circ f)(x) = $ \ans{$x^2 - 3$}

\item \textbf{Back at the checkout counter.} The \$10 coupon is
      $c(p) = p - 10$ and the 6\% sales tax is $t(x) = 1.06x$. A \$45 shirt
      goes through two different registers.

\begin{work}
  t(c(45)) &= t(35) \\
           &= 1.06(35) \\
           &= 37.10 \\
  c(t(45)) &= c(47.70) \\
           &= 47.70 - 10 \\
           &= 37.70
\end{work}

\begin{enumerate}[label=(\alph*), itemsep=4pt]
  \item Coupon first, then tax: \$\ans{37.10}
  \item Tax first, then coupon: \$\ans{37.70}
  \item The shopper saves \$\ans{0.60} by having the coupon rung up
        first.
\end{enumerate}

\item \textbf{Taking functions apart.}

\begin{enumerate}[label=(\alph*), itemsep=6pt]
  \item Decompose $h(x) = \sqrt{4x - 7}$. \quad
        Inner: $g(x) = $ \ans{$4x - 7$} \quad Outer: $f(u) = $ \ans{$\sqrt{u}$}
  \item In Lesson 1.4 you shifted a graph with $k(x) = f(x + 6)$. Write it as
        a composition: $k(x) = f(g(x))$ where $g(x) = $ \ans{$x + 6$}
  \item Was that change made \textbf{inside} or \textbf{outside} $f$?
        \quad \ans{inside}
\end{enumerate}

\item Let $f(x) = 3x - 2$ and $g(x) = x^2$. Which expression is
      $(f \circ g)(x)$?

\begin{enumerate}[label=\Alph*., itemsep=3pt]
  \item $3x^2 - 2$ \quad \textcolor{keyred}{\textbf{$\leftarrow$ correct}}
  \item $(3x - 2)^2$
  \item $9x^2 - 12x + 4$
  \item $3x^3 - 2x^2$
\end{enumerate}

\end{enumerate}

\boxguard
\begin{extensionbox}
The \textbf{identity function} is $I(x) = x$. Let $f(x) = 5x - 2$.

\smallskip
First check that $I$ really does nothing: $f(I(x)) = $ \ans{$5x - 2$} and
$I(f(x)) = $ \ans{$5x - 2$}.

\smallskip
Now the interesting question: is there a function $g$ with
$f(g(x)) = x$ --- a function that \textit{undoes} $f$? Try to find it, and
explain how you searched.

\ansline{Yes: $g(x) = \dfrac{x+2}{5}$, since $f(g(x)) = 5 \cdot \dfrac{x+2}{5} - 2 = x$.}
\ansline{Undo $f$'s steps backwards: $f$ multiplies by 5 then subtracts 2, so $g$ adds 2 then divides by 5.}
\end{extensionbox}

\boxguard
\begin{notesbox}{Preview of Lesson 1.6}
The extension asked for a function that undoes $f$. Lesson 1.6,
\textit{Inverse Functions}, gives that function a name and a notation:
$f^{-1}$. The test for it is a composition --- $f(f^{-1}(x)) = x$ --- which is
exactly the identity function you met today.
\end{notesbox}
