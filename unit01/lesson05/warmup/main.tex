\item From Lesson 1.1: let $f(x) = 3x - 2$. Whatever you put in the parentheses
      is what gets tripled --- even when it is not a number.

\begin{work}
   f(5) &= 3(5) - 2 \\
        &= 13 \\
  f(-1) &= 3(-1) - 2 \\
        &= -5 \\
 f(x^2) &= 3(x^2) - 2 \\
        &= 3x^2 - 2
\end{work}

\begin{enumerate}[label=(\alph*), itemsep=5pt]
  \item $f(5) = $ \blank{2.2cm} \qquad $f(-1) = $ \blank{2.2cm}
  \item $f(x^2) = $ \blank{3.2cm} \quad (a small question today, and the whole
        mechanic of Part 3)
\end{enumerate}
