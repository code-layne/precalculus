\begin{enumerate}[leftmargin=1.5em, itemsep=10pt]

\item Factor $p(x)=x^3-4x$ completely and identify all zeros.
      For each zero, state whether it has multiplicity 1 (simple) or is a repeated zero.

      \vspace{0.06in}
      Factored form: \ans{$x(x-2)(x+2)$} \hfill Zeros: \ans{$x=0,\;x=2,\;x=-2$}

      \vspace{0.06in}
      Multiplicity of each zero: \ans{Each zero has multiplicity 1 (all simple).}

\item Write a polynomial (in factored form) that has zeros at $x=-3$ (multiplicity 1),
      $x=0$ (multiplicity 2), and $x=4$ (multiplicity 1).
      What is the degree of your polynomial?

      \vspace{0.06in}
      $p(x) = {}$ \ans{$(x+3)\,x^2\,(x-4)$} \hfill Degree: \ans{4}

\item Let $f(x)=3x^4-x^2$.

      \vspace{0.04in}
      (a) Find $f(-x)$: \ans{$3(-x)^4-(-x)^2 = 3x^4 - x^2$}

      \vspace{0.06in}
      (b) Is $f$ even, odd, or neither?
      Circle one: \quad {\color{keyred}\textbf{Even}} \quad \textbf{Odd} \quad \textbf{Neither}

      \vspace{0.04in}
      Explain how you know: \ansline{$f(-x) = f(x)$ for all $x$, so $f$ is \textbf{even}.}

\end{enumerate}
