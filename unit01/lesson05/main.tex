\begin{minipage}[t]{0.48\linewidth}\vspace{0pt}
    \textbf{Key Understandings}\\
    \begin{itemize}[leftmargin=1.2em, itemsep=2pt]
        \item Factor $(x-a)$ $\Leftrightarrow$ zero at $a$; multiplicity $=$ how many times repeated.
        \item A degree-$n$ polynomial has exactly $n$ complex zeros (counting multiplicities).
        \item Odd multiplicity $\Rightarrow$ graph \emph{crosses}; even multiplicity $\Rightarrow$
              graph \emph{touches and bounces}.
        \item Complex zeros come in conjugate pairs for polynomials with real coefficients.
        \item Even function: $f(-x)=f(x)$, symmetric about $y$-axis.
              Odd function: $f(-x)=-f(x)$, symmetric about origin.
    \end{itemize}
\end{minipage}
\end{skillbox}

\begin{skillbox}[{Vocabulary, Concepts, \& Theorems}]{greenbox}
\begin{multicols}{2}
\begin{itemize}[leftmargin=1.2em, itemsep=3pt]
    \item \textbf{Zero (root)} — value $a$ such that $p(a)=0$
    \item \textbf{Multiplicity} — how many times a linear factor is repeated
    \item \textbf{Complex zero} — a zero of the form $a+bi$, $b\neq 0$
    \item \textbf{Complex Conjugate Pairs Theorem} — if $p$ has real coefficients and
          $a+bi$ is a zero, then $a-bi$ is also a zero
    \item \textbf{Even function} — $f(-x)=f(x)$; symmetric over $y$-axis
    \item \textbf{Odd function} — $f(-x)=-f(x)$; symmetric about origin
    \item \textbf{Successive differences} — method for identifying polynomial degree from
          a table of equally-spaced values
\end{itemize}
\end{multicols}
\end{skillbox}

\begin{skillbox}[Hook]{sky}
    \textbf{Display on board:} $p(x)=(x-2)^2(x+1)$

    \medskip
    Ask students:
    \begin{enumerate}[leftmargin=1.5em, itemsep=3pt]
        \item Where does the graph of $y=p(x)$ cross the $x$-axis?
        \item Does the graph cross the $x$-axis at $x=2$, or does it simply touch it?
              How do you know just by looking at the factor $(x-2)^2$?
        \item If we multiply out $p(x)$, what is the degree? How many zeros should
              there be in total?
    \end{enumerate}
    \textit{Goal:} Students notice that the exponent on the factor predicts crossing vs.\
    bouncing — launching the multiplicity concept.
\end{skillbox}

\begin{skillbox}[Lesson]{sky}
    \textbf{1. Zeros and Linear Factors (LO 1.5.A.1--A.2)}

    If $p(a)=0$, then $a$ is a zero of $p$, and $(x-a)$ is a linear factor.
    Revisit hook polynomial and list all zeros and their multiplicities.
    Stress: a degree-$n$ polynomial has \emph{exactly} $n$ complex zeros counting multiplicities.

    \medskip
    \textbf{2. Graph Behavior at Real Zeros (LO 1.5.A.3, A.5)}

    Build a table: \quad
    \begin{tabular}{|l|l|l|}
    \hline
    \textbf{Multiplicity} & \textbf{Graph behavior} & \textbf{Example} \\\hline
    Odd (1, 3, \ldots) & Crosses $x$-axis & $(x+1)$ in hook poly \\\hline
    Even (2, 4, \ldots) & Tangent (touches, bounces) & $(x-2)^2$ in hook poly \\\hline
    \end{tabular}

    Work additional example: $q(x)=x^2(x-3)(x+1)^3$. Ask students to predict
    crossing/touching at each zero before graphing with technology.

    \medskip
    \textbf{3. Exactly $n$ Complex Zeros (LO 1.5.A.2)}

    Introduce the Fundamental Theorem of Algebra (not proved, just stated).
    Example: a degree-4 polynomial must have 4 zeros; some may be complex.

    \medskip
    \textbf{4. Complex Conjugate Pairs (LO 1.5.A.4)}

    State theorem: for polynomials with real coefficients, complex zeros come in
    conjugate pairs $a\pm bi$.
    Example: if $p$ has real coefficients and $2+3i$ is a zero, then $2-3i$ must
    also be a zero. Build minimal polynomial: $(x-(2+3i))(x-(2-3i))=x^2-4x+13$.

    \medskip
    \textbf{5. Even and Odd Functions (LO 1.5.B.1--B.2)}

    Algebraic test: substitute $-x$ and simplify.
    \begin{itemize}[leftmargin=1.2em]
        \item If $f(-x)=f(x)$ $\Rightarrow$ \textbf{even}; graph symmetric over $y$-axis.
        \item If $f(-x)=-f(x)$ $\Rightarrow$ \textbf{odd}; graph symmetric about origin.
        \item Otherwise $\Rightarrow$ \textbf{neither}.
    \end{itemize}
    Examples: $f(x)=x^4-2x^2$ (even); $g(x)=x^3-x$ (odd); $h(x)=x^2+x$ (neither).

    \medskip
    \textbf{6. Successive Differences (LO 1.5.A.6)} (\textit{brief introduction})

    Show a table of values for a cubic over equally-spaced $x$-values. Take first,
    second, third differences. Third differences are constant $\Rightarrow$ degree 3.
    Degree $=$ least $n$ for which $n$th differences are constant.
\end{skillbox}

\begin{skillbox}[Active Monitoring]{redbox}
    \begin{itemize}[leftmargin=1.2em, itemsep=3pt]
        \item Circulate during the multiplicity table: watch for students who think
              even multiplicity means the graph doesn't touch the axis at all.
        \item Cold-call: ``If $p(x)=(x-5)^3(x+2)^2$, how many zeros does $p$ have?
              What is the degree? Does the graph cross or bounce at $x=5$?''
        \item During even/odd: watch for students who only check a single value of $x$
              instead of proving algebraically for all $x$.
        \item Ensure students don't confuse \emph{even/odd function} with
              \emph{even/odd degree}.
    \end{itemize}
\end{skillbox}
