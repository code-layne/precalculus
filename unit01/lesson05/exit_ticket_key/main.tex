\textit{Let $f(x) = 2x$ and $g(x) = x - 6$ for items 1 and 2.}

\vspace{0.1in}

\begin{enumerate}[itemsep=12pt]

\item Evaluate both composites at $x = 10$. \textbf{Write the handoff down.}

\begin{work}
  f(g(10)) &= f(4) \\
           &= 8 \\
  g(f(10)) &= g(20) \\
           &= 14
\end{work}

handoff $= $ \ans{4}, \; so $f(g(10)) = $ \ans{8}

\smallskip
handoff $= $ \ans{20}, \; so $g(f(10)) = $ \ans{14}

\item Write each composite as a simplified formula.

\begin{work}
  (f \circ g)(x) &= f(x - 6) \\
                 &= 2(x - 6) \\
                 &= 2x - 12 \\
  (g \circ f)(x) &= g(2x) \\
                 &= 2x - 6
\end{work}

$(f \circ g)(x) = $ \ans{$2x - 12$} \qquad $(g \circ f)(x) = $ \ans{$2x - 6$}

\smallskip
Are they the same function? \ans{no}

\item Decompose $h(x) = (x + 7)^4$.

\begin{enumerate}[label=(\alph*), itemsep=5pt]
  \item Inner: $g(x) = $ \ans{$x + 7$} \qquad Outer: $f(u) = $ \ans{$u^4$}
  \item To compute $h(3)$, the handoff is the number \ans{10}.
\end{enumerate}

\end{enumerate}
