% SECTION 1 — THE HANDOFF
\begin{frame}[t, plain]
  \plumheader{1. The Handoff}
  \sectionlabel[redacc]{The Whole Lesson in One Picture}

  \((f \circ g)(x) = f\big(g(x)\big)\), read ``\(f\) of \(g\) of \(x\).'' The
  circle is \textbf{not} multiplication.

  \medskip
  \begin{center}
  \begin{tikzpicture}[x=1cm, y=1cm]
    \draw[thick, plum, fill=lilac, rounded corners=2pt]
      (-0.8,-0.5) rectangle (0.8,0.5);
    \node at (0,0) {\large \(g\)};
    \draw[thick, plum, fill=lilac, rounded corners=2pt]
      (4.2,-0.5) rectangle (5.8,0.5);
    \node at (5,0) {\large \(f\)};
    \draw[->, very thick, plum] (-2.4,0) -- (-0.9,0);
    \node[above, font=\small] at (-1.65,0.08) {\(x\)};
    \draw[->, very thick, goldacc] (0.9,0) -- (4.1,0);
    \node[above, font=\small] at (2.5,0.10) {\(g(x)\)};
    \draw[->, very thick, plum] (5.9,0) -- (7.9,0);
    \node[above, font=\small] at (6.9,0.08) {\(f(g(x))\)};
    \node[above, font=\small\itshape, text=slate] at (0,0.58) {runs first};
    \node[above, font=\small\itshape, text=slate] at (5,0.58) {runs second};
    \node[below, font=\small\bfseries, goldacc] at (2.5,-0.10) {THE HANDOFF};
    \node[below, font=\small, text=slate] at (2.5,-0.55)
      {\(g\)'s answer \textit{is} \(f\)'s question};
  \end{tikzpicture}
  \end{center}

  \medskip
  \begin{center}
    \textbf{One number, two jobs --- an answer first, then a question.
    Find it, and write it down.}
  \end{center}
\end{frame}

% SECTION 2 — FROM A TABLE
\begin{frame}[t, plain]
  \plumheader{2. The Handoff from a Table}
  \sectionlabel[goldacc]{Two Lookups, No Formula}

  \begin{columns}[T, onlytextwidth]
    \column{0.44\textwidth}
      \begin{center}
      \renewcommand{\arraystretch}{1.3}
      \begin{tabular}{|l|c|c|c|c|c|}
      \hline
      \(x\) & 0 & 1 & 2 & 3 & 4 \\
      \hline
      \(f(x)\) & 3 & 4 & 0 & 2 & 1 \\
      \hline
      \(g(x)\) & 2 & 3 & 7 & 0 & 4 \\
      \hline
      \end{tabular}
      \end{center}

      \medskip
      \(f(g(1))\): handoff \(g(1) = \mathbf{3}\), then \(f(3) = 2\).

      \smallskip
      \(g(f(1))\): handoff \(f(1) = \mathbf{4}\), then \(g(4) = 4\).

    \column{0.52\textwidth}
      \begin{block}{Swap the machines, swap the handoff}
        Same two functions, same input, a different middle number --- so a
        different answer. Composition is \textbf{not commutative}.
      \end{block}

      \medskip
      \begin{block}{And sometimes there is no answer at all}
        \(f(g(2))\): the handoff is \(7\), and the table has no \(f(7)\).
        A handoff must be a question the \textbf{outer} function accepts ---
        that is the \textbf{domain of a composite}.
      \end{block}
  \end{columns}
\end{frame}

% SECTION 3 — AN EXPRESSION
\begin{frame}[t, plain]
  \plumheader{3. Handing Over a Whole Expression}

  \begin{block}{The recipe}
    To build \(f(g(x))\), substitute \(g(x)\) for \textbf{every} \(x\) in
    \(f(x)\). The handoff is no longer a number --- it is an expression.
    Nothing else changes.
  \end{block}

  \medskip
  \begin{columns}[T, onlytextwidth]
    \column{0.48\textwidth}
      \(f(x) = 3x - 2\), \quad \(g(x) = x^2 + 1\):
      \begin{align*}
        f(g(x)) &= f(x^2+1) = 3(x^2+1) - 2 \\
                &= 3x^2 + 1 \\[4pt]
        g(f(x)) &= g(3x-2) = (3x-2)^2 + 1 \\
                &= 9x^2 - 12x + 5
      \end{align*}

    \column{0.48\textwidth}
      \begin{block}{Two routes, one number}
        \(f(g(2))\) step by step: handoff \(g(2) = 5\), so \(f(5) = 13\).

        \smallskip
        Straight from the formula: \(3(2)^2 + 1 = 13\).
      \end{block}

      \smallskip
      Not even the same \textit{shape} of expression --- order matters.
  \end{columns}
\end{frame}

% SECTION 4 — TWO REGISTERS
\begin{frame}[t, plain]
  \plumheader{4. Two Registers, Two Handoffs}
  \sectionlabel[redacc]{Why the Order Costs Sixty Cents}

  Coupon \(c(p) = p - 10\), \qquad tax \(t(x) = 1.06x\). Neither one alone
  answers ``what do I pay?''

  \medskip
  \begin{columns}[T, onlytextwidth]
    \column{0.48\textwidth}
      \begin{block}{At \(p = 60\)}
        \(t(c(60)) = t(\mathbf{50}) = 53.00\)

        \smallskip
        \(c(t(60)) = c(\mathbf{63.60}) = 53.60\)
      \end{block}

    \column{0.48\textwidth}
      \begin{block}{At every price}
        \(t(c(p)) = 1.06p - 10.60\)

        \smallskip
        \(c(t(p)) = 1.06p - 10\)
      \end{block}
  \end{columns}

  \medskip
  \begin{center}
    Taxing first charges 6\% of the \$10 you never paid: \(0.06 \times 10 =
    \$0.60\), at \textbf{any} price.
  \end{center}
\end{frame}

% SECTION 5 — DECOMPOSITION, IDENTITY, 1.4
\begin{frame}[t, plain]
  \plumheader{5. Decomposition, the Identity, and Lesson 1.4}

  \begin{columns}[T, onlytextwidth]
    \column{0.48\textwidth}
      \begin{block}{Decomposition = name the handoff}
        \(h(x) = (5x-1)^3\). What would your calculator do first?

        \smallskip
        Inner \(g(x) = 5x-1\), \quad outer \(f(u) = u^3\).
      \end{block}

      \medskip
      \begin{block}{The identity function}
        \(I(x) = x\) makes the handoff the number itself, so
        \(f(I(x)) = I(f(x)) = f(x)\).
      \end{block}

    \column{0.48\textwidth}
      \begin{block}{Lesson 1.4 was composition all along}
        \renewcommand{\arraystretch}{1.3}
        \begin{tabular}{@{}lll@{}}
          \textbf{1.4 rule} & \textbf{is} & \textbf{handoff} \\
          \(f(x+3)\) & \(f(h(x))\) & \(x+3\) \\
          \(f(2x)\)  & \(f(h(x))\) & \(2x\) \\
        \end{tabular}
      \end{block}

      \medskip
      The handoff is what \(f\) gets \textbf{asked} --- so an inside change can
      only rewrite the question, never the answers.
  \end{columns}
\end{frame}

% CLASS PRACTICE
\begin{frame}[t, plain]
  \plumheader{Class Practice}
  \sectionlabel[redacc]{Name the Handoff First}

  \begin{enumerate}\setlength{\itemsep}{10pt}
    \item Using the table from section 2, find \(f(g(3))\) and \(g(f(3))\).
          Write both handoffs down.

    \item Let \(f(x) = x^2\) and \(g(x) = x - 4\). Build \((f \circ g)(x)\) and
          \((g \circ f)(x)\), simplified. Are they the same function?

    \item Decompose \(h(x) = \sqrt{4x+1}\) into an inner and an outer function,
          then check by substituting back.
  \end{enumerate}
\end{frame}
