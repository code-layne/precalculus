\begin{enumerate}[itemsep=10pt]

\item Identify the input and output quantities in each situation.

\begin{enumerate}[label=(\alph*), itemsep=6pt]
  \item A pizzeria charges by the pizza; an order's total cost depends on how many
        pizzas are ordered.

        Input: \blank{3.2cm} \quad Output: \blank{3.2cm}

  \item A cup of hot chocolate cools on the counter as the minutes pass.

        Input: \blank{3.2cm} \quad Output: \blank{3.2cm}
\end{enumerate}

\item Decide whether each rule is a function. Circle yes or no, and give a short
      reason.

\begin{enumerate}[label=(\alph*), itemsep=6pt]
  \item Each person $\to$ their right thumbprint. \quad \textbf{Yes} / \textbf{No}

        \vspace{0.05in}
        \writeline

  \item Each teacher $\to$ the students in their first-period class. \quad \textbf{Yes} / \textbf{No}

        \vspace{0.05in}
        \writeline

  \item Each number $\to$ its square. \quad \textbf{Yes} / \textbf{No}

        \vspace{0.05in}
        \writeline
\end{enumerate}

\item A candle's height is measured each hour after it is lit.

\smallskip
\begin{tabular}{|c|c|c|c|c|}
\hline
Hours burning & 0 & 1 & 2 & 3 \\
\hline
Height (cm) & 20 & 17 & 14 & 11 \\
\hline
\end{tabular}
\smallskip

\begin{enumerate}[label=(\alph*), itemsep=4pt]
  \item After 2 hours, the height is \blank{2.2cm} cm.
  \item As the hours increase, the height \blank{2.8cm}.
  \item Input quantity: \blank{2.8cm} \quad Output quantity: \blank{2.8cm}
\end{enumerate}

\item A delivery driver parks at a stoplight for 2 minutes, then drives away from
      it for 3 minutes. Describe the driver's distance from the stoplight on each
      interval.

\begin{enumerate}[label=(\alph*), itemsep=4pt]
  \item First 2 minutes: the distance is \blank{2.8cm}.
  \item Next 3 minutes: the distance is \blank{2.8cm}.
\end{enumerate}

\item Which of the following rules is \textbf{NOT} a function?

\begin{enumerate}[label=\Alph*., itemsep=3pt]
  \item Each person $\to$ their birthdate
  \item Each book $\to$ its number of pages
  \item Each month $\to$ the students born that month
  \item Each number $\to$ its square
\end{enumerate}

\end{enumerate}
