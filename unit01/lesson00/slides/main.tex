% SECTION 1 — QUANTITIES THAT CHANGE TOGETHER
\begin{frame}[t, plain]
  \plumheader{1. Quantities That Change Together}

  This whole unit is about two quantities linked so that one \textbf{responds} to
  the other.

  \medskip
  \begin{block}{Input and Output}
    The quantity we \textbf{choose or track} is the \textbf{input} (independent
    variable). The quantity that \textbf{responds} is the \textbf{output}
    (dependent variable). Ordered pairs are written (input, output).
  \end{block}

  \medskip
  \begin{tabular}{|p{0.45\textwidth}|c|c|}
  \hline
  \textbf{Situation} & \textbf{Input} & \textbf{Output} \\
  \hline
  Pay-per-minute phone plan & minutes talked & cost of the bill \\
  \hline
  A lit candle burning down & ? & ? \\
  \hline
  A ball dropped from a rooftop & ? & ? \\
  \hline
  Pumping gas into a car & ? & ? \\
  \hline
  \end{tabular}

  \medskip
  \textbf{The ``you-control'' test:} which quantity do \textit{you} choose or
  control? That one is the input.
\end{frame}

% SECTION 2 — WHAT IS A FUNCTION
\begin{frame}[t, plain]
  \plumheader{2. What Is a Function? (First Look)}

  \begin{block}{Definition (informal)}
    A rule is a \textbf{function} when each input produces \textbf{exactly one}
    output --- same input, same output, every time. \textit{(Notation arrives in
    Lesson 1.1.)}
  \end{block}

  \bigskip
  Function or not? Test each rule aloud:

  \medskip
  \begin{tabular}{|p{0.62\textwidth}|c|}
  \hline
  \textbf{Rule} & \textbf{Function?} \\
  \hline
  each student \(\to\) their student ID number & yes \\
  \hline
  each month \(\to\) the students born that month & ? \\
  \hline
  each number \(\to\) its double & ? \\
  \hline
  each shoe size \(\to\) a student who wears that size & ? \\
  \hline
  \end{tabular}
\end{frame}

% SECTION 2, CONT. — THE FINE PRINT
\begin{frame}[t, plain]
  \plumheader{The Fine Print: Many-to-One vs.\ One-to-Many}

  \begin{columns}[T]
    \column{0.48\textwidth}
      \begin{block}{Allowed: many-to-one}
        \begin{itemize}\setlength{\itemsep}{6pt}
          \item Two inputs \textbf{may share} an output
          \item Both twins \(\to\) the same birthday
          \item Each input still has exactly one output --- still a function
        \end{itemize}
      \end{block}

    \column{0.48\textwidth}
      \begin{block}{Broken: one-to-many}
        \begin{itemize}\setlength{\itemsep}{6pt}
          \item One input with \textbf{two outputs}
          \item Shoe size 9 \(\to\) two different students
          \item The machine is no longer trustworthy --- not a function
        \end{itemize}
      \end{block}
  \end{columns}

  \bigskip
  \textbf{In one sentence:} two inputs may share an output, but one input may never
  have two outputs. The rule restricts each \textbf{input} --- it says nothing
  about how often an output may repeat.

  \medskip
  \textbf{Challenge:} Give a rule from your life that IS a function. Now break it
  so it isn't.
\end{frame}

% SECTION 3 — A GRAPH TELLS A STORY
\begin{frame}[t, plain]
  \plumheader{3. A Graph Tells a Story}

  A cyclist rides away from home, rests, and rides back.

  \medskip
  \begin{columns}[T]
    \column{0.5\textwidth}
      \begin{center}
      \begin{tikzpicture}[x=0.5cm, y=0.42cm]
        \draw[linegray, very thin, step=1] (0,0) grid (10,6);
        \draw[->, thick] (0,0) -- (10.6,0) node[below right] {\small time (min)};
        \draw[->, thick] (0,0) -- (0,6.6) node[above] {\small distance (blocks)};
        \foreach \x in {2,4,6,8,10} \node[below] at (\x,0) {\small \x};
        \foreach \y in {1,3,5} \node[left] at (0,\y) {\small \y};
        \draw[very thick, plum] (0,0) -- (4,5) node[midway, above left] {\small\bfseries A};
        \draw[very thick, plum] (4,5) -- (6,5) node[midway, above] {\small\bfseries B};
        \draw[very thick, plum] (6,5) -- (10,0) node[midway, above right] {\small\bfseries C};
      \end{tikzpicture}
      \end{center}

    \column{0.46\textwidth}
      Read it left to right, like a story:
      \begin{itemize}\setlength{\itemsep}{6pt}
        \item \textbf{A} (\(t=0\) to \(4\)): rising --- \textbf{increasing} --- riding away
        \item \textbf{B} (\(t=4\) to \(6\)): flat --- \textbf{constant} --- resting
        \item \textbf{C} (\(t=6\) to \(10\)): falling --- \textbf{decreasing} --- riding home
      \end{itemize}
  \end{columns}

  \bigskip
  \textit{No formula needed:} Where is the rider at \(t = 5\)? The graph \textit{is}
  the information. A flat segment is not ``nothing happening'' --- time keeps
  moving; the \textbf{output} holds steady. And each time has exactly one distance,
  so distance \textbf{is} a function of time.
\end{frame}

% SECTION 4 — A TABLE TELLS THE SAME STORY
\begin{frame}[t, plain]
  \plumheader{4. A Table Tells the Same Story}

  A candle is lit and measured each hour.

  \medskip
  \begin{tabular}{|c|c|c|c|c|c|}
  \hline
  Hours burning & 0 & 1 & 2 & 3 & 4 \\
  \hline
  Height (cm) & 20 & 17 & 14 & 14 & 11 \\
  \hline
  \end{tabular}

  \bigskip
  \begin{itemize}\setlength{\itemsep}{8pt}
    \item Input: hours burning \quad Output: candle height
    \item Hour 0 \(\to\) 2: height \textbf{decreasing} \quad
          Hour 2 \(\to\) 3: height \textbf{constant}
    \item That flat stretch is this table's version of \textbf{segment B} on the
          bike-ride graph --- same story, different representation.
    \item Read a graph \textit{or} a table the same way: watch what the
          \textbf{output} does as the \textbf{input} moves forward.
    \item Each hour has exactly one height \(\to\) the height \textbf{is} a
          function of the hours burning.
  \end{itemize}
\end{frame}

  \sectionlabel[redacc]{Try It}

  \begin{enumerate}\setlength{\itemsep}{8pt}
    \item A movie theater tracks tickets sold and money collected each night.
          Name the input and the output. Is the money collected a function of the
          tickets sold?
    \item The table shows quiz scores after different amounts of studying.

          \smallskip
          \begin{tabular}{|c|c|c|c|c|}
          \hline
          Hours studied & 0 & 1 & 2 & 3 \\
          \hline
          Quiz score & 60 & 70 & 80 & 80 \\
          \hline
          \end{tabular}

          \smallskip
          \begin{enumerate}\setlength{\itemsep}{5pt}
            \item Describe the score's behavior from 0 to 2 hours, then from 2 to 3 hours.
            \item Inputs 2 and 3 share the output 80. Is the score still a function
                  of hours studied? Explain.
          \end{enumerate}
  \end{enumerate}
\end{frame}
