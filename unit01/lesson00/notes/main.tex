\begin{notesbox}{1. Evaluating Expressions}
\textbf{Order of operations:} parentheses, \blank{2.4cm}, multiplication and
division, then \blank{2.4cm} and subtraction.

\textbf{Substitution habit:} when substituting a negative value, always put it inside
\blank{3cm}.

\medskip
\textbf{Example 1.} Evaluate $3x^2 - 2x + 1$ at $x = -2$.

\begin{work}
  3(-2)^2 - 2(-2) + 1 &= 3(4) + 4 + 1 \\
                      &= 12 + 4 + 1 \\
                      &= 17
\end{work}

Value: \blank{2.5cm}

\smallskip
\textit{Check yourself:} $(-2)^2 = $ \blank{1.5cm} but $-2^2 = $ \blank{1.5cm} ---
the parentheses are the whole difference.
\end{notesbox}

\vspace{0.1in}

\boxguard
\begin{notesbox}{2. Solving Equations \& Inequalities}
\textbf{Golden rule:} undo operations in \blank{2.5cm} order, doing the same thing
to both sides.

\medskip
\textbf{Example 2.} Solve $4x - 7 = 13$.

\begin{work}
  4x - 7 &= 13 \\
      4x &= 20 \\
       x &= 5
\end{work}

\textbf{Example 3.} Solve $5 - 2x > 11$.

\begin{work}
  5 - 2x &> 11 \\
     -2x &> 6 \\
       x &< -3
\end{work}

\textbf{The flip rule:} multiplying or dividing both sides by a negative number
\blank{2.5cm} the inequality symbol.

\medskip
