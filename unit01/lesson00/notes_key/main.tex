\par\vspace{2pt}
\noindent\textbf{\textcolor{navy}{expression:}}\\[1pt]
\ansline{Numbers, variables, and operations combined; it has a value but no equals sign.}\\[3pt]
\noindent\textbf{\textcolor{navy}{equation:}}\\[1pt]
\ansline{A statement that two expressions are equal; it can be solved.}\\[3pt]

\ansline{A value of the variable that makes the equation or inequality true.}\\[3pt]

\ansline{A way to write a set of numbers with endpoints, e.g.\ $(-2, 5]$.}\\[3pt]

\ansline{A point $(x, y)$: input listed first, output second.}

\boxguard
\begin{notesbox}{1. Evaluating Expressions}
\textbf{Order of operations:} parentheses, \ans{exponents}, multiplication and
division, then \ans{addition} and subtraction.

\textbf{Substitution habit:} when substituting a negative value, always put it inside
\ans{parentheses}.

\medskip
\textbf{Example 1.} Evaluate $3x^2 - 2x + 1$ at $x = -2$.

\begin{work}
  3(-2)^2 - 2(-2) + 1 &= 3(4) + 4 + 1 \\
                      &= 12 + 4 + 1 \\
                      &= 17
\end{work}

Value: \ans{17}

\smallskip
\textit{Check yourself:} $(-2)^2 = $ \ans{4} but $-2^2 = $ \ans{$-4$} ---
the parentheses are the whole difference.
\end{notesbox}

\vspace{0.1in}

\boxguard
\begin{notesbox}{2. Solving Equations \& Inequalities}
\textbf{Golden rule:} undo operations in \ans{reverse} order, doing the same thing
to both sides.

\medskip
\textbf{Example 2.} Solve $4x - 7 = 13$.

\begin{work}
  4x - 7 &= 13 \\
      4x &= 20 \\
       x &= 5
\end{work}

\textbf{Example 3.} Solve $5 - 2x > 11$.

\begin{work}
  5 - 2x &> 11 \\
     -2x &> 6 \\
       x &< -3
\end{work}

\textbf{The flip rule:} multiplying or dividing both sides by a negative number
\ans{reverses} the inequality symbol.

\medskip
\textbf{Interval notation.} Complete the table.

\smallskip
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{|c|c|}
\hline
\textbf{Inequality} & \textbf{Interval notation} \\
\hline
$x < -3$ & \ans{$(-\infty, -3)$} \\
\hline
$-1 \le x < 4$ & \ans{$[-1, 4)$} \\
\hline
$x \ge 0$ & \ans{$[0, \infty)$} \\
\hline
\end{tabular}
\end{notesbox}

\boxguard
\begin{practicebox}
\begin{enumerate}[itemsep=8pt]
  \item Evaluate $x^2 - 3x$ at $x = -4$.

\begin{work}
  (-4)^2 - 3(-4) &= 16 + 12 \\
                 &= 28
\end{work}

        Value: \ans{28}

  \item Solve $7 - 3x \ge 22$ and write the solution set in interval notation.

\begin{work}
  7 - 3x &\ge 22 \\
     -3x &\ge 15 \\
       x &\le -5
\end{work}

        Interval notation: \ans{$(-\infty, -5]$}
\end{enumerate}
\end{practicebox}
