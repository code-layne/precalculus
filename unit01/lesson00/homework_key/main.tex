\begin{enumerate}[itemsep=10pt]

\item Evaluate each expression at the given value.

\begin{enumerate}[label=(\alph*), itemsep=6pt]
  \item $x^2 + 4x$ at $x = -3$

\begin{work}
  (-3)^2 + 4(-3) &= 9 - 12 \\
                 &= -3
\end{work}

        Value: \ans{$-3$}

  \item $2x^2$ at $x = -3$

\begin{work}
  2(-3)^2 &= 2(9) \\
          &= 18
\end{work}

        Value: \ans{18}
\end{enumerate}

\item Solve for $x$: \quad $5x + 8 = -12$

\begin{work}
  5x + 8 &= -12 \\
      5x &= -20 \\
       x &= -4
\end{work}

$x = $ \ans{$-4$}

\item Solve for $x$: \quad $4(x + 2) = 28$

\begin{work}
  4(x + 2) &= 28 \\
    4x + 8 &= 28 \\
        4x &= 20 \\
         x &= 5
\end{work}

$x = $ \ans{5}

\item Solve $9 - 2x < 3$ and write the solution set in interval notation.

\begin{work}
  9 - 2x &< 3 \\
     -2x &< -6 \\
       x &> 3
\end{work}

Interval notation: \ans{$(3, \infty)$}

\item Which of the following is the value of $3x^2 - x$ when $x = -2$?

\begin{enumerate}[label=\Alph*., itemsep=3pt]
  \item $-14$
  \item $10$
  \item $14$ \textcolor{keyred}{\textbf{$\leftarrow$ correct}}
  \item $-10$
\end{enumerate}

\end{enumerate}

\begin{extensionbox}
The perimeter of a rectangle is $P = 2\ell + 2w$. Solve this formula for $w$ ---
that is, rewrite it as $w = \cdots$ using $P$ and $\ell$.

\begin{work}
          P &= 2\ell + 2w \\
  P - 2\ell &= 2w \\
          w &= \frac{P - 2\ell}{2}
\end{work}

$w = $ \ans{$(P - 2\ell)/2$}
\end{extensionbox}
