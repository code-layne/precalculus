\begin{enumerate}[itemsep=10pt]

\item Identify the input and output quantities in each situation.

\begin{enumerate}[label=(\alph*), itemsep=6pt]
  \item A pizzeria charges by the pizza; an order's total cost depends on how many
        pizzas are ordered.

        Input: \ans{number of pizzas} \quad Output: \ans{total cost}

  \item A cup of hot chocolate cools on the counter as the minutes pass.

        Input: \ans{time (minutes)} \quad Output: \ans{temperature}
\end{enumerate}

\item Decide whether each rule is a function. Circle yes or no, and give a short
      reason.

\begin{enumerate}[label=(\alph*), itemsep=6pt]
  \item Each person $\to$ their right thumbprint. \quad \ans{Yes} / \textbf{No}

        \vspace{0.02in}
        \ansline{Each person has exactly one right thumbprint --- one output per input.}

  \item Each teacher $\to$ the students in their first-period class. \quad \textbf{Yes} / \ans{No}

        \vspace{0.02in}
        \ansline{One teacher (input) is paired with many students (outputs) --- one input, many outputs.}

  \item Each number $\to$ its square. \quad \ans{Yes} / \textbf{No}

        \vspace{0.02in}
        \ansline{Each number has exactly one square. (Two numbers sharing a square, like $3$ and $-3$, is allowed.)}
\end{enumerate}

\item A candle's height is measured each hour after it is lit.

\smallskip
\begin{tabular}{|c|c|c|c|c|}
\hline
Hours burning & 0 & 1 & 2 & 3 \\
\hline
Height (cm) & 20 & 17 & 14 & 11 \\
\hline
\end{tabular}
\smallskip

\begin{enumerate}[label=(\alph*), itemsep=4pt]
  \item After 2 hours, the height is \ans{14} cm.
  \item As the hours increase, the height \ans{decreases}.
  \item Input quantity: \ans{hours burning} \quad Output quantity: \ans{height}
\end{enumerate}

\item A delivery driver parks at a stoplight for 2 minutes, then drives away from
      it for 3 minutes. Describe the driver's distance from the stoplight on each
      interval.

\begin{enumerate}[label=(\alph*), itemsep=4pt]
  \item First 2 minutes: the distance is \ans{constant}.
  \item Next 3 minutes: the distance is \ans{increasing}.
\end{enumerate}

\item Which of the following rules is \textbf{NOT} a function?

\begin{enumerate}[label=\Alph*., itemsep=3pt]
  \item Each person $\to$ their birthdate
  \item Each book $\to$ its number of pages
  \item Each month $\to$ the students born that month \textcolor{keyred}{\textbf{$\leftarrow$ correct}}
  \item Each number $\to$ its square
\end{enumerate}

\end{enumerate}

\boxguard
\begin{extensionbox}
Two \textit{different} buttons on a vending machine both dispense the same brand of
water. Is the machine still a function (button in, snack out)? Explain using the
words \textit{input} and \textit{output}.

\vspace{0.05in}
\ansline{Yes. Each button (input) still gives exactly one snack (output). The function rule only forbids one input giving two different outputs --- two inputs sharing an output is perfectly fine.}
\end{extensionbox}
