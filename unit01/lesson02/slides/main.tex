% SECTION 1 — TANDEM
\begin{frame}[t, plain]
  \plumheader{1. Two Quantities in Tandem}

  \begin{block}{Vary in tandem}
    Change the input and the output answers. A function is the rule that ties
    the two together.
  \end{block}

  \bigskip
  \begin{columns}[T]
    \column{0.46\textwidth}
      \textbf{A hose fills a tank.}

      \medskip
      \begin{tabular}{|c|c|c|c|c|}
      \hline
      \(t\) (min) & 0 & 2 & 4 & 6 \\
      \hline
      \(V(t)\) (gal) & 0 & 8 & 16 & 24 \\
      \hline
      \end{tabular}

      \medskip
      Every 2 minutes: \textbf{8 more gallons}, the same amount each time.

    \column{0.48\textwidth}
      \begin{itemize}\setlength{\itemsep}{7pt}
        \item Input quantity: \textbf{time}, in minutes
        \item Output quantity: \textbf{volume}, in gallons
        \item A table shows a handful of pairs. A \textbf{graph} shows them
              \textbf{all at once} --- which is why trends become things you
              can see.
      \end{itemize}
  \end{columns}
\end{frame}

% SECTION 2 — WHICH WAY
\begin{frame}[t, plain]
  \plumheader{2. Question 1 --- Which Way Are the Outputs Going?}

  \begin{block}{Definitions with teeth}
    \(f\) is \textbf{increasing} on an interval when: whenever \(a < b\),
    \(f(a) < f(b)\). \\[3pt]
    \(f\) is \textbf{decreasing} on an interval when: whenever \(a < b\),
    \(f(a) > f(b)\).
  \end{block}

  \smallskip
  \begin{columns}[T]
    \column{0.46\textwidth}
      \begin{center}
      \begin{tikzpicture}[x=0.52cm, y=0.34cm]
        \draw[linegray, very thin, step=1] (0,-3) grid (8,3);
        \draw[->, thick] (0,-3.5) -- (0,3.7) node[above] {\small \(h(x)\)};
        \draw[->, thick] (-0.5,0) -- (8.6,0) node[below right] {\small \(x\)};
        \foreach \x in {1,2,3,4,5,6,7,8} \node[below right, font=\tiny] at (\x,0) {\x};
        \foreach \y in {-3,-1,1,3} \node[left, font=\tiny] at (0,\y) {\y};
        \draw[very thick, plum] (0,-2) -- (2,2);
        \draw[very thick, plum] (2,2) -- (3,3);
        \draw[very thick, plum] (3,3) -- (5,0);
        \draw[very thick, plum] (5,0) -- (6,-3);
        \draw[very thick, plum] (6,-3) -- (8,-1);
      \end{tikzpicture}
      \end{center}

    \column{0.48\textwidth}
      \begin{itemize}\setlength{\itemsep}{6pt}
        \item Increasing on \(0 < x < 3\) and \(6 < x < 8\)
        \item Decreasing on \(3 < x < 6\)
      \end{itemize}

      \medskip
      \begin{block}{Two traps}
        Intervals are named with \textbf{inputs}, never outputs. \\[3pt]
        Decreasing \(\ne\) negative: cocoa falling from \(90^\circ\) to
        \(40^\circ\) is decreasing the whole way.
      \end{block}
  \end{columns}
\end{frame}

% SECTION 3 — CONCAVITY
\begin{frame}[t, plain]
  \plumheader{3. Question 2 --- How Is the Change Changing?}
  \sectionlabel[redacc]{The Missing Word}

  Both tanks fill for 6 minutes. Both end at 24 gallons. They do \textbf{not}
  fill the same way --- compute the gains.

  \medskip
  \begin{columns}[T]
    \column{0.48\textwidth}
      \textbf{Tank A}

      \smallskip
      \begin{tabular}{|c|c|c|c|c|}
      \hline
      \(t\) & 0 & 2 & 4 & 6 \\
      \hline
      \(A(t)\) & 0 & 14 & 20 & 24 \\
      \hline
      gain & --- & 14 & 6 & 4 \\
      \hline
      \end{tabular}

      \smallskip
      Gains \textbf{shrinking} \(\Rightarrow\) rate of change decreasing
      \(\Rightarrow\) \textbf{concave down}.

    \column{0.48\textwidth}
      \textbf{Tank B}

      \smallskip
      \begin{tabular}{|c|c|c|c|c|}
      \hline
      \(t\) & 0 & 2 & 4 & 6 \\
      \hline
      \(B(t)\) & 0 & 4 & 10 & 24 \\
      \hline
      gain & --- & 4 & 6 & 14 \\
      \hline
      \end{tabular}

      \smallskip
      Gains \textbf{growing} \(\Rightarrow\) rate of change increasing
      \(\Rightarrow\) \textbf{concave up}.
  \end{columns}

  \bigskip
  Both tanks are increasing the whole time. Concavity is a \textbf{second}
  question --- and it is about the gains, not the outputs.
\end{frame}

% SECTION 5 — ZEROS
\begin{frame}[t, plain]
  \plumheader{5. Zeros --- Where the Output Is 0}

  \begin{block}{Definition}
    A \textbf{zero} of a function is an input whose output is \textbf{0}. On a
    graph, the curve meets the horizontal axis there.
  \end{block}

  \bigskip
  \begin{columns}[T]
    \column{0.48\textwidth}
      \textbf{It is a \textit{solve} question.} (From 1.1.)

      \smallskip
      Finding a zero means solving \(h(x) = 0\) --- you are given the output
      and asked for the input(s). There can be none, one, or many.

      \medskip
      From the graph in Part 2: zeros at \(x = 1\) and \(x = 5\).

    \column{0.48\textwidth}
      \begin{block}{A zero is an event}
        Let \(D(t)\) be a drone's height in meters, \(t\) seconds after
        takeoff. \\[4pt]
        \(D(48) = 0\) means: forty-eight seconds after takeoff, the drone is
        back on the ground.
      \end{block}

      \medskip
      \textbf{Spiral:} that Part 2 graph has domain \(0 \le x \le 8\) and
      range \(-3 \le y \le 3\).
  \end{columns}
\end{frame}

          \end{enumerate}
