% HOOK SLIDE
\begin{frame}[t, plain]
  \plumheader{Hook: The Car and the Stop Sign}
  \sectionlabel[goldacc]{Think About It}

  A car pulls away from a stop sign. Two quantities start changing at once:

  \bigskip
  \begin{itemize}\setlength{\itemsep}{8pt}
    \item Its \textbf{speed} climbs from 0 --- quickly at first, then levels
          off near the limit.
    \item Its \textbf{distance} from the sign grows --- slowly at first, then
          faster and faster.
  \end{itemize}

  \bigskip
  Both are increasing. \textbf{In words: what makes the two stories
  different?}

  \bigskip
  \textit{Today we build the vocabulary that tells them apart.}
\end{frame}

% SECTION 1 — TWO QUANTITIES IN TANDEM
\begin{frame}[t, plain]
  \plumheader{1. Two Quantities in Tandem}

  A hose fills a tank: as time passes, the volume responds. The two
  quantities \textbf{vary in tandem}, linked by the function rule.

  \bigskip
  \begin{tabular}{|c|c|c|c|c|}
  \hline
  \(t\) (min) & 0 & 2 & 4 & 6 \\
  \hline
  \(V\) (gal) & 0 & 8 & 16 & 24 \\
  \hline
  \end{tabular}

  \bigskip
  \begin{itemize}\setlength{\itemsep}{8pt}
    \item Input (independent): \textbf{time}. Output (dependent):
          \textbf{volume}.
    \item Every 2 minutes, 8 more gallons --- a perfectly steady climb.
    \item As \(t\) increases, \(V\) \textbf{increases}: the graph rises left
          to right.
  \end{itemize}
\end{frame}

% SECTION 2 — INCREASING AND DECREASING
\begin{frame}[t, plain]
  \plumheader{2. Increasing and Decreasing}

  \begin{block}{The definitions, with teeth}
    \(f\) is \textbf{increasing} on an interval: whenever \(a < b\),
    \(f(a) < f(b)\) --- larger inputs give larger outputs.\\[4pt]
    \(f\) is \textbf{decreasing} on an interval: whenever \(a < b\),
    \(f(a) > f(b)\) --- larger inputs give smaller outputs.
  \end{block}

  \bigskip
  \begin{columns}[T]
    \column{0.5\textwidth}
      \begin{center}
      \begin{tikzpicture}[x=0.5cm, y=0.45cm]
        \draw[linegray, very thin, step=1] (0,0) grid (8,5);
        \draw[->, thick] (0,0) -- (8.6,0) node[below right] {\small \(x\)};
        \draw[->, thick] (0,0) -- (0,5.6) node[above] {\small \(f(x)\)};
        \foreach \x in {1,2,3,4,5,6,7,8} \node[below] at (\x,0) {\small \x};
        \foreach \y in {1,2,3,4,5} \node[left] at (0,\y) {\small \y};
        \draw[very thick, plum] (0,1) -- (3,5);
        \draw[very thick, plum] (3,5) -- (6,2);
        \draw[very thick, plum] (6,2) -- (8,4);
        \fill[plum] (0,1) circle (2.5pt);
        \fill[plum] (8,4) circle (2.5pt);
      \end{tikzpicture}
      \end{center}

    \column{0.46\textwidth}
      \begin{itemize}\setlength{\itemsep}{8pt}
        \item Increasing for \(0 < x < 3\) and \(6 < x < 8\)
        \item Decreasing for \(3 < x < 6\)
        \item Intervals are named by \textbf{inputs} (\(x\)-values), never
              outputs.
      \end{itemize}
  \end{columns}

  \bigskip
  \textbf{Watch out:} decreasing \(\ne\) negative --- falling from 5 to 2
  stays positive the whole way.
\end{frame}

% SECTION 3 — CONCAVITY, TABLES
\begin{frame}[t, plain]
  \plumheader{3. Concavity --- How the Change Changes}

  \begin{columns}[T]
    \column{0.48\textwidth}
      \begin{block}{Tank, drain half-open}
        \begin{tabular}{|c|c|c|c|c|}
        \hline
        \(t\) & 0 & 2 & 4 & 6 \\
        \hline
        \(V\) & 0 & 14 & 20 & 24 \\
        \hline
        \end{tabular}\\[6pt]
        Gains: 14, 6, 4 --- rate of change \textbf{decreasing}\\
        \(\Rightarrow\) \textbf{concave down}
      \end{block}

    \column{0.48\textwidth}
      \begin{block}{Savings account}
        \begin{tabular}{|c|c|c|c|c|}
        \hline
        \(m\) & 0 & 1 & 2 & 3 \\
        \hline
        \(S\) & 100 & 120 & 145 & 175 \\
        \hline
        \end{tabular}\\[6pt]
        Gains: 20, 25, 30 --- rate of change \textbf{increasing}\\
        \(\Rightarrow\) \textbf{concave up}
      \end{block}
  \end{columns}

  \bigskip
  The concavity call comes from the \textbf{arithmetic of the gains}, not
  from squinting at a picture.
\end{frame}

% SECTION 3, CONT. — THE TWO BENDS
\begin{frame}[t, plain]
  \plumheader{The Two Bends}

  \begin{columns}[T]
    \column{0.48\textwidth}
      \begin{center}
      \begin{tikzpicture}[x=1.1cm, y=0.8cm]
        \draw[->, thick] (0,0) -- (3.4,0);
        \draw[->, thick] (0,0) -- (0,2.8);
        \draw[very thick, plum] (0,0.2) .. controls (1.8,0.5) .. (3,2.4);
      \end{tikzpicture}\\[4pt]
      slow, then fast --- \textbf{concave up} (bowl)
      \end{center}

    \column{0.48\textwidth}
      \begin{center}
      \begin{tikzpicture}[x=1.1cm, y=0.8cm]
        \draw[->, thick] (0,0) -- (3.4,0);
        \draw[->, thick] (0,0) -- (0,2.8);
        \draw[very thick, plum] (0,0.2) .. controls (1.2,2.1) .. (3,2.4);
      \end{tikzpicture}\\[4pt]
      fast, then slow --- \textbf{concave down} (dome)
      \end{center}
  \end{columns}

  \bigskip
  \begin{block}{Independent questions}
    ``Increasing or decreasing?'' and ``concave up or down?'' are
    \textbf{separate calls} --- the half-open-drain tank is increasing
    \textit{and} concave down at the same time. All four combinations exist.
  \end{block}
\end{frame}

% SECTION 4 — ZEROS
\begin{frame}[t, plain]
  \plumheader{4. Zeros}

  \begin{columns}[T]
    \column{0.5\textwidth}
      \begin{center}
      \begin{tikzpicture}[x=0.55cm, y=0.5cm]
        \draw[linegray, very thin, step=1] (0,0) grid (8,4);
        \draw[->, thick] (0,0) -- (8.6,0) node[below right] {\small \(x\)};
        \draw[->, thick] (0,0) -- (0,4.6) node[above] {\small \(g(x)\)};
        \foreach \x in {1,2,3,4,5,6,7,8} \node[below] at (\x,0) {\small \x};
        \foreach \y in {1,2,3,4} \node[left] at (0,\y) {\small \y};
        \draw[very thick, plum] (0,4) -- (2,0);
        \draw[very thick, plum] (2,0) -- (4,3);
        \draw[very thick, plum] (4,3) -- (6,0);
        \draw[very thick, plum] (6,0) -- (8,2);
        \fill[plum] (0,4) circle (2.5pt);
        \fill[plum] (8,2) circle (2.5pt);
      \end{tikzpicture}
      \end{center}

    \column{0.46\textwidth}
      \begin{itemize}\setlength{\itemsep}{8pt}
        \item A \textbf{zero}: an input whose output is 0 --- the graph meets
              the horizontal axis.
        \item Zeros of \(g\): \(x = 2\) and \(x = 6\).
      \end{itemize}
  \end{columns}

  \bigskip
  \begin{block}{A zero is an event}
    Drone: \(h(8) = 0\) --- it lands. Tank: \(w(7) = 0\) --- it's empty.
    Translate the zero into a sentence, just like \(f(3) = 7\) in Lesson 1.1.
  \end{block}
\end{frame}

% CLASS PRACTICE
\begin{frame}[t, plain]
  \plumheader{Class Practice}
  \sectionlabel[redacc]{Try It}

  \begin{enumerate}\setlength{\itemsep}{10pt}
    \item A tank is draining:

          \smallskip
          \begin{tabular}{|c|c|c|c|c|}
          \hline
          \(t\) (min) & 0 & 2 & 4 & 6 \\
          \hline
          \(w\) (gal) & 30 & 24 & 16 & 6 \\
          \hline
          \end{tabular}

          \smallskip
          \begin{enumerate}\setlength{\itemsep}{5pt}
            \item Increasing or decreasing?
            \item Find the drops. Concave up or concave down?
          \end{enumerate}
    \item If \(w(7) = 0\), say the sentence that zero is telling us.
    \item In one sentence: how can a function be increasing and concave down
          at the same time?
  \end{enumerate}
\end{frame}
