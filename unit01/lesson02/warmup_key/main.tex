\item Let $f(x) = 2x - 7$. Find $f(-4)$.

\begin{work}
  f(-4) &= 2(-4) - 7 \\
        &= -15
\end{work}

$f(-4) = $ \ans{$-15$}

\item The table shows a function $p$.

\smallskip
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{|c|c|c|c|c|}
\hline
$x$ & 1 & 2 & 3 & 4 \\
\hline
$p(x)$ & 8 & 5 & 2 & $-1$ \\
\hline
\end{tabular}
\smallskip

\begin{enumerate}[label=(\alph*), itemsep=4pt]
  \item $p(2) = $ \ans{5}
  \item Solve $p(x) = 8$: \quad $x = $ \ans{1}
  \item In your own words: as the inputs $x$ get bigger, what do the outputs
        $p(x)$ do?

        \vspace{0.02in}
        \ansline{They get smaller --- every step right drops the output by 3. (Today we name this: $p$ is decreasing.)}
\end{enumerate}

  \item Range: \ans{$\{1, 3, 6\}$ (the repeated 6 is listed once)}
\end{enumerate}

\end{enumerate}
