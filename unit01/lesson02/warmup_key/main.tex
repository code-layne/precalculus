\begin{enumerate}[itemsep=12pt]

\item Let $f(x) = 2x - 9$. Find $f(-3)$.

\begin{work}
  f(-3) &= 2(-3) - 9 \\
        &= -15
\end{work}

$f(-3) = $ \ans{$-15$}

\item The table shows a function $p$.

\smallskip
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{|c|c|c|c|c|c|}
\hline
$x$ & 1 & 2 & 3 & 4 & 5 \\
\hline
$p(x)$ & 20 & 14 & 10 & 7 & 5 \\
\hline
\end{tabular}
\smallskip

\begin{enumerate}[label=(\alph*), itemsep=6pt]
  \item $p(2) = $ \ans{14} \qquad Solve $p(x) = 10$: \quad $x = $ \ans{3}
  \item In plain words --- no vocabulary needed --- what do the outputs do as
        the inputs climb from 1 to 5?

        \vspace{0.05in}
        \ansline{They go down, but by less each time: first a drop of 6, then 4, then 3, then 2.}
\end{enumerate}

\item State the domain and range of the function
      $\{(0, 7),\ (2, 4),\ (5, 4),\ (6, 1)\}$.

\vspace{0.05in}
Domain: \ans{$\{0, 2, 5, 6\}$} \qquad Range: \ans{$\{1, 4, 7\}$}

\end{enumerate}
