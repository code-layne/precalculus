\begin{enumerate}[itemsep=10pt]

\item A pumpkin is planted at $t = 0$ and harvested at $t = 16$ weeks. Its
      weight grows the whole time: slowly at first, then faster and faster
      until week 10 --- after week 10 it keeps gaining, but the weekly gains
      shrink.

\begin{enumerate}[label=(\alph*), itemsep=6pt]
  \item Circle one: over $0 < t < 16$, the weight is \ans{increasing} /
        \textbf{decreasing}.
  \item The weight function is concave \textbf{up} on \ans{0}
        $< t <$ \ans{10} and concave \textbf{down} on \ans{10}
        $< t <$ \ans{16}.
\end{enumerate}

\item The graph of a function $f$ is shown.

\begin{center}
\begin{tikzpicture}[x=0.6cm, y=0.55cm]
  \draw[linegray, very thin, step=1] (0,0) grid (7,4);
  \draw[->, thick] (0,0) -- (7.6,0) node[below right] {\small $x$};
  \draw[->, thick] (0,0) -- (0,4.6) node[above] {\small $f(x)$};
  \foreach \x in {1,2,3,4,5,6,7} \node[below] at (\x,0) {\small \x};
  \foreach \y in {1,2,3,4} \node[left] at (0,\y) {\small \y};
  \draw[very thick, plum] (0,4) -- (4,0);
  \draw[very thick, plum] (4,0) -- (7,3);
  \fill[plum] (0,4) circle (2.5pt);
  \fill[plum] (7,3) circle (2.5pt);
\end{tikzpicture}
\end{center}

\begin{enumerate}[label=(\alph*), itemsep=4pt]
  \item Decreasing for \ans{0} $< x <$ \ans{4} \qquad
        Increasing for \ans{4} $< x <$ \ans{7}
  \item The zero of $f$ is $x = $ \ans{4}.
  \item Domain: \ans{0} $\le x \le$ \ans{7} \qquad
        Range: \ans{0} $\le y \le$ \ans{4}
\end{enumerate}

\item A savings account balance $S(m)$, in dollars, is shown for the first
      months of the year.

\smallskip
\begin{tabular}{|c|c|c|c|c|}
\hline
$m$ (months) & 0 & 1 & 2 & 3 \\
\hline
$S(m)$ (\$) & 400 & 440 & 490 & 550 \\
\hline
\end{tabular}
\smallskip

\begin{enumerate}[label=(\alph*), itemsep=4pt]
  \item The monthly gains are \ans{40} , \ans{50} ,
        \ans{60} dollars.
  \item Circle one: $S$ is \ans{concave up} / \textbf{concave down},
        because the rate of change is \ans{increasing}.
\end{enumerate}

\item Can a function be \textbf{decreasing} and \textbf{concave up} at the
      same time? If yes, describe a real context that behaves this way; if
      no, explain why not.

\vspace{0.02in}
\ansline{Yes. Cooling cocoa: the temperature falls, but each minute it drops by less than the minute before. The outputs decrease while the rate of change is increasing (drops shrinking) --- decreasing and concave up.}

\item A function is increasing and concave down on an interval. Which
      description fits?

\begin{enumerate}[label=\Alph*., itemsep=3pt]
  \item The outputs fall, dropping faster and faster.
  \item The outputs climb, but the gains shrink as the input grows. \quad \textcolor{keyred}{\textbf{$\leftarrow$ correct}}
  \item The outputs climb, and the gains grow as the input grows.
  \item The outputs fall, but the drops shrink as the input grows.
\end{enumerate}

\end{enumerate}

\begin{extensionbox}
A hiker climbs for 2 hours, rests on a ledge for 1 hour, then climbs the
rest of the way. During the resting hour, her elevation function is neither
increasing nor decreasing. Why does neither definition apply? (Hint: compare
$E(a)$ and $E(b)$ for two moments $a < b$ during the rest.)

\vspace{0.02in}
\ansline{During the rest, any two moments $a < b$ give the same elevation, $E(a) = E(b)$. Increasing requires $E(a) < E(b)$ and decreasing requires $E(a) > E(b)$ --- equality satisfies neither, so the function is constant there.}
\end{extensionbox}
