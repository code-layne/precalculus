\item \textbf{The same pen, the other question.} In class Ana chose the width.
      Ben chose the \textbf{length}, and wrote the same constraint
      $2w + \ell = 80$.

\begin{work}
   2w + \ell &= 80 \\
           w &= 40 - \frac{\ell}{2} \\
     A(\ell) &= \ell\left(40 - \frac{\ell}{2}\right) \\
             &= 40\ell - \frac{\ell^2}{2} \\
       A(40) &= 1600 - 800 \\
             &= 800
\end{work}

\begin{enumerate}[label=(\alph*), itemsep=4pt]
  \item Solve the constraint for $w$: \qquad $w = $ \blank{3.4cm}
  \item $A(\ell) = $ \blank{4cm} \qquad Domain: \blank{1cm} $< \ell <$
        \blank{1cm}
  \item $A(40) = $ \blank{2cm} ft$^2$. Ana's best pen was $w = 20$ ft. Is Ben's
        best length describing the same pen? \blank{1.6cm}
  \item Ana's domain was $0 < w < 40$; Ben's is different. In one sentence, why
        do two correct models of one pen have different domains?

\item \textbf{Using Ana's model.} Hers is $A(w) = 80w - 2w^2$ on $0 < w < 40$.

\begin{work}
   A(5) &= 80(5) - 2(5)^2 \\
        &= 400 - 50 \\
        &= 350 \\
  A(35) &= 80(35) - 2(35)^2 \\
        &= 2800 - 2450 \\
        &= 350 \\
  \frac{A(15) - A(5)}{15 - 5} &= \frac{750 - 350}{10} \\
                              &= 40
\end{work}

\item \textbf{An inverse-proportion model.} The force $F$ between two magnets is
      inversely proportional to the square of the distance $d$ between them, so
      $F = \dfrac{k}{d^2}$. At $d = 3$ cm the force measures 20 newtons.

\begin{work}
    20 &= \frac{k}{3^2} \\
     k &= 180 \\
  F(6) &= \frac{180}{6^2} \\
       &= 5
\end{work}

\begin{enumerate}[label=(\alph*), itemsep=4pt]
  \item $k = $ \blank{2cm} \qquad so $F(d) = $ \blank{3cm}
  \item $F(6) = $ \blank{2cm} newtons
  \item Domain restriction: $d$ \blank{2cm}, because \blank{5cm}
\end{enumerate}

\item Two AP-style multiple-choice questions.

\begin{enumerate}[label=(\alph*), itemsep=6pt]
  \item A table of equally spaced inputs has second differences that are all
        equal to 4. Which statement is correct?

        \begin{enumerate}[label=\Alph*., itemsep=2pt]
          \item Linear, because the differences are constant.
          \item Quadratic, because the second differences are constant.
          \item Degree 5, because the table has five rows.
          \item No polynomial model fits, because the second differences are
                not zero.
        \end{enumerate}

  \item A rectangular flag has a \textbf{fixed height of 3 feet}. Its area is
        modeled as a function of its width. The model is

        \begin{enumerate}[label=\Alph*., itemsep=2pt]
          \item linear.
          \item quadratic, because area always is.
          \item cubic.
          \item piecewise.
        \end{enumerate}
\end{enumerate}

\end{enumerate}
