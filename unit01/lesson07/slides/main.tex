% SECTION 1 — PICK THE QUESTION
\begin{frame}[t, plain]
  \plumheader{1. Pick the Question}
  \sectionlabel[redacc]{The Whole Lesson in One Decision}

  \begin{columns}[T, onlytextwidth]
    \column{0.46\textwidth}
      \begin{center}
      \begin{tikzpicture}[x=1.1cm, y=1.1cm]
        \fill[lilac] (0,0) rectangle (3.4,1.3);
        \draw[very thick, slate] (-0.3,1.3) -- (3.7,1.3);
        \node[slate, above, font=\scriptsize] at (1.7,1.35) {barn wall --- no fence};
        \draw[very thick, plum] (0,1.3) -- (0,0) -- (3.4,0) -- (3.4,1.3);
        \node[plum, left,  font=\small] at (0,0.65) {\(w\)};
        \node[plum, right, font=\small] at (3.4,0.65) {\(w\)};
        \node[plum, below, font=\small] at (1.7,0) {\(\ell\)};
        \node[slate, font=\scriptsize\itshape] at (1.7,0.65) {area \(A\)};
      \end{tikzpicture}
      \end{center}

      \smallskip
      Four numbers: \(80\) (fixed), \(w\), \(\ell\), \(A\). A function accepts
      \textbf{one} question.

    \column{0.50\textwidth}
      \begin{block}{A model starts with a decision}
        Naming the question is the first move --- and everything follows from
        it: which differences you may take, what the constraint has to do,
        which inputs are allowed, what the units say.
      \end{block}

      \medskip
      \begin{block}{The hook, answered}
        Ana's question was the \textbf{width}. Ben's was the
        \textbf{length}. Neither is wrong.

        \smallskip
        The 80 is neither question nor answer --- it is the
        \textbf{constraint}: \(2w + \ell = 80\).
      \end{block}
  \end{columns}
\end{frame}

% SECTION 2 — THE TABLE PICKED IT
\begin{frame}[t, plain]
  \plumheader{2. When the Table Already Picked the Question}
  \sectionlabel[goldacc]{Top Row Is the Question --- Check It First}

  \begin{columns}[T, onlytextwidth]
    \column{0.47\textwidth}
      \begin{block}{Why equal spacing is required}
        A difference compares two answers across \textit{one step of the
        question}. Fair only if every step is the same size.

        \smallskip
        \renewcommand{\arraystretch}{1.15}
        \begin{tabular}{@{}lccc@{}}
          \(x\)          & 1 & 2 & 4 \\
          \(g(x)\)       & 3 & 7 & 15 \\
          change         & --- & 4 & 8 \\
          \textbf{per unit} & --- & \textbf{4} & \textbf{4} \\
        \end{tabular}

        \smallskip
        \(g\) \textit{is} linear. The raw differences lied.
      \end{block}

    \column{0.49\textwidth}
      \begin{block}{Ana's table --- widths step by 5}
        \renewcommand{\arraystretch}{1.15}
        \begin{tabular}{@{}lccccc@{}}
          \(w\)  & 5 & 10 & 15 & 20 & 25 \\
          \(A\)  & 350 & 600 & 750 & 800 & 750 \\
          1st    & --- & 250 & 150 & 50 & \(-50\) \\
          2nd    & --- & --- & \(-100\) & \(-100\) & \(-100\) \\
        \end{tabular}

        \smallskip
        Second differences constant \(\Rightarrow\) \textbf{quadratic}.
      \end{block}

      \medskip
      \begin{block}{A warning about tables}
        Any \(n+1\) points fit some degree-\(n\) polynomial. ``It fits'' is not
        a reason --- the context gets a vote.
      \end{block}
  \end{columns}
\end{frame}

% SECTION 3 — READING THE CONTEXT
\begin{frame}[t, plain]
  \plumheader{3. When Nobody Picked It: Reading the Context}
  \sectionlabel[redacc]{How Does the Answer Move as the Question Grows?}

  \begin{columns}[T, onlytextwidth]
    \column{0.50\textwidth}
      \renewcommand{\arraystretch}{1.25}
      \begin{tabular}{@{}p{0.68\linewidth}l@{}}
        \hline
        Same amount each step & linear \\
        \hline
        \textbf{Two} lengths multiplied & quadratic \\
        \hline
        \textbf{Three} lengths multiplied & cubic \\
        \hline
        Answer turns several times & polynomial \\
        \hline
        Question sits underneath & rational \\
        \hline
        Rule changes at a threshold & piecewise \\
        \hline
      \end{tabular}

    \column{0.46\textwidth}
      \begin{block}{The clue is not the noun}
        ``Area'' does not mean quadratic by itself.

        \smallskip
        Length always \(2\times\) width: \(A(w) = 2w^2\) ---
        \textbf{quadratic}, two lengths move.

        \smallskip
        Length fixed at 9 ft: \(A(w) = 9w\) --- \textbf{linear}, one length
        moves.
      \end{block}

      \medskip
      \begin{center}
        Count the lengths that move with the question.
      \end{center}
  \end{columns}
\end{frame}

% SECTION 4 — BUILD IT
\begin{frame}[t, plain]
  \plumheader{4. Build It: the Constraint Answers for the Rest}
  \sectionlabel[goldacc]{Two Formulas, One Pen}

  \begin{columns}[T, onlytextwidth]
    \column{0.30\textwidth}
      \begin{block}{The four steps}
        \begin{enumerate}\setlength{\itemsep}{2pt}
          \item Name the \textbf{question}.
          \item Write the \textbf{constraint}.
          \item Solve it for the other quantity.
          \item \textbf{Substitute}.
        \end{enumerate}
      \end{block}

    \column{0.33\textwidth}
      \textbf{Ana} --- question \(w\):
      \begin{align*}
        \ell &= 80 - 2w \\
        A(w) &= w(80 - 2w) \\
             &= 80w - 2w^2
      \end{align*}
      \(A(20) = 800\) \checkmark

    \column{0.33\textwidth}
      \textbf{Ben} --- question \(\ell\):
      \begin{align*}
        w &= \frac{80 - \ell}{2} \\
        A(\ell) &= \frac{80-\ell}{2}\cdot \ell \\
                &= 40\ell - \frac{\ell^2}{2}
      \end{align*}
      \(A(40) = 800\) \checkmark
  \end{columns}

  \medskip
  \begin{center}
    Same constraint, solved for a different letter. Neither notebook is wrong.
  \end{center}
\end{frame}

% SECTION 5 — USE IT
\begin{frame}[t, plain]
  \plumheader{5. Use the Model}
  \sectionlabel[redacc]{The Units Name the Question You Chose}

  \begin{columns}[T, onlytextwidth]
    \column{0.48\textwidth}
      \begin{block}{Average rate of change, with units}
        Ana on \([10,20]\): \(+20\); on \([20,30]\): \(-20\)

        \smallskip
        \textbf{square feet per foot of width}

        \smallskip
        Ben straddling the same pen, on \([30,40]\) and \([40,50]\): \(+5\) and
        \(-5\), in square feet per foot of \textbf{length}.
      \end{block}

    \column{0.48\textwidth}
      \begin{block}{The sign change locates the best pen}
        \(+\) then \(-\), so the maximum sits between.

        \smallskip
        \(w = -\dfrac{b}{2a} = -\dfrac{80}{2(-2)} = \mathbf{20}\), so
        \(\ell = 40\) and \(A = 800\) ft\(^2\).
      \end{block}

      \medskip
      \begin{center}
        The class vote said \textit{square}. The best pen is
        \(20 \times 40\) --- one side is free.
      \end{center}
  \end{columns}
\end{frame}

% SECTION 6 — THE FINE PRINT
\begin{frame}[t, plain]
  \plumheader{6. The Fine Print: Which Questions Are Allowed}
  \sectionlabel[goldacc]{A Domain Is a List of Legal Questions}

  \begin{columns}[T, onlytextwidth]
    \column{0.48\textwidth}
      \begin{block}{Different questions, different domains}
        Ana: \(0 < w < 40\). \quad Ben: \(0 < \ell < 80\).

        \smallskip
        Same pen. The domains differ because the \textit{questions} differ.

        \smallskip
        \(A(-3) = -258\) is not bad arithmetic --- it answers a question the
        situation does not allow.
      \end{block}

    \column{0.48\textwidth}
      \begin{block}{Say the assumption out loud}
        All 80 ft is used, the barn is at least 40 ft long, the pen is exactly
        rectangular, no gate gap.
      \end{block}

      \medskip
      \begin{block}{When one rule is not enough}
        If the barn is only 30 ft, the constraint changes past a threshold ---
        the model becomes \textbf{piecewise}.
      \end{block}
  \end{columns}
\end{frame}

% CLASS PRACTICE
\begin{frame}[t, plain]
  \plumheader{Class Practice}
  \sectionlabel[redacc]{Name the Question First, Every Time}

  \begin{enumerate}\setlength{\itemsep}{10pt}
    \item A table has \(x = 0,1,2,3,4\) and \(y = 0,1,8,27,64\). Take
          differences until they are constant and name the model type.

    \item A garden bed against a wall uses 30 ft of edging on the other three
          sides. Take the width as your question: build \(A(w)\) and state its
          domain.

    \item Two rugs. One has length always 3 times its width; the other has a
          fixed length of 8 feet. Both areas, as functions of the width ---
          why is only one of them quadratic?
  \end{enumerate}
\end{frame}
