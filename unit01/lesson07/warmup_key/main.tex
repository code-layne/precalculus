\begin{enumerate}[label=(\alph*), itemsep=4pt]
  \item Is $f$'s rate of change constant? \ans{Yes --- it is 3 per step.}
  \item Look at the \textbf{$x$ row} of the $g$ table. What is different about it?

        \ansline{The inputs are not equally spaced --- they step by 1, then by 2.}
\end{enumerate}

\item One equation, solved two different ways.

\begin{work}
   2w + \ell &= 80 \\
        \ell &= 80 - 2w \\
          2w &= 80 - \ell \\
           w &= \frac{80 - \ell}{2} \\
  w(80 - 2w) &= 10\big(80 - 2(10)\big) \\
             &= 600
\end{work}

\begin{enumerate}[label=(\alph*), itemsep=4pt]
  \item Solve $2w + \ell = 80$ for $\ell$: \qquad $\ell = $ \ans{$80 - 2w$}
  \item Solve the \textbf{same} equation for $w$: \qquad $w = $ \ans{$(80 - \ell)/2$}
  \item Using $w = 10$: \quad $w(80 - 2w) = $ \ans{600}
\end{enumerate}

\item A tank is filling. The table gives the volume $V$, in gallons, after
      $t$ minutes.

\smallskip
\renewcommand{\arraystretch}{1.4}
\begin{center}
\begin{tabular}{|c|c|c|c|c|}
\hline
$t$ (minutes) & 0 & 2 & 4 & 6 \\
\hline
$V(t)$ (gallons) & 12 & 20 & 32 & 44 \\
\hline
\end{tabular}
\end{center}

\smallskip
Find the average rate of change of $V$ from $t = 2$ to $t = 6$.

\begin{work}
  \frac{V(6) - V(2)}{6 - 2} &= \frac{44 - 20}{4} \\
                            &= 6
\end{work}

Average rate of change $= $ \ans{6} \qquad Units: \ans{gallons per minute}

\end{enumerate}
