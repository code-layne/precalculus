\begin{enumerate}[itemsep=18pt]

\item Consider the rational function $r(x) = \dfrac{5x^2 - 3}{2x^2 + x - 1}$.

\begin{enumerate}[label=(\alph*), itemsep=10pt]
  \item What is the quotient of the leading terms?

        \medskip
        Leading term of numerator: \ans{$5x^2$} \quad
        Leading term of denominator: \ans{$2x^2$}

        \medskip
        Quotient: \ans{$5x^2 / 2x^2 = 5/2$}

  \item Write the end behavior in limit notation:

        \medskip
        $\displaystyle\lim_{x \to \infty} r(x) = $ \ans{$\dfrac{5}{2}$}
        \qquad
        $\displaystyle\lim_{x \to -\infty} r(x) = $ \ans{$\dfrac{5}{2}$}

  \item Does the graph of $r$ have a horizontal asymptote?
        Circle one: \quad \ans{Yes} \quad / \quad No

        \medskip
        If yes, the horizontal asymptote is at $y = $ \ans{$\dfrac{5}{2}$}.
\end{enumerate}

\item A rational function has a numerator of degree 3 and a denominator of degree 5.

\begin{enumerate}[label=(\alph*), itemsep=8pt]
  \item What type of horizontal asymptote does it have?
        \ans{Horizontal asymptote at $y = 0$ (deg num $<$ deg denom)}
  \item Write the limit statements:

        \medskip
        $\displaystyle\lim_{x \to \infty} r(x) = $ \ans{$0$}
        \qquad
        $\displaystyle\lim_{x \to -\infty} r(x) = $ \ans{$0$}
\end{enumerate}

\end{enumerate}
