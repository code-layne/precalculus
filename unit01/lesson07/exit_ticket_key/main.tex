\begin{enumerate}[itemsep=10pt]

\item Complete the difference table and identify the best model type.

      \medskip
      \renewcommand{\arraystretch}{1.4}
      \begin{tabular}{|c|c|c|c|}
      \hline
      $x$ & $y$ & $\Delta y$ & $\Delta^2 y$ \\
      \hline
      0 & 5  & ---         & ---         \\
      \hline
      1 & 9  & \ans{4} & ---      \\
      \hline
      2 & 17 & \ans{8} & \ans{4} \\
      \hline
      3 & 29 & \ans{12} & \ans{4} \\
      \hline
      4 & 45 & \ans{16} & \ans{4} \\
      \hline
      \end{tabular}

      \medskip
      The \ans{second} differences are constant, so the best model type is
      \ans{quadratic} (degree \ans{2}).

\item A shipping company charges a flat rate of \$8 for packages up to 5 lb,
      then adds \$1.50 per pound above 5 lb. Circle the best model type and give one reason.

      \medskip
      \textbf{A.} Linear \qquad \textbf{B.} Quadratic \qquad \textbf{C.} Cubic \qquad \textbf{D.} Piecewise-defined \textcolor{keyred}{\textbf{$\leftarrow$ correct}}

      \medskip
      One reason: \ansline{The pricing rule changes at 5 lb, so different formulas apply on different intervals.}

\item A quadratic function models the area of a rectangular garden with one fixed wall
      (using 60 m of fencing for three sides). State one \textbf{domain restriction}
      and one \textbf{model assumption}.

      \medskip
      Domain restriction: \ansline{$0 < w < 30$ where $w$ is the width (width must be positive and less than half the available fencing).}

      Model assumption: \ansline{The garden is a perfect rectangle and exactly 60 m of fencing is used on the three open sides.}

\end{enumerate}
