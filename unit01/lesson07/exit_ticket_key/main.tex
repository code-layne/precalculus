\item A dog run is built against the side of a garage using \textbf{40 feet} of
      fencing on the other three sides. Let $w$ be the width (the two fenced
      ends) and $\ell$ the length along the garage.

\begin{work}
   2w + \ell &= 40 \\
        \ell &= 40 - 2w \\
        A(w) &= w(40 - 2w) \\
             &= 40w - 2w^2
\end{work}

\begin{enumerate}[label=(\alph*), itemsep=4pt]
  \item Constraint: \ans{2} $w \; + \;$ \ans{$\ell$} $= 40$
  \item $A(w) = $ \ans{$40w - 2w^2$}
  \item Domain: \ans{0} $< w <$ \ans{20}
\end{enumerate}
