\begin{notesbox}{Zeros of Rational Functions (EK 1.8.A.1)}
\textbf{Rule:} For $r(x) = \dfrac{p(x)}{q(x)}$, we have $r(x) = 0$ when:
\begin{itemize}[itemsep=3pt]
  \item $p(x) = 0$ (numerator equals \blank{2cm}), AND
  \item $q(x) \ne 0$ ($x$ is in the \blank{2cm} of $r$).
\end{itemize}

\medskip
\textbf{Worked Example.} Find all zeros of $r(x) = \dfrac{x^2 - 4}{x - 3}$.

\medskip
\renewcommand{\arraystretch}{1.5}
\begin{tabular}{@{} r l @{\qquad} l @{}}
\textbf{Step 1} & Find domain restrictions. & $q(x) = 0 \Rightarrow$ \blank{3.5cm} \\
                &                           & Domain: all reals except $x =$ \blank{1.5cm} \\[4pt]
\textbf{Step 2} & Set numerator $= 0$.      & $x^2 - 4 = 0 \Rightarrow (x - \blank{1cm})(x + \blank{1cm}) = 0$ \\
                &                           & Solutions: $x =$ \blank{1.5cm} and $x =$ \blank{1.5cm} \\[4pt]
\textbf{Step 3} & Check domain.             & Is $x = 2$ in the domain? \blank{1.5cm} \\
                &                           & Is $x = -2$ in the domain? \blank{1.5cm} \\[4pt]
\textbf{Step 4} & State zeros.              & Zeros of $r$: $x =$ \blank{3cm} \\
\end{tabular}
\end{notesbox}
