\begin{enumerate}[itemsep=18pt]

\item A table of values for $f(x)$ is given below (where $x$ is in seconds and $f(x)$ is in feet).

      \smallskip
      \renewcommand{\arraystretch}{1.3}
      \begin{tabular}{|c|c|c|}
      \hline
      $x$ & 2 & 10 \\
      \hline
      $f(x)$ & 8 & 32 \\
      \hline
      \end{tabular}
      \smallskip

      Compute the average rate of change of $f$ on the interval $[2, 10]$.
      Include appropriate units.

      \vspace{0.08in}
      $\text{ARC} = \dfrac{f(10) - f(2)}{10 - 2} = \dfrac{{\color{keyred}\mathbf{24}}}{{\color{keyred}\mathbf{8}}} = {}$ \ans{3}

      \vspace{0.04in}
      Units: \ans{feet per second (ft/s)}

\item Evaluate $p(x) = -3x^3 + 2x^2 - x + 5$ at $x = -2$. Show each step.

      \vspace{0.08in}
      $p(-2) = -3({\color{keyred}\mathbf{-2}})^3 + 2({\color{keyred}\mathbf{-2}})^2 - ({\color{keyred}\mathbf{-2}}) + 5 = {}$ \ans{$-3(-8)+2(4)+2+5 = 24+8+2+5 = 39$}

\item A scientist says: ``As the distance between two magnets doubles, the force between them
      drops to one-fourth of its original value.'' Circle the model type that best fits
      this description and explain in one sentence.

      \medskip
      \textbf{A.}~Linear \qquad \textbf{B.}~Quadratic \qquad \textbf{C.}~\ans{Rational (inverse proportion)} \qquad \textbf{D.}~Cubic

      \vspace{0.08in}
      Explanation: \ansline{Force $\propto 1/d^2$; doubling $d$ multiplies $d^2$ by 4, so force becomes $1/4$ --- this is an inverse-square (rational) model.}

\end{enumerate}
