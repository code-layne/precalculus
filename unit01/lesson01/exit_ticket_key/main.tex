\item Let $f(x) = 6x - 5$.

\begin{work}
  f(3) &= 6(3) - 5 \\
       &= 13
\end{work}

$f(3) = $ \ans{13} \qquad $f(-1) = $ \ans{$-11$}

\item The table shows a function $g$.

\smallskip
\begin{tabular}{|c|c|c|c|c|}
\hline
$x$ & 0 & 1 & 2 & 3 \\
\hline
$g(x)$ & 5 & 8 & 11 & 14 \\
\hline
\end{tabular}
\smallskip

\begin{enumerate}[label=(\alph*), itemsep=4pt]
  \item $g(2) = $ \ans{11}
  \item Solve $g(x) = 8$: \quad $x = $ \ans{1}
\end{enumerate}
