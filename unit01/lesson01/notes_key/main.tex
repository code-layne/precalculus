\begin{objectivebox}
By the end of this lesson, I will be able to\ldots
\begin{enumerate}[itemsep=2pt]
  \item Read and write function \ans{notation}, and explain the three facts a
        statement like $f(3) = 7$ carries.
  \item \ans{Evaluate} a function given by a formula, a table, or a graph.
  \item State the \ans{domain} and \ans{range} of a function.
\end{enumerate}
\end{objectivebox}

\vspace{0.1in}

\boxguard
\begin{vocabbox}
Fill in each term as we define it together.
\par\vspace{2pt}
\noindent\textbf{\textcolor{plum}{Function (from 1.0):}}\\[1pt]
\ansline{A rule that sends each input to exactly one output.}\\[3pt]
\noindent\textbf{\textcolor{plum}{Function notation:}}\\[1pt]
\ansline{The symbol $f(x)$, read ``$f$ of $x$'': $f$ names the rule, $x$ is the input, $f(x)$ is the output.}\\[3pt]
\noindent\textbf{\textcolor{plum}{Evaluate:}}\\[1pt]
\ansline{Find the output a function produces for a given input.}\\[3pt]
\noindent\textbf{\textcolor{plum}{Domain:}}\\[1pt]
\ansline{The set of all inputs a function uses.}\\[3pt]
\noindent\textbf{\textcolor{plum}{Range:}}\\[1pt]
\ansline{The set of all outputs a function produces; a repeated output is listed once.}
\end{vocabbox}

\vspace{0.1in}

\boxguard
\begin{hookbox}
A ball is launched from the ground, and $h(t)$ gives its height in feet
$t$ seconds after launch. On the board is one short equation:
$h(2) = 61$.

What is that equation \textit{saying}? Write it as a full sentence.
\ansline{Two seconds after launch, the ball is 61 feet above the ground. (Input 2 seconds, output 61 feet.)}
\end{hookbox}

\vspace{0.1in}

\boxguard
\begin{notesbox}{1. Function Notation --- Naming the Machine}
Yesterday's trustworthy machine gets a name. The notation $f(x)$ has three
parts:

\begin{itemize}[itemsep=4pt]
  \item $f$ is the \ans{name} of the function (the machine itself),
  \item $x$ is the \ans{input} (what goes in the slot),
  \item $f(x)$ is the \ans{output} (what comes out).
\end{itemize}

$f(x)$ is read ``$f$ \ans{of} $x$'' --- it is \textbf{not} $f$ times $x$.

\medskip
\textbf{In context.} A phone plan charges by the minute: $C(m) = 0.10m + 5$.

\begin{itemize}[itemsep=4pt]
  \item The function's name is \ans{$C$}; the input $m$ stands for
        \ans{minutes used}; the output $C(m)$ stands for \ans{the cost of the bill}.
\end{itemize}

\textbf{One statement, three facts.} The single statement $f(3) = 7$ tells us:

\begin{itemize}[itemsep=4pt]
  \item the input is \ans{3}, \quad the output is \ans{7},
  \item and the point ( \ans{3} , \ans{7} ) is on the graph of $f$.
\end{itemize}
\end{notesbox}

\vspace{0.1in}

\boxguard
\begin{notesbox}{2. Evaluating from a Formula}
To \textbf{evaluate} a function, substitute the input into the formula ---
\textit{always inside parentheses} --- then do the arithmetic.

\medskip
\textbf{Example 1.} Let $f(x) = 2x + 5$. Find $f(4)$.

\begin{work}
  f(4) &= 2(4) + 5 \\
       &= 13
\end{work}

\textbf{Example 2.} Same $f$. Find $f(-3)$.

\begin{work}
  f(-3) &= 2(-3) + 5 \\
        &= -1
\end{work}

\textbf{Example 3.} Let $g(x) = x^2 - 4$. Find $g(-2)$.

\begin{work}
  g(-2) &= (-2)^2 - 4 \\
        &= 4 - 4 \\
        &= 0
\end{work}

\textbf{Watch the trap:} $(-2)^2 = $ \ans{4}, but $-2^2 = $ \ans{$-4$}.
The parentheses habit is what saves you.
\end{notesbox}

\vspace{0.1in}

\boxguard
\begin{notesbox}{3. Evaluating from a Table and a Graph}
A table \textit{is} a function: each column is an input--output pair.

\smallskip
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{|c|c|c|c|c|}
\hline
$x$ & 0 & 2 & 4 & 6 \\
\hline
$g(x)$ & 3 & 7 & 11 & 15 \\
\hline
\end{tabular}
\smallskip

\begin{itemize}[itemsep=4pt]
  \item $g(4) = $ \ans{11} \quad (input 4, read the output)
  \item Solve $g(x) = 3$: \quad $x = $ \ans{0} \quad (output 3, find the
        input)
\end{itemize}

\textbf{From a graph.} The graph of $f$ is below.

\begin{center}
\begin{tikzpicture}[x=0.6cm, y=0.55cm]
  \draw[linegray, very thin, step=1] (0,0) grid (9,5);
  \draw[->, thick] (0,0) -- (9.6,0) node[below right] {\small $x$};
  \draw[->, thick] (0,0) -- (0,5.6) node[above] {\small $f(x)$};
  \foreach \x in {1,2,3,4,5,6,7,8,9} \node[below] at (\x,0) {\small \x};
  \foreach \y in {1,2,3,4,5} \node[left] at (0,\y) {\small \y};
  \draw[very thick, plum] (0,1) -- (3,4);
  \draw[very thick, plum] (3,4) -- (6,4);
  \draw[very thick, plum] (6,4) -- (9,1);
  \fill[plum] (0,1) circle (2.5pt);
  \fill[plum] (9,1) circle (2.5pt);
\end{tikzpicture}
\end{center}

\begin{itemize}[itemsep=4pt]
  \item $f(2) = $ \ans{3} \quad (start at $x = 2$, walk up to the graph,
        read across)
  \item $f(5) = $ \ans{4}
  \item Solve $f(x) = 2$: \quad $x = $ \ans{1} and $x = $ \ans{8}
        \quad (two answers --- many-to-one is allowed!)
\end{itemize}

\textbf{The direction distinction.} ``Find $f(2)$'' hands you an
\ans{input} and asks for the \ans{output}. ``Solve $f(x) = 2$'' hands you
an \ans{output} and asks for the \ans{input(s)}.
\end{notesbox}

\vspace{0.1in}

\boxguard
\begin{notesbox}{4. Domain and Range}
The \textbf{domain} is the set of all \ans{inputs} a function uses; the
\textbf{range} is the set of all \ans{outputs} it produces. A repeated output
is listed \ans{once}.

\medskip
\textbf{From a list of points.} Let $f = \{(1, 3),\ (2, 5),\ (4, 5),\ (6, 8)\}$.

\begin{itemize}[itemsep=4pt]
  \item Domain: \ans{$\{1, 2, 4, 6\}$} \quad Range: \ans{$\{3, 5, 8\}$}
\end{itemize}

\textbf{From a graph.} Look back at the graph in Part 3. The graph runs from
$x = 0$ to $x = 9$ and its heights run from $1$ to $4$.

\begin{itemize}[itemsep=4pt]
  \item Domain: \ans{0} $\le x \le$ \ans{9} \quad
        Range: \ans{1} $\le y \le$ \ans{4}
\end{itemize}

\textbf{From a formula (first look).} Let $g(x) = \dfrac{10}{x - 2}$.
At $x = 2$ the denominator is \ans{0}, and division by zero is
\ans{undefined} --- so $2$ is excluded. The domain is all real numbers except
\ans{2}.
\end{notesbox}

\boxguard
\begin{practicebox}
\begin{enumerate}[itemsep=8pt]
  \item Let $f(x) = 3x - 7$. Evaluate each.

        \begin{work}
          f(5) &= 3(5) - 7 \\
               &= 8
        \end{work}

        $f(5) = $ \ans{8}

        \begin{work}
          f(-1) &= 3(-1) - 7 \\
                &= -10
        \end{work}

        $f(-1) = $ \ans{$-10$}

  \item The table shows a function $p$.

        \smallskip
        \renewcommand{\arraystretch}{1.4}
        \begin{tabular}{|c|c|c|c|c|}
        \hline
        $x$ & 1 & 2 & 3 & 4 \\
        \hline
        $p(x)$ & 9 & 6 & 6 & 3 \\
        \hline
        \end{tabular}
        \smallskip

        \begin{enumerate}[label=(\alph*), itemsep=4pt]
          \item $p(2) = $ \ans{6}
          \item Solve $p(x) = 3$: \quad $x = $ \ans{4}
          \item Domain of $p$: \ans{$\{1, 2, 3, 4\}$} \quad Range of $p$: \ans{$\{3, 6, 9\}$}
        \end{enumerate}
\end{enumerate}
\end{practicebox}
