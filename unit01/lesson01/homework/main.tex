\item The function $W$ gives a puppy's weight in pounds $d$ days after it is
      born.

\begin{enumerate}[label=(\alph*), itemsep=6pt]
  \item Write the statement $W(30) = 9$ as a full sentence.

        \vspace{0.05in}
        \writeline

  \item Write this sentence in function notation: ``after 60 days the puppy
        weighs 15 pounds.'' \quad \blank{3cm}
\end{enumerate}

\item Let $f(x) = 4x + 7$ and $g(x) = x^2 - 5x$. Evaluate each.

\begin{enumerate}[label=(\alph*), itemsep=6pt]
  \item $f(6) = $ \blank{2cm}

  \item Find $f(-3)$.

        \begin{work}
          f(-3) &= 4(-3) + 7 \\
                &= -5
        \end{work}

        $f(-3) = $ \blank{2cm}

  \item Find $g(-2)$.

        \begin{work}
          g(-2) &= (-2)^2 - 5(-2) \\
                &= 4 + 10 \\
                &= 14
        \end{work}

        $g(-2) = $ \blank{2cm}
\end{enumerate}

\item Which of the following is the \textbf{range} of the function
      $\{(2, 3),\ (4, 3),\ (6, 1),\ (8, 5)\}$?
