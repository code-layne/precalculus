\begin{enumerate}[itemsep=10pt]

\item The function $p$ gives a plant's height in centimeters $t$ weeks after it
      is planted.

\begin{enumerate}[label=(\alph*), itemsep=6pt]
  \item Write the statement $p(3) = 14$ as a full sentence.

        \vspace{0.02in}
        \ansline{Three weeks after planting, the plant is 14 centimeters tall.}

  \item Write this sentence in function notation: ``after 5 weeks, the plant is
        20 centimeters tall.'' \quad \ans{$p(5) = 20$}
\end{enumerate}

\item Let $f(x) = 2x + 5$ and $g(x) = x^2 - 1$. Evaluate each.

\begin{enumerate}[label=(\alph*), itemsep=6pt]
  \item $f(3) = $ \ans{11}

  \item Find $f(-4)$.

        \begin{work}
          f(-4) &= 2(-4) + 5 \\
                &= -3
        \end{work}

        $f(-4) = $ \ans{$-3$}

  \item Find $g(-2)$.

        \begin{work}
          g(-2) &= (-2)^2 - 1 \\
                &= 3
        \end{work}

        $g(-2) = $ \ans{3}
\end{enumerate}

\item The table shows a function $h$.

\smallskip
\begin{tabular}{|c|c|c|c|c|c|}
\hline
$x$ & 0 & 1 & 2 & 3 & 4 \\
\hline
$h(x)$ & 6 & 4 & 2 & 0 & $-2$ \\
\hline
\end{tabular}
\smallskip

\begin{enumerate}[label=(\alph*), itemsep=4pt]
  \item $h(3) = $ \ans{0}
  \item Solve $h(x) = 4$: \quad $x = $ \ans{1}
  \item Domain of $h$: \ans{$\{0, 1, 2, 3, 4\}$}
\end{enumerate}

\item The graph of a function $g$ is shown.

\begin{center}
\begin{tikzpicture}[x=0.6cm, y=0.55cm]
  \draw[linegray, very thin, step=1] (0,0) grid (8,5);
  \draw[->, thick] (0,0) -- (8.6,0) node[below right] {\small $x$};
  \draw[->, thick] (0,0) -- (0,5.6) node[above] {\small $g(x)$};
  \foreach \x in {1,2,3,4,5,6,7,8} \node[below] at (\x,0) {\small \x};
  \foreach \y in {1,2,3,4,5} \node[left] at (0,\y) {\small \y};
  \draw[very thick, plum] (0,0) -- (2,4);
  \draw[very thick, plum] (2,4) -- (5,4);
  \draw[very thick, plum] (5,4) -- (8,1);
  \fill[plum] (0,0) circle (2.5pt);
  \fill[plum] (8,1) circle (2.5pt);
\end{tikzpicture}
\end{center}

\begin{enumerate}[label=(\alph*), itemsep=4pt]
  \item $g(2) = $ \ans{4} \qquad $g(7) = $ \ans{2}
  \item Solve $g(x) = 2$: \quad $x = $ \ans{1} and $x = $ \ans{7}
  \item Domain: \ans{0} $\le x \le$ \ans{8} \qquad
        Range: \ans{0} $\le y \le$ \ans{4}
\end{enumerate}

\item Which of the following is the \textbf{domain} of the function
      $\{(2, 3),\ (4, 3),\ (6, 1),\ (8, 5)\}$?

\begin{enumerate}[label=\Alph*., itemsep=3pt]
  \item $\{2, 4, 6, 8\}$ \quad \textcolor{keyred}{\textbf{$\leftarrow$ correct}}
  \item $\{3, 1, 5\}$
  \item $\{2, 3\}$
  \item all real numbers
\end{enumerate}

\end{enumerate}

\boxguard
\begin{extensionbox}
The formula $y = \dfrac{12}{x - 5}$ refuses one input. Which $x$ is forbidden,
and why? Then write the domain in words.

\vspace{0.02in}
\ansline{$x = 5$ is forbidden: it makes the denominator 0, and division by zero is undefined. The domain is all real numbers except 5.}
\end{extensionbox}
