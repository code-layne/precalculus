\begin{enumerate}[itemsep=10pt]

\item The function $W$ gives a puppy's weight in pounds $d$ days after it is
      born.

\begin{enumerate}[label=(\alph*), itemsep=6pt]
  \item Write the statement $W(30) = 9$ as a full sentence.

        \vspace{0.05in}
        \ansline{Thirty days after birth, the puppy weighs 9 pounds.}

  \item Write this sentence in function notation: ``after 60 days the puppy
        weighs 15 pounds.'' \quad \ans{$W(60) = 15$}
\end{enumerate}

\item Let $f(x) = 4x + 7$ and $g(x) = x^2 - 5x$. Evaluate each.

\begin{enumerate}[label=(\alph*), itemsep=6pt]
  \item $f(6) = $ \ans{31}

  \item Find $f(-3)$.

        \begin{work}
          f(-3) &= 4(-3) + 7 \\
                &= -5
        \end{work}

        $f(-3) = $ \ans{$-5$}

  \item Find $g(-2)$.

        \begin{work}
          g(-2) &= (-2)^2 - 5(-2) \\
                &= 4 + 10 \\
                &= 14
        \end{work}

        $g(-2) = $ \ans{14}
\end{enumerate}

\item The table shows a function $h$.

\smallskip
\begin{tabular}{|c|c|c|c|c|c|}
\hline
$x$ & 0 & 1 & 2 & 3 & 4 \\
\hline
$h(x)$ & 10 & 7 & 4 & 4 & 1 \\
\hline
\end{tabular}
\smallskip

\begin{enumerate}[label=(\alph*), itemsep=4pt]
  \item $h(3) = $ \ans{4}
  \item Solve $h(x) = 7$: \quad $x = $ \ans{1}
  \item Domain of $h$: \ans{$\{0,1,2,3,4\}$} \quad Range of $h$:
        \ans{$\{1, 4, 7, 10\}$}
\end{enumerate}

\item The graph of a function $g$ is shown.

\begin{center}
\begin{tikzpicture}[x=0.6cm, y=0.45cm]
  \draw[linegray, very thin, step=1] (0,0) grid (8,8);
  \draw[->, thick] (0,0) -- (8.6,0) node[below right] {\small $x$};
  \draw[->, thick] (0,0) -- (0,8.6) node[above] {\small $g(x)$};
  \foreach \x in {1,2,3,4,5,6,7,8} \node[below] at (\x,0) {\small \x};
  \foreach \y in {2,4,6,8} \node[left] at (0,\y) {\small \y};
  \draw[very thick, plum] (0,2) -- (3,8);
  \draw[very thick, plum] (3,8) -- (8,3);
  \fill[plum] (0,2) circle (2.5pt);
  \fill[plum] (8,3) circle (2.5pt);
\end{tikzpicture}
\end{center}

\begin{enumerate}[label=(\alph*), itemsep=4pt]
  \item $g(1) = $ \ans{4} \qquad $g(6) = $ \ans{5}
  \item Solve $g(x) = 6$: \quad $x = $ \ans{2} and $x = $ \ans{5}
  \item Domain: \ans{0} $\le x \le$ \ans{8} \qquad
        Range: \ans{2} $\le y \le$ \ans{8}
  \item Rewrite the domain from part (c) in bracket notation. \quad
        \ans{$[\,0, 8\,]$}
\end{enumerate}

\item Which of the following is the \textbf{range} of the function
      $\{(2, 3),\ (4, 3),\ (6, 1),\ (8, 5)\}$?

\begin{enumerate}[label=\Alph*., itemsep=3pt]
  \item $\{2, 4, 6, 8\}$
  \item $\{1, 3, 5\}$ \quad \textcolor{keyred}{\textbf{$\leftarrow$ correct}}
  \item $\{3, 3, 1, 5\}$
  \item all real numbers
\end{enumerate}

\end{enumerate}

The formula $y = \dfrac{6}{2x - 8}$ refuses one input. Which $x$ is forbidden,

\vspace{0.05in}

\ansline{The domain is every real number except 4.}
