\begin{frame}[t, plain]
  \plumheader{Hook: One Symbol, a Whole Sentence}
  \sectionlabel[goldacc]{Think About It}

  A ball is launched from the ground, and \(h(t)\) gives its height in feet
  \(t\) seconds after launch.

  \bigskip
  \begin{center}
    {\LARGE \(h(2) = 61\)}
  \end{center}

  \bigskip
  \begin{itemize}\setlength{\itemsep}{8pt}
    \item What is this equation \textbf{saying}? Say it as a full sentence.
    \item What would \(h(0) = 0\) say?
    \item What would \(h(5) = 0\) say?
  \end{itemize}

  \bigskip
  \textit{Yesterday's trustworthy machine is about to get a name.}

% SECTION 1 — FUNCTION NOTATION
\begin{frame}[t, plain]
  \plumheader{1. Function Notation --- Naming the Machine}

  \begin{block}{The three parts of \(f(x)\)}
    \(f\) is the \textbf{name} of the function (the machine itself),
    \(x\) is the \textbf{input} (what goes in the slot), and
    \(f(x)\) is the \textbf{output} (what comes out).
  \end{block}

  \medskip
  Read \(f(x)\) as ``\(f\) \textbf{of} \(x\)'' --- it is \textbf{not}
  \(f\) times \(x\).

  \medskip
  \begin{block}{In context}
    A phone plan charges by the minute: \(C(m) = 0.10m + 5\).
    The machine's name is \(C\); the input \(m\) is the minutes used;
    the output \(C(m)\) is the cost of the bill.
  \end{block}

  \medskip
  \textbf{One statement, three facts.} \quad \(f(3) = 7\) says:
  the input is 3, the output is 7, and the point \((3, 7)\) is on the graph
  of \(f\).
\end{frame}

% SECTION 2 — EVALUATING FROM A FORMULA
\begin{frame}[t, plain]
  \plumheader{2. Evaluating from a Formula}

  To \textbf{evaluate}: substitute the input --- \textit{always inside
  parentheses} --- then do the arithmetic.

  \medskip
  \begin{columns}[T]
    \column{0.48\textwidth}
      \begin{block}{\(f(x) = 2x + 5\)}
        \(f(4) = 2(4) + 5 = 13\)\\[6pt]
        \(f(-3) = 2(-3) + 5 = -1\)
      \end{block}

    \column{0.48\textwidth}
      \begin{block}{\(g(x) = x^2 - 4\)}
        \(g(-2) = (-2)^2 - 4\)\\[6pt]
        \(\phantom{g(-2)} = 4 - 4 = 0\)
      \end{block}
  \end{columns}

  \bigskip
  \begin{block}{Watch the trap}
    \((-2)^2 = 4\) \quad but \quad \(-2^2 = -4\).
    Write the parentheses \textbf{first}, empty, then fill them.
  \end{block}
\end{frame}

% SECTION 3 — TABLES
\begin{frame}[t, plain]
  \plumheader{3. Evaluating from a Table}

  A table \textit{is} a function: each column is an input--output pair.

  \bigskip
  \begin{tabular}{|c|c|c|c|c|}
  \hline
  \(x\) & 0 & 2 & 4 & 6 \\
  \hline
  \(g(x)\) & 3 & 7 & 11 & 15 \\
  \hline
  \end{tabular}

  \bigskip
  \begin{itemize}\setlength{\itemsep}{8pt}
    \item Find \(g(4)\): input 4, read the output \(\to\) \(g(4) = 11\)
    \item Solve \(g(x) = 3\): output 3, find the input \(\to\) \(x = 0\)
  \end{itemize}

  \bigskip
  \textbf{Same table, opposite directions.}
\end{frame}

\begin{frame}[t, plain]
  \plumheader{Evaluating from a Graph}

  \begin{columns}[T]
    \column{0.5\textwidth}
      \begin{center}
      \begin{tikzpicture}[x=0.55cm, y=0.5cm]
        \draw[linegray, very thin, step=1] (0,0) grid (9,5);
        \draw[->, thick] (0,0) -- (9.6,0) node[below right] {\small \(x\)};
        \draw[->, thick] (0,0) -- (0,5.6) node[above] {\small \(f(x)\)};
        \foreach \x in {1,2,3,4,5,6,7,8,9} \node[below] at (\x,0) {\small \x};
        \foreach \y in {1,2,3,4,5} \node[left] at (0,\y) {\small \y};
        \draw[very thick, plum] (0,1) -- (3,4);
        \draw[very thick, plum] (3,4) -- (6,4);
        \draw[very thick, plum] (6,4) -- (9,1);
        \fill[plum] (0,1) circle (2.5pt);
        \fill[plum] (9,1) circle (2.5pt);
      \end{tikzpicture}
      \end{center}

    \column{0.46\textwidth}
      \begin{itemize}\setlength{\itemsep}{8pt}
        \item \(f(2)\): start at \(x = 2\), walk \textbf{up}, read across
              \(\to\) \(f(2) = 3\)
        \item \(f(5) = 4\)
        \item Solve \(f(x) = 2\): start at height 2, walk \textbf{across}
              \(\to\) \(x = 1\) and \(x = 8\)
      \end{itemize}
  \end{columns}

  \bigskip
  \begin{block}{The direction distinction}
    ``Find \(f(2)\)'' hands you an \textbf{input} and asks for the
    \textbf{output}. ``Solve \(f(x) = 2\)'' hands you an \textbf{output} and
    asks for the \textbf{input(s)} --- and two answers is fine (many-to-one!).
  \end{block}
\end{frame}

% SECTION 4 — DOMAIN AND RANGE
\begin{frame}[t, plain]
  \plumheader{4. Domain and Range}

  \begin{block}{Definitions}
    The \textbf{domain} is the set of all \textbf{inputs} a function uses;
    the \textbf{range} is the set of all \textbf{outputs} it produces.
    A repeated output is listed \textbf{once}.
  \end{block}

  \medskip
  \begin{columns}[T]
    \column{0.48\textwidth}
      \textbf{From a list of points}\\[4pt]
      \(f = \{(1,3),\ (2,5),\ (4,5),\ (6,8)\}\)
      \begin{itemize}\setlength{\itemsep}{6pt}
        \item Domain: \(\{1, 2, 4, 6\}\)
        \item Range: \(\{3, 5, 8\}\) --- the repeated 5 is listed once
      \end{itemize}

    \column{0.48\textwidth}
      \textbf{From the graph} (previous slide)
      \begin{itemize}\setlength{\itemsep}{6pt}
        \item Left--right reach: domain \(0 \le x \le 9\)
        \item Bottom--top reach: range \(1 \le y \le 4\)
      \end{itemize}
  \end{columns}
\end{frame}

% SECTION 4, CONT. — FORMULA FINE PRINT
\begin{frame}[t, plain]
  \plumheader{Domain from a Formula (First Look)}

  \begin{block}{\(g(x) = \dfrac{10}{x - 2}\)}
    Try \(x = 2\): the denominator becomes \(0\), and division by zero is
    \textbf{undefined}. The machine jams --- so \(2\) is \textbf{excluded}.
  \end{block}

  \bigskip
  \begin{itemize}\setlength{\itemsep}{8pt}
    \item Domain of \(g\): all real numbers \textbf{except} 2.
    \item Context can shrink a domain too: a drone flight that lasts 8 minutes
          only makes sense for \(0 \le t \le 8\).
    \item That is the whole story for now --- more excluded-input detective
          work comes with rational functions later in the course.
  \end{itemize}
\end{frame}

          \end{tabular}

          \smallskip
          \begin{enumerate}\setlength{\itemsep}{5pt}
            \item Find \(p(2)\), then solve \(p(x) = 3\).
            \item State the domain and the range of \(p\).
          \end{enumerate}
    \item In one sentence: what is the difference between ``find \(f(2)\)''
          and ``solve \(f(x) = 2\)''?
  \end{enumerate}
