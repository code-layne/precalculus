\begin{frame}[t, plain]
  \navyheader{Hook: A Car Leaves a Stop Sign}
  \sectionlabel[goldacc]{Think About It}

  A car accelerates away from a stop sign. Consider two quantities, each as a function of time:

  \bigskip
  \begin{columns}[T]
    \column{0.48\textwidth}
      \begin{block}{Speed vs.\ Time}
        \begin{itemize}\setlength{\itemsep}{6pt}
          \item Speed starts at \textbf{0} mph.
          \item As the driver presses the gas, speed \textbf{increases}.
          \item Does it increase slowly at first, or quickly?
        \end{itemize}
      \end{block}

    \column{0.48\textwidth}
      \begin{block}{Distance vs.\ Time}
        \begin{itemize}\setlength{\itemsep}{6pt}
          \item Distance also starts at \textbf{0}.
          \item As the car moves, distance \textbf{increases}.
          \item Does it grow slowly at first, or quickly?
        \end{itemize}
      \end{block}
  \end{columns}

  \bigskip
  \textit{Discussion:} How does the shape of one graph relate to the behavior of the other?
\end{frame}

% WHAT IS A FUNCTION
\begin{frame}[t, plain]
  \navyheader{What Is a Function?}

  \begin{block}{Definition}
    A \textbf{function} is a rule that assigns each \textbf{input} (independent variable)
    exactly \textbf{one output} (dependent variable).
  \end{block}

  \bigskip
  \begin{columns}[T]
    \column{0.48\textwidth}
      \textbf{Domain \& Range}
      \begin{itemize}\setlength{\itemsep}{6pt}
        \item \textbf{Domain} = set of all allowable inputs
        \item \textbf{Range} = set of all possible outputs
        \item Inputs plotted on the \textit{horizontal} axis
        \item Outputs plotted on the \textit{vertical} axis
      \end{itemize}

    \column{0.48\textwidth}
      \textbf{Water Tank Example}

      \smallskip
      \begin{tabular}{|c|c|c|c|c|}
      \hline
      $t$ (min) & 0 & 2 & 4 & 6 \\
      \hline
      $V$ (gal)  & 0 & 8 & 16 & 24 \\
      \hline
      \end{tabular}

      \smallskip
      Domain: $[0, 6]$ \quad Range: $[0, 24]$
  \end{columns}
\end{frame}

% INCREASING AND DECREASING
\begin{frame}[t, plain]
  \navyheader{Increasing \& Decreasing Functions}

  \begin{columns}[T]
    \column{0.48\textwidth}
      \begin{block}{Increasing}
        $f$ is \textbf{increasing} on an interval if:\\[4pt]
        for any $a < b$ in that interval,
        \[ f(a) < f(b) \]
        \textit{As the input grows, the output grows.}
      \end{block}

    \column{0.48\textwidth}
      \begin{block}{Decreasing}
        $f$ is \textbf{decreasing} on an interval if:\\[4pt]
        for any $a < b$ in that interval,
        \[ f(a) > f(b) \]
        \textit{As the input grows, the output shrinks.}
      \end{block}
  \end{columns}

  \bigskip
  \textbf{Hiker Example:} Elevation rises from hour 0 to hour 4 \textbf{(increasing)}, then
  falls from hour 4 to hour 7 \textbf{(decreasing)}.

  \medskip
  \textit{Key:} A function can be increasing on one interval and decreasing on another.
\end{frame}

% CONCAVE UP AND CONCAVE DOWN
\begin{frame}[t, plain]
  \navyheader{Concave Up \& Concave Down}

  Concavity describes whether the \textbf{rate of change} is itself increasing or decreasing.

  \bigskip
  \begin{columns}[T]
    \column{0.48\textwidth}
      \begin{block}{Concave Up}
        \begin{itemize}\setlength{\itemsep}{5pt}
          \item Rate of change is \textbf{increasing}
          \item Graph curves upward like a bowl
          \item Example: distance as a car accelerates
        \end{itemize}
      \end{block}

    \column{0.48\textwidth}
      \begin{block}{Concave Down}
        \begin{itemize}\setlength{\itemsep}{5pt}
          \item Rate of change is \textbf{decreasing}
          \item Graph curves downward like a hill
          \item Example: hiker ascending (gains less elevation per hour)
        \end{itemize}
      \end{block}
  \end{columns}

  \bigskip
  \textbf{Important:} Concavity $\neq$ increasing/decreasing.\\
  A function can be \textit{increasing and concave down} at the same time!
\end{frame}

% ZEROS OF A FUNCTION
\begin{frame}[t, plain]
  \navyheader{Zeros of a Function}

  \begin{block}{Definition}
    The \textbf{zeros} of a function $f$ are the input values where $f(x) = 0$.\\
    On a graph, zeros are where the graph \textbf{crosses or touches the $x$-axis}.
  \end{block}

  \bigskip
  \textbf{Example:} A ball is thrown upward. Its height $h(t)$ in feet satisfies:
  \begin{itemize}\setlength{\itemsep}{6pt}
    \item $h(0) = 0$ \quad (starts at ground level)
    \item $h(2) = 64$ \quad (maximum height)
    \item $h(4) = 0$ \quad (returns to ground)
  \end{itemize}

  \medskip
  Zeros of $h$: \quad $t = 0$ \quad and \quad $t = 4$

  \medskip
  $h$ is increasing on $[0, 2]$ and decreasing on $[2, 4]$.
\end{frame}

\begin{frame}[t, plain]
  \navyheader{Class Practice}
  \sectionlabel[redacc]{Try It}

  Use the hiker's elevation table below.

  \smallskip
  \begin{tabular}{|c|c|c|c|c|c|c|c|c|}
  \hline
  Hour $t$ & 0 & 1 & 2 & 3 & 4 & 5 & 6 & 7 \\
  \hline
  Elev.\ $E(t)$ (ft) & 500 & 1700 & 2400 & 2700 & 2800 & 2400 & 1400 & 500 \\
  \hline
  \end{tabular}

  \bigskip
  \begin{enumerate}\setlength{\itemsep}{7pt}
    \item State the domain and range.
    \item On what interval(s) is $E$ increasing? Decreasing?
    \item Compute the change in elevation per hour on $[0,1]$, $[1,2]$, and $[2,3]$.
          Is the rate of change increasing or decreasing on $[0, 4]$?
          What does that tell you about the concavity?
    \item Does $E$ have any zeros based on this data? Explain.
  \end{enumerate}
\end{frame}
