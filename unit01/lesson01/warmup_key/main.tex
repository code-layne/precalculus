\begin{enumerate}[itemsep=10pt]

\item Evaluate $3x + 4$ when $x = -2$.

\vspace{0.1in}
Value: \ans{$3(-2) + 4 = -2$}

\item Evaluate $x^2 - 5$ when $x = -3$.

\begin{work}
  (-3)^2 - 5 &= 9 - 5 \\
             &= 4
\end{work}

Value: \ans{4}

\item Solve for $x$: \quad $2x + 1 = 9$

\begin{work}
  2x + 1 &= 9 \\
      2x &= 8 \\
       x &= 4
\end{work}

$x = $ \ans{4}
