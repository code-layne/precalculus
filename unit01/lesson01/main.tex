\begin{skillbox}[Priority Ideas \& Skills]{goldbox}
\begin{tabularx}{\linewidth}{@{}X X@{}}
\textbf{Priority Skills} &
\textbf{Key Understandings} \\
\midrule
Read and write function notation. &
\begin{itemize}[leftmargin=1.2em,itemsep=2pt,topsep=0pt]
  \item $f(x)$ is read ``$f$ of $x$'' --- never ``$f$ times $x$.''
  \item One statement $f(3) = 7$ carries three facts: input 3, output 7,
        and the point $(3, 7)$ on the graph.
\end{itemize} \\
Evaluate a function from a formula, a table, or a graph. &
\begin{itemize}[leftmargin=1.2em,itemsep=2pt,topsep=0pt]
  \item Substitute with parentheses; negative inputs are the error magnet.
  \item ``Find $f(2)$'' asks for an output; ``solve $f(x) = 2$'' asks for the
        input(s) --- opposite directions.
\end{itemize} \\
State the domain and range of a function. &
\begin{itemize}[leftmargin=1.2em,itemsep=2pt,topsep=0pt]
  \item Domain = every input the function uses; range = every output it
        produces (repeats listed once).
  \item From a graph with endpoints, both are read as inequalities,
        e.g.\ $0 \le x \le 9$.
\end{itemize} \\
\end{tabularx}
\end{skillbox}

\begin{skillbox}[{Vocabulary, Concepts, \& Theorems}]{greenbox}
\renewcommand{\arraystretch}{1.4}
\begin{tabularx}{\linewidth}{@{} >{\leavevmode\bfseries\color{plum}}p{0.24\linewidth} X @{}}
Function          & A rule that sends each input to exactly one output (from Lesson 1.0). \\
Function notation & The symbol $f(x)$: $f$ names the rule, $x$ is the input, $f(x)$ is the output at $x$. Read ``$f$ of $x$.'' \\
Evaluate          & Find the output a function produces for a given input. \\
Domain            & The set of all inputs a function uses. \\
Range             & The set of all outputs a function produces; a repeated output is listed once. \\
Undefined         & What we call an expression with no value, such as a division by zero --- inputs causing it are excluded from the domain. \\
\end{tabularx}
\end{skillbox}

\begin{skillbox}[Hook]{lilac}
    \textbf{One symbol, a whole sentence.} A ball is launched from the ground, and
    $h(t)$ gives its height in feet $t$ seconds after launch.
    \begin{enumerate}[itemsep=2pt]
        \item On the board, write only: $h(2) = 61$. Ask: what is this equation
              \textit{saying}? Push for a full sentence --- ``two seconds after
              launch, the ball is 61 feet high.''
        \item Follow-up: what would $h(0) = 0$ say? And $h(5) = 0$?
    \end{enumerate}
    \textit{Discussion prompt:} yesterday's trustworthy machine now has a
    \textit{name}. One short symbol packs in the machine, the input, and the
    output --- that efficiency is the whole point of function notation.
\end{skillbox}

\begin{skillbox}[Lesson]{lilac}
\begin{multicols}{2}

\textbf{Part 1: Function Notation} (10 min)

\begin{itemize}
  \item Name the parts of $f(x)$: $f$ is the machine's name, $x$ is the input
        slot, $f(x)$ is the output. Insist on ``$f$ of $x$,'' and confront the
        misreading directly: $f(x)$ is \textit{not} multiplication.
  \item Context example: $C(m) = 0.10m + 5$ for a phone bill --- the name, input,
        and output all mean something.
  \item The three-facts reading: $f(3) = 7$ says input 3, output 7, point
        $(3,7)$. Students translate notation to sentences and back.
\end{itemize}

\textbf{Part 2: Evaluating from a Formula} (12 min)

\begin{itemize}
  \item $f(x) = 2x + 5$: find $f(4)$, then $f(-3)$. Model the parentheses habit:
        every substitution goes in $(\ \ )$ before arithmetic.
  \item $g(x) = x^2 - 4$: find $g(-2)$. The classic trap $-2^2$ vs.\ $(-2)^2$ ---
        make the error visible on the board once, on purpose.
  \item Emphasize: evaluating never changes the rule; it just runs the machine
        once.
\end{itemize}

\columnbreak

\textbf{Part 3: Tables and Graphs} (12 min)

\begin{itemize}
  \item A table \textit{is} a function: each column is an input--output pair.
        Read $g(4)$ from a table; then reverse it --- for which $x$ is
        $g(x) = 3$?
  \item Project the pre-drawn graph from the notes. Find $f(2)$ by walking
        up from the axis; solve $f(x) = 2$ by walking across --- two answers,
        and that is fine (many-to-one, from 1.0).
  \item Drill the direction distinction: $f(2)$ is an output; $f(x) = 2$ asks
        for inputs. This confusion is the lesson's biggest hazard.
\end{itemize}

\textbf{Part 4: Domain and Range} (12 min)

\begin{itemize}
  \item Definitions: domain = all inputs, range = all outputs. From a list of
        points, both are just lists --- repeated outputs written once.
  \item From the graph: domain is how far the graph reaches left--right
        ($0 \le x \le 9$), range how far bottom--top ($1 \le y \le 4$).
  \item Light first look at formulas: $g(x) = 10/(x-2)$ has no value at
        $x = 2$ --- division by zero is \textit{undefined}, so 2 is excluded.
        One example only; interval notation and harder cases come later.
\end{itemize}
\end{multicols}
\end{skillbox}

\begin{skillbox}[Active Monitoring]{redbox}
While students work on guided notes and practice, circulate and check:
\begin{itemize}
  \item \textbf{Common error:} reading $f(x)$ as multiplication --- watch for
        students ``solving for $f$.'' Cold-call: ``In $C(m)$, what is the machine's
        name? What goes in the slot?''
  \item \textbf{Common error:} dropping parentheses on negative inputs, so
        $g(-2) = -2^2 - 4$. Ask the student to say the substitution aloud with the
        word ``quantity.''
  \item \textbf{Common error:} answering $f(x) = 2$ with an output instead of the
        input(s). Push: ``Is the 2 going \textit{into} the machine or coming
        \textit{out}?''
  \item \textbf{Domain/range:} students listing a repeated output twice, or
        swapping the two sets. Cold-call: ``Domain --- inputs or outputs?''
\end{itemize}
\end{skillbox}

\begin{skillbox}[Group Work \& Differentiation]{redbox}
Students work on the \textbf{Group Activity: The Drone Flight} in leveled tiers.
Every tier reads the same pre-drawn flight graph --- no tier sketches anything.

\begin{itemize}
  \item \textbf{Tier R (Remediate):} Read outputs straight off the graph ---
        $h(0)$, $h(3)$, the landing time --- and translate $h(1) = 4$ into a
        sentence.
  \item \textbf{Tier A (Apply):} Evaluate and solve against the graph
        ($h(6)$, then $h(t) = 4$ with its two answers), and state the domain and
        range as inequalities.
  \item \textbf{Tier E (Extend):} Flip the machine --- is \textit{time} a function
        of \textit{height}? --- plus a first formula-domain question
        ($g(x) = 12/(x-5)$) and the meaning of a realistic domain.
\end{itemize}
\end{skillbox}

\begin{skillbox}[Individual Work \& Assessment]{redbox}
\textbf{Exit Ticket} (last 5--7 minutes, individual, collected):
\begin{enumerate}
  \item Evaluate $f(x) = 4x - 3$ at $x = 2$ (worked space provided) and at $x = 0$.
  \item From a small table, read $g(2)$ and find the $x$ with $g(x) = 10$.
  \item State the domain and range of a function given as four ordered pairs,
        with one repeated output.
\end{enumerate}
\textbf{Assessment note:} Item 2(b) is the direction check --- students who answer
it with an output have missed the lesson's central distinction. Sort tickets by
2(b) and reteach ``into the machine vs.\ out of the machine'' in the 1.2 warm-up
window if more than a handful miss it. On item 3, listing the repeated 7 twice is
a minor slip; swapping domain and range is not.
\end{skillbox}

\begin{skillbox}[Reinforcement \& Extension]{goldbox}
\textbf{Homework:} 5 problems covering notation-to-sentence translation both
directions, evaluating two formulas (negative input included), a table read in
both directions, a full graph read (evaluate, solve, domain, range), and one
SOL-style multiple-choice domain item.

\textbf{Extension (optional):} Why does $y = 12/(x-5)$ forbid $x = 5$? Students
name the excluded input and write the domain in words.

\textbf{Preview:} Next lesson (1.2, Change in Tandem) puts two changing
quantities side by side: as the input climbs, how does the output respond ---
and how do we describe that behavior precisely?
\end{skillbox}

\begin{teachernote}[Warm-Up]
Five minutes, silent start, no calculators. Items 1--3 are the exact algebra that
evaluation runs on today --- a student who misses item 1 or 2 will make the same
sign error inside $f(-3)$ within the hour, so note names. Item 4 is the promised
revisit of the 1.0 function decision; it should be quick.
\end{teachernote}

\begin{teachernote}[Guided Notes]
Part 1 lives or dies on the language: make students \textit{say} ``$f$ of $x$''
aloud several times, and translate $f(3)=7$ into a sentence before any computing
starts. In Part 2, write the parentheses first, empty, then fill them --- the
habit matters more than the answers. The graph in Part 3 is pre-drawn; students
only read it. If time runs short, Part 4's formula-domain example ($10/(x-2)$) can
be pushed to the homework extension --- do not cut the direction distinction.
\end{teachernote}

\begin{teachernote}[Group Activity]
All three tiers work the same drone graph, so groups can be regrouped mid-activity
without new setup. Tier R should say answers as sentences (``at 3 minutes the
drone is 6 meters up''). The Tier A solve $h(t)=4$ having two answers is the hinge
--- connect it back to many-to-one from 1.0. If a Tier E pair resolves ``time is
not a function of height'' cleanly, have them present it in the closing minutes;
it previews inverse functions in 1.6.
\end{teachernote}

\begin{teachernote}[Exit Ticket]
Collected, completion credit only. Scan item 2(b) first --- an answer of 13 (an
output) instead of 1 (the input) is the diagnostic miss. Item 3: accept range
$\{4, 7, 9\}$ with or without braces, but a doubled 7 earns a margin note, not a
deduction.
\end{teachernote}

\begin{teachernote}[Homework]
Problem 5 is the SOL-style multiple-choice item; distractor B is the range and
distractor C mixes coordinates of one pair --- both worth two minutes of
whole-class autopsy tomorrow. The extension is the only formula-domain item, so if
Part 4 ran long in class, assign it explicitly rather than optionally.
\end{teachernote}
