\item Verify that $f(x) = \log_5 x$ and $g(x) = 5^x$ are inverse functions by computing
      both compositions. Show all steps.

\vspace{4pt}
$f(g(x)) = \log_5(5^x) = $ \smash{\gradedblank[id=hw2a, type=short, points=1, answer={x}]{3cm}{13pt}}

\vspace{6pt}
$g(f(x)) = 5^{\log_5 x} = $ \smash{\gradedblank[id=hw2b, type=short, points=1, answer={x}]{3cm}{13pt}}

\vspace{4pt}
Conclusion:
\rubricbox{\gradedblank[id=hw2c, type=rubric, points=1,
  display={Both compositions equal $x$, so $f$ and $g$ are inverses.}]{\linewidth}{13pt}}

\item A biologist models algae coverage with $A(t) = 10^t$ cm$^2$, where $t$ is time in days.

\begin{enumerate}[leftmargin=1.5em, itemsep=6pt, label=\alph*.]
  \item Write the inverse function $t(A)$ that gives the number of days from the area.

  \vspace{4pt}
  $t(A) = $ \smash{\gradedblank[id=hw3a, type=short, points=1, answer={\log_{10} A}]{4cm}{13pt}}

  \item Evaluate $t(1000)$. What does this value represent in context?

  \vspace{4pt}

  \vspace{4pt}
  In context:
  \rubricbox{\gradedblank[id=hw3b2, type=rubric, points=1,
  display={After 3 days the algae covers 1000 cm$^2$.}]{\linewidth}{13pt}}
\end{enumerate}

\item \textbf{Conceptual question.} Explain in your own words why the logarithmic function's
      growth pattern (output changes additively as input changes multiplicatively) is the
      ``opposite'' of the exponential function's growth pattern.

      \rubricbox{\gradedblank[id=hw4, type=rubric, points=2,
  display={Exponential: adding to the input multiplies the output by $b$. Logarithm: multiplying the input by $b$ adds 1 to the output. Inverses swap inputs and outputs, so the patterns are reversed.}]{\linewidth}{41pt}}

\item Let $h(x) = 2\log_3 x$.

\begin{enumerate}[leftmargin=1.5em, itemsep=6pt, label=\alph*.]
  \item Identify the base $b$ and the value of $a$.

  \vspace{4pt}
  $b = $ \smash{\gradedblank[id=hw5a1, type=short, points=1, answer={3}]{2cm}{13pt}} \qquad $a = $ \smash{\gradedblank[id=hw5a2, type=short, points=1, answer={2}]{2cm}{13pt}}

  \item Evaluate $h(9)$ and $h(27)$.

  \vspace{4pt}
  $h(9) = $ \smash{\gradedblank[id=hw5b1, type=short, points=1, answer={4}]{3cm}{13pt}} \qquad $h(27) = $ \smash{\gradedblank[id=hw5b2, type=short, points=1, answer={6}]{3cm}{13pt}}

  \item How do the outputs of $h(x)$ compare to the outputs of $\log_3 x$ for the same

        \rubricbox{\gradedblank[id=hw5c, type=rubric, points=2,
  display={Each output of $h$ is twice that of $\log_3 x$: a vertical stretch by 2.}]{\linewidth}{13pt}}
\end{enumerate}
