      \vspace{0.06in}
      $I / I_0 = $ \ans{$10^{-5}/10^{-12} = 10^{7}$} \qquad
      $\log_{10}(I / I_0) = $ \ans{7} \qquad
      $L = $ \ans{70 dB}

\item \textbf{(AP-Style Multiple Choice)} Which of the following is equivalent to $\log_b c = 5$?

      \medskip
      \begin{enumerate}[label=(\Alph*), itemsep=4pt]
        \item $b^c = 5$
        \item $c^5 = b$
        \item $b^5 = c$
        \item $5^b = c$
      \end{enumerate}

      \vspace{0.04in}
      Answer: \ans{(C)} \quad By the definition $\log_b c = 5 \Leftrightarrow b^5 = c$.

\end{enumerate}
