\item A data set contains the points $f(1) = 0$ and $f(e^3) = 6$. Assume
      $f(x) = a\ln(x)$.

\begin{enumerate}[leftmargin=1.5em, itemsep=8pt, label=\alph*.]
  \item From $f(1) = 0$: verify that $\ln(1) = 0$, so any value of $a$ satisfies this.
        Use $f(e^3) = 6$ to find $a$.

  \vspace{4pt}
  $a\ln(e^3) = 6 \;\Rightarrow\; a \cdot \smash{\gradedblank[id=hw2a1, type=short, points=1, answer={3}]{1.5cm}{13pt}} = 6 \;\Rightarrow\; a = $ \smash{\gradedblank[id=hw2a2, type=short, points=1, answer={2}]{2cm}{13pt}}

  \item Write the complete model.

  \vspace{4pt}
  $f(x) = $ \smash{\gradedblank[id=hw2b, type=short, points=1, answer={2\ln(x)}]{4cm}{13pt}}

  \item Predict $f(e^5)$.

  \vspace{4pt}
  $f(e^5) = $ \smash{\gradedblank[id=hw2c, type=short, points=1, answer={10}]{5cm}{13pt}}
\end{enumerate}

\item \textbf{Extension.} Using the change-of-base property $\log_b x = \dfrac{\ln x}{\ln b}$,
      explain why $f(x) = a\ln(x)$ and $g(x) = a\log_{10}(x)$ always produce models that
      are vertical dilations of each other. Write the exact dilation factor that converts
      one into the other.

\writegap\\
\writegap\\
\writebox{\gradedblank[id=hw6, type=rubric, points=2,
  display={Change of base: $\log_{10}(x) = \dfrac{\ln x}{\ln 10}$, so $a\log_{10}(x) = \dfrac{1}{\ln 10}\cdot a\ln(x)$: $g$ is a vertical dilation of $f$ by $\dfrac{1}{\ln 10} \approx 0.4343$ (equivalently $f$ is $g$ stretched by $\ln 10 \approx 2.303$).}]{\linewidth}{42pt}}\\
