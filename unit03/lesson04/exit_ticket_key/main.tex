Consider the rational function $\displaystyle r(x) = \frac{x^2 - x - 2}{x^2 - 4}$.

\begin{enumerate}[itemsep=14pt]

\item Factor the numerator and denominator completely.

      \vspace{0.08in}
      Numerator: $x^2 - x - 2 = $ \ans{$(x-2)(x+1)$}

      \vspace{0.08in}
      Denominator: $x^2 - 4 = $ \ans{$(x-2)(x+2)$}

\item List any shared zeros of the numerator and denominator.

      \vspace{0.08in}
      Shared zero(s): \ans{$x = 2$}

\item For each shared zero, compare multiplicities and classify as a \textbf{hole} or
      a \textbf{vertical asymptote}. Then state any additional VAs from non-shared zeros.

      \vspace{0.08in}
      \renewcommand{\arraystretch}{1.6}
      \begin{tabular}{|c|c|c|c|}
      \hline
      Zero & Num.\ mult.\ $m$ & Denom.\ mult.\ $n$ & Hole or VA? \\
      \hline
      \ans{$x=2$} & \ans{1} & \ans{1} & \ans{Hole ($m = n \geq 1$)} \\
      \hline
      \ans{$x=-2$} & \ans{0} & \ans{1} & \ans{VA (denom only)} \\
      \hline
      \end{tabular}

\item Find the coordinates $(c, L)$ of any holes. Show the simplification and substitution.

      \vspace{0.08in}
      Simplified form: \ans{$\dfrac{x+1}{x+2}$}

      \vspace{0.08in}
      $L = \dfrac{2+1}{2+2}$ \ans{$= \dfrac{3}{4}$} \qquad Hole at: $\Big($\ans{2}$,$ \ans{$\dfrac{3}{4}$}$\Big)$

\item Write the limit statement for the hole using proper notation.

      \vspace{0.08in}
      $\displaystyle \lim_{x \to \ans{2}} r(x) = $ \ans{$\dfrac{3}{4}$}

\end{enumerate}
