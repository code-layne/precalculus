\begin{enumerate}[itemsep=20pt]

\item Factor each trinomial.

\begin{enumerate}[label=(\alph*), itemsep=30pt]
  \item $x^2 - 5x - 14$ \hfill \blank{3.4cm}
  \item $x^2 - 9x + 20$ \hfill \blank{3.4cm}
\end{enumerate}

\item Factor out the greatest common factor.

\begin{enumerate}[label=(\alph*), itemsep=30pt]
  \item $4x^3 - 10x^2$ \hfill \blank{3.4cm}
  \item $3x^4 + 12x^2$ \hfill \blank{3.4cm}
\end{enumerate}

\item Simplify. Write each answer in terms of $i$.

\begin{enumerate}[label=(\alph*), itemsep=30pt]
  \item $\sqrt{-16}$ \hfill \blank{2.2cm}
  \item $\sqrt{-12}$

\begin{work}
  \sqrt{-12} &= \sqrt{-1}\cdot\sqrt{4}\cdot\sqrt{3} \\
             &= i \cdot 2 \cdot \sqrt{3}
\end{work}

  \hfill \blank{2.2cm}
\end{enumerate}

\item $f(x) = (x - 4)^2(x + 1)$.

\begin{enumerate}[label=(\alph*), itemsep=30pt]
  \item List each zero of $f$ and its multiplicity. \hfill \blank{4.8cm}
  \item Where does the graph \emph{touch} the $x$-axis, and where does it \emph{cross}? \hfill \blank{4.8cm}
  \item The degree of $f$, and its number of zeros, counting multiplicity. \hfill \blank{4.8cm}
\end{enumerate}

\end{enumerate}
