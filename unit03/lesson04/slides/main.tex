% ROW 1 — FACTOR FIRST
\begin{frame}[t, plain]
  \plumheader{Factor First --- the Zero Product Property}
  \sectionlabel[goldacc]{Notes row FACTOR FIRST $\cdot$ I Do: problem 1 $\cdot$ We Do: problem 2}

  \begin{columns}[T]
    \column{0.50\textwidth}
      \begin{center}
      \begin{tikzpicture}[x=0.42cm, y=0.07cm, >=stealth]
        \draw[linegray, very thin, xstep=1, ystep=5] (-5,-15) grid (4,25);
        \draw[->, thick] (-5,0) -- (4.5,0) node[right] {\small $x$};
        \draw[->, thick] (0,-15) -- (0,27) node[above] {\small $p(x)$};
        \foreach \x in {-4,-3,-2,-1,1,2,3} \node[below, font=\tiny] at (\x,0) {\x};
        \draw[very thick, plum, domain=-4.45:3.55, samples=70, smooth, variable=\t]
             plot (\t, {\t*(\t+4)*(\t-3)});
        \foreach \z in {-4,0,3} \fill[goldacc] (\z,0) circle (3pt);
      \end{tikzpicture}
      \end{center}
      \centering $p(x) = x(x + 4)(x - 3)$: zeros $0$, $-4$, $3$

    \column{0.46\textwidth}
      \begin{block}{Zero Product Property}
        If $ab = 0$, then $a = 0$ or $b = 0$.
      \end{block}

      \medskip
      \textbf{To solve $p(x) = 0$:}
      \begin{enumerate}\setlength{\itemsep}{4pt}
        \item Move every term to one side.
        \item Factor --- \textbf{GCF first}, including a power of $x$.
        \item Set each factor equal to zero.
      \end{enumerate}

      \medskip
      $2x^2 - 10x = 0 \;\Rightarrow\; 2x(x - 5) = 0 \;\Rightarrow\; x = 0,\ 5$
  \end{columns}
\end{frame}

% ROW 2 — FACTORING STRATEGY
\begin{frame}[t, plain]
  \plumheader{Factoring Strategy --- a Checklist}
  \sectionlabel[goldacc]{Notes row FACTORING STRATEGY $\cdot$ I Do: problem 3 $\cdot$ We Do: problems 4--6}

  \begin{columns}[T]
    \column{0.42\textwidth}
      \begin{enumerate}\setlength{\itemsep}{6pt}
        \item \textbf{GCF} first.
        \item \textbf{Count the terms.}
          \begin{itemize}\setlength{\itemsep}{2pt}
            \item 2: squares, cubes
            \item 3: trinomial, quadratic form ($u = x^2$)
            \item 4: grouping
          \end{itemize}
        \item \textbf{Factor completely} --- test every factor again.
        \item \textbf{Check} by multiplying back.
      \end{enumerate}

    \column{0.54\textwidth}
      \begin{block}{Two-term patterns}
        $a^2 - b^2 = (a - b)(a + b)$\\[3pt]
        $a^3 - b^3 = (a - b)(a^2 + ab + b^2)$\\[3pt]
        $a^3 + b^3 = (a + b)(a^2 - ab + b^2)$\\[3pt]
        $a^2 + b^2$ is \textbf{prime} over the reals
      \end{block}

      \medskip
      \textbf{Grouping:} $x^3 + 2x^2 - 9x - 18$\\[2pt]
      $= x^2(x + 2) - 9(x + 2)$\\[2pt]
      $= (x + 2)(x^2 - 9) = (x + 2)(x - 3)(x + 3)$
  \end{columns}
\end{frame}

% ROW 3 — EVERY SOLUTION
\begin{frame}[t, plain]
  \plumheader{Every Solution --- Real or Not}
  \sectionlabel[goldacc]{Notes row EVERY SOLUTION $\cdot$ teacher works 7 $\cdot$ We Do: problems 8--10}

  \begin{columns}[T]
    \column{0.44\textwidth}
      \begin{center}
      \begin{tikzpicture}[x=0.75cm, y=0.12cm, >=stealth]
        \draw[linegray, very thin, xstep=1, ystep=2] (-3,-8) grid (3,10);
        \draw[->, thick] (-3,0) -- (3.5,0) node[right] {\small $x$};
        \draw[->, thick] (0,-8) -- (0,11.5) node[above] {\small $y$};
        \foreach \x in {-2,-1,1,2} \node[below, font=\tiny] at (\x,0) {\x};
        \draw[very thick, plum, domain=-2.33:2.33, samples=70, smooth, variable=\t]
             plot (\t, {\t*\t*\t*\t - 3*\t*\t - 4});
        \foreach \z in {-2,2} \fill[goldacc] (\z,0) circle (3pt);
      \end{tikzpicture}
      \end{center}
      \centering $y = x^4 - 3x^2 - 4$: two crossings, four solutions

    \column{0.52\textwidth}
      \begin{block}{Set every factor to zero}
        $(x - 2)(x + 2)(x^2 + 1) = 0$\\[3pt]
        $x = \pm 2$ \quad (the $x$-intercepts)\\[3pt]
        $x^2 = -1$, so $x = \pm i$ \quad (a conjugate pair)
      \end{block}

      \medskip
      \begin{itemize}\setlength{\itemsep}{4pt}
        \item Degree $n$ $\Rightarrow$ $n$ solutions, counting multiplicity.
        \item Only the real ones are $x$-intercepts.
        \item A prime quadratic factor: use the quadratic formula.
      \end{itemize}
  \end{columns}
\end{frame}

% ROW 4 — THE CRUX
\begin{frame}[t, plain]
  \plumheader{Two Paths --- Divide, or Factor?}
  \sectionlabel[redacc]{Notes row TWO PATHS $\cdot$ teacher works 11 $\cdot$ We Do: 12--14}

  \begin{columns}[T]
    \column{0.31\textwidth}
      \begin{block}{Path A: divide by $x$}
        $x^3 = 9x$\\[2pt]
        $x^2 = 9$\\[2pt]
        $x = \pm 3$\\[2pt]
        \textit{two solutions}
      \end{block}
    \column{0.31\textwidth}
      \begin{block}{Path B: one side, factor}
        $x^3 - 9x = 0$\\[2pt]
        $x(x - 3)(x + 3) = 0$\\[2pt]
        $x = 0,\ 3,\ -3$\\[2pt]
        \textit{three solutions}
      \end{block}
    \column{0.32\textwidth}
      \begin{itemize}\setlength{\itemsep}{5pt}
        \item Check $x = 0$: \ $0^3 = 9(0)$. True.
        \item Dividing by $x$ assumes $x \ne 0$ --- it deletes a solution.
        \item Never divide by $x$. Factor it out.
      \end{itemize}
  \end{columns}

  \bigskip
  \textbf{And finish:} $x^4 - 16 = (x^2 - 4)(x^2 + 4)$ is \emph{not} done ---
  $(x - 2)(x + 2)(x^2 + 4)$ gives four solutions, $\pm 2$ and $\pm 2i$.
\end{frame}

% DEBRIEF
\begin{frame}[t, plain]
  \plumheader{Debrief --- The Headline}
  \sectionlabel[goldacc]{Share out 15--18, one answer each $\cdot$ then say it back}

  \begin{columns}[T]
    \column{0.48\textwidth}
      \begin{block}{Factor, never divide}
        Move everything to one side and factor. Dividing by $x$ throws away $x = 0$.
      \end{block}
    \column{0.48\textwidth}
      \begin{block}{Finish, then count}
        Factor completely and solve every factor. A degree-$n$ equation has $n$ solutions.
      \end{block}
  \end{columns}
