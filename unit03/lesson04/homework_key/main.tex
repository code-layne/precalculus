\item For each rational function below, find all holes and vertical asymptotes.

      \begin{enumerate}[label=(\alph*), itemsep=10pt]

        \item $\displaystyle r(x) = \dfrac{x^2 - 5x + 6}{x - 3} = \dfrac{(x-3)(x-2)}{x-3}$

              Shared zero $x=3$, $m=n=1$ so \ans{Hole}. Simplified: $x-2$. $L = 3-2 = 1$.

              Factor: \smash{\gradedblank[id=hw1a1, type=short, points=1, answer={\frac{(x-3)(x-2)}{x-3}}, display={$(x-3)(x-2)/(x-3)$}]{6cm}{11pt}}

              Holes: \smash{\gradedblank[id=hw1a2, type=short, check=text, points=2, answer={(3, 1)}, display={$(3,\ 1)$}]{4cm}{11pt}} \qquad VAs: \smash{\gradedblank[id=hw1a3, type=short, check=text, points=1, answer={none}]{4cm}{11pt}}\vphantom{\blank{1pt}}

        \item $\displaystyle r(x) = \dfrac{x^2 - x - 12}{(x-4)(x+1)} = \dfrac{(x-4)(x+3)}{(x-4)(x+1)}$

              Shared zero $x=4$, $m=n=1$ so \ans{Hole}. Simplified: $\dfrac{x+3}{x+1}$. $L = \dfrac{7}{5}$.

              Factor: \smash{\gradedblank[id=hw1b1, type=short, points=1, answer={\frac{(x-4)(x+3)}{(x-4)(x+1)}}, display={$(x-4)(x+3)/((x-4)(x+1))$}]{6cm}{11pt}}

              Holes: \smash{\gradedblank[id=hw1b2, type=short, check=text, points=2, answer={(4, 7/5)}, display={$(4,\ 7/5)$}]{4cm}{11pt}} \qquad VAs: \smash{\gradedblank[id=hw1b3, type=short, points=1, answer={x = -1}]{4cm}{11pt}}\vphantom{\blank{1pt}}

        \item $\displaystyle r(x) = \dfrac{x^2 + 3x}{x^2 + 3x + 2} = \dfrac{x(x+3)}{(x+1)(x+2)}$

              No shared zeros (numerator zeros: $0, -3$; denominator zeros: $-1, -2$).

              Factor: \smash{\gradedblank[id=hw1c1, type=short, points=1, answer={\frac{x(x+3)}{(x+1)(x+2)}}, display={$x(x+3)/((x+1)(x+2))$}]{6cm}{11pt}}

              Holes: \smash{\gradedblank[id=hw1c2, type=short, check=text, points=1, answer={none}]{4cm}{11pt}} \qquad VAs: \smash{\gradedblank[id=hw1c3, type=short, check=set, points=1, answer={-1, -2}, display={$x = -1$ and $x = -2$}]{4cm}{11pt}}

      \end{enumerate}

\item \textbf{Explanation (sample):}

If $m < n$ (or $a$ is a zero of the denominator only), a factor stays in the denominator, the outputs grow without bound, and there is a vertical asymptote.}]{\linewidth}{51pt}}\\

\item $\displaystyle r(x) = \dfrac{x^3 - x^2}{x^2 - x}$

      \begin{enumerate}[label=(\alph*), itemsep=6pt]
        \item Numerator: \smash{\gradedblank[id=hw4a1, type=short, points=1, answer={x^2(x-1)}]{5cm}{13pt}} \qquad Denominator: \smash{\gradedblank[id=hw4a2, type=short, points=1, answer={x(x-1)}]{5cm}{13pt}}

        \item
              \renewcommand{\arraystretch}{1.6}
              \begin{tabular}{|c|c|c|c|}
              \hline
              Zero & $m$ & $n$ & Hole or VA? \\
              \hline
              \smash{\gradedblank[id=hw4b1, type=short, points=1, answer={x = 0}]{1.5cm}{14pt}} & \smash{\gradedblank[id=hw4b2, type=short, points=1, answer={2}]{1cm}{14pt}} & \smash{\gradedblank[id=hw4b3, type=short, points=1, answer={1}]{1cm}{14pt}} & \smash{\gradedblank[id=hw4b4, type=short, check=text, points=1, answer={hole}, display={Hole ($m > n$)}]{3cm}{14pt}} \\

\item \textbf{(AP-Style)} $r(x) = \dfrac{x^2-4}{x^2-x-2} = \dfrac{(x-2)(x+2)}{(x-2)(x+1)}$

      Shared zero $x=2$, $m=n=1 \Rightarrow$ \ans{Hole}. Simplified: $\dfrac{x+2}{x+1}$. $L = \dfrac{4}{3}$.

      Denominator zero $x=-1$ (not shared) $\Rightarrow$ VA at $x=-1$.

      \leavevmode\rlap{\textit{Letter:}}\phantom{Answer:}\spacefactor2000{} \smash{\gradedblank[id=hw5, type=choice, points=1, answer=A]{1.5cm}{13pt}} \quad (Hole at $x=2$ with $L = \dfrac{4}{3}$; VA at $x=-1$.)

\end{enumerate}
