\begin{enumerate}[itemsep=10pt]

\item The polynomial $p(x) = -2x(x - 1)(x + 3)$ is written in factored form.

\begin{enumerate}[label=(\alph*), itemsep=6pt]
  \item Multiply only the leading terms of the factors, $-2 \cdot x \cdot x \cdot x$. What is the
        degree of $p$, and what is its leading coefficient?

        Degree: \ans{3} \qquad Leading coefficient: \ans{$-2$}
  \item Complete the end behavior.
        \quad As $x \to \infty$, $p(x) \to$ \ans{$-\infty$} \qquad
        As $x \to -\infty$, $p(x) \to$ \ans{$\infty$}
\end{enumerate}

\item A polynomial has degree 5. At most how many real zeros can it have? At most how many
turning points?

\vspace{0.1in}
Real zeros, at most: \ans{5} \qquad Turning points, at most: \ans{4}

\item Write each number using $i$, and simplify.

\begin{enumerate}[label=(\alph*), itemsep=6pt]
  \item $\sqrt{-9} = $ \ans{$3i$}
  \item $\sqrt{-20} = $ \ans{$2i\sqrt{5}$}
\end{enumerate}

\item The numbers $2 + 3i$ and $2 - 3i$ are a \textbf{conjugate pair}.

\begin{enumerate}[label=(\alph*), itemsep=6pt]
  \item Add them.

\begin{work}
  (2 + 3i) + (2 - 3i) &= (2 + 2) + (3i - 3i) \\
                      &= 4
\end{work}

        Sum: \ans{4}
  \item Multiply them.

\begin{work}
  (2 + 3i)(2 - 3i) &= 4 - 6i + 6i - 9i^2 \\
                   &= 4 - 9(-1) \\
                   &= 13
\end{work}

        Product: \ans{13}
  \item Is either answer a real number, or does each still have an $i$ in it?

        \ansline{Both are real: the $i$ parts cancel in the sum and in the product.}
\end{enumerate}

\end{enumerate}
