\begin{notesbox}{Vertical Asymptotes}
A rational function $r(x) = \dfrac{p(x)}{q(x)}$ has a \textbf{vertical asymptote at $x = a$} when:

\begin{itemize}
  \item $q(a) =$ \blank{1.5cm} \quad (the denominator is \blank{2.5cm} at $a$), \textbf{AND}
  \item $p(a)$ \blank{0.8cm} $0$ \quad (the numerator is \textbf{not} zero at $a$).
\end{itemize}

\textit{Multiplicity rule (EK 1.9.A.1):} A VA also occurs at $x = a$ when the multiplicity of
$a$ in the \blank{3cm} is \emph{greater} than its multiplicity in the \blank{3cm}.

\medskip
\textbf{Example 1.} Find any vertical asymptotes of $r(x) = \dfrac{x + 2}{x - 3}$.

\begin{enumerate}[label=\textbf{Step \arabic*.}, itemsep=4pt]
  \item Set the denominator equal to zero and solve: \blank{5cm}

  \item Check the numerator at each solution: at $x = 3$, numerator $=$ \blank{2cm}

  \item Conclusion: \blank{7cm}
\end{enumerate}

\medskip
\textbf{Example 2.} Find any vertical asymptotes of $r(x) = \dfrac{x - 1}{(x-1)^2(x+2)}$.

\begin{enumerate}[label=\textbf{Step \arabic*.}, itemsep=4pt]
  \item Denominator zeros: \blank{5cm}

  \item At $x = 1$: numerator multiplicity \blank{0.8cm}, denominator multiplicity \blank{0.8cm}.
        Since \blank{0.8cm} $>$ \blank{0.8cm}: \blank{4cm}

  \item At $x = -2$: numerator value $=$ \blank{2cm}. Since it is nonzero: \blank{4cm}
\end{enumerate}
\end{notesbox}

From the table, write the one-sided limits:

\[
  \lim_{x \to 2^-} r(x) = \blank{2cm}
  \qquad\qquad
  \lim_{x \to 2^+} r(x) = \blank{2cm}
\]

\textbf{Sign analysis.} When $x$ is slightly greater than 2, $(x - 2)$ is \blank{2cm},
so $r(x) \to$ \blank{2cm}.\\

so $r(x) \to$ \blank{2cm}.
\end{notesbox}
