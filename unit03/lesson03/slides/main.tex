% ROW 1 — ZEROS
\begin{frame}[t, plain]
  \plumheader{One Fact, Four Ways}
  \sectionlabel[goldacc]{Notes row ZEROS $\cdot$ I Do: problem 1 $\cdot$ We Do: problems 2--4}

  \begin{columns}[T]
    \column{0.5\textwidth}
      \begin{block}{For a real number \(a\), these say the same thing}
        \(a\) is a \textbf{zero} of \(p\): \(p(a) = 0\)\\[3pt]
        \(a\) is a \textbf{root} of \(p(x) = 0\)\\[3pt]
        \((a, 0)\) is an \textbf{\(x\)-intercept}\\[3pt]
        \((x - a)\) is a \textbf{linear factor} of \(p(x)\)
      \end{block}

      \medskip
      \textbf{Watch the sign:} the zero \(-2\) goes with the factor \((x + 2)\).

    \column{0.46\textwidth}
      \begin{center}
      \begin{tikzpicture}[x=0.72cm, y=0.12cm]
        \draw[linegray, very thin] (-3,-12) grid[xstep=1, ystep=4] (5,12);
        \draw[->, thick] (-3,0) -- (5.4,0) node[right] {\small \(x\)};
        \draw[->, thick] (0,-12) -- (0,13) node[above] {\small \(p(x)\)};
        \foreach \x in {-2,-1,1,2,3,4} \node[below, font=\scriptsize] at (\x,0) {\x};
        \begin{scope}\clip (-3,-12) rectangle (5,12);
          \draw[very thick, plum, domain=-2.5:4.55, samples=120, smooth]
            plot (\x, {(\x+2)*(\x-1)*(\x-4)});
        \end{scope}
        \foreach \x in {-2,1,4} \fill[plum] (\x,0) circle (2.5pt);
      \end{tikzpicture}

      \(p(x) = (x + 2)(x - 1)(x - 4)\)
      \end{center}
  \end{columns}
\end{frame}

% ROW 2 — MULTIPLICITY
\begin{frame}[t, plain]
  \plumheader{Multiplicity --- Counting Every Zero}
  \sectionlabel[goldacc]{Notes row MULTIPLICITY $\cdot$ I Do: problem 5 $\cdot$ We Do: problem 6}

  \begin{block}{Two rules}
    If \((x - a)\) appears \(k\) times, \(a\) is a zero of \textbf{multiplicity} \(k\).\\[3pt]
    \textbf{Fundamental Theorem of Algebra:} degree \(n\) \(\Rightarrow\) exactly \(n\) complex
    zeros, counting multiplicity.
  \end{block}

  \medskip
  \begin{columns}[T]
    \column{0.5\textwidth}
      \begin{center}
      {\large \(f(x) = 2(x - 3)^2(x + 1)^3\)}

      \medskip
      \renewcommand{\arraystretch}{1.3}
      \begin{tabular}{|c|c|c|}
        \hline
        \textbf{Factor} & \textbf{Zero} & \textbf{Mult.} \\ \hline
        \((x - 3)^2\) & 3 & 2 \\ \hline
        \((x + 1)^3\) & \(-1\) & 3 \\ \hline
      \end{tabular}

      \medskip
      \(2 + 3 = 5 =\) the degree. The 2 out front: no zero.
      \end{center}

    \column{0.46\textwidth}
      \textbf{Your turn.} Degree, zeros, multiplicities:
      \begin{itemize}\setlength{\itemsep}{6pt}
        \item \(g(x) = (x + 4)(x - 2)^3\)
        \item \(h(x) = -3x^2(x - 5)(x + 1)^2\)
      \end{itemize}
  \end{columns}
\end{frame}

% ROW 3 — CONJUGATE PAIRS
\begin{frame}[t, plain]
  \plumheader{Complex Zeros Come in Conjugate Pairs}
  \sectionlabel[goldacc]{Notes row CONJUGATE PAIRS $\cdot$ I Do: problem 7 $\cdot$ We Do: problems 8--10}

  \begin{columns}[T]
    \column{0.52\textwidth}
      \begin{block}{Real coefficients \(\Rightarrow\) pairs}
        If \(a + bi\) is a zero, so is \(a - bi\).\\[3pt]
        A non-real zero is \textbf{not} an \(x\)-intercept --- but it still counts toward the
        degree.
      \end{block}

      \medskip
      \(x^2 + 4 = 0\) \(\Rightarrow\) \(x^2 = -4\) \(\Rightarrow\) \(x = \pm 2i\)

      \smallskip
      Two zeros, \textbf{no} \(x\)-intercepts.

    \column{0.44\textwidth}
      \begin{itemize}\setlength{\itemsep}{6pt}
        \item \(2 + 3i\) is a zero. Name another.
        \item Every zero of \((x - 1)(x^2 + 9)\)?
        \item Degree 4: 3 (mult.\ 2) and \(1 + i\). All four?
        \item Why must a cubic have a real zero?
      \end{itemize}
  \end{columns}
\end{frame}

% DEBRIEF
\begin{frame}[t, plain]
  \plumheader{Debrief --- The Headline}
  \sectionlabel[goldacc]{Share out 15--18, one answer each $\cdot$ then say it back}

  \begin{columns}[T]
    \column{0.48\textwidth}
      \begin{block}{Count}
        Every factor is a zero. The exponents count them --- to exactly the
        \textbf{degree}.
      \end{block}
    \column{0.48\textwidth}
      \begin{block}{Cross or touch}
        Only an \textbf{odd} exponent crosses. An \textbf{even} one touches and turns back ---
        no sign change.
      \end{block}
  \end{columns}

  \bigskip
  \textbf{The hook, answered:} \(H\) never lost a zero. Its other two are a non-real conjugate
  pair --- the graph cannot show them.

  \bigskip

  factoring ourselves.
