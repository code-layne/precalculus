\item Vertical asymptotes:

\begin{enumerate}[label=(\alph*), itemsep=10pt]
  \item $f(x) = \dfrac{x + 3}{x^2 - 1} = \dfrac{x+3}{(x-1)(x+1)}$

  \item $g(x) = \dfrac{x^2 - 9}{x - 3} = \dfrac{(x-3)(x+3)}{x-3}$

        Factor the numerator: \smash{\gradedblank[id=hw1b1, type=short, points=1, answer={(x-3)(x+3)}]{4cm}{11pt}}

        Denominator zero: \smash{\gradedblank[id=hw1b2, type=short, points=1, answer={x = 3}]{2cm}{11pt}} \quad Numerator at $x=3$: \smash{\gradedblank[id=hw1b3, type=short, points=1, answer={0}, display={$0$ (numerator also zero)}]{3cm}{11pt}}

        Vertical asymptote(s): \smash{\gradedblank[id=hw1b4, type=short, check=text, points=1, answer={none}, display={None (hole at $x=3$)}]{4cm}{11pt}} The factor $(x-3)$ cancels; there is a hole at $x=3$.

  \item $h(x) = \dfrac{5}{(x - 1)(x + 4)}$

        Denominator zeros: \smash{\gradedblank[id=hw1c1, type=short, check=set, points=1, answer={1, -4}, display={$x = 1$ and $x = -4$}]{3.5cm}{11pt}}

        Numerator check: \smash{\gradedblank[id=hw1c2, type=rubric, points=1, display={$5 \ne 0$ at both}]{3.5cm}{11pt}}

        Vertical asymptote(s): \smash{\gradedblank[id=hw1c3, type=short, check=set, points=1, answer={1, -4}, display={$x = 1$ and $x = -4$}]{4cm}{11pt}}
\end{enumerate}

\item Sign analysis for $r(x) = \dfrac{2}{x + 1}$:

\begin{enumerate}[label=(\alph*), itemsep=6pt]
  \item Vertical asymptote at $x =$ \smash{\gradedblank[id=hw2a, type=short, points=1, answer={-1}]{1.5cm}{13pt}}.

  \item When $x$ is slightly greater than $-1$, $(x+1)$ is \smash{\gradedblank[id=hw2b1, type=short, check=text, points=1, answer={positive}]{2cm}{13pt}},
        so $r(x)$ is \smash{\gradedblank[id=hw2b2, type=short, check=text, points=1, answer={positive}]{2cm}{13pt}}.
        Therefore $\displaystyle\lim_{x \to -1^+} r(x) =$ \smash{\gradedblank[id=hw2b3, type=short, points=1, answer={\infty}, display={$+\infty$}]{2cm}{13pt}}.

  \item When $x$ is slightly less than $-1$, $(x+1)$ is \smash{\gradedblank[id=hw2c1, type=short, check=text, points=1, answer={negative}]{2cm}{13pt}},
        so $r(x)$ is \smash{\gradedblank[id=hw2c2, type=short, check=text, points=1, answer={negative}]{2cm}{13pt}}.
        Therefore $\displaystyle\lim_{x \to -1^-} r(x) =$ \smash{\gradedblank[id=hw2c3, type=short, points=1, answer={-\infty}]{2cm}{13pt}}.
\end{enumerate}

\item AP-Style Multiple Choice. \textit{Letter:}~\smash{\gradedblank[id=hw5, type=choice, points=1, answer=C]{0.42in}{13pt}} $x = 3$ only.

\textit{Work:} $r(x) = \dfrac{x^2-9}{x^2-6x+9} = \dfrac{(x-3)(x+3)}{(x-3)^2}$.
Numerator zero at $x = 3$ with multiplicity 1; denominator zero at $x = 3$ with multiplicity 2.
Since denominator multiplicity ($2$) $>$ numerator multiplicity ($1$), there is a VA at $x = 3$.
At $x = -3$: the denominator $(-3-3)^2 = 36 \ne 0$, so $x=-3$ is not a VA (not even a zero
of the denominator).

\end{enumerate}

\vspace{0.1in}

\begin{extensionbox}
\textbf{Extension answers.}
\begin{enumerate}[label=(\alph*), itemsep=4pt]
  \item Numerator: $(x-3)(x+3)$. Denominator: $(x-3)^2$.
  \item At $x = 3$: numerator multiplicity 1, denominator multiplicity 2. Since $2 > 1$,
        there is a vertical asymptote at $x = 3$ (the zero does NOT cancel).
  \item $x = -3$: the denominator $(x-3)^2 = (-3-3)^2 = 36 \ne 0$, so $x = -3$ is not
        a zero of the denominator and is therefore not a vertical asymptote.
\end{enumerate}
\end{extensionbox}
