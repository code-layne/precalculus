\begin{enumerate}[itemsep=14pt]

\item Vertical asymptotes:

\begin{enumerate}[label=(\alph*), itemsep=10pt]
  \item $f(x) = \dfrac{x + 3}{x^2 - 1} = \dfrac{x+3}{(x-1)(x+1)}$

        Denominator zeros: \ans{$x = 1$ and $x = -1$}

        Numerator check: \ans{at $x=1$: $1+3=4\ne0$;\quad at $x=-1$: $-1+3=2\ne0$}

        Vertical asymptote(s): \ans{$x = 1$ and $x = -1$}

  \item $g(x) = \dfrac{x^2 - 9}{x - 3} = \dfrac{(x-3)(x+3)}{x-3}$

        Factor the numerator: \ans{$(x-3)(x+3)$}

        Denominator zero: \ans{$x = 3$} \quad Numerator at $x=3$: \ans{$0$ (numerator also zero)}

        Vertical asymptote(s): \ans{None. The factor $(x-3)$ cancels; there is a hole at $x=3$.}

  \item $h(x) = \dfrac{5}{(x - 1)(x + 4)}$

        Denominator zeros: \ans{$x = 1$ and $x = -4$}

        Numerator check: \ans{numerator is $5 \ne 0$ at both points}

        Vertical asymptote(s): \ans{$x = 1$ and $x = -4$}
\end{enumerate}

\item Sign analysis for $r(x) = \dfrac{2}{x + 1}$:

\begin{enumerate}[label=(\alph*), itemsep=6pt]
  \item Vertical asymptote at $x =$ \ans{$-1$}.

  \item When $x$ is slightly greater than $-1$, $(x+1)$ is \ans{positive},
        so $r(x)$ is \ans{positive}.
        Therefore $\displaystyle\lim_{x \to -1^+} r(x) =$ \ans{$+\infty$}.

  \item When $x$ is slightly less than $-1$, $(x+1)$ is \ans{negative},
        so $r(x)$ is \ans{negative}.
        Therefore $\displaystyle\lim_{x \to -1^-} r(x) =$ \ans{$-\infty$}.
\end{enumerate}

\item Conceptual question.

\ans{No. The number of vertical asymptotes of a rational function cannot exceed the degree of
the denominator, because the denominator can have at most as many distinct real zeros as its
degree. Each VA requires a distinct zero of the denominator.}

\item Interpreting limit notation.

\ans{Both one-sided limits equal $+\infty$, so $r(x)$ grows without bound on \emph{both}
sides of $x = 2$. Yes, $x = 2$ is a vertical asymptote. On the graph, $r(x) \to +\infty$
as $x$ approaches 2 from either side --- the graph rises steeply toward the asymptote from
both the left and the right. (This occurs when the denominator factor at $x=2$ is raised to
an even power.)}

\item AP-Style Multiple Choice. \ans{(C) $x = 3$ only.}

\textit{Work:} $r(x) = \dfrac{x^2-9}{x^2-6x+9} = \dfrac{(x-3)(x+3)}{(x-3)^2}$.
Numerator zero at $x = 3$ with multiplicity 1; denominator zero at $x = 3$ with multiplicity 2.
Since denominator multiplicity ($2$) $>$ numerator multiplicity ($1$), there is a VA at $x = 3$.
At $x = -3$: the denominator $(-3-3)^2 = 36 \ne 0$, so $x=-3$ is not a VA (not even a zero
of the denominator).

\end{enumerate}

\vspace{0.1in}

\begin{extensionbox}
\textbf{Extension answers.}
\begin{enumerate}[label=(\alph*), itemsep=4pt]
  \item Numerator: $(x-3)(x+3)$. Denominator: $(x-3)^2$.
  \item At $x = 3$: numerator multiplicity 1, denominator multiplicity 2. Since $2 > 1$,
        there is a vertical asymptote at $x = 3$ (the zero does NOT cancel).
  \item $x = -3$: the denominator $(x-3)^2 = (-3-3)^2 = 36 \ne 0$, so $x = -3$ is not
        a zero of the denominator and is therefore not a vertical asymptote.
\end{enumerate}
\end{extensionbox}
