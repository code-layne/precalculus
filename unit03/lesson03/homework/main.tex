\item \textbf{One-sided limits via sign analysis.} For $r(x) = \dfrac{2}{x + 1}$:

\begin{enumerate}[label=(\alph*), itemsep=6pt]
  \item The vertical asymptote is at $x =$ \smash{\gradedblank[id=hw2a, type=short, points=1, answer={-1}]{1.5cm}{13pt}}.

  \item When $x$ is slightly greater than $-1$, $(x + 1)$ is \smash{\gradedblank[id=hw2b1, type=short, check=text, points=1, answer={positive}]{2cm}{13pt}} (positive/negative),
        so $r(x)$ is \smash{\gradedblank[id=hw2b2, type=short, check=text, points=1, answer={positive}]{2cm}{13pt}}. Therefore $\displaystyle\lim_{x \to -1^+} r(x) =$ \smash{\gradedblank[id=hw2b3, type=short, points=1, answer={\infty}, display={$+\infty$}]{2cm}{13pt}}.

  \item When $x$ is slightly less than $-1$, $(x + 1)$ is \smash{\gradedblank[id=hw2c1, type=short, check=text, points=1, answer={negative}]{2cm}{13pt}} (positive/negative),
        so $r(x)$ is \smash{\gradedblank[id=hw2c2, type=short, check=text, points=1, answer={negative}]{2cm}{13pt}}. Therefore $\displaystyle\lim_{x \to -1^-} r(x) =$ \smash{\gradedblank[id=hw2c3, type=short, points=1, answer={-\infty}]{2cm}{13pt}}.
\end{enumerate}

\item \textbf{Conceptual question.} Can a rational function have more vertical asymptotes than
      the degree of its denominator? Explain your reasoning in one or two sentences.

      \rubricbox[0.3in]{\gradedblank[id=hw3, type=rubric, points=2,
  display={No. Each vertical asymptote needs a distinct real zero of the denominator, and a polynomial has at most as many distinct real zeros as its degree.}]{\linewidth}{49pt}}

\item \textbf{Interpreting limit notation.} Suppose $\displaystyle\lim_{x \to 2^+} r(x) = +\infty$
      and $\displaystyle\lim_{x \to 2^-} r(x) = +\infty$.

      What can you conclude about the behavior of $r$ near $x = 2$? Is $x = 2$ a vertical
      asymptote? Describe how the graph looks on both sides of $x = 2$.

      \rubricbox[0.3in]{\gradedblank[id=hw4, type=rubric, points=3,
  display={Yes, $x = 2$ is a vertical asymptote: $r(x)$ grows without bound on both sides. The graph rises steeply toward the asymptote from the left and from the right.}]{\linewidth}{50pt}}

\item \textbf{AP-Style Multiple Choice.} Write the letter of the single best answer in its Letter box. For the rational function

\begin{enumerate}[label=(\Alph*), itemsep=3pt]
  \item $x = 3$ and $x = -3$
  \item $x = -3$ only
  \item $x = 3$ only
  \item There are no vertical asymptotes. \hfill\textit{Letter:}~\gradedblank[id=hw5, type=choice, points=1, answer=C]{0.42in}{13pt}
\end{enumerate}

\end{enumerate}
