% ROW 1 — SUCCESSIVE DIFFERENCES
\begin{frame}[t, plain]
  \plumheader{Successive Differences --- What Degree Is It?}
  \sectionlabel[goldacc]{Notes row SUCCESSIVE DIFFERENCES $\cdot$ I Do: problem 1 $\cdot$ We Do: problems 2--4}

  \begin{columns}[T]
    \column{0.52\textwidth}
      \begin{center}
      $p(x) = x^3 - 6x^2 + 9x + 2$\par\medskip
      {\small
      \renewcommand{\arraystretch}{1.25}
      \setlength{\tabcolsep}{3pt}
      \begin{tabular}{r|*{11}{c}}
      $x$    & 0 &   & 1 &    & 2 &    & 3 &   & 4 &    & 5  \\
      $p(x)$ & 2 &   & 6 &    & 4 &    & 2 &   & 6 &    & 22 \\ \hline
      1st    &   & 4 &   & $-2$ &   & $-2$ &   & 4 &   & 16 &    \\
      2nd    &   &   & $-6$ &  & 0 &    & 6 &   & 12 &   &    \\
      3rd    &   &   &   & \textbf{6}  &   & \textbf{6}  &   & \textbf{6} &   &    &    \\
      \end{tabular}}
      \end{center}

    \column{0.44\textwidth}
      \begin{block}{The rule}
        \textbf{Equal input steps first.} Subtract neighbors (right minus left), then again.
        The first row of constant differences is the \textbf{degree}.
      \end{block}

      \medskip
      \begin{itemize}\setlength{\itemsep}{4pt}
        \item Constant 1st: linear $\cdot$ 2nd: quadratic $\cdot$ 3rd: cubic
        \item $p$: third differences 6, 6, 6 --- \textbf{degree 3}
        \item Steps of 1, 2, 1, 2? The differences say nothing.
      \end{itemize}
  \end{columns}
\end{frame}

% ROW 2 — RATES OF CHANGE
\begin{frame}[t, plain]
  \plumheader{Rates of Change --- the Secant Slope}
  \sectionlabel[goldacc]{Notes row RATES OF CHANGE $\cdot$ I Do: problem 5 $\cdot$ We Do: problems 6--8}

  \begin{columns}[T]
    \column{0.48\textwidth}
      \begin{center}
      \begin{tikzpicture}[x=0.75cm, y=0.19cm, >=stealth]
        \paxes
        \draw[very thick, plum, <->] \pcurve;
        \draw[very thick, goldacc, dashed] (-0.45,0.2) -- (1.6,8.4);
        \draw[thick, redacc] (0,2) -- (1,2) -- (1,6);
        \foreach \p in {(0,2),(1,6),(2,4),(3,2),(4,6)} \fill[plum] \p circle (2.3pt);
      \end{tikzpicture}
      \end{center}
      {\footnotesize Run 1, rise 4: the secant slope is 4 --- the first difference.}

    \column{0.48\textwidth}
      \textbf{Average rate of change on $[a, b]$} --- the secant slope:
      \[ \frac{p(b) - p(a)}{b - a} \]
      \textbf{Difference quotient} --- the rate on $[x, x + h]$:
      \[ \frac{f(x + h) - f(x)}{h}, \quad h \ne 0 \]
      Replace \emph{every} $x$ with $(x + h)$ first.

      \medskip
      $[0, 2]$: $\frac{4 - 2}{2} = 1$. \qquad $[1, 3]$: $\frac{2 - 6}{2} = -2$.
  \end{columns}
\end{frame}

  \sectionlabel[goldacc]{Notes row SHAPE OF $p$ $\cdot$ I Do: problem 9 $\cdot$ We Do: problem 10}

        \foreach \p in {(1,6),(3,2),(4,6)} \fill[plum] \p circle (2.3pt);
        \draw[redacc, very thick] (0,2) circle (3.5pt);
        \draw[goldacc, very thick] (2,4) circle (5pt);
      \end{tikzpicture}
      \end{center}

    \column{0.48\textwidth}
      \begin{itemize}\setlength{\itemsep}{5pt}
        \item \textbf{$y$-intercept:} $p(0) = 2$, the constant term --- $(0, 2)$
        \item \textbf{Increasing} on $(-\infty, 1)$ and $(3, \infty)$; \textbf{decreasing} on
              $(1, 3)$ --- intervals in $x$
        \item \textbf{Concave down} on $(-\infty, 2)$, \textbf{up} on $(2, \infty)$
        \item \textbf{Inflection point} $(2, 4)$: the rates 4, $-2$, $-2$, 4, 16 stop falling
              and start rising
      \end{itemize}
  \end{columns}
\end{frame}

    \column{0.36\textwidth}
      \begin{itemize}\setlength{\itemsep}{5pt}
        \item Kai: ``The relative max of $p$ is 6, so 6 is the highest $p$ goes.'' \quad
              $p(5) = $ \textbf{?}
        \item $q$ has two valleys. Which one is the \textbf{absolute} minimum?
        \item Odd degree: \textbf{no} absolute max or min. Even degree: one kind.
        \item Restricted domain? Check the \textbf{endpoints}.
      \end{itemize}
  \end{columns}

  \medskip
  \textbf{A relative maximum is the highest point nearby --- not the highest point of all.}

% DEBRIEF
\begin{frame}[t, plain]
  \plumheader{Debrief --- The Headline}
  \sectionlabel[goldacc]{Share out 15--18, one answer each $\cdot$ then say it back}

  \begin{columns}[T]
    \column{0.48\textwidth}
      \begin{block}{Highest nearby, or highest of all?}
        A \textbf{relative} maximum is the highest point \emph{nearby}.\\[3pt]
        The \textbf{absolute} maximum is the highest of all --- check the ends.
      \end{block}
    \column{0.48\textwidth}
      \begin{block}{What degree?}
        \textbf{Equal steps first.} Then the first constant row of differences is the
        degree.
      \end{block}
  \end{columns}

  polynomial go --- the arrows that decided today's absolute extrema.
