\item Explain why $\displaystyle\lim_{x \to \infty} \frac{7x^3 + 2}{x^3 - 1} = 7$.

\vspace{0.28in}
\writegap\\[4pt]
\writegap\\[4pt]
\writebox{\gradedblank[id=hw2, type=rubric, points=2,
  display={For large $x$, $7x^3$ dominates the numerator and $x^3$ the denominator; the other terms are negligible. The quotient of leading terms is $7x^3/x^3 = 7$, so the outputs approach 7.}]{\linewidth}{56pt}}

\item A rational function $r$ satisfies $\displaystyle\lim_{x \to \infty} r(x) = 0$.
What can you conclude about the degrees of the numerator and denominator?

\vspace{0.28in}
\writegap\\[4pt]
\writebox{\gradedblank[id=hw3, type=rubric, points=2,
  display={The numerator's degree is strictly less than the denominator's: the denominator dominates, so the quotient of leading terms approaches 0.}]{\linewidth}{38pt}}

\item A rational function has a slant asymptote.

\begin{enumerate}[label=(\alph*), itemsep=6pt]
  \item What must be true about the degrees of the numerator and denominator?

        \vspace{0.1in}
        \writebox{\gradedblank[id=hw4a, type=rubric, points=1,
  display={deg numerator $=$ deg denominator $+ 1$.}]{\linewidth}{16pt}}

  \item Can you write the end behavior as a single limit value $\lim_{x\to\infty} r(x) = k$?

        \vspace{0.1in}
        \writegap\\[4pt]
        \writebox{\gradedblank[id=hw4b, type=rubric, points=2,
  display={No. The quotient of leading terms is a non-constant linear expression, so the outputs grow without bound: $\lim_{x\to\infty} r(x) = \pm\infty$.}]{\linewidth}{34pt}}
\end{enumerate}

\item \textbf{AP-Style Multiple Choice.} Which function has a HA at $y = 3/2$?

\textit{Letter:}~\smash{\gradedblank[id=hw5, type=choice, points=1, answer=B]{0.42in}{13pt}} $\dfrac{3x^2 - 1}{2x^2 + 7}$ --- the quotient of leading terms is $3x^2/2x^2 = 3/2$.

\textit{(A): deg num $<$ deg denom $\Rightarrow$ HA at $y = 0$.
(C): quotient is $2/3$.
(D): deg num $>$ deg denom $\Rightarrow$ polynomial end behavior.)}

\end{enumerate}
