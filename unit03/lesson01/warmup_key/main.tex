\item Let $f(x) = x^3 - 2x^2 + 3$. Find each value.

\begin{enumerate}[label=(\alph*), itemsep=8pt]
  \item $f(2)$ \hfill \ans{$3$}
  \item $f(-1)$ \hfill \ans{$0$}
\end{enumerate}

\begin{work}
  f(2)  &= 2^3 - 2(2)^2 + 3 \\
        &= 3 \\
  f(-1) &= (-1)^3 - 2(-1)^2 + 3 \\
        &= 0
\end{work}

\item Find the average rate of change of $g(x) = x^2 + 1$ on the interval $[1, 4]$.

\begin{work}
  \frac{g(4) - g(1)}{4 - 1} &= \frac{17 - 2}{3} \\
                            &= 5
\end{work}

Average rate of change: \ans{$5$}
