\begin{enumerate}[itemsep=14pt]

\item Consider the polynomial $p(x) = -4x^3 + 2x - 9$.

\begin{enumerate}[label=(\alph*), itemsep=5pt]
  \item What is the degree of $p$? \ans{3} \quad
        What is its leading coefficient? \ans{$-4$}
  \item State the end behavior of $p$ using limit notation.

        \medskip
        $\displaystyle\lim_{x \to \infty} p(x) = $ \ans{$-\infty$}
        \qquad
        $\displaystyle\lim_{x \to -\infty} p(x) = $ \ans{$+\infty$}
\end{enumerate}

\item For the polynomial $p(x) = 5x^4 - 3x^2 + 1$:

\begin{enumerate}[label=(\alph*), itemsep=5pt]
  \item As $x \to \infty$, which term dominates? \ans{$5x^4$}
  \item Complete using limit notation:

        \medskip
        $\displaystyle\lim_{x \to \infty} p(x) = $ \ans{$+\infty$}
        \qquad
        $\displaystyle\lim_{x \to -\infty} p(x) = $ \ans{$+\infty$}
\end{enumerate}

\item A polynomial $q$ satisfies
$\displaystyle\lim_{x\to\infty} q(x) = -\infty$
and
$\displaystyle\lim_{x\to -\infty} q(x) = -\infty$.

What can you conclude about the degree and leading coefficient of $q$?
Circle all that apply, then explain briefly.

\medskip
\begin{tabular}{@{}llll@{}}
\ans{(A)} Even degree &
\textbf{(B)} Odd degree &
\textbf{(C)} Positive leading coefficient &
\ans{(D)} Negative leading coefficient
\end{tabular}

\vspace{0.15in}
\ansline{Both limits equal $-\infty$, so the end behavior is the same in both directions.
This means the degree is \textbf{even} and the leading coefficient is \textbf{negative}
(an even-degree poly with negative leading coeff falls on both ends).}

\end{enumerate}
