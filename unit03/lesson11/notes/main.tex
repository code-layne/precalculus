\begin{vocabbox}
\begin{description}[leftmargin=0pt, itemsep=6pt]
  \item[\termblanklong{secant}]
    $\sec\theta = \dfrac{1}{\underline{\hspace{2.2cm}}}$;
    the reciprocal of cosine.
    Undefined when $\cos\theta = $ \underline{\hspace{1cm}}.

  \item[\termblanklong{cosecant}]
    $\csc\theta = \dfrac{1}{\underline{\hspace{2.2cm}}}$;
    the reciprocal of sine.
    Undefined when $\sin\theta = $ \underline{\hspace{1cm}}.

  \item[\termblanklong{cotangent}]
    $\cot\theta = \dfrac{\underline{\hspace{1.8cm}}}{\underline{\hspace{1.8cm}}}$;
    the reciprocal of tangent.
    Undefined when $\sin\theta = $ \underline{\hspace{1cm}}.

  \item[\termblanklong{vertical asymptote}]
    A vertical line $\theta = a$ where the function value grows without
    \underline{\hspace{2.5cm}}; caused by a \underline{\hspace{1.5cm}} in the denominator.

  \item[\termblanklong{reciprocal function}]
    A function formed by $\dfrac{1}{f(\theta)}$.
    Zeros of $f$ become \underline{\hspace{2.5cm}} of the reciprocal,
    and $f(\theta) = \pm 1$ are \underline{\hspace{2cm}} points (outputs unchanged).
\end{description}
\end{vocabbox}

\begin{notesbox}{LO 3.11.A --- Part 1: Secant and Cosecant Functions}

\hline

\begin{notesbox}{LO 3.11.A --- Part 2: Cotangent Function}

\textbf{Cotangent:}
$\cot\theta = \dfrac{\cos\theta}{\sin\theta} = \dfrac{1}{\tan\theta}$
\quad (where $\sin\theta \neq 0$)

\vspace{6pt}
\renewcommand{\arraystretch}{1.7}
\begin{tabular}{|c|c|c|c|}
\hline
$\theta$ & $\sin\theta$ & $\cos\theta$ & $\cot\theta = \dfrac{\cos\theta}{\sin\theta}$ \\
\hline
$0^+$ (approaching 0) & $\approx 0^+$ & $1$ & \\
\hline
$\dfrac{\pi}{4}$ & $\dfrac{\sqrt{2}}{2}$ & $\dfrac{\sqrt{2}}{2}$ & \\
\hline
$\dfrac{\pi}{2}$ & $1$ & $0$ & \\
\hline
$\dfrac{3\pi}{4}$ & $\dfrac{\sqrt{2}}{2}$ & $-\dfrac{\sqrt{2}}{2}$ & \\
\hline
$\pi^-$ (approaching $\pi$) & $\approx 0^+$ & $-1$ & \\
\hline
\end{tabular}

\vspace{6pt}
As $\theta$ increases from $0$ to $\pi$, cotangent is \underline{\hspace{3cm}}
(circle: \textit{increasing} / \textit{decreasing}).

\vspace{4pt}
Vertical asymptotes of $\cot\theta$: $\theta = \underline{\hspace{4cm}}$ for integer $n$

\vspace{4pt}
Period of $\cot\theta$: \underline{\hspace{2.5cm}}
\hspace{1.5cm}
Range of $\cot\theta$: \underline{\hspace{4.5cm}}

\end{notesbox}
