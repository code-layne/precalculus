\begin{enumerate}[itemsep=10pt]

\item Divide $157 \div 12$ by long division. Then write the result in the form
      dividend $=$ divisor $\cdot$ quotient $+$ remainder.

\begin{work}
  &\begin{array}{r@{\,}ccc}
               &     & 1      & 3      \\ \cline{2-4}
    12\,\big)  & 1   & 5      & 7      \\
               & -(1 & 2)     &        \\ \cline{2-3}
               &     & 3      & 7      \\
               &     & -(3    & 6)     \\ \cline{3-4}
               &     &        & 1
  \end{array}
\end{work}

$157 = 12 \cdot$ \ans{13} $+$ \ans{1}

\item Evaluate $p(x) = x^3 - 4x + 5$ at $x = -2$. Put the $-2$ in parentheses first.

\begin{work}
  p(-2) &= (-2)^3 - 4(-2) + 5 \\
        &= -8 + 8 + 5 \\
        &= 5
\end{work}

$p(-2) = $ \ans{5}

\item Factor $x^2 - x - 6$. Then name both zeros of $x^2 - x - 6$.

\begin{work}
  x^2 - x - 6 &= (x - 3)(x + 2) \\
            x &= 3 \text{ or } x = -2
\end{work}

Factored: \ans{$(x - 3)(x + 2)$} \qquad Zeros: \ans{$3$ and $-2$}

\item Simplify and write in standard form (highest power first):
      \quad $(x^3 + 4x - 1) - (x^3 - 2x^2 + x)$.
      Which power of $x$ is \textbf{missing} from $x^3 + 4x - 1$?

\begin{work}
  (x^3 + 0x^2 + 4x - 1) - (x^3 - 2x^2 + x) &= x^3 - x^3 + 0x^2 + 2x^2 + 4x - x - 1 \\
                                            &= 2x^2 + 3x - 1
\end{work}

Result: \ans{$2x^2 + 3x - 1$} \qquad Missing power: \ans{$x^2$}

\end{enumerate}
