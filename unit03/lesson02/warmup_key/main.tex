\begin{enumerate}[itemsep=14pt]

\item For the rational function $\displaystyle r(x) = \frac{4x^2 + 1}{2x^2 - 8}$,
      find $\displaystyle\lim_{x \to \infty} r(x)$.
      Show your reasoning by identifying the leading terms.

      \vspace{0.06in}
      Leading term of numerator: \ans{$4x^2$} \quad
      Leading term of denominator: \ans{$2x^2$}

      \vspace{0.08in}
      $\displaystyle\lim_{x \to \infty} r(x) = $ \ans{$\dfrac{4}{2} = 2$}

\item A rational function has a horizontal asymptote at $y = 0$.

      \begin{enumerate}[label=(\alph*), itemsep=6pt]
        \item What does this tell you about the \textit{end behavior} of the function?

              \vspace{0.06in}
              \ansline{As $x \to \pm\infty$, the output approaches $0$; the graph gets closer and closer to the $x$-axis but never reaches it (in general).}

        \item Under what degree condition on the numerator and denominator does this occur?

              \vspace{0.06in}
              \ansline{The degree of the numerator is \textbf{less than} the degree of the denominator.}
      \end{enumerate}

\item Factor completely: $x^2 - 5x + 6$.

      \vspace{0.08in}
      $x^2 - 5x + 6 = $ \ans{$(x - 2)(x - 3)$}

      What are the solutions to $x^2 - 5x + 6 = 0$? \quad $x = $ \ans{$2$ and $x = 3$}

\end{enumerate}
