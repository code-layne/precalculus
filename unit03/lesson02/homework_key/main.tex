        \item $\displaystyle f(x) = \dfrac{x^2 - 9}{x + 5}$.
              Numerator: $(x-3)(x+3)$.

              \vspace{0.06in}
              Domain restriction(s): \smash{\gradedblank[id=hw1a1, type=short, points=1, answer={x = -5}]{3.5cm}{13pt}} \qquad Zeros: \smash{\gradedblank[id=hw1a2, type=short, check=set, points=1, answer={3, -3}, display={$x = 3$ and $x = -3$}]{4cm}{13pt}}

        \item $\displaystyle g(x) = \dfrac{x^2 - 2x - 8}{x^2 - 4} = \dfrac{(x-4)(x+2)}{(x-2)(x+2)}$.
              Domain restrictions: $x = 2$ and $x = -2$.
              Numerator zero $x = 4$ is in the domain; $x = -2$ is a domain restriction (not a zero).

              \vspace{0.06in}
              Domain restriction(s): \smash{\gradedblank[id=hw1b1, type=short, check=set, points=1, answer={2, -2}, display={$x = 2$ and $x = -2$}]{3.5cm}{13pt}} \qquad Zeros: \smash{\gradedblank[id=hw1b2, type=short, points=1, answer={x = 4}, display={$x = 4$ only}]{4cm}{13pt}}

        \item $\displaystyle h(x) = \dfrac{3x + 6}{x^2 + 1} = \dfrac{3(x+2)}{x^2+1}$.
              $x^2 + 1 > 0$ for all real $x$, so domain is all reals.

              \vspace{0.06in}
              Domain restriction(s): \smash{\gradedblank[id=hw1c1, type=short, check=text, points=1, answer={none}, display={None}]{3.5cm}{13pt}} \qquad Zeros: \smash{\gradedblank[id=hw1c2, type=short, points=1, answer={x = -2}]{4cm}{13pt}}
      \end{enumerate}

\item $r(x) = \dfrac{2x^2 + x - 3}{x - 5} = \dfrac{(2x + 3)(x - 1)}{x - 5}$.
      Numerator zeros: $x = 1$ and $x = -\tfrac{3}{2}$. Domain restriction: $x = 5$.
      Both $x = 1$ and $x = -\tfrac{3}{2}$ are in the domain.

      \textit{Letter:}~\gradedblank[id=hw5, type=choice, points=1, answer=B]{0.42in}{13pt} (B) $x = 1$. (A) is also a zero; (B) is the intended answer.

      \textit{Note to teacher: $2(1)^2 + 1 - 3 = 0$ and $1 \ne 5$, so $x = 1$ is a zero.
      Also $x = -3/2$ satisfies $2(9/4) - 3/2 - 3 = 9/2 - 3/2 - 6/2 = 0$, so (A) is also a zero.
      If the MC has a unique answer, (B) $x = 1$ is the cleaner choice.}
