\begin{scenariobox}[Break-Even Analysis]{plumlight}
\small
A factory models its profit (or loss) per unit using the rational function
\[
  r(x) = \frac{x^2 - 7x + 10}{x - 4},
\]
where $x > 0$ represents units produced (in hundreds) and $x \ne 4$.
Positive values of $r(x)$ mean profit; negative values mean loss.
\end{scenariobox}

\vspace{0.08in}

\begin{tcolorbox}[colback=white, colframe=black!40,
    title=\textbf{Tier R --- Remediate},
    fonttitle=\bfseries, arc=2mm, left=3mm, right=3mm, top=2mm, bottom=2mm]

\begin{enumerate}[itemsep=10pt]

  \item Factor the numerator $x^2 - 7x + 10$ completely.

        $x^2 - 7x + 10 = $ \ans{$(x - 2)(x - 5)$}

  \item At what values of $x$ does $r(x) = 0$?

        $x = $ \ans{$2$ and $x = 5$}

  \item Is $x = 4$ in the domain of $r$? \quad \textbf{No} (circle).

        Why? \ansline{$x = 4$ makes the denominator $x - 4 = 0$, so $r(4)$ is undefined.}

  \item Critical values on the number line: numerator zeros $x = 2$ (\textbullet) and $x = 5$ (\textbullet);
        denominator zero $x = 4$ (open dot).

        \vspace{0.18in}
        \begin{center}
        \begin{tikzpicture}
          \draw[->] (-0.5,0) -- (7.5,0) node[right] {$x$};
          \foreach \x in {0,1,2,3,4,5,6,7} {
            \draw (\x,0.08) -- (\x,-0.08) node[below] {\small\x};
          }
          \fill (2,0) circle (3pt);
          \fill (5,0) circle (3pt);
          \draw (4,0) circle (3pt);
        \end{tikzpicture}
        \end{center}
\end{enumerate}
\end{tcolorbox}

\vspace{0.08in}

\begin{tcolorbox}[colback=white, colframe=black!40,
    title=\textbf{Tier A --- Apply},
    fonttitle=\bfseries, arc=2mm, left=3mm, right=3mm, top=2mm, bottom=2mm]

\begin{enumerate}[itemsep=10pt]

  \item Domain restriction: \ans{$x = 4$} \qquad
        Numerator zeros: \ans{$x = 2,\; x = 5$} \qquad
        Zeros of $r$: \ans{$x = 2$ and $x = 5$}

  \item Sign chart:

        \medskip
        \begin{center}
        \renewcommand{\arraystretch}{1.6}
        \begin{tabular}{|c|c|c|c|c|}
        \hline
        \textbf{Interval} & $(0,\,2)$ & $(2,\,4)$ & $(4,\,5)$ & $(5,\,\infty)$ \\
        \hline
        Test value  & \ans{$1$}  & \ans{$3$}  & \ans{$4.5$} & \ans{$6$} \\
        \hline
        Sign of $r$ & \ans{$(+)$} & \ans{$(-)$} & \ans{$(-)$} & \ans{$(+)$} \\
        \hline
        \end{tabular}
        \end{center}

        \textit{Check at $x=1$: $r(1) = (1-2)(1-5)/(1-4) = (-1)(-4)/(-3) = 4/(-3) < 0$\ldots}
        \textit{Wait --- recheck: $(1-7+10)/(1-4) = 4/(-3) < 0$. Hmm.}

        \textit{Correction: at $x = 1$ (in $(0,2)$): $r(1) = (1-7+10)/(1-4) = 4/(-3) \approx -1.33 < 0$, so $(0,2)$ is $(-)$.
        At $x = 3$ (in $(2,4)$): $r(3)=(9-21+10)/(3-4) = -2/(-1) = 2 > 0$, so $(2,4)$ is $(+)$.
        At $x = 4.5$: $r(4.5) = (20.25-31.5+10)/(0.5) = -1.25/0.5 < 0$, so $(4,5)$ is $(-)$.
        At $x = 6$: $r(6) = (36-42+10)/(2) = 4/2 = 2 > 0$, so $(5,\infty)$ is $(+)$.}

        \medskip
        \begin{center}
        \renewcommand{\arraystretch}{1.6}
        \begin{tabular}{|c|c|c|c|c|}
        \hline
        \textbf{Interval} & $(0,\,2)$ & $(2,\,4)$ & $(4,\,5)$ & $(5,\,\infty)$ \\
        \hline
        Test value  & \ans{$1$}  & \ans{$3$}  & \ans{$4.5$} & \ans{$6$} \\
        \hline
        Sign of $r$ & \ans{$(-)$} & \ans{$(+)$} & \ans{$(-)$} & \ans{$(+)$} \\
        \hline
        \end{tabular}
        \end{center}

  \item $r(x) \geq 0$ on intervals where sign is $(+)$, plus the zeros $x = 2$ and $x = 5$:

        Solution: \ans{$[2,\,4)\cup[5,\,\infty)$}

\end{enumerate}
\end{tcolorbox}

\vspace{0.08in}

\begin{tcolorbox}[colback=white, colframe=black!40,
    title=\textbf{Tier E --- Extend},
    fonttitle=\bfseries, arc=2mm, left=3mm, right=3mm, top=2mm, bottom=2mm]

\begin{enumerate}[itemsep=10pt]

  \item $s(x) = \dfrac{x^2-4}{x^2-9} = \dfrac{(x-2)(x+2)}{(x-3)(x+3)}$.

        Domain restrictions: \ans{$x = 3$ and $x = -3$} \qquad
        Numerator zeros: \ans{$x = 2$ and $x = -2$} \qquad
        Zeros of $s$: \ans{$x = 2$ and $x = -2$}

  \item Sign chart (test values: $x = -4,\,-2.5,\,0,\,2.5,\,4$):

        \medskip
        \begin{center}
        \renewcommand{\arraystretch}{1.6}
        \begin{tabular}{|c|c|c|c|c|c|}
        \hline
        \textbf{Interval} & $(-\infty,-3)$ & $(-3,-2)$ & $(-2,2)$ & $(2,3)$ & $(3,\infty)$ \\
        \hline
        Test value  & \ans{$-4$} & \ans{$-2.5$} & \ans{$0$} & \ans{$2.5$} & \ans{$4$} \\
        \hline
        Sign of $s$ & \ans{$(+)$} & \ans{$(-)$} & \ans{$(+)$} & \ans{$(-)$} & \ans{$(+)$} \\
        \hline
        \end{tabular}
        \end{center}

  \item $s(x) > 0$ (strict inequality, so zeros $x = \pm 2$ are excluded):

        Solution: \ans{$(-\infty,\,-3)\cup(-2,\,2)\cup(3,\,\infty)$}

  \item \ansline{No. $x = 3$ and $x = -3$ are NOT zeros of $s$ because they are not in the domain of $s$; $s(3)$ and $s(-3)$ are undefined, so the function cannot equal zero at those values.}

\end{enumerate}
\end{tcolorbox}
